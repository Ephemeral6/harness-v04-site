def W : Nat := 64
def bt (r c : Nat) : Nat := 1 <<< (W * r + c)
def dif (a b : Nat) : Nat := a ^^^ (a &&& b)          -- 集合差 a \ b

/-- 从最高位往下逐个摘位，结果按位序升序（= 先行后列）。fuel 只是结构递归的上界。 -/
def bitsAsc : Nat → Nat → List Nat
  | 0, _ => []
  | n + 1, x => if x == 0 then [] else
      let p := Nat.log2 x
      bitsAsc n (x ^^^ (1 <<< p)) ++ [p]

def popc : Nat → Nat → Nat
  | 0, _ => 0
  | n + 1, x => if x == 0 then 0 else 1 + popc n (x ^^^ (1 <<< Nat.log2 x))

def pc (x : Nat) : Nat := popc 4096 x

/-- 左上角 (r,c)、边长 k 的实心方块。 -/
def sqm (r c k : Nat) : Nat :=
  (List.range k).foldl (fun acc i => acc ||| (((1 <<< k) - 1) <<< (W * (r + i) + c))) 0

/-- 闭区间矩形 [r0,r1] × [c0,c1]。 -/
def rect (r0 r1 c0 c1 : Nat) : Nat :=
  (List.range (r1 + 1 - r0)).foldl (fun acc i => acc ||| (((1 <<< (c1 + 1 - c0)) - 1) <<< (W * (r0 + i) + c0))) 0

/-- 跨度 span 的方框四角（左上角在 (r,c)）。 -/
def corners (r c span : Nat) : Nat :=
  bt r c ||| bt r (c + span - 1) ||| bt (r + span - 1) c ||| bt (r + span - 1) (c + span - 1)

/-- 四角都落在 M 里的那些跨度 span 方框的左上角。 -/
def cornerHits (M span : Nat) : Nat :=
  let d := span - 1
  M &&& (M >>> d) &&& (M >>> (W * d)) &&& (M >>> (W * d + d))

def expandC (X span : Nat) : Nat :=
  let d := span - 1
  X ||| (X <<< d) ||| (X <<< (W * d)) ||| (X <<< (W * d + d))

def pairs (k : Nat) : List (Nat × Nat) := (List.range k).flatMap (fun i => (List.range k).map (fun j => (i, j)))

/-- 标记块压上 (r,c,k)：记下每格底色（与旧方区重叠的格沿用旧记忆），并把这些格从场地集合里拿掉。 -/
def takeUnder (f : Fld) (r c k orr occ ok ou : Nat) : Nat × Fld :=
  let u := (pairs k).foldl (fun acc ij =>
    let rr := r + ij.1
    let cc := c + ij.2
    let v := if orr ≤ rr && rr < orr + ok && occ ≤ cc && cc < occ + ok then nib ou (rr - orr) (cc - occ)
             else f.colorAt (bt rr cc)
    acc ||| (v <<< (4 * (4 * ij.1 + ij.2)))) 0
  (u, f.clear (sqm r c k))

/-- 熄灭的记号格：本关早先是地板的回到地板，其余成虚空。 -/
def revert (f : Fld) (hfl m : Nat) : Fld :=

/-- mark_boxes 的合法性：框里除地板与该色之外至多一类别的色、且不是虚空；也不能整块都是该色。标记块压着的部分算 9、10 两色。 -/
def insideOk (f : Fld) (br bc k col mr mc mk : Nat) : Bool :=
  let q := sqm br bc k
  let cls := (f.classes.filter (fun p => p.2 &&& q != 0)).map (·.1) ++
             (if q &&& sqm mr mc mk != 0 then [9, 10] else [])
  let extra := (cls.filter (fun x => x != 0 && x != col)).eraseDups
  !(!cls.isEmpty && cls.all (· == col)) && extra.length ≤ 1 && !(extra.contains 5)

/-- 记号色 col 在场上圈出的、边长 k 的合法框（四角画在框上，或画在外扩一圈上），按先行后列排。 -/
def boxesOf (f : Fld) (col k mr mc mk : Nat) : List (Nat × Nat) :=
  let M := f.mark col
  let hits := cornerHits M k ||| (cornerHits M (k + 2) <<< (W + 1))
  (bitsAsc 4096 hits).filterMap (fun p =>
    let r := p / W
    let c := p % W
    if r + k ≤ 64 && c + k ≤ 64 && insideOk f r c k col mr mc mk then some (r, c) else none)

/-- 抹标本：正前方唯一的合法色 6 框（带子与框环互相套住）、框里恰有一种标本色、带子里至少一条车道全是地板或记号。
    返回 (框左上, 标本色, 框环左上)。 -/
def eraseTarget (s : State) : Option ((Nat × Nat) × Nat × (Nat × Nat)) :=
  match meOf s, s.facing_code with
  | some (r0, c0, k), some fc =>
    let f := Fld.ofS s
    let fd := fc.toNat
    let hits := (boxesOf f 6 k r0 c0 k).filterMap (fun b =>
      let (br, bc) := b
      let ring := (bitsAsc 4096 (f.m6 &&& rect (br - 1) (br + k) (bc - 1) (bc + k))).map (fun p => (p / W, p % W))
      if ring.isEmpty then none else
      let rs := ring.map (·.1)
      let cs := ring.map (·.2)
      let (lo, hi, mlo, mhi, ahead) :=
        if fd ≤ 1 then (cs.foldl min 64, cs.foldl max 0, c0, c0 + k - 1, if fd == 0 then decide (br + k ≤ r0) else decide (r0 + k ≤ br))
        else (rs.foldl min 64, rs.foldl max 0, r0, r0 + k - 1, if fd == 2 then decide (bc + k ≤ c0) else decide (c0 + k ≤ bc))
      let nested := (lo ≤ mlo && mhi ≤ hi) || (mlo ≤ lo && hi ≤ mhi)
      if nested && ahead then some (b, (rs.foldl min 64, cs.foldl min 64)) else none)
    match hits with
    | [((br, bc), tl)] =>
      let q := sqm br bc k
      let extra := ([(5, f.vo), (4, f.bk), (11, f.m11), (15, f.m15), (12, f.c12), (13, f.c13), (14, f.fu)].filter
                      (fun p => p.2 &&& q != 0)).map (·.1)
      match extra with
      | [sp] =>
        if sp != 12 && sp != 13 then none else
        let ok := f.fl ||| marksAll s f
        let (lo, hi) := if fd == 0 then (br + k, r0) else if fd == 1 then (r0 + k, br)
                        else if fd == 2 then (bc + k, c0) else (c0 + k, bc)
        let lanes := (List.range k).map (fun i =>
          if fd ≤ 1 then rect lo (hi - 1) (c0 + i) (c0 + i) else rect (r0 + i) (r0 + i) lo (hi - 1))
        if lo < hi && !(lanes.any (fun l => l &&& ok == l)) then none else some ((br, bc), sp, tl)
      | _ => none
    | _ => none
  | _, _ => none

  else 0

/-- 吃燃料：新方区压到前帧场地上的燃料格时，与之 4-连通的整坨（前帧燃料集合 fu0 里）变地板。返回 (场地, 吃没吃)。 -/
def eatFuel (f : Fld) (fu0 newsq : Nat) : Fld × Bool :=
  let x0 := fu0 &&& newsq
  if x0 == 0 then (f, false) else
  let blob := (List.range 16).foldl (fun x _ => (x ||| (x <<< 1) ||| (x >>> 1) ||| (x <<< W) ||| (x >>> W)) &&& fu0) x0
  (f.paint blob 0, true)

/-- 吃一坨燃料退还的计费动作数（_lib.FUEL_REFUND；由两段吃过燃料的尾巴反推）。 -/
def fuelRefund : Int := 10

def refund (ate : Bool) : Int := if ate then fuelRefund else 0

-- ---------- 前向函数 ----------

def cnt (n : Int) (old new : Nat) : Int := n + (pc (dif new old) : Int) - (pc (dif old new) : Int)

/-- 把场地集合写回状态，并更新计数、fl 记忆、竖条与 marker_*。 -/
def finish (s0 s : State) (f : Fld) : State :=
  let o := Fld.ofS s0
  { s with
    arena_m := f.fl, arena_n := cnt s0.arena_n o.fl f.fl,
    g_void_m := f.vo, g_void_n := cnt s0.g_void_n o.vo f.vo,
    g_block_m := f.bk, g_block_n := cnt s0.g_block_n o.bk f.bk,
    g_sock_m := f.so, g_sock_n := cnt s0.g_sock_n o.so f.so,
    g_m6_m := f.m6, g_m6_n := cnt s0.g_m6_n o.m6 f.m6,
    g_m11_m := f.m11, g_m11_n := cnt s0.g_m11_n o.m11 f.m11,
    g_m15_m := f.m15, g_m15_n := cnt s0.g_m15_n o.m15 f.m15,
    g_c12_m := f.c12, g_c12_n := cnt s0.g_c12_n o.c12 f.c12,
    g_c13_m := f.c13, g_c13_n := cnt s0.g_c13_n o.c13 f.c13,
    fuel_cells_m := f.fu, fuel_cells_n := cnt s0.fuel_cells_n o.fu f.fu,
    h_fresh := 0,
    h_fl := ((s0.h_fl.toNat ||| o.fl) &&& f.vis : Nat),
    lit_grid_n := (pc s.lit_grid_m.toNat : Nat),
    bar_blocks := match s.k_B with | some b => 32 - (32 * s.h_n) / b | none => s.bar_blocks,
    marker_0 := s.me_r, marker_1 := s.me_c, marker_2 := s.side }

def moveStep (s : State) (a : Act) : Option State :=
  match meOf s with
  | none => none
  | some (r, c, k) =>
    if voidStep s then some { s with h_fresh := 0 } else
    let (_, _, face) := dirOf a
    -- 计费：会话首条（h_fresh = 1）且整帧不变（走不动、朝向也没变）的那一下不计，与 hidden_step 同一条
    let still := s.h_fresh == 1 && !canEnter s a && s.facing_code == some (face : Int)
    let s1 := { s with h_n := s.h_n + (if still then 0 else 1), facing_code := some (face : Int) }
    let f := Fld.ofS s
    match targetSq s a with
    | some (tr, tc, T) =>
      if canEnter s a then
        let f1 := reveal f s.h_under.toNat r c k 0
        let (f1', ate) := eatFuel f1 f.fu T
        let (u, f2) := takeUnder f1' tr tc k r c k s.h_under.toNat
        some (finish s { s1 with me_r := some (tr : Int), me_c := some (tc : Int), h_under := (u : Int),
                                  h_n := s1.h_n - refund ate } f2)
      else some (finish s s1 f)
    | none => some (finish s s1 f)

def machine (s s1 : State) (col r c k : Nat) : State :=
  let f := Fld.ofS s
  let hfl := s.h_fl.toNat
  let u := s.h_under.toNat
  if col == 15 && k == 4 then
    let hid := sqm r c 4
    let f1 := reveal f u r c 4 (sqm r c 2)
    let f2 := revert f1 hfl (dif (span6Off s f) hid)
    let f3 := f2.paint (dif (shrinkOn f) hid) 15
    finish s { s1 with side := some 2, h_under := ((u &&& 16711935 : Nat) : Int) } f3
  else if col == 15 && k == 2 then
    match growDest s with
    | some (rr, cc) =>
      let (f0, ate) := eatFuel f f.fu (sqm rr cc 4)
      let (u2, f1) := takeUnder f0 rr cc 4 r c 2 u
      let hid := sqm r c 2 ||| sqm rr cc 4
      let f2 := revert f1 hfl (dif (span6Off s f) hid)
      let f3 := f2.paint (dif (growOn s f u r c) hid) 11
      finish s { s1 with side := some 4, me_r := some (rr : Int), me_c := some (cc : Int), h_under := (u2 : Int),
                         h_n := s1.h_n - refund ate } f3
    | none => finish s s1 f
  else if col == 11 then
    match teleportSets s with
    | (same, [d]) =>
      let (f1, ate) := eatFuel (reveal f u r c k 0) f.fu (sqm d.1 d.2 k)
      let (u2, f2) := takeUnder f1 d.1 d.2 k r c k 0
      let others := same.filter (· != d)
      let spot := [(0, 0), (0, k - 1), (k - 1, 0), (k - 1, k - 1)].all (fun ij => nib u ij.1 ij.2 == 11)
      let other := match others with
        | [o] => some o
        | [] => if spot then some (r, c) else none
        | _ => none
      let newsq := sqm d.1 d.2 k
      let f3 := match other with
        | some o =>
          if same.length ≤ 2 then
            (revert f2 hfl (dif (corners (d.1 - 1) (d.2 - 1) (k + 2)) newsq)).paint
              (dif (corners (o.1 - 1) (o.2 - 1) (k + 2)) newsq) 11
          else f2
        | none => f2
      finish s { s1 with me_r := some (d.1 : Int), me_c := some (d.2 : Int), h_under := (u2 : Int),
                         h_n := s1.h_n - refund ate } f3
    | _ => finish s s1 f
  else if col == 6 then
    match eraseTarget s with
    | some ((br, bc), sp, tl) =>
      let zone := rect (br - 1) (br + k) (bc - 1) (bc + k)
      let spm := if sp == 12 then f.c12 else f.c13
      let f1 := f.paint (dif spm zone) 0
      let spm1 := if sp == 12 then f1.c12 else f1.c13
      let zs := (spm1 ||| f1.m6) &&& zone
      let fuel := if s.k_erase_fuel == some 1 then sqm tl.1 tl.2 2 else 0
      let f2 := (f1.paint (dif zs fuel) 0).paint (zs &&& fuel) 14
      finish s s1 f2
    | none => finish s s1 f
  else finish s s1 f

def clickStep (s : State) (a : Act) : Option State :=
  match meOf s with
  | none => none
  | some (r, c, k) =>
    if voidStep s then some { s with h_fresh := 0 } else
    match clickHit s a with
    | none => some { s with h_fresh := 0 }
    | some (b, none) =>
      let s1 : State := { s with
        h_n := s.h_n + 1
        lit_grid_m := ((s.lit_grid_m.toNat ||| b : Nat) : Int)
        panel_grid_m_c14 := ((s.panel_grid_m_c14.toNat ||| b : Nat) : Int)
        panel_grid_m_c0 := ((dif s.panel_grid_m_c0.toNat b : Nat) : Int)
        panel_grid_m_c2 := ((dif s.panel_grid_m_c2.toNat b : Nat) : Int) }
      some (finish s s1 (Fld.ofS s))
    | some (_, some col) =>
      let s1 : State := { s with
        h_n := s.h_n + 2
        lit_grid_m := 0
        panel_grid_m_c14 := 0
        panel_grid_m_c0 := 0
        panel_grid_m_c2 := (full9 : Int) }
      some (machine s s1 col r c k)

def stepRules (s : State) (a : Act) : Option State :=
  match a with
  | .a1 | .a2 | .a3 | .a4 => moveStep s a
  | .a6 _ _ => clickStep s a
  | _ => none

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

def isGoal (s : State) : Bool := s.level == 6
