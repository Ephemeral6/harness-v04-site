structure Slot where
  r : Nat
  c : Nat
  col : Nat
  role : Nat
  g : Nat          -- 形状族 * 8 + 相位；族 3 = 不在已量到的三族里

def decodeSlot (n : Nat) : Slot :=
  { r := (n >>> 17) &&& 63, c := (n >>> 11) &&& 63, col := (n >>> 7) &&& 15,
    role := (n >>> 5) &&& 3, g := n &&& 31 }

def slotsOf (s : State) : List Slot :=
  (codesOf s).filterMap (fun o => match o with
    | some (Int.ofNat n) => some (decodeSlot n)
    | _ => none)

def cellOf (r c : Option Int) : List (Nat × Nat) :=
  match r, c with
  | some (Int.ofNat r), some (Int.ofNat c) => [(r, c)]
  | _, _ => []

-- 焦点串的格（串头、第二格）
def focusCells (s : State) : List (Nat × Nat) :=
  cellOf s.focus_run_0_r s.focus_run_0_c ++ cellOf s.focus_run_1_r s.focus_run_1_c

def idxIn (x : Nat) : List Nat → Option Nat
  | [] => none
  | y :: ys => if x == y then some 0 else (idxIn x ys).map (· + 1)

def toI (n : Nat) : Option Int := some (Int.ofNat n)

-- ---------- shape_step：焦点串里每一格相位 +d（模 7），形状族与别的格不变 ----------
def bumpCode (fc : List (Nat × Nat)) (d : Nat) (o : Option Int) : Option Int :=
  match o with
  | some (Int.ofNat n) =>
    let x := decodeSlot n
    if fc.contains (x.r, x.c) && (x.g >>> 3) != 3 then
      let ph := x.g &&& 7
      some (Int.ofNat (n - ph + (ph + d) % 7))
    else o
  | _ => o

-- ---------- focus_move：可停的串 ----------
-- 图例里的串：同一行里列距恰好 7 相连的槽（输入已按 (行, 列) 排序）
def groupRuns : List Slot → List (List Slot)
  | [] => []
  | x :: xs =>
    match groupRuns xs with
    | (y :: ys) :: rest =>
      if y.r == x.r && y.c == x.c + 7 then (x :: y :: ys) :: rest else [x] :: (y :: ys) :: rest
    | _ => [[x]]

-- 焦点在答题行上：可停的是答题行各格；否则是图例的各个串
def cycleOf (sl : List Slot) (r0 : Nat) : List (List Slot) :=
  let ans := sl.filter (fun x => x.role == 2)
  if ans.any (fun x => x.r == r0) then ans.map (fun x => [x])
  else groupRuns (sl.filter (fun x => x.role == 0))

def moveFocus (s : State) (fwd : Bool) : Option State :=
  let sl := slotsOf s
  match focusCells s with
  | [] => none
  | p :: _ =>
    let cyc := cycleOf sl p.1
    let heads := cyc.map (fun run => match run with | x :: _ => x.r * 64 + x.c | [] => 0)
    match idxIn (p.1 * 64 + p.2) heads with
    | none => none
    | some i =>
      let n := cyc.length
      let j := if fwd then (i + 1) % n else (i + n - 1) % n
      match cyc[j]? with
      | some (x :: rest) =>
        let ans := sl.filter (fun y => y.role == 2)
        let onAns := ans.any (fun y => y.r == x.r)
        let cur := if onAns then idxIn x.c (ans.map (fun y => y.c)) else none
        let second := rest.head?
        some { s with
          focus_run_n := toI (rest.length + 1),
          focus_run_0_r := toI x.r, focus_run_0_c := toI x.c,
          focus_run_1_r := second.map (fun y => Int.ofNat y.r),
          focus_run_1_c := second.map (fun y => Int.ofNat y.c),
          cursor_cells_n := Int.ofNat (14 * (rest.length + 1)),
          cursor := cur.map Int.ofNat,
          me_r := if onAns then toI x.r else none,
          me_c := if onAns then toI x.c else none }
      | _ => none

-- ---------- 目标形状：题面沿颜色链反复改写（_lib.parses）----------
-- 图例一行按列距 > 10 断句
def groupChunks : List Slot → List (List Slot)
  | [] => []
  | x :: xs =>
    match groupChunks xs with
    | (y :: ys) :: rest =>
      if y.r == x.r && y.c ≤ x.c + 10 then (x :: y :: ys) :: rest else [x] :: (y :: ys) :: rest
    | _ => [[x]]

-- 一段 = 键串（与段首同色的格）+ 值串（其余格，须同色）：(键色, 键, 值色, 值)
def chunkRule (ch : List Slot) : Option (Nat × List Nat × Nat × List Nat) :=
  match ch with
  | [] => none
  | x :: _ =>
    let ks := ch.filter (fun y => y.col == x.col)
    let rest := ch.drop ks.length
    match rest with
    | [] => none
    | z :: _ =>
      if rest.all (fun y => y.col == z.col) then some (x.col, ks.map (·.g), z.col, rest.map (·.g))
      else none

def legendRules (sl : List Slot) : List (Nat × List Nat × Nat × List Nat) :=
  (groupChunks (sl.filter (fun x => x.role == 0))).filterMap chunkRule

-- 把 seq 切段、每段换成某条规则的值串，全部结果（长度不超过 cap）
def rwAll (rs : List (List Nat × List Nat)) (cap : Nat) : Nat → List Nat → List Nat → List (List Nat)
  | 0, _, _ => []
  | fuel + 1, sq, acc =>
    if acc.length > cap then []
    else match sq with
      | [] => [acc]
      | _ => rs.foldr (fun kv out =>
          if !kv.1.isEmpty && kv.1.isPrefixOf sq then
            rwAll rs cap fuel (sq.drop kv.1.length) (acc ++ kv.2) ++ out
          else out) []

def dedupL : List (List Nat) → List (List Nat)
  | [] => []
  | x :: xs => let r := dedupL xs; if r.contains x then r else x :: r

def dedupP : List (Nat × List Nat) → List (Nat × List Nat)
  | [] => []
  | x :: xs => let r := dedupP xs; if r.contains x then r else x :: r

def dedupN : List Nat → List Nat
  | [] => []
  | x :: xs => let r := dedupN xs; if r.contains x then r else x :: r

def expand (rules : List (Nat × List Nat × Nat × List Nat)) (cap : Nat) (it : Nat × List Nat) :
    List (Nat × List Nat) :=
  let mine := rules.filter (fun r => r.1 == it.1)
  (dedupN (mine.map (fun r => r.2.2.1))).foldr (fun vc out =>
    let rs := (mine.filter (fun r => r.2.2.1 == vc)).map (fun r => (r.2.1, r.2.2.2))
    (rwAll rs cap (it.2.length + 1) it.2 []).map (fun res => (vc, res)) ++ out) []

def parseLoop (rules : List (Nat × List Nat × Nat × List Nat)) (acol n : Nat) :
    Nat → List (Nat × List Nat) → List (Nat × List Nat) → List (List Nat) → List (List Nat)
  | 0, _, _, out => out
  | k + 1, frontier, seen, out =>
    let fr := (dedupP frontier).filter (fun it => !seen.contains it)
    let out' := ((fr.filter (fun it => it.1 == acol && it.2.length == n)).map (·.2)) ++ out
    let nxt := (fr.filter (fun it => it.1 != acol)).foldr (fun it acc => expand rules (2 * n) it ++ acc) []
    parseLoop rules acol n k nxt (fr ++ seen) out'

def parsesOf (sl : List Slot) : Option (List (List Nat)) :=
  let ans := sl.filter (fun x => x.role == 2)
  let pr := sl.filter (fun x => x.role == 1)
  match ans, pr with
  | a :: _, p :: _ =>
    some (dedupL (parseLoop (legendRules sl) a.col ans.length 4 [(p.col, pr.map (·.g))] [] []))
  | _, _ => none

-- ---------- 由整块版面重算的派生量 ----------
def famOf (x : Slot) : Option Int := if (x.g >>> 3) == 3 then none else toI (x.g >>> 3)
def phOf (x : Slot) : Option Int := if (x.g >>> 3) == 3 then none else toI (x.g &&& 7)

def gapOf (x : Slot) (w : Nat) : Option Nat :=
  if (x.g >>> 3) == 3 || (w >>> 3) != (x.g >>> 3) then none
  else some (((w &&& 7) + 7 - (x.g &&& 7)) % 7)

def derive (s : State) : State :=
  let sl := slotsOf s
  let fc := focusCells s
  let slotAt := fun (p : Nat × Nat) => sl.find? (fun x => x.r == p.1 && x.c == p.2)
  let f0 := (fc[0]?).bind slotAt
  let f1 := (fc[1]?).bind slotAt
  let ans := sl.filter (fun x => x.role == 2)
  let ansCols := ans.map (fun x => x.c)
  let ansRow := ans.head?.map (fun x => x.r)
  let boxed := (fc.filter (fun p => some p.1 == ansRow)).map (fun p => p.2)
  let ps := parsesOf sl
  let want : Option (List Nat) := match ps with
    | some [w] => some w
    | _ => none
  let gaps : Option (List (Option Nat)) := match want with
    | some w => if w.length == ans.length then some (List.zipWith gapOf ans w) else none
    | none => none
  let gap0 : Option Int := match gaps, (boxed.filter (fun c => ansCols.contains c)).head? with
    | some gs, some c => match idxIn c ansCols with
      | some i => (gs[i]?).bind (fun g => g.map Int.ofNat)
      | none => none
    | _, _ => none
  { s with
    focus_fam_0 := f0.bind famOf, focus_phase_0 := f0.bind phOf,
    focus_fam_1 := f1.bind famOf, focus_phase_1 := f1.bind phOf,
    parses_n := ps.map (fun l => Int.ofNat l.length),
    shapes_n := want.map (fun w => Int.ofNat w.length),
    focus_gap_0 := gap0,
    rest_wrong_n := gaps.map (fun gs =>
      Int.ofNat ((List.zip ansCols gs).filter (fun cg => !boxed.contains cg.1 && cg.2 != some 0)).length) }

-- ---------- 前向函数 ----------
-- budget_bar：本关步数 +1，计数条已吃格数 = 本关步数 / k_hud_period
def stepRules (s : State) (a : Act) : Option State :=
  match s.k_hud_period with
  | some p =>
    let h := s.h_steps + 1
    let s1 := { s with h_steps := h, hud_filled := h / p }
    match a with
    | Act.a1 => some (derive (mapCodes (bumpCode (focusCells s) 1) s1))
    | Act.a2 => some (derive (mapCodes (bumpCode (focusCells s) 6) s1))
    | Act.a3 => (moveFocus s1 false).map derive
    | Act.a4 => (moveFocus s1 true).map derive
    | _ => none
  | none => none

def inLevel (s s' : State) : Bool := s.level == s'.level

-- ---------- 规则（每条 = 一条概念的可判错主张）----------

-- cursor_lr：光标在答题条上按 ±1 mod（答题格数）走，带环绕；角标格数不因平移而变。
def cursor_lr_rule : Rule :=
  { name := "cursor_lr",
    guard := fun s a => (a == Act.a3 || a == Act.a4) && s.cursor.isSome && s.ans_cols_n.isSome,
    effect := fun s a s' =>
      match s.cursor, s.ans_cols_n with
      | some i, some n =>
        n != 0 &&
        s'.cursor == some (((if a == Act.a3 then i - 1 else i + 1) % n + n) % n) &&
        s'.cursor_cells_n == s.cursor_cells_n
      | _, _ => false }

-- cursor_shape：角标格数 = 14 × 串长，前后两帧都成立。
def cursor_shape_rule : Rule :=
  { name := "cursor_shape",
    guard := fun s _a => s.focus_run_n.isSome,
    effect := fun s _a s' =>
      match s.focus_run_n with
      | some k =>
        s.cursor_cells_n == 14 * k &&
        (match s'.focus_run_n with
         | some k' => s'.cursor_cells_n == 14 * k'
         | none => true)
      | none => false }

-- focus_move：同关内 ACTION3/4 把焦点挪到「可停的串」名单（答题条各格，或图例各串）里阅读序的上/下一串，带环绕。
def focus_move_rule : Rule :=
  { name := "focus_move",
    guard := fun s a => (a == Act.a3 || a == Act.a4) && !(focusCells s).isEmpty,
    effect := fun s a s' =>
      !inLevel s s' ||
      (match moveFocus s (a == Act.a4) with
       | some t => focusCells t == focusCells s' && t.cursor == s'.cursor && t.focus_run_n == s'.focus_run_n
       | none => false) }

-- glyph_local：同关内 ACTION1/2 不挪光标、不挪焦点串、不改角标格数。
def glyph_local_rule : Rule :=
  { name := "glyph_local",
    guard := fun _s a => a == Act.a1 || a == Act.a2,
    effect := fun s _a s' =>
      !inLevel s s' ||
      (s.cursor == s'.cursor && s.me_r == s'.me_r && s.me_c == s'.me_c &&
       focusCells s == focusCells s' && s.focus_run_n == s'.focus_run_n &&
       s.cursor_cells_n == s'.cursor_cells_n) }

-- only_focus：同关内，焦点串（前后两帧）之外的每个字形槽一格不变；版面结构计数不变。
def touches (fc : List (Nat × Nat)) (o : Option Int) : Bool :=
  match o with
  | some (Int.ofNat n) => let x := decodeSlot n; fc.contains (x.r, x.c)
  | _ => false

def only_focus_rule : Rule :=
  { name := "only_focus",
    guard := fun _s _a => true,
    effect := fun s _a s' =>
      !inLevel s s' ||
      ((List.zip (codesOf s) (codesOf s')).all (fun oo => oo.1 == oo.2 || touches (focusCells s ++ focusCells s') oo.1) &&
       s.ans_cols_n == s'.ans_cols_n && s.board_n == s'.board_n && s.legend_n == s'.legend_n &&
       s.slots_n == s'.slots_n && s.runs_of_n == s'.runs_of_n && s.word_ans_n == s'.word_ans_n &&
       s.word_prompt_n == s'.word_prompt_n && s.focus_cycle_n == s'.focus_cycle_n) }

-- shape_step：同关内 ACTION1 让焦点串每一格相位 +1、ACTION2 −1（模 7），族不变；整块版面其余各槽不变。
def shape_step_rule : Rule :=
  { name := "shape_step",
    guard := fun s a => (a == Act.a1 || a == Act.a2) && !(focusCells s).isEmpty,
    effect := fun s a s' =>
      !inLevel s s' ||
      codesOf s' == codesOf (mapCodes (bumpCode (focusCells s) (if a == Act.a1 then 1 else 6)) s) }

-- budget_bar：同关内每一步本关步数 +1，计数条已吃格数 = 本关步数 / 每关常数 k_hud_period
-- （第 0 关即概念原文的「第 s 步之后 s//2 格」）。
def budget_bar_rule : Rule :=
  { name := "budget_bar",
    guard := fun s a => !(a == Act.reset) && s.k_hud_period.isSome,
    effect := fun s _a s' =>
      !inLevel s s' ||
      (match s.k_hud_period with
       | some p => s'.h_steps == s.h_steps + 1 && s'.hud_filled == (s.h_steps + 1) / p
       | none => false) }

-- win_rule：焦点卡在答题条唯一一格时，这一按过关 ⇔ 其余格已全部同余且这一格按完差值归零。
def win_rule_rule : Rule :=
  { name := "win_rule",
    guard := fun s a =>
      (a == Act.a1 || a == Act.a2) && s.rest_wrong_n.isSome && s.focus_gap_0.isSome &&
      s.focus_run_n == some 1,
    effect := fun s a s' =>
      match s.rest_wrong_n, s.focus_gap_0 with
      | some rw, some g =>
        let g' := if a == Act.a1 then (g - 1 + 7) % 7 else (g + 1) % 7
        (rw == 0 && g' == 0) == (s.level != s'.level)
      | _, _ => false }

-- win_prev：在答题条上拨字形而过关的那一步，前一帧除光标那格外其余各格都已同余。
def win_prev_rule : Rule :=
  { name := "win_prev",
    guard := fun s a => (a == Act.a1 || a == Act.a2) && s.cursor.isSome && s.rest_wrong_n.isSome,
    effect := fun s _a s' => inLevel s s' || s.rest_wrong_n == some 0 }
