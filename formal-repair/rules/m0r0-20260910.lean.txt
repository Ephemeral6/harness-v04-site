-- ---------- 位运算小工具 ----------
def nat (x : Int) : Nat := x.toNat
def onat (x : Option Int) : Nat := (x.getD 0).toNat
def bitRC (r c : Nat) : Nat := 1 <<< (r * 64 + c)
def testRC (m r c : Nat) : Bool := (m >>> (r * 64 + c)) % 2 == 1
def andNot (a b : Nat) : Nat := a ^^^ (a &&& b)
def subsetOf (a b : Nat) : Bool := andNot a b == 0
-- 四邻（列不会越过 63：本局格点列数 ≤ 16）
def nbr (m : Nat) : Nat := (m <<< 1) ||| (m >>> 1) ||| (m <<< 64) ||| (m >>> 64)
-- 孤立格：四邻里没有同一集合的格
def isolated (m : Nat) : Nat := andNot m (nbr m)
def popc : Nat → Nat → Nat
  | 0, _ => 0
  | f + 1, m => if m == 0 then 0 else 1 + popc f (m &&& (m - 1))
def cnt (m : Nat) : Nat := popc 4096 m
def single (m : Nat) : Bool := m != 0 && (m &&& (m - 1)) == 0
def lowBit (m : Nat) : Nat := m ^^^ (m &&& (m - 1))
def cellOfIdx (k : Nat) : Int × Int := (Int.ofNat (k / 64), Int.ofNat (k % 64))

-- 带符号格点
def inM (m : Nat) (p : Int × Int) : Bool := 0 ≤ p.1 && 0 ≤ p.2 && p.2 < 64 && testRC m p.1.toNat p.2.toNat
def bitP (p : Int × Int) : Nat := bitRC p.1.toNat p.2.toNat
def addP (p d : Int × Int) : Int × Int := (p.1 + d.1, p.2 + d.2)
-- 列序：列小在前，同列行小在前（与 _lib.pieces 的排序一致）
def colLt (a b : Int × Int) : Bool := a.2 < b.2 || (a.2 == b.2 && a.1 < b.1)
def byCol (a b : Int × Int) : (Int × Int) × (Int × Int) := if colLt b a then (b, a) else (a, b)
def lexLt (a b : Int × Int) : Bool := a.1 < b.1 || (a.1 == b.1 && a.2 < b.2)
def near1 (a b : Int × Int) : Bool := (a.1 - b.1).natAbs + (a.2 - b.2).natAbs ≤ 1

-- ---------- 几何：像素 <-> 格点 ----------
def side (s : State) : Nat := onat s.geom_0
def org (s : State) : Nat × Nat := (onat s.geom_1, onat s.geom_2)
def toCell (s : State) (r c : Option Int) : Int × Int :=

-- 两块的格点，按列序
def cellsOf (s : State) : (Int × Int) × (Int × Int) :=
  byCol (toCell s s.pieces_0_r s.pieces_0_c) (toCell s s.pieces_1_r s.pieces_1_c)
-- 下一帧（s'）的两块格点，按 s 的几何换算（关内几何不变）
def cellsWith (s t : State) : (Int × Int) × (Int × Int) :=
  byCol (toCell s t.pieces_0_r t.pieces_0_c) (toCell s t.pieces_1_r t.pieces_1_c)
def pmOf (a b : Int × Int) : Nat := bitP a ||| bitP b
def gridCols (s : State) : Nat := (64 - (org s).2) / side s
def gridRows (s : State) : Nat := (64 - (org s).1) / side s

-- ---------- 动作 ----------
def dirOf : Act → Option (Int × Int)
  | Act.a1 => some (-1, 0)
  | Act.a2 => some (1, 0)
  | Act.a3 => some (0, -1)
  | Act.a4 => some (0, 1)
  | _ => none
def isMove (a : Act) : Bool := (dirOf a).isSome
def dOf (a : Act) : Int × Int := (dirOf a).getD (0, 0)

-- ---------- 记号 ----------
def lit (s : State) : Nat := onat s.markers_m_c11
def selected (s : State) : Bool := lit s != 0
-- 点中哪个格点（x 列 y 行）；点中的是记号格就点亮它，否则全部复原（click_selects）
def clickSel (s : State) (x y : Nat) : Nat :=
  let (pr, pc) := org s
  if pr ≤ y && pc ≤ x && side s != 0 then
    let r := (y - pr) / side s
    let c := (x - pc) / side s
    if testRC (onat s.markers_m) r c then bitRC r c else 0
  else 0
-- 点亮的记号按 DELTA（不取镜像）挪一格：落点须是可走格、且没有块占着（push_door）
def pushTarget (s : State) (d : Int × Int) : Int × Int := addP (cellOfIdx (Nat.log2 (lit s))) d
def pushOk (s : State) (d : Int × Int) : Bool :=
  let t := pushTarget s d
  let (p0, p1) := cellsOf s
  inM (nat s.free_cells_m) t && !inM (pmOf p0 p1) t

-- ---------- 两块的移动 ----------
-- 标法 k：列序第 k 块按 DELTA 走（direct），另一块列取反（mirror）
def intentsBy (s : State) (a : Act) (k : Nat) : (Int × Int) × (Int × Int) :=
  let (p0, p1) := cellsOf s
  let d := dOf a
  let dm : Int × Int := (d.1, -d.2)
  if k == 0 then (addP p0 d, addP p1 dm) else (addP p0 dm, addP p1 d)
-- 真身份：本关开局是第 k_dir 块，此后跟着块走（每对调一次列序就换一次下标）
def dirIdx (s : State) : Nat := (nat s.k_dir + nat s.h_flip) % 2
def hitsHazard (s : State) (t : (Int × Int) × (Int × Int)) : Bool :=
  inM (nat s.hazard_cells_m) t.1 || inM (nat s.hazard_cells_m) t.2
def destBy (s : State) (a : Act) (k : Nat) : (Int × Int) × (Int × Int) :=
  let (p0, p1) := cellsOf s
  let t := intentsBy s a k
  let fr := nat s.free_cells_m
  (if inM fr t.1 then t.1 else p0, if inM fr t.2 then t.2 else p1)
def hzTrue (s : State) (a : Act) : Bool := hitsHazard s (intentsBy s a (dirIdx s))
def hzAny (s : State) (a : Act) : Bool := hitsHazard s (intentsBy s a 0) || hitsHazard s (intentsBy s a 1)
def destTrue (s : State) (a : Act) : (Int × Int) × (Int × Int) := destBy s a (dirIdx s)
def collides (s : State) (a : Act) : Bool := let d := destTrue s a; d.1 == d.2
-- 这一步是「未点亮态的方向键」（两块真的会迈步）
def freeMove (s : State) (a : Act) : Bool := isMove a && !selected s && s.pieces_n == some 2
def spawnCells (s : State) : (Int × Int) × (Int × Int) :=
  let sp := nat s.k_spawn
  byCol (cellOfIdx (Nat.log2 (lowBit sp))) (cellOfIdx (Nat.log2 sp))

-- ---------- 物件：开关与门 ----------
def lay12 (s : State) : Nat := nat s.k_o12
def lay14 (s : State) : Nat := nat s.k_o14
def lay15 (s : State) : Nat := nat s.k_o15
def swOf (lay : Nat) : Nat := isolated lay
def gtOf (lay : Nat) : Nat := andNot lay (isolated lay)
-- 有块站在同色开关上，这色的门就开（switch_opens_gate / gate_closes_on_leave）
def openBy (lay pm : Nat) : Bool := (swOf lay &&& pm) != 0
-- 帧上看得见的物件格：开关 + 关着的门，减去块盖住的格
def visOf (lay pm : Nat) : Nat := andNot (if openBy lay pm then swOf lay else lay) pm

-- ---------- 由「两块格点 + 记号 + 点亮」重算全部派生字段 ----------
def derive (s : State) (n : Nat) (flip : Nat) (a b : Int × Int) (mk c11 : Nat) : State :=
  let pm := pmOf a b
  let v12 := visOf (lay12 s) pm
  let v14 := visOf (lay14 s) pm
  let v15 := visOf (lay15 s) pm
  let allv := v12 ||| v14 ||| v15
  let i12 := isolated v12
  let i14 := isolated v14
  let i15 := isolated v15
  let fl := nat s.free_cells_m ||| lay12 s ||| lay14 s ||| lay15 s ||| onat s.markers_m
  let fr := pm ||| andNot (andNot fl mk) allv ||| i12 ||| i14 ||| i15
  let fr0 := nat s.free_cells_m
  let ob0 := onat s.objects_m
  let sel := c11 != 0
  let sd := side s
  let (pr, pc) := org s
  let pa := (Int.ofNat pr + Int.ofNat sd * a.1, Int.ofNat pc + Int.ofNat sd * a.2)
  let pb := (Int.ofNat pr + Int.ofNat sd * b.1, Int.ofNat pc + Int.ofNat sd * b.2)
  let (q0, q1) := if lexLt pb pa then (pb, pa) else (pa, pb)
  { s with
    h_steps := Int.ofNat n
    h_flip := Int.ofNat flip
    hud_len := Int.ofNat ((128 * n + 150) / 300)
    free_cells_m := Int.ofNat fr
    free_cells_n := s.free_cells_n + Int.ofNat (cnt (andNot fr fr0)) - Int.ofNat (cnt (andNot fr0 fr))
    objects_m := some (Int.ofNat allv)
    objects_m_c12 := some (Int.ofNat v12)
    objects_m_c14 := some (Int.ofNat v14)
    objects_m_c15 := some (Int.ofNat v15)
    objects_n := some ((s.objects_n.getD 0) + Int.ofNat (cnt (andNot allv ob0)) - Int.ofNat (cnt (andNot ob0 allv)))
    switches_n := some (Int.ofNat ((if i12 != 0 then 1 else 0) + (if i14 != 0 then 1 else 0) + (if i15 != 0 then 1 else 0)))
    gates_n := some (Int.ofNat ((if andNot v12 i12 != 0 then 1 else 0) + (if andNot v14 i14 != 0 then 1 else 0)
                  + (if andNot v15 i15 != 0 then 1 else 0)))
    markers_m := some (Int.ofNat mk)
    markers_m_c11 := some (Int.ofNat c11)
    markers_m_c9 := some (Int.ofNat (andNot mk c11))
    piece_color := some (if sel then 1 else 10)
    sel_dim := some sel
    pieces_0_r := some q0.1
    pieces_0_c := some q0.2
    pieces_1_r := some q1.1
    pieces_1_c := some q1.2
    pieces_m := some (Int.ofNat (pmOf q0 q1)) }

-- 未点亮态的方向键：危险 -> 整关重置；相撞 -> 说不出（换关，交给查表）；否则按真身份走
def moveStep (s : State) (a : Act) (n : Nat) : Option State :=
  let (p0, p1) := cellsOf s
  let mk := onat s.markers_m
  let c11 := lit s
  if selected s then
    let d := dOf a
    if pushOk s d then
      let tb := bitP (pushTarget s d)
      some (derive s n (nat s.h_flip) p0 p1 (andNot mk c11 ||| tb) tb)
    else some (derive s n (nat s.h_flip) p0 p1 mk c11)
  else if hzTrue s a then
    let sp := spawnCells s
    some (derive s n 0 sp.1 sp.2 (nat s.k_mark0) 0)
  else
    let d := destTrue s a
    if d.1 == d.2 then none
    else
      let fl := if colLt d.2 d.1 then (nat s.h_flip + 1) % 2 else nat s.h_flip
      some (derive s n fl d.1 d.2 mk c11)

def stepRules (s : State) (a : Act) : Option State :=
  let n := nat s.h_steps + 1
  let (p0, p1) := cellsOf s
  match a with
  | Act.a5 => some (derive s n (nat s.h_flip) p0 p1 (onat s.markers_m) (lit s))
  | Act.a6 x y => some (derive s n (nat s.h_flip) p0 p1 (onat s.markers_m) (clickSel s x y))
  | Act.reset => none
  | Act.other => none
  | _ => moveStep s a n

-- ---------- 规则用的判定 ----------
def sameLevel (s t : State) : Bool := s.level == t.level
def pmNow (s : State) : Nat := let c := cellsOf s; pmOf c.1 c.2
def pmNext (s t : State) : Nat := let c := cellsWith s t; pmOf c.1 c.2
-- 格点位移是 0 或给定位移
def stepOk (p q d : Int × Int) : Bool := q == p || q == addP p d
-- 被挡块数（某种标法下）
def blockedBy (s : State) (a : Act) (k : Nat) : Nat :=
  let (p0, p1) := cellsOf s
  let d := destBy s a k
  (if d.1 == p0 then 1 else 0) + (if d.2 == p1 then 1 else 0)
def explainsBy (s t : State) (a : Act) (k : Nat) : Bool :=
  let d := destBy s a k
  pmNext s t == pmOf d.1 d.2
def colGapOdd (p : (Int × Int) × (Int × Int)) : Bool := (p.1.2 - p.2.2) % 2 != 0
def mirrorPair (s : State) (p : (Int × Int) × (Int × Int)) : Bool :=
  p.1.1 == p.2.1 && p.1.2 + p.2.2 == Int.ofNat (gridCols s) - 1
-- 通行图左右对称（逐行逐列比，第 j 列与第 nc-1-j 列）
def symRow (fr : Nat) (nc r : Nat) : Nat → Bool
  | 0 => true
  | j + 1 => testRC fr r j == testRC fr r (nc - 1 - j) && symRow fr nc r j
def symBoard (s : State) : Bool :=
  let fr := nat s.free_cells_m
  let nc := gridCols s
  (List.range (gridRows s)).all (fun r => symRow fr nc r nc)
-- 所有开关与门（三色合起来）
def swAll (s : State) : Nat := swOf (lay12 s) ||| swOf (lay14 s) ||| swOf (lay15 s)
def gtAll (s : State) : Nat := gtOf (lay12 s) ||| gtOf (lay14 s) ||| gtOf (lay15 s)
-- 预测的下一步两块格点（未点亮态方向键，无危险）
def pmDest (s : State) (a : Act) : Nat := let d := destTrue s a; pmOf d.1 d.2
-- 这一步有块从开关上走开（开关色会露出来）
def leavesSwitch (s : State) (a : Act) : Bool :=
  freeMove s a && !hzTrue s a && andNot (swAll s &&& pmNow s) (pmDest s a) != 0
def leftGate (lay : Nat) (s : State) (a : Act) : Nat :=
  if openBy lay (pmNow s) && !openBy lay (pmDest s a) then gtOf lay else 0

  { name := "gate_closes_on_leave"
    guard := fun s a => leavesSwitch s a && !collides s a
    effect := fun s a t =>
      let g := leftGate (lay12 s) s a ||| leftGate (lay14 s) s a ||| leftGate (lay15 s) s a
      subsetOf (andNot g (pmNext s t)) (onat t.objects_m) && (andNot g (pmNext s t) &&& nat t.free_cells_m) == 0 },
  { name := "gate_stays_open"
    guard := fun s a => gtAll s != 0 && !(leavesSwitch s a) && a != Act.reset
    effect := fun s _ t => !sameLevel s t || subsetOf (onat t.objects_m) (onat s.objects_m ||| pmNow s) },
  { name := "piece_survives_on_gate"
    guard := fun s a => freeMove s a && !hzTrue s a && !collides s a && andNot (gtAll s &&& pmNow s) (pmDest s a) != 0
    effect := fun s _ t =>
      let (p0, p1) := cellsOf s
      let (q0, q1) := cellsWith s t
      sameLevel s t && t.pieces_n == some 2
        && ((near1 p0 q0 && near1 p1 q1) || (near1 p0 q1 && near1 p1 q0)) }
]

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

-- 目标：通关后的第 6 关画面（WIN）
def isGoal (s : State) : Bool := s.level == 6

-- ==== 定理段（脚本生成）====
-- 规则与全部观测一致。内核证明（decide +kernel），不依赖 ofReduceBool。
theorem consistent : checkRules rules corpus = true := by decide +kernel

-- 逐步重算整条路线。它换掉了旧的 solvable —— 那条只看末态 isGoal，中间几百步一步不查，
-- 「直接把目标字段拍成真」的 step 照样能过。faithful 每一步都要对上观测，规则段没写完之前编不过，那正是要求。
theorem faithful : (faithfulRun step start routeObs).map isGoal = some true := by decide +kernel

-- 每条规则的 guard 在语料上命中多少条。checkRules 是 `!guard || effect`，guard 恒 false 就**空真**——
-- 一条永不触发的规则能让 consistent 无条件成立。verify 读这一行，有 0 就判红。
#eval IO.println ("GUARD_HITS " ++ String.intercalate "," (rules.map (fun r => r.name ++ "=" ++ toString (corpus.filter (fun t => r.guard t.s t.a)).length)))
