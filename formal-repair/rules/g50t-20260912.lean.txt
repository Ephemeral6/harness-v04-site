def oneB (i : Int) : Nat := if 0 ≤ i then 1 <<< i.toNat else 0

def popc : Nat → Nat → Nat
  | 0, _ => 0
  | fuel + 1, n => if n == 0 then 0 else n % 2 + popc fuel (n >>> 1)
def cellOf (r c : Int) : Int := (r / 6) * 16 + c / 6

def pxC (i : Int) : Int := 2 + 6 * (i % 16)

-- ---------- 轨道与车 ----------
structure Tr where
  home : Int
  fwd : Int
  btn : Int
  kind : Int      -- 1 = 色 8 弹簧，2 = 色 11 闩
def decTr (v : Int) : Tr := ⟨v % 256, (v / 256) % 256, (v / 65536) % 256, v / 16777216⟩
def tracksOf (s : State) : List Tr := ([s.k_tr0, s.k_tr1, s.k_tr2, s.k_tr3, s.k_tr4].filterMap id).map decTr
def trMask (m8 m11 : Int) (t : Tr) : Int := if t.kind == 1 then m8 else m11
def trAt (m8 m11 : Int) (t : Tr) : Int := if bitI (trMask m8 m11 t) t.home then t.home else t.fwd
def trPut (m8 m11 : Int) (t : Tr) (p : Int) : Int × Int :=
  let f := fun (mm : Int) => Int.ofNat (clearB mm (oneB t.home ||| oneB t.fwd) ||| oneB p)
  if t.kind == 1 then (f m8, m11) else (m8, f m11)

-- cart_shifts_on_press：弹簧车按钮上升沿进到前一格、下降沿回家；闩车上升沿在家与前一格之间翻一次；其余不动
def springResp (t : Tr) (at' : Int) (rise fall : Bool) : Int := if rise then t.fwd else if fall then t.home else at'
def latchResp (t : Tr) (at' : Int) (rise : Bool) : Int := if rise then (if at' == t.home then t.fwd else t.home) else at'
def cartsOnMove (s : State) (o0 o1 : List Int) : Int × Int :=
  (tracksOf s).foldl (fun (acc : Int × Int) t =>
    let at' := trAt acc.1 acc.2 t
    let rise := o1.contains t.btn && !o0.contains t.btn
    let fall := o0.contains t.btn && !o1.contains t.btn
    trPut acc.1 acc.2 t (if t.kind == 1 then springResp t at' rise fall else latchResp t at' rise))
    (s.cart_cells_0, s.cart_cells_1)

-- switch_rewinds_carts：换枚时弹簧车回家；闩车回到这一盘开局时的格（h_r0；-1 = 本关开局，即家），按钮此刻被压着则不动
def cartsOnSwitch (s : State) (o0 : List Int) : Int × Int :=
  (tracksOf s).foldl (fun (acc : Int × Int) t =>
    let at' := trAt acc.1 acc.2 t
    let r0home := decide (s.h_r0 < 0) || bitI s.h_r0 t.home
    trPut acc.1 acc.2 t (if t.kind == 1 then t.home else if o0.contains t.btn then at' else if r0home then t.home else t.fwd))
    (s.cart_cells_0, s.cart_cells_1)

-- reset 之后车全在家（统计车 = 色 15 那一掩码不动）
def cartsHome (s : State) : Int × Int :=
  (tracksOf s).foldl (fun (acc : Int × Int) t => trPut acc.1 acc.2 t t.home) (s.cart_cells_0, s.cart_cells_1)

structure Link where

  btn : Int

def linksOf (s : State) : List Link := ([s.k_pt0, s.k_pt1].filterMap id).map decLink
def isPortal (s : State) (i : Int) : Bool := (linksOf s).any (fun l => l.pa == i || l.pb == i)
def partnerOf (s : State) (i : Int) : Int :=
  match (linksOf s).find? (fun l => l.pa == i || l.pb == i) with

def rising (o0 o1 : List Int) (b : Int) : Bool := o1.contains b && !o0.contains b
-- portal_teleport：走进门且**同一张网上**的按钮这一步被压下（上升沿）就到搭档门，否则停在门里

  | none => p

-- ---------- 一步里的各个活物 ----------
def occ0 (s : State) : List Int := [meI s, s.h_gA, s.h_gB, wpOf s].filter (fun x => decide (0 ≤ x))
def meTarget (s : State) (d : Nat) : Int := meI s + dStep d
-- move_iff_free：目标格进得去才走，否则原地
def me1 (s : State) (d : Nat) : Int := if enterable (kPass s) (cartsAll s) (meTarget s d) then meTarget s d else meI s
-- 替身：从本盘起点重放带子，本盘第 rt 步取带子第 rt 个动作（八进制第 rt 位，0 = 带子已放完、停在原地），进得去才走
def tapeDigit (tp rt : Int) : Nat := (tp.toNat >>> (3 * rt.toNat)) &&& 7
def ghostAdv (s : State) (g tp : Int) : Int :=
  let dg := tapeDigit tp s.h_rt
  if decide (0 ≤ g) && dg != 0 && enterable (kPass s) (cartsAll s) (g + dStep dg) then g + dStep dg else g
-- 环：棋子挪了窝它才走一格；朝原方向进得去就直走，否则拐进唯一进得去的侧路，两侧都通就原地，两侧都不通就掉头
def walkChoice (pass carts w : Int) (wd : Nat) : Nat :=
  let ok := fun d => enterable pass carts (w + dStep d)
  if wd != 0 && ok wd then wd else
  let back := dBack wd
  match [1, 2, 3, 4].filter (fun d => d != wd && d != back && ok d) with
  | [d] => d
  | [] => if back != 0 && ok back then back else 0
  | _ => 0
def walkerAdv (s : State) (d : Nat) : Int × Int :=
  let w := wpOf s
  if decide (0 ≤ w) && me1 s d != meI s then
    let c := walkChoice (kPass s) (cartsAll s) w s.h_wd.toNat
    if c != 0 then (w + dStep c, Int.ofNat c) else (w, s.h_wd)
  else (w, s.h_wd)
def occ1 (s : State) (d : Nat) : List Int :=
  [me1 s d, ghostAdv s s.h_gA s.h_tpA, ghostAdv s s.h_gB s.h_tpB, (walkerAdv s d).1].filter (fun x => decide (0 ≤ x))
-- walker_walks_alone / walker_dies_only_to_moving_cart：落脚格被（挪完之后的）车占着就被压掉
def walkerFinal (s : State) (d : Nat) : Int :=
  let w := (walkerAdv s d).1
  let cs := cartsOnMove s (occ0 s) (occ1 s d)
  if decide (0 ≤ w) && bitI (Int.ofNat (cs.1.toNat ||| cs.2.toNat ||| s.cart_cells_2.toNat)) w then -1 else w
def tapeLen (t : Int) : Nat := if t == 0 then 0 else t.toNat.log2 / 3 + 1

-- ---------- 画面：由真位置推出看得见的字段 ----------
def barSpent (tick gap : Int) : Int := max 0 (tick / 2 - gap)
def visGhosts (me g1 g2 w : Int) : List Int :=
  let vs := [g1, g2].filter (fun g => decide (0 ≤ g) && g != me && g != w)
  match vs with
  | [x, y] => if x == y then [x] else if x < y then [x, y] else [y, x]
  | l => l
def occMask (xs : List Int) : Nat := xs.foldl (fun acc x => acc ||| oneB x) 0
def render (s : State) (me g1 g2 w cur tick m8 m11 : Int) : State :=
  let vs := visGhosts me g1 g2 w
  let occ := occMask [me, g1, g2, w]
  let sp := barSpent tick (s.k_gap.getD 0)
  let g0 := vs.head?
  let g1' := (vs.drop 1).head?
  { s with
    me_r := some (pxR me), me_c := some (pxC me),
    ghosts_n := Int.ofNat vs.length, ghosts_m := Int.ofNat (vs.foldl (fun acc g => acc ||| pixBit g) 0),
    ghosts_0_r := g0.map pxR, ghosts_0_c := g0.map pxC,
    ghosts_1_r := g1'.map pxR, ghosts_1_c := g1'.map pxC,
    walker_r := if decide (0 ≤ w) && w != me then some (pxR w) else none,
    walker_c := if decide (0 ≤ w) && w != me then some (pxC w) else none,
    hud_cursor := cur, bar_spent := sp, bar := s.bar + s.bar_spent - sp, h_tick := tick,
    cart_cells_0 := m8, cart_cells_1 := m11,
    buttons_n := Int.ofNat (popc 200 (clearB (s.k_btnv.getD 0) occ)),
    pressed_colors_n := Int.ofNat ([s.k_de8, s.k_de11, s.k_de15].filter (fun m => (m.getD 0).toNat &&& occ != 0)).length,
    h_gA := g1, h_gB := g2, h_wp := w }

-- ---------- 一步 ----------
def stepMove (s : State) (d : Nat) (extra : Int) : State :=
  let me := meI s
  let o0 := occ0 s
  let o1 := occ1 s d
  let cs := cartsOnMove s o0 o1
  let me2 := hop s o0 o1 (me1 s d)
  let ga := ghostAdv s s.h_gA s.h_tpA
  let gb := ghostAdv s s.h_gB s.h_tpB
  let ga2 := if decide (0 ≤ ga) then hop s o0 o1 ga else ga
  let gb2 := if decide (0 ≤ gb) then hop s o0 o1 gb else gb
  let rec' := if me2 != me then Int.ofNat (s.h_rec.toNat ||| (d <<< (3 * tapeLen s.h_rec))) else s.h_rec
  let s1 := { s with h_wd := (walkerAdv s d).2, h_rt := s.h_rt + 1, h_rec := rec' }
  render s1 me2 ga2 gb2 (walkerFinal s d) s.hud_cursor (s.h_tick + 1 + extra) cs.1 cs.2

-- switch_hides_ghosts + hud_cursor_cycle + switch_rewinds_carts：换枚
def nextCursor (s : State) : Int := if s.hud_slots_n > 0 then (s.hud_cursor + 1) % s.hud_slots_n else 0
def stepSwitch (s : State) (extra : Int) : State :=
  let cur := nextCursor s
  let cs := cartsOnSwitch s (occ0 s)
  let st := s.k_st.getD 0
  let tpA := if cur == 0 then 0 else if cur == 1 then s.h_rec else s.h_tpA
  let tpB := if cur == 2 then s.h_rec else 0
  let s1 := { s with h_tpA := tpA, h_tpB := tpB, h_wd := 0, h_rt := 0, h_rec := 0, h_r0 := cs.2 }
  render s1 st (if cur ≥ 1 then st else -1) (if cur ≥ 2 then st else -1) (s.k_wk.getD (-1)) cur (s.h_tick + 1 + extra) cs.1 cs.2

def stepRules (s : State) (a : Act) : Option State :=
  if !ready s then none else
  match actCode a with
  | some (5, e) => some (stepSwitch s e)
  | some (d, e) => some (stepMove s d e)
  | none => none

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

    guard := fun s a => ready s && isMove a && s.walker_r.isSome
    effect := fun s _ s' => lvChg s s' ||
      (match s'.walker_r, s'.walker_c, s.walker_r, s.walker_c with
       | some r, some c, some r0, some c0 => (meI s' != meI s) == (r != r0 || c != c0)
       | _, _, _, _ => true) },
  { name := "walker_walks_alone"
    guard := fun s a => ready s && isMove a && decide (0 ≤ wpOf s)
    effect := fun s a s' => lvChg s s' ||
      (s'.h_wp == walkerFinal s (dirOf a) && s'.h_wd == (walkerAdv s (dirOf a)).2
        && s'.walker_r.isSome == (decide (0 ≤ s'.h_wp) && s'.h_wp != meI s')) }
]

-- 规则与全部观测一致。用 native_decide：语料几百条，kernel 上 decide 太慢；代价是多依赖 Lean.ofReduceBool 这条公理。
theorem consistent : checkRules rules corpus = true := by decide +kernel

-- 逐步重算整条路线。它换掉了旧的 solvable —— 那条只看末态 isGoal，中间几百步一步不查，

theorem faithful : (faithfulRun step start routeObs).map isGoal = some true := by decide +kernel
