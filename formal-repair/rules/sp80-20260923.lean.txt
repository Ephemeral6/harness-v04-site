structure Pc where

def Pc.cells (q : Pc) : Nat := q.sh <<< q.p
def Pc.fours (q : Pc) : Nat := q.fo <<< q.p
def Pc.body (q : Pc) : Nat := (minus q.sh q.fo) <<< q.p
def Pc.row (q : Pc) : Nat := q.p / 32
def Pc.col (q : Pc) : Nat := q.p % 32
def Pc.height (q : Pc) : Nat := Nat.log2 q.sh / 32 + 1

  | 0, _, _ => []
  | f + 1, pm, m4 =>

def findSel (m9 : Nat) : List Pc → Nat → Option Nat
  | [], _ => none
  | q :: qs, i => if (q.cells &&& m9) != 0 then some i else findSel m9 qs (i + 1)
def segment (s : State) : Option Lay :=
  let body := blocks s
  let pm := body ||| (mFour s &&& dil4 body)
  let ps := segPieces 5 pm (mFour s)
  match findSel (mSel s) ps 0 with
  | some i => some ⟨i, ps⟩
  | none => none

structure Vis where
  v4 : Nat
  v8 : Nat
  v9 : Nat
  v15 : Nat

def paintGo (L : Lay) : List Nat → Nat → Vis → Vis
  | [], _, v => v
  | i :: rest, cov, v =>
    let q := pcAt L i
    let vis := minus q.cells cov
    let body := minus vis q.fours
    let v1 : Vis :=
      if i == L.sel then { v with v9 := v.v9 ||| body }
      else if q.straight then { v with v8 := v.v8 ||| body }
      else { v with v15 := v.v15 ||| body }
    paintGo L rest (cov ||| q.cells) { v1 with v4 := v1.v4 ||| (q.fours &&& vis) }
def paint (static4 : Nat) (L : Lay) : Vis :=
  let v := paintGo L (zorder L) 0 ⟨0, 0, 0, 0⟩
  { v with v4 := v.v4 ||| static4 }

def ballRowsGo : Nat → Nat → Nat → Nat
  | 0, _, _ => 0
  | f + 1, r, m => (if ((m >>> (32 * r)) &&& rowM) != 0 then rowM <<< (32 * r) else 0) ||| ballRowsGo f (r + 1) m
-- 禁区：墙；每个杯格的八邻光环；球所在行及其上下各一行（整行）
def forb (s : State) : Nat :=
  let b := ballRowsGo (nRows s) 0 (mBall s)
  mWall s ||| dil8 (mCup s) ||| b ||| (b <<< 32) ||| (b >>> 32)

def moveLay (s : State) (L : Lay) (dr dc : Int) : Lay :=
  let q := selPc L
  let r : Int := (q.row : Int) + dr
  let c : Int := (q.col : Int) + dc
  if r < 0 || c < 0 || r + (q.height : Int) > (nRows s : Int) || c + (q.width : Int) > (nCols s : Int) then L
  else
    let np := r.toNat * 32 + c.toNat
    if ((q.sh <<< np) &&& forb s) != 0 then L
    else { L with ps := L.ps.set L.sel { q with p := np } }

def clickCell (s : State) (x y : Nat) : Option (Nat × Nat) :=
  let cell := s.geometry_0.toNat
  let r0 := s.geometry_1.toNat
  let c0 := s.geometry_2.toNat
  if y < r0 || x < c0 then none else
    let r := (y - r0) / cell

def inBox (q : Pc) (r c : Nat) : Bool :=
  q.row ≤ r && r < q.row + q.height && q.col ≤ c && c < q.col + q.width
def hitGo (L : Lay) (r c : Nat) : List Nat → Option Nat
  | [] => none
  | i :: rest => if inBox (pcAt L i) r c then some i else hitGo L r c rest

def clickLay (s : State) (L : Lay) (x y : Nat) : Lay :=

    | none => L

def stepOnce (s : State) (a : Act) : Option State :=

  | none => none
  | some L0 =>

-- 零偏折：球沿行进方向（背离同列最近的色4）的整条前路上没有件格（含挨着件色的色4 记号格）

def nearFourGo (m4 : Nat) (br bc n : Nat) : Nat → Nat → Option Int

  | f + 1, k =>

    else if br + k < n && ((m4 >>> ((br + k) * 32 + bc)) &&& 1) == 1 then some ((br : Int) + k)
    else nearFourGo m4 br bc n f (k + 1)
def zeroDefl (s : State) : Bool :=
  if pop (mBall s) != 1 then false else
    let i := lowIdx (mBall s)
    let br := i / 32
    let bc := i % 32
    let n := nRows s
    match nearFourGo (mFour s) br bc n n 1 with
    | none => false
    | some lr =>
      let col := colMaskGo n 0 bc
      -- 前路：背离发射器那一侧、球以外的同列格
      let front := if (br : Int) < lr then (col &&& ((1 <<< (br * 32)) - 1)) else (col >>> ((br + 1) * 32)) <<< ((br + 1) * 32)

      (front &&& (body ||| (mFour s &&& dil4 body))) == 0

  let cand := othersOf L
  if cand.isEmpty || cand.any (fun i => !(pcAt L i).straight) then none else
    let key := fun i => if s.travel_dir == 1 then ((pcAt L i).row : Int) else -(((pcAt L i).row + (pcAt L i).height - 1 : Nat) : Int)
    let best := cand.foldl (fun acc i => min acc (key i)) (key (cand.getD 0 0))
    let near := cand.filter (fun i => key i == best)
    let right := fun i => (pcAt L i).col + (pcAt L i).width - 1
    let m := near.foldl (fun acc i => max acc (right i)) 0
    match near.filter (fun i => right i == m) with
    | [i] => some i
    | _ => none

-- ---------------- 规则 ----------------
def moveRule (nm : String) (act : Act) (d : Nat) (vacate : Bool) : Rule :=
  { name := nm
    guard := fun s a => a == act && mSel s != 0 && (layout s).isSome
    effect := fun s _ s' =>
      !sameLv s s' ||
        (mSel s' == moved9 s d &&
          (!vacate ||
            match layout s with
            | some L =>
              let cs := (selPc L).cells
              let nw := if blocked9 s d then cs else shiftM cs d
              ((minus (minus cs nw) (othersCells L)) &&& occ s') == 0
            | none => true)) }
