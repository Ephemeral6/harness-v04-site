-- ---------- 位掩码与格 ----------
def ALLM : Nat := (1 <<< 256) - 1
def COL0 : Nat := ALLM / 65535                 -- 每行第 0 列那一位：Σ 2^(16r)
def NOTC0 : Nat := ALLM ^^^ COL0
def NOTC15 : Nat := ALLM ^^^ (COL0 <<< 15)
def NONE : Nat := 300                          -- 「没有这一格」（格号只到 255）
def bitN (p : Nat) : Nat := if p < 256 then 1 <<< p else 0
def hasN (m p : Nat) : Bool := p < 256 && Nat.testBit m p
def andNot (a b : Nat) : Nat := a ^^^ (a &&& b)
def absd (x y : Nat) : Nat := if y ≤ x then x - y else y - x
def distC (a b : Nat) : Nat := absd (a / 16) (b / 16) + absd (a % 16) (b % 16)
def adjN (a b : Nat) : Bool := a < 256 && b < 256 && distC a b == 1

-- 方向码 1 上 2 下 3 左 4 右
def stepCell (p d : Nat) : Nat :=
  if 256 ≤ p then NONE else
  match d with
  | 1 => if p < 16 then NONE else p - 16
  | 2 => if 256 ≤ p + 16 then NONE else p + 16
  | 3 => if p % 16 == 0 then NONE else p - 1
  | 4 => if p % 16 == 15 then NONE else p + 1
  | _ => NONE
def dirTo (p c : Nat) : Nat :=
  if stepCell p 1 == c then 1 else if stepCell p 2 == c then 2
  else if stepCell p 3 == c then 3 else if stepCell p 4 == c then 4 else 0

-- 四邻并集；pre x d = {q : q 往 d 走一格落在 x 里}
def nbrs (x : Nat) : Nat :=
  ((x <<< 16) ||| (x >>> 16) ||| ((x &&& NOTC15) <<< 1) ||| ((x &&& NOTC0) >>> 1)) &&& ALLM
def pre (x d : Nat) : Nat :=
  match d with
  | 1 => (x <<< 16) &&& ALLM
  | 2 => x >>> 16
  | 3 => ((x &&& NOTC15) <<< 1) &&& ALLM
  | _ => (x &&& NOTC0) >>> 1

-- 最短路下坡一步：从 goals 往外逐层扩（只在 ok 里扩），扩到 pos 时取 pos 在上一层里的邻格（次序 左 右 上 下）。
-- pos 就在 goals 里、或根本够不着 goals：原地。
def pickIn (pos layer : Nat) : List Nat → Nat
  | [] => pos
  | d :: ds => if hasN layer (stepCell pos d) then stepCell pos d else pickIn pos layer ds
def descendAux (pos ok : Nat) : Nat → Nat → Nat → Nat
  | 0, _, _ => pos
  | fuel + 1, layer, seen =>
    let nxt := andNot (nbrs layer &&& ok) seen
    if nxt == 0 then pos
    else if hasN nxt pos then pickIn pos layer [3, 4, 1, 2]
    else descendAux pos ok fuel nxt (seen ||| nxt)
def descend (pos goals ok : Nat) : Nat :=
  if hasN (goals &&& ok) pos then pos else descendAux pos ok 256 (goals &&& ok) (goals &&& ok)

-- 位掩码里的格，升序（逐次取最低位）
def lowBits : Nat → Nat → List Nat
  | 0, _ => []
  | fuel + 1, x => if x == 0 then [] else
      Nat.log2 (x ^^^ (x &&& (x - 1))) :: lowBits fuel (x &&& (x - 1))

-- 鼹鼠身份槽（关头一帧 h_m0 = -2：按读序现编）
structure Mole where
  slot : Nat
  pos : Nat
  skin : Nat      -- 0 帮手（皮12），1 贼（皮15）
  cargo : Nat     -- 扛着的货所在格；NONE = 空手
def slotCodes (s : State) : List Int :=
  if s.h_m0 == -2 then
    let cs := ((mlList s).filter (fun x => decide (0 ≤ x))).map
      (fun x => x % 256 + 256 * (if x / 256 == 1 || x / 256 == 2 then 1 else 0))
    cs ++ List.replicate (4 - cs.length) (-1)
  else hmList s
def decodeMoles (s : State) : List Mole :=
  ((slotCodes s).zip (List.range 4)).filterMap (fun vi =>
    if vi.1 < 0 then none else
    let n := vi.1.toNat
    some ⟨vi.2, n % 256, (n / 256) % 2, if n / 512 == 0 then NONE else stepCell (n % 256) (n / 512)⟩)
def moleMask (ms : List Mole) : Nat := ms.foldl (fun m o => m ||| bitN o.pos) 0
def cargoMask (ms : List Mole) : Nat := ms.foldl (fun m o => m ||| bitN o.cargo) 0
def freeOf (s : State) (ms : List Mole) : Nat :=
  (dnC s).foldl (fun m c => if c.2 != 0 && !hasN (cargoMask ms) c.1 then m ||| bitN c.1 else m) 0
def setMole (ms : List Mole) (o : Mole) : List Mole := ms.map (fun x => if x.slot == o.slot then o else x)

-- 玩家这一步（方向键 d）会不会整步不走：目的格是墙 / 围栏 / 出界 / 有货有鼹鼠有扛着的货；拖着货时货的目的格同样查（货可进围栏）
def blockedMove (s : State) (d : Nat) : Bool :=
  let w := world s
  let ms := decodeMoles s
  let occ := freeOf s ms ||| moleMask ms ||| cargoMask ms
  !(walkable w (stepCell (meC s) d)) || hasN occ (stepCell (meC s) d)
    || (heldC s < 256 && (!(donutOk w (stepCell (heldC s) d)) || hasN occ (stepCell (heldC s) d)))

structure PS where
  me : Nat
  face : Nat
  held : Nat          -- NONE = 空手
  free : Nat          -- 地上货（不含攥着的、不含被扛的）
  moles : List Mole
  tug : Nat           -- 被玩家对着它的货按 A5 拉住的贼的槽号；NONE = 没有

def playerPhase (w : Wd) (st : PS) (a : Nat) : PS :=
  if 1 ≤ a && a ≤ 4 then
    let blk := st.free ||| moleMask st.moles ||| cargoMask st.moles
    if st.held < 256 then
      if walkable w (stepCell st.me a) && donutOk w (stepCell st.held a)
          && !hasN blk (stepCell st.me a) && !hasN blk (stepCell st.held a)
      then { st with me := stepCell st.me a, held := stepCell st.held a } else st
    else
      if walkable w (stepCell st.me a) && !hasN blk (stepCell st.me a)
      then { st with me := stepCell st.me a, face := a } else { st with face := a }
  else if a == 5 then
    let f := stepCell st.me st.face
    if st.held < 256 then { st with free := st.free ||| bitN st.held, held := NONE }
    else if hasN st.free f then { st with free := andNot st.free (bitN f), held := f }
    else if st.moles.any (fun o => o.skin == 0 && o.cargo == f) then
      { st with moles := st.moles.map (fun o => if o.skin == 0 && o.cargo == f then { o with cargo := NONE } else o), held := f }
    else match st.moles.find? (fun o => o.skin == 1 && o.cargo == f) with
      | some o => { st with tug := o.slot }
      | none => match st.moles.find? (fun o => o.skin == 1 && o.pos == f) with
        | some o => { st with moles := st.moles.filter (fun x => x.slot != o.slot),
                              free := if o.cargo < 256 then st.free ||| bitN o.cargo else st.free }
        | none => st
  else st

structure MS where
  st : PS
  skip0 : Bool
  skip1 : Bool
  carried0 : List Nat       -- 本步开头扛着货的帮手槽号

def occOf (st : PS) (m : Mole) : Nat :=
  st.free ||| cargoMask st.moles ||| moleMask (st.moles.filter (fun x => x.slot != m.slot)) ||| bitN st.me ||| bitN st.held

-- 扛着货的一只：到了目标地形就放下；否则整对挪一格
def carryTurn (w : Wd) (acc : MS) (m : Mole) : MS :=
  let st := acc.st
  let goal := if m.skin == 0 then w.sock else w.den
  if hasN goal m.cargo then
    { acc with st := { st with free := st.free ||| bitN m.cargo, moles := setMole st.moles { m with cargo := NONE } } }
  else
    let d := dirTo m.pos m.cargo
    let occ2 := andNot (occOf st m) (bitN m.cargo)
    let okm := (andNot (moleOkM w) occ2 &&& pre (andNot (donutOkM w) occ2) d) ||| bitN m.pos
    let np := descend m.pos (pre (andNot goal occ2) d) okm
    if np == m.pos then acc
    else { acc with st := { st with moles := setMole st.moles { m with pos := np, cargo := stepCell np d } } }

-- 空手的贼：贴着「本步开头就被帮手扛着」的货就夺过来（次序 下 上 右 左）
def robFind (acc : MS) (m : Mole) : List Nat → Option Mole
  | [] => none
  | d :: ds =>
    match acc.st.moles.find? (fun o => o.skin == 0 && acc.carried0.contains o.slot && o.cargo == stepCell m.pos d
                                        && stepCell m.pos d < 256) with
    | some v => some v
    | none => robFind acc m ds

def firstIn (pos tg : Nat) : List Nat → Nat
  | [] => NONE
  | d :: ds => if hasN tg (stepCell pos d) then stepCell pos d else firstIn pos tg ds

def freeTurn (w : Wd) (acc : MS) (m : Mole) : MS :=
  let st := acc.st
  match (if m.skin == 1 then robFind acc m [2, 1, 4, 3] else none) with
  | some v =>
    { acc with st := { st with moles := setMole (setMole st.moles { v with cargo := NONE }) { m with cargo := v.cargo } } }
  | none =>
    let tg := if m.skin == 0 then andNot st.free w.sock else andNot (st.free ||| bitN st.held) w.den
    let pk := firstIn m.pos tg (if m.skin == 0 then [1, 2, 3, 4] else [2, 1, 4, 3])
    if pk < 256 then
      let st2 := if pk == st.held then { st with held := NONE } else { st with free := andNot st.free (bitN pk) }
      { acc with st := { st2 with moles := setMole st2.moles { m with cargo := pk } },
                 skip0 := acc.skip0 || m.skin == 0, skip1 := acc.skip1 || m.skin == 1 }
    else

        (st.moles.filter (fun o => o.skin == 0 && acc.carried0.contains o.slot)).foldl (fun x o => x ||| bitN o.cargo) 0 else 0
      let np := descend m.pos (nbrs (tg ||| chase)) (andNot (moleOkM w) (occOf st m) ||| bitN m.pos)
      { acc with st := { st with moles := setMole st.moles { m with pos := np } } }

def moleTurn (w : Wd) (acc : MS) (i : Nat) : MS :=
  match acc.st.moles.find? (fun x => x.slot == i) with
  | none => acc
  | some m =>
    if m.skin == 0 && acc.skip0 then { acc with skip0 := false }
    else if m.skin == 1 && acc.skip1 then { acc with skip1 := false }
    else if acc.st.tug == m.slot then { acc with skip0 := acc.skip0 || m.skin == 0, skip1 := acc.skip1 || m.skin == 1 }
    else if m.cargo < 256 then carryTurn w acc m
    else freeTurn w acc m

def molePhase (w : Wd) (st : PS) : PS :=
  let order := (st.moles.filter (fun o => o.skin == 0)).map (·.slot) ++ (st.moles.filter (fun o => o.skin == 1)).map (·.slot)
  let c0 := (st.moles.filter (fun o => o.skin == 0 && o.cargo < 256)).map (·.slot)
  (order.foldl (moleTurn w) ⟨st, false, false, c0⟩).st

-- ---------- 渲染回状态 ----------
def padTo (n : Nat) (l : List Int) : List Int := l ++ List.replicate (n - l.length) (-1)
def ringOf (st : PS) (f p : Nat) : Nat :=
  if p == st.held then 0 else if hasN (cargoMask st.moles) p then (if p == f then 3 else 5) else (if p == f then 3 else 4)
def slotCode (ms : List Mole) (i : Nat) : Int :=
  match ms.find? (fun o => o.slot == i) with
  | some o => Int.ofNat (o.pos + 256 * o.skin + 512 * (if o.cargo < 256 then dirTo o.pos o.cargo else 0))
  | none => -1
def moleKind (st : PS) (f p : Nat) : Nat :=
  match st.moles.find? (fun o => o.pos == p) with
  | some o => if o.skin == 0 then 0 else if p == f then 2 else 1
  | none => 0

def render (s : State) (st : PS) : State :=
  let f := stepCell st.me st.face
  let ds := padTo 14 ((lowBits 20 (st.free ||| bitN st.held ||| cargoMask st.moles)).map (fun p => Int.ofNat (p + 256 * ringOf st f p)))
  let mk := padTo 4 ((lowBits 8 (moleMask st.moles)).map (fun p => Int.ofNat (p + 256 * moleKind st f p)))
  let n := s.h_n + 1
  let b := s.k_B.toNat
  { s with
    dn_0 := ds.getD 0 (-1), dn_1 := ds.getD 1 (-1), dn_2 := ds.getD 2 (-1), dn_3 := ds.getD 3 (-1),
    dn_4 := ds.getD 4 (-1), dn_5 := ds.getD 5 (-1), dn_6 := ds.getD 6 (-1), dn_7 := ds.getD 7 (-1),
    dn_8 := ds.getD 8 (-1), dn_9 := ds.getD 9 (-1), dn_10 := ds.getD 10 (-1), dn_11 := ds.getD 11 (-1),
    dn_12 := ds.getD 12 (-1), dn_13 := ds.getD 13 (-1),
    ml_0 := mk.getD 0 (-1), ml_1 := mk.getD 1 (-1), ml_2 := mk.getD 2 (-1), ml_3 := mk.getD 3 (-1),
    h_m0 := slotCode st.moles 0, h_m1 := slotCode st.moles 1, h_m2 := slotCode st.moles 2, h_m3 := slotCode st.moles 3,
    h_n := n,
    timer_px := if b == 0 then 64 else Int.ofNat (64 - (n.toNat * 128 + b) / (2 * b)),
    pl_0 := Int.ofNat (st.me / 16), pl_1 := Int.ofNat (st.me % 16), pl_2 := Int.ofNat (st.face - 1) }

def initPS (s : State) : PS :=
  let ms := decodeMoles s
  ⟨meC s, faceD s, heldC s, freeOf s ms, ms, NONE⟩

def stepRules (s : State) (a : Act) : Option State :=
  if 1 ≤ actN a && actN a ≤ 5 then
    let w := world s
    some (render s (molePhase w (playerPhase w (initPS s) (actN a))))
  else none

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

-- 第 8 关之后是通关画面（level 9 = WIN）
def isGoal (s : State) : Bool := s.level == 9

-- ---------- 概念规则（每条对应 concepts/ 里同名的一条；guard = 概念开口的局面，effect = 概念的断言）----------

-- action1_up … action4_right：方向键让玩家整块往那边走恰好一格，或整块不动；不动恰好是被挡的时候
-- （墙 / 围栏 / 出界 / 目的格有货、鼹鼠或被扛的货；拖着货时货的目的格同样查，货进得去围栏）。
def moveRule (nm : String) (d : Nat) : Rule :=
  { name := nm
    guard := fun _ a => actN a == d
    effect := fun s _ s' => lvChg s s' || meC s' == (if blockedMove s d then meC s else stepCell (meC s) d) }

-- head_faces_last_move：头朝最后按的方向键（走没走成都转）；拖着货时、以及 A5 时头不转
def headRule : Rule :=
  { name := "head_faces_last_move"
    guard := fun _ a => 1 ≤ actN a && actN a ≤ 5
    effect := fun s a s' => lvChg s s' ||
      (if heldC s < 256 || actN a == 5 then s'.pl_2 == s.pl_2 else faceD s' == actN a) }

-- capsule_walkable：第 0/1 关玩家走得进容器格（2 芯）；第 2 关走不进含 2 的格（那一关只有围栏柱挡人）。
-- 只看目的格是容器或围栏、上面没有物件、手上没攥货的方向键。
def capsuleRule : Rule :=
  { name := "capsule_walkable"
    guard := fun s a => isMove a && s.level ≤ 2 && heldC s == NONE &&
      (let t := stepCell (meC s) (actN a)
       let ms := decodeMoles s
       hasN (s.k_sock.toNat ||| s.k_fence.toNat) t && !hasN (freeOf s ms ||| moleMask ms ||| cargoMask ms) t)
    effect := fun s a s' => lvChg s s' ||
      (if s.level ≤ 1 then meC s' == stepCell (meC s) (actN a) else meC s' != stepCell (meC s) (actN a)) }

-- donut_drag_follows：拖着的货随玩家走，货减人的偏移不变
def dragRule : Rule :=
  { name := "donut_drag_follows"
    guard := fun s a => isMove a && heldC s < 256
    effect := fun s _ s' => lvChg s s' || heldC s' == NONE ||
      (Int.ofNat (heldC s') - Int.ofNat (meC s') == Int.ofNat (heldC s) - Int.ofNat (meC s)) }

-- donut_drop_on_leave：玩家走开那一步，松手态（环3）的货原格落成环4；贴着鼹鼠的那颗不表态
def dropLeaveRule : Rule :=
  { name := "donut_drop_on_leave"
    guard := fun s a => isMove a && (dnC s).any (fun c => c.2 == 3)
    effect := fun s _ s' => lvChg s s' || meC s' == meC s ||
      (dnC s).all (fun c => c.2 != 3 || (mlC s).any (fun m => m.1 == c.1 || adjN m.1 c.1) || ringAt s' c.1 == 4) }

-- donut_enters_capsule：拖着的货进得去容器 / 围栏格（玩家自己的目的格不含 2、这一步没被挡时）
def entersRule : Rule :=
  { name := "donut_enters_capsule"
    guard := fun s a => isMove a && heldC s < 256 &&
      !hasN (s.k_sock.toNat ||| s.k_fence.toNat ||| s.k_den.toNat) (stepCell (meC s) (actN a)) &&
      hasN (s.k_sock.toNat ||| s.k_fence.toNat) (stepCell (heldC s) (actN a)) && !blockedMove s (actN a)
    effect := fun s a s' => lvChg s s' || heldC s' == stepCell (heldC s) (actN a) }

-- donut_grab_toggle：A5 抓 / 松都在原格；攥着就松成原格环3，没攥着而正前方是地上货（不是谁扛着的）就抓成环0
def grabRule : Rule :=
  { name := "donut_grab_toggle"
    guard := fun _ a => actN a == 5
    effect := fun s _ s' => lvChg s s' ||
      ((dnC s).all (fun c => !((c.2 == 3 && ringAt s' c.1 == 0) || (c.2 == 0 && ringAt s' c.1 == 3)) || hasDn s' c.1) &&
       (if heldC s < 256 then ringAt s' (heldC s) == 3
        else !hasN (freeOf s (decodeMoles s)) (facedC s) || ringAt s' (facedC s) == 0)) }

-- donut_push_activate：环4→3 只发生在走完之后玩家正对的那一格上（顶住或走到紧邻）
def pushRule : Rule :=
  { name := "donut_push_activate"
    guard := fun s a => isMove a && !(dnC s).isEmpty
    effect := fun s _ s' => lvChg s s' ||
      (dnC s).all (fun c => !(c.2 == 4 && ringAt s' c.1 == 3) || c.1 == facedC s') }

-- donut_take_back_from_mole：方向键之后，玩家正前方若坐着一颗刚从搬运里来的货（上一帧是环5，或贴着一颗环5），它这一帧是环3
def takeBackRule : Rule :=
  { name := "donut_take_back_from_mole"
    guard := fun s a => isMove a && heldC s == NONE && (dnC s).any (fun c => c.2 == 5)
    effect := fun s _ s' => lvChg s s' ||
      (let t := facedC s'
       t ≥ 256 || (hasDn s t && ringAt s t != 5) ||
       !((dnC s).any (fun c => c.2 == 5 && (c.1 == t || adjN c.1 t))) ||
       !hasDn s' t || ringAt s' t == 3) }

-- drag_released_by_thief_block：拖着的货的目的格被鼹鼠或被扛的货挡住（玩家自己的目的格没挡）：
-- 挡路的是贼 → 就地脱手（不再攥着、原格仍有货）；是帮手或被扛的货 → 仍攥在原格
def thiefBlockRule : Rule :=
  { name := "drag_released_by_thief_block"
    guard := fun s a => isMove a && heldC s < 256 &&
      (let bl := fun p => (mlC s).any (fun m => m.1 == p) || (dnC s).any (fun c => c.2 == 5 && c.1 == p)
       !bl (stepCell (meC s) (actN a)) && bl (stepCell (heldC s) (actN a)))
    effect := fun s a s' => lvChg s s' ||
      (if isThiefK (moleAt s (stepCell (heldC s) (actN a))) then heldC s' == NONE && hasDn s' (heldC s)
       else heldC s' == heldC s) }

-- grab_refused_by_still_thief：A5 去抓唯一一颗贴着玩家的环3，而它贴着贼：抓到 ⟺ 贴着的贼这一步都挪了窝
def grabRefusedRule : Rule :=
  { name := "grab_refused_by_still_thief"
    guard := fun s a => actN a == 5 && heldC s == NONE &&
      (let adj3 := (dnC s).filter (fun c => c.2 == 3 && adjN c.1 (meC s))
       adj3.length == 1 && (mlC s).any (fun m => isThiefK m.2 && adj3.any (fun c => adjN m.1 c.1)))
    effect := fun s _ s' => lvChg s s' ||
      (let cell := ((dnC s).filter (fun c => c.2 == 3 && adjN c.1 (meC s))).head?.elim NONE (·.1)
       let thieves := (mlC s).filter (fun m => isThiefK m.2 && adjN m.1 cell)
       let stayed := thieves.any (fun m => (mlC s').any (fun m2 => isThiefK m2.2 && m2.1 == m.1))
       (ringAt s' cell == 0) == !stayed) }

-- killed_mole_drops_cargo：A5 清掉正对的贼时，它扛着的那颗（与它相邻的唯一环5）原格变环4
def killRule : Rule :=
  { name := "killed_mole_drops_cargo"
    guard := fun s a => actN a == 5 && isThiefK (moleAt s (facedC s)) &&
      ((dnC s).filter (fun c => c.2 == 5 && adjN c.1 (facedC s))).length == 1
    effect := fun s _ s' => lvChg s s' || moleAt s' (facedC s) != NONE ||
      (((dnC s).filter (fun c => c.2 == 5 && adjN c.1 (facedC s))).all (fun c => ringAt s' c.1 == 4)) }

-- action5_vanishes_adjacent_mole：有鼹鼠贴着玩家时按 A5：正对的若是贼，它从场上消失；按皮点名数，只少那一只
def vanishRule : Rule :=
  { name := "action5_vanishes_adjacent_mole"
    guard := fun s a => actN a == 5 && (mlC s).any (fun m => adjN m.1 (meC s))
    effect := fun s _ s' => lvChg s s' ||
      (let doomed := isThiefK (moleAt s (facedC s))
       let cnt := fun (t : State) (th : Bool) => ((mlC t).filter (fun m => isThiefK m.2 == th)).length
       (!doomed || moleAt s' (facedC s) == NONE) &&
       cnt s' false == cnt s false && cnt s' true + (if doomed then 1 else 0) == cnt s true) }

-- action5_inert_without_target：手空、四邻没有环3 / 环5 / 鼹鼠、正前方不是货时按 A5：玩家不动不转，也没抓起任何东西
def inertRule : Rule :=
  { name := "action5_inert_without_target"
    guard := fun s a => actN a == 5 && heldC s == NONE && !hasDn s (facedC s) &&
      !((dnC s).any (fun c => (c.2 == 3 || c.2 == 5) && adjN c.1 (meC s))) && !((mlC s).any (fun m => adjN m.1 (meC s)))
    effect := fun s _ s' => lvChg s s' || (meC s' == meC s && s'.pl_2 == s.pl_2 && heldC s' == NONE) }

-- levelup_on_release_iff_all_seated：场上没有环5 时按 A5：换关 ⟺ 按之前每一颗（含攥着的）都坐在容器格上
def levelupRule : Rule :=
  { name := "levelup_on_release_iff_all_seated"
    guard := fun s a => actN a == 5 && !(dnC s).any (fun c => c.2 == 5) && !(dnC s).isEmpty
    effect := fun s _ s' => lvChg s s' == (dnC s).all (fun c => hasN s.k_sock.toNat c.1) }

-- win_needs_all_donuts_home：还有不被扛着的货（环0/3/4）不在容器格上，这一步就不换关
def winRule : Rule :=
  { name := "win_needs_all_donuts_home"
    guard := fun s a => actN a != 9 && (dnC s).any (fun c => c.2 != 5 && !hasN s.k_sock.toNat c.1)
    effect := fun s _ s' => !lvChg s s' }

-- mole_alert_ring_when_adjacent：一只鼹鼠外圈是警觉色 ⟺ 它是贼、且正好在玩家头朝的那一格上
def alertRule : Rule :=
  { name := "mole_alert_ring_when_adjacent"
    guard := fun _ a => actN a != 9
    effect := fun s _ s' => lvChg s s' ||
      (mlC s').all (fun m => (m.2 == 2 || m.2 == 3) == (isThiefK m.2 && m.1 == facedC s')) }

-- mole_blocks_move：玩家或所拖货的目的格上蹲着鼹鼠：玩家停在原格，拖着的那颗仍在原格（攥没攥住不管）
def moleBlockRule : Rule :=
  { name := "mole_blocks_move"
    guard := fun s a => isMove a &&
      ((mlC s).any (fun m => m.1 == stepCell (meC s) (actN a)) ||
       (heldC s < 256 && (mlC s).any (fun m => m.1 == stepCell (heldC s) (actN a))))
    effect := fun s _ s' => lvChg s s' || (meC s' == meC s && (heldC s == NONE || hasDn s' (heldC s))) }

-- 平移一格（dr, dc ∈ {-1,0,1}）；出界回 NONE
def shiftC (p : Nat) (dr dc : Int) : Nat :=
  let r := Int.ofNat (p / 16) + dr
  let c := Int.ofNat (p % 16) + dc
  if r < 0 || r > 15 || c < 0 || c > 15 then NONE else (r * 16 + c).toNat

-- mole_carry_above：每颗环5 都随某只鼹鼠（这一步至多走一格）平移同样的位移；脱离搬运（原格或随行后那格变环3/4）的跳过
def carryRule : Rule :=
  { name := "mole_carry_above"
    guard := fun s a => actN a != 9 && (dnC s).any (fun c => c.2 == 5)
    effect := fun s _ s' => lvChg s s' ||
      (!(mlC s').isEmpty &&
       (let deltas := (mlC s).foldl (fun acc m => acc ++ ((mlC s').filter (fun n => distC m.1 n.1 ≤ 1)).map
                          (fun n => (Int.ofNat (n.1 / 16) - Int.ofNat (m.1 / 16), Int.ofNat (n.1 % 16) - Int.ofNat (m.1 % 16)))) []
        (dnC s).all (fun c => c.2 != 5 ||
          (ringAt s' c.1 != 5 && ((0, 0) :: deltas).any (fun dd => let q := ringAt s' (shiftC c.1 dd.1 dd.2); q == 3 || q == 4)) ||
          deltas.any (fun dd => ringAt s' (shiftC c.1 dd.1 dd.2) == 5)))) }

-- mole_drop_in_capsule：环5 全部消失的那一步（没有鼹鼠被清掉、不是落到玩家正前方成环3）：
-- 第一颗在原格落地、有鼹鼠这一步没动，且落点是目标地形（容器或巢）
def dropCapRule : Rule :=
  { name := "mole_drop_in_capsule"
    guard := fun s a => actN a != 9 && (dnC s).any (fun c => c.2 == 5)
    effect := fun s _ s' => lvChg s s' || (dnC s').any (fun c => c.2 == 5) || (mlC s').length < (mlC s).length ||
      (let p := (((dnC s).filter (fun c => c.2 == 5)).map (·.1)).foldl min 255
       let fr := facedC s'
       (fr < 256 && (fr == p || adjN fr p) && ringAt s' fr == 3) ||
       (hasDn s' p && (mlC s).any (fun m => (mlC s').any (fun n => n.1 == m.1)) &&
        hasN (s.k_sock.toNat ||| s.k_den.toNat) p)) }

-- mole_drop_target_by_skin：投放（环5 原格变环4，投放者是原地没动、同皮、与落点相邻的唯一一只）：帮手落在容器，贼落在巢
def dropSkinRule : Rule :=
  { name := "mole_drop_target_by_skin"
    guard := fun s a => actN a != 9 && (dnC s).any (fun c => c.2 == 5)
    effect := fun s _ s' => lvChg s s' ||
      (dnC s).all (fun c => c.2 != 5 || ringAt s' c.1 != 4 ||
        (let cands := (mlC s').filter (fun n => (mlC s).any (fun m => m.1 == n.1 && isThiefK m.2 == isThiefK n.2) && adjN n.1 c.1)
         cands.length != 1 ||
         (match cands.head? with
          | some n => if isThiefK n.2 then hasN s.k_den.toNat c.1 else hasN s.k_sock.toNat c.1
          | none => true))) }

-- mole_steal_on_block：环4 原格变环5（被偷）时，存在一只这一步没动、与它相邻的鼹鼠
def stealRule : Rule :=
  { name := "mole_steal_on_block"
    guard := fun s a => actN a != 9 && (dnC s).any (fun c => c.2 == 4 && (mlC s).any (fun m => adjN m.1 c.1))
    effect := fun s _ s' => lvChg s s' ||
      (dnC s).all (fun c => !(c.2 == 4 && ringAt s' c.1 == 5) ||
        (mlC s).any (fun m => adjN m.1 c.1 && (mlC s').any (fun n => n.1 == m.1))) }

-- stolen_donut_blocks_move：玩家或所拖货的目的格上是环5：玩家与拖着的货都不动
def stolenBlockRule : Rule :=
  { name := "stolen_donut_blocks_move"
    guard := fun s a => isMove a &&
      ((dnC s).any (fun c => c.2 == 5 && (c.1 == stepCell (meC s) (actN a) ||
                                          (heldC s < 256 && c.1 == stepCell (heldC s) (actN a)))))
    effect := fun s _ s' => lvChg s s' || (meC s' == meC s && heldC s' == heldC s) }

-- 一一配对：A 的每只按次序在 B 里挑一只距离 ≤1 的、没被挑过的（回溯，至多 4 只）
def matchLe1 : List Nat → List Nat → Bool
  | [], _ => true
  | a :: as, bs => bs.any (fun b => distC a b ≤ 1 && matchLe1 as (bs.erase b))

-- thief_step_rule：贼数不变（1~4 只）时，存在一一配对使每只贼至多挪一格
def thiefStepRule : Rule :=
  { name := "thief_step_rule"
    guard := fun s a => actN a != 9 && (mlC s).any (fun m => isThiefK m.2)
    effect := fun s _ s' => lvChg s s' ||
      (let A := ((mlC s).filter (fun m => isThiefK m.2)).map (·.1)
       let B := ((mlC s').filter (fun m => isThiefK m.2)).map (·.1)
       A.length != B.length || A.length > 4 || matchLe1 A B) }

def thiefCells (t : State) : List Nat := ((mlC t).filter (fun m => isThiefK m.2)).map (·.1)

-- thief_patrol_next：第 8 关贼只有一只时，它下一拍的格由当前局面决定、与玩家的意图无关——
-- 不抄巡逻表，而是说：s' 的贼格集合 = stepRules 从 s 推出的贼格集合（追货最短路 + 偷 / 放 / 夺 / 被清除）
def patrolRule : Rule :=
  { name := "thief_patrol_next"
    guard := fun s a => s.level == 8 && 1 ≤ actN a && actN a ≤ 5 && (thiefCells s).length == 1
    effect := fun s a s' => lvChg s s' ||
      (match stepRules s a with
       | some p => thiefCells p == thiefCells s'
       | none => false) }

-- thief_strips_drag：拖着货走、落点旁边有贼：货脱手 ⟺ 落点旁边那只贼这一步留在原格
def stripRule : Rule :=
  { name := "thief_strips_drag"
    guard := fun s a => isMove a && heldC s < 256 && (mlC s).any (fun m => isThiefK m.2)
    effect := fun s _ s' => lvChg s s' ||
      (let after := (mlC s').filter (fun m => isThiefK m.2)
       let pc := meC s'
       let loose := (dnC s').filter (fun c => c.2 == 3)
       let pick := loose.foldl (fun b c => if b == NONE || distC c.1 pc < distC b pc || (distC c.1 pc == distC b pc && c.1 < b) then c.1 else b) NONE
       let cell := if heldC s' < 256 then heldC s' else pick
       after.isEmpty || cell == NONE ||
       (let touching := after.filter (fun m => distC cell m.1 ≤ 1)
        touching.isEmpty ||
        (touching.any (fun m => (mlC s).any (fun n => isThiefK n.2 && n.1 == m.1)) == !(heldC s' < 256)))) }

-- timer_bar_drains：同一关内每一步计时条缩 0 或 1 像素；加强：长度 = 64 - round(本关步数*64/B)，步数加一
def timerRule : Rule :=
  { name := "timer_bar_drains"
    guard := fun _ a => actN a != 9
    effect := fun s _ s' => lvChg s s' ||
      ((s'.timer_px == s.timer_px || s'.timer_px == s.timer_px - 1) && s'.h_n == s.h_n + 1 &&
       decide (0 < s.k_B) && s'.timer_px == 64 - ((s.h_n + 1) * 128 + s.k_B) / (2 * s.k_B)) }

   thiefBlockRule, grabRefusedRule, killRule, vanishRule, inertRule, levelupRule, winRule, alertRule,
   moleBlockRule, carryRule, dropCapRule, dropSkinRule, stealRule, stolenBlockRule, thiefStepRule, stripRule, timerRule,
   patrolRule]

-- 规则与全部观测一致。证明用 decide +kernel（内核求值，不依赖 Lean.ofReduceBool）。
theorem consistent : checkRules rules corpus = true := by decide +kernel

-- 逐步重算整条路线。它换掉了旧的 solvable —— 那条只看末态 isGoal，中间几百步一步不查，
-- 「直接把目标字段拍成真」的 step 照样能过。faithful 每一步都要对上观测，规则段没写完之前编不过，那正是要求。
theorem faithful : (faithfulRun step start routeObs).map isGoal = some true := by decide +kernel

-- 每条规则的 guard 在语料上命中多少条。checkRules 是 `!guard || effect`，guard 恒 false 就**空真**——
-- 一条永不触发的规则能让 consistent 无条件成立。verify 读这一行，有 0 就判红。
#eval IO.println ("GUARD_HITS " ++ String.intercalate "," (rules.map (fun r => r.name ++ "=" ++ toString (corpus.filter (fun t => r.guard t.s t.a)).length)))
