-- ---------- 几何 ----------
def side (t : Int) : Int := if t ≤ 5 then t else t + 1

def clampX (w nx : Int) : Int := max 0 (min nx (64 - w))
def clampY (h ny : Int) : Int := min (min (max 10 ny) 63) (64 - h)

def inReach (px py ox oy w h : Int) : Bool :=
  let qx := min (max px ox) (ox + w - 1)
  let qy := min (max py oy) (oy + h - 1)
  (px - qx) * (px - qx) + (py - qy) * (py - qy) ≤ 64

def boxHit (ax ay aw ah bx bY bw bh : Int) : Bool :=
  ax < bx + bw && bx < ax + aw && ay < bY + bh && bY < ay + ah

def step4 (d : Int) : Int := if d > 0 then min 4 d else if d < 0 then max (-4) d else 0
def sgnStep (k d : Int) : Int := if d > 0 then k else if d < 0 then -k else 0

-- a·√D ≤ b（整数精确比较）
def sqrtLe (a b d : Int) : Bool :=
  if a ≤ 0 then (if b ≥ 0 then true else a * a * d ≥ b * b) else (if b < 0 then false else a * a * d ≤ b * b)

def scanQ (num den d : Int) : Nat → Int → Int
  | 0, q => q
  | f + 1, q => if sqrtLe ((2 * q - 1) * den) (2 * num) d then q else scanQ num den d f (q - 1)

-- 四舍五入 p + num / (den·√d)：q = 满足 q - 1/2 ≤ num/(den·√d) 的最大整数
def roundAdd (p num den d : Int) : Int :=
  if d ≤ 0 then p else p + scanQ num den d 40 20

structure Pc where
  t : Int
  x : Int
  y : Int
  sel : Bool
  cap : Bool
  cnt : Nat
  sx : Int
  sy : Int
  dx : Int
  dy : Int

structure Cr where

  x : Int
  y : Int

  dx : Int
  dy : Int

def pcx (p : Pc) : Int := p.x + side p.t / 2
def pcy (p : Pc) : Int := p.y + side p.t / 2
def ccx (c : Cr) : Int := c.x + 2
def ccy (c : Cr) : Int := c.y + 2
def pcHitCr (c : Cr) (p : Pc) : Bool := boxHit c.x c.y 5 4 p.x p.y (side p.t) (side p.t)
def pcHitPc (p q : Pc) : Bool := boxHit p.x p.y (side p.t) (side p.t) q.x q.y (side q.t) (side q.t)
def crHitCr (a b : Cr) : Bool := boxHit a.x a.y 5 4 b.x b.y 5 4

-- ---------- pieces_merge_at_click / mover14_pounces_when_close：选中者走向点击点 ----------
def movePc (x y : Int) (p : Pc) : Pc :=
  let s := side p.t
  { p with x := clampX s (p.x + step4 (x - pcx p)), y := clampY s (p.y + step4 (y - pcy p)) }

def moveCr (x y : Int) (c : Cr) : Cr :=
  if c.k == 1 then
    let d := c.dx * c.dx + c.dy * c.dy
    { c with x := clampX 5 (roundAdd c.x (17 * c.dx) 5 d), y := clampY 4 (roundAdd c.y (17 * c.dy) 5 d) }
  else
    { c with x := clampX 5 (c.x + step4 (x - ccx c)), y := clampY 4 (c.y + step4 (y - ccy c)) }

def moveSel (x y : Int) (w : W) : W :=
  { w with ps := w.ps.map (fun p => if p.sel && !p.cap then movePc x y p else p),
           cs := w.cs.map (fun c => if c.sel then moveCr x y c else c) }

-- ---------- captured_piece_flung_ten：停 3 子步，再降一级、一次甩出 10 格 ----------
def flingOne (p : Pc) : Pc :=
  if p.cnt < 3 then { p with cnt := p.cnt + 1, cap := !(p.dest && p.cnt + 1 ≥ 3) } else
  let t' := p.t - 1
  let s := side t'
  let dd := p.dx * p.dx + p.dy * p.dy
  { p with t := t', x := clampX s (roundAdd p.sx (10 * p.dx) 1 dd), y := clampY s (roundAdd p.sy (10 * p.dy) 1 dd),
           cnt := p.cnt + 1, cap := false }

def fling (w : W) : W :=
  let ps := w.ps.map (fun p => if p.cap then flingOne p else p)
  { w with ps := ps.filter (fun p => !(p.dest && !p.cap)) }

-- ---------- mover_captures_piece：盒子压上就咬 ----------
def bite (c : Cr) (p : Pc) : Pc :=
  let dx0 := pcx p - ccx c
  let dy0 := pcy p - ccy c
  let z := dx0 == 0 && dy0 == 0
  { p with cap := true, cnt := 0, sx := p.x, sy := p.y, dx := (if z then 0 else dx0), dy := (if z then -1 else dy0),
           dest := p.t ≤ 1 }

def bitesBy (c : Cr) (ps : List Pc) : List Pc × Bool :=
  ps.foldr (fun p (acc : List Pc × Bool) =>
    if !p.cap && pcHitCr c p then (bite c p :: acc.1, true) else (p :: acc.1, acc.2)) ([], false)

def biteAll (w : W) : W :=
  let r := w.cs.foldl (fun (acc : List Pc × List Cr × Nat) c =>
    if c.sel || c.stop then (acc.1, acc.2.1 ++ [c], acc.2.2) else
    let (ps', hit) := bitesBy c acc.1
    (ps', acc.2.1 ++ [if hit then { c with stop := true } else c], acc.2.2 + (if hit then 1 else 0))) (w.ps, [], 0)
  { w with ps := r.1, cs := r.2.1, ncap := w.ncap + r.2.2 }

-- ---------- 并（同种/同级相交并成高一级；不同种/不同级相交 = 驳回）----------
def relabel (hit : Nat → Nat → Bool) (n : Nat) (lab : List Nat) : List Nat :=
  (List.range n).map (fun i =>
    (List.range n).foldl (fun m j => if hit i j && lab.getD j j < m then lab.getD j j else m) (lab.getD i i))

def labelsOf (hit : Nat → Nat → Bool) (n : Nat) : Nat → List Nat → List Nat
  | 0, lab => lab
  | f + 1, lab => labelsOf hit n f (relabel hit n lab)

def groupsOf (n : Nat) (lab : List Nat) : List (List Nat) :=
  ((List.range n).filter (fun i => lab.getD i i == i)).map (fun r => (List.range n).filter (fun i => lab.getD i i == r))

def creMerge (w : W) : W :=
  let cs := w.cs
  let n := cs.length
  let hit := fun i j => i != j && crHitCr (cs.getD i default) (cs.getD j default)
  if n < 2 then w else
  if (List.range n).any (fun i => (List.range n).any (fun j => hit i j && (cs.getD i default).k != (cs.getD j default).k))
  then { w with ref := true } else
  let gs := (groupsOf n (labelsOf hit n n (List.range n))).filter (fun g => g.length ≥ 2)
  if gs.isEmpty then w else
  let born := gs.filterMap (fun g =>
    let k := (cs.getD (g.headD 0) default).k
    if k ≥ 3 then none else
    let m : Int := g.length
    let sx := (g.map (fun i => ccx (cs.getD i default))).foldl (· + ·) 0 / m
    let sy := (g.map (fun i => ccy (cs.getD i default))).foldl (· + ·) 0 / m
    some ({ k := k + 1, x := clampX 5 (sx - 2), y := clampY 4 (sy - 2), sel := false, dx := 0, dy := 0, stop := true } : Cr))
  let gone := gs.foldl (· ++ ·) []
  { w with cs := ((List.range n).filter (fun i => !gone.contains i)).map (fun i => cs.getD i default) ++ born,
           nmerge := w.nmerge + gs.length }

def pieceMerge (w : W) : W :=
  let sel := w.ps.filter (·.sel)
  let n := sel.length
  let hit := fun i j => i != j && pcHitPc (sel.getD i default) (sel.getD j default)
  if n < 2 then w else
  if (List.range n).any (fun i => (List.range n).any (fun j => hit i j && (sel.getD i default).t != (sel.getD j default).t))
  then { w with ref := true } else
  let gs := (groupsOf n (labelsOf hit n n (List.range n))).filter (fun g => g.length ≥ 2)
  if gs.isEmpty then w else
  let born := gs.map (fun g =>
    let t := (sel.getD (g.headD 0) default).t
    let m : Int := g.length
    let s := side (t + 1)
    let sx := (g.map (fun i => pcx (sel.getD i default))).foldl (· + ·) 0 / m
    let sy := (g.map (fun i => pcy (sel.getD i default))).foldl (· + ·) 0 / m
    ({ t := t + 1, x := clampX s (sx - s / 2), y := clampY s (sy - s / 2), sel := false, cap := false, cnt := 0,
       sx := 0, sy := 0, dx := 0, dy := 0, dest := false, uid := 0 } : Pc))
  let goneU := (gs.foldl (· ++ ·) []).map (fun i => (sel.getD i default).uid)
  { w with ps := w.ps.filter (fun p => !goneU.contains p.uid) ++ born }

-- ---------- mover7_clamp_far_from_click_no_capture：没被选中的去追最近的件 ----------
def nearest (c : Cr) (ps : List Pc) : Option Pc :=
  (ps.foldl (fun (acc : Option (Int × Pc)) p =>
    let d := (pcx p - ccx c) * (pcx p - ccx c) + (pcy p - ccy c) * (pcy p - ccy c)
    match acc with
    | none => some (d, p)
    | some (bd, _) => if d < bd then some (d, p) else acc) none).map (·.2)

def chaseOne (free : List Pc) (c : Cr) : Cr :=
  match nearest c free with
  | none => c
  | some p =>
    let v := if c.k == 3 then 2 else 1
    { c with x := clampX 5 (c.x + sgnStep v (pcx p - ccx c)), y := clampY 4 (c.y + sgnStep v (pcy p - ccy c)) }

def chaseStep (xw : Nat) (w0 : W) : W :=
  let w := if w0.ps.any (·.cap) then fling w0 else w0
  if xw ≥ 4 || w.cs.isEmpty || w.ps.isEmpty then w else
  let free := w.ps.filter (fun p => !p.cap)
  let w := { w with cs := w.cs.map (fun c => if c.sel || c.stop then c else chaseOne free c) }
  creMerge (biteAll w)

def clickLoop (x y : Int) : Nat → Nat → Bool → W → W
  | 0, _, _, w => w
  | f + 1, xw, mg, w =>
    let w := chaseStep xw (moveSel x y w)
    if w.ref then w else
    let cont := xw + 1 < 4 || w.ps.any (·.cap)
    let w := if !mg && xw + 1 ≥ 4 then (let w2 := creMerge w; if w2.ref then w2 else pieceMerge w2) else w
    if w.ref then w else if cont then clickLoop x y f (xw + 1) (mg || xw + 1 ≥ 4) w else w

def budgetOf (t clicks frozen level : Int) : Int :=
  let n := clicks + (if level ≥ 1 then frozen * frozen else 0)
  (max 0 (t - n) * 64 + t - 1) / t

def render (s : State) (w : W) : Option State :=
  let ps := sortBy (fun p => p.y * 64 + p.x) w.ps
  let pm := ps.foldl (fun m p => m ||| pcMask p) 0
  let vis := sortBy (fun c => c.y * 64 + c.x) (w.cs.filter (fun c => minus (crMask c) pm != 0))
  if ps.length > 9 || vis.length > 4 then none else
  let pl := ps.map encP
  let cl := vis.map encC
  let ix := ps.zip (List.range ps.length)
  let iso := ix.filter (fun (p, i) => !(ix.any (fun (q, j) => i != j && q.t == p.t && touching p q)))
  let c3 := iso.filter (fun (p, _) => p.t == 3 && p.y + 2 ≤ 62)
  let cur : Option (Int × Int) := match c3 with | [(p, _)] => some (p.y + 1, p.x + 1) | _ => none
  let mv := minus (w.cs.foldl (fun m c => if c.k == 1 then m ||| crMask c else m) 0) pm &&& boardRows
  let kd := (s.k_dots.getD 0).toNat
  let dots := minus kd pm
  let n := s.h_clicks + 1
  let same := pl == (pcsL s).filter (· ≥ 0) && cl == (cresL s).filter (· ≥ 0)
  let fz := s.h_frozen + (if same then 1 else 0)
  some { s with
    pcs_0 := pl.getD 0 (-1), pcs_1 := pl.getD 1 (-1), pcs_2 := pl.getD 2 (-1), pcs_3 := pl.getD 3 (-1),
    pcs_4 := pl.getD 4 (-1), pcs_5 := pl.getD 5 (-1), pcs_6 := pl.getD 6 (-1), pcs_7 := pl.getD 7 (-1),
    pcs_8 := pl.getD 8 (-1),
    cres_0 := cl.getD 0 (-1), cres_1 := cl.getD 1 (-1), cres_2 := cl.getD 2 (-1), cres_3 := cl.getD 3 (-1),
    pieces_n := iso.length,
    cursor_r := cur.map (·.1), cursor_c := cur.map (·.2),
    me_r := cur.map (·.1 - 1), me_c := cur.map (·.2 - 1),
    mover_m := (if mv == 0 then none else some (mv : Int)), mover_n := (if mv == 0 then none else some (popc 64 mv : Int)),
    hunters_n := (vis.filter (fun c => c.k == 1 || c.k == 2)).length,
    diamonds_n := (vis.filter (fun c => c.k == 2)).length,
    dot_cells_m := (dots : Int), dot_cells_n := (popc 64 dots : Int),
    budget := budgetOf (s.k_T.getD 32) n fz s.level,
    h_clicks := n,
    h_frozen := fz }

-- 超时重发（修订 R2）：本局唯一的重发点击 seq 66（ACTION6:42,42）预算只扣一步，件的结果与点一次相同
-- （这一击点两次与点一次结果也相同，数据分不出应用了一次还是两次）；按一次处理。
def baseAct : Act → Act
  | Act.retried a => baseAct a
  | a => a

def boardClick (a : Act) : Option (Int × Int) :=
  match baseAct a with
  | Act.a6 x y => if 10 ≤ y && y ≤ 62 then some ((x : Int), (y : Int)) else none
  | _ => none

def stepRules (s : State) (a : Act) : Option State :=
  match baseAct a with
  | Act.a6 x y =>
    if y ≤ 9 || y ≥ 63 then some s else
    let w := simClick s (x : Int) (y : Int)
    -- 世界驳回：棋盘一格不动（对象照原样渲染），h_frozen 加一，预算按 budgetOf 扣
    if w.ref then render s (initW s (x : Int) (y : Int)) else render s w
  | _ => none

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

-- 第 9 关打完 = 整局 WIN（level 从 8 跳到 9）
def isGoal (s : State) : Bool := s.level == 9

-- ---------- 规则（每条对应一条概念；effect 只断言该概念主张的那一部分）----------
-- 「世界接受了这次点击」：没换关、且点完第 10..62 行不是一格没动（h_frozen 没加）。概念的 SCOPE 都只管这一档。
def accepted (s s' : State) : Bool := s'.level == s.level && s'.h_frozen == s.h_frozen
def capOf (l : List Int) : Int := (l.filterMap decObj).foldl (fun m (t, _, _) => m + 2 ^ (t - 1).toNat) 0
def kindsAt (l : List Int) (k : Int) : List Int := l.filter (fun v => v ≥ 0 && v / 4096 == k)
def simP (w : W) : List Int := (sortBy (fun p => p.y * 64 + p.x) w.ps).map encP
def simVisC (w : W) : List Int :=
  let pm := w.ps.foldl (fun m p => m ||| pcMask p) 0
  (sortBy (fun c => c.y * 64 + c.x) (w.cs.filter (fun c => minus (crMask c) pm != 0))).map encC
-- 起手时每只没被选中的会动物件，最近的件唯一（并列时世界挑哪件帧上看不出，见 _lib.mover_target_ambiguous）
def noTie (s : State) (x y : Int) : Bool :=
  let w := initW s x y
  w.cs.all (fun c => c.sel ||
    let ds := w.ps.map (fun p => (pcx p - ccx c) * (pcx p - ccx c) + (pcy p - ccy c) * (pcy p - ccy c))
    let m := ds.foldl min (ds.headD 0)
    (ds.filter (· == m)).length ≤ 1)
-- 起手帧上会动物件一只都没藏：可见「质量」（色7 记 1、色14 记 2、色13 记 4，并不改质量）等于本关常数 k_mass
-- （hidden_hunter_revealed_by_its_bite：藏在件底下的那只帧上看不见，它的并与咬说不出来）。
-- 可见质量只在有物件被件整只盖住时下降，露出或撤销时回升，从不超过 k_mass（语料里 12 次下降、8 次回升：
-- 点击后露出 seq 150、500、502，撤销 seq 188、504、729、731、745）。seq 184、219、797、812、903 等处藏起的物件之后
-- 再没露出，它们在件底下的位置没有任何检验。
def visMass (s : State) : Int :=
  ((cresL s).filterMap decObj).foldl (fun m (k, _, _) => m + (if k == 1 then 1 else if k == 2 then 2 else 4)) 0
def noHide (s : State) : Bool := visMass s == s.k_mass.getD 0
