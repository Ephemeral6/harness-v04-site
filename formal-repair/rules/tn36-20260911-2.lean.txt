def popc : Nat → Nat → Nat

  | f + 1, m => if m == 0 then 0 else m % 2 + popc f (m / 2)

def rpGroups (s : State) : List (Int × Int) :=
  [(kv s.rpanel_geom_0, kv s.rpanel_geom_3), (kv s.rpanel_geom_1, kv s.rpanel_geom_4),

  [kv s.rpanel_geom_6, kv s.rpanel_geom_7, kv s.rpanel_geom_8, kv s.rpanel_geom_9,

def nSlots (s : State) : Nat := (rpCols s).countP (fun c => decide (0 ≤ c))

/-- _lib.panel_slot_at 的行：横条行 ±1 算横条、竖杆那三行算竖杆，后一组覆盖前一组。返回 2·组号 + kind。 -/
def rowHit (r : Int) : List (Int × Int) → Nat → Option Nat → Option Nat
  | [], _, acc => acc
  | (b, st) :: rest, i, acc =>
    let acc1 := if 0 ≤ b && (r - b).natAbs ≤ 1 then some (2 * i) else acc
    let acc2 := if 0 ≤ st && st ≤ r && r ≤ st + 2 then some (2 * i + 1) else acc1
    rowHit r rest (i + 1) acc2

/-- 右下面板命中哪盏灯：位号 6·槽 + 2·组 + kind。 -/
def rHit (s : State) (r c : Int) : Option Nat :=
  match rowHit r (rpGroups s) 0 none, findCol c (rpCols s) 0 with
  | some rb, some k => some (6 * k + rb)
  | _, _ => none

-- ---------- 码本（_lib.OP_CODE + CODE_ALIAS）----------
-- 一槽的码 = 这一竖列暗着的灯：横条 A=1 竖杆 A=2 横条 B=4 竖杆 B=8 横条 C=16 竖杆 C=32。
-- 指令：0=北 1=东 2=南 3=西（与缺口码同序）4=放大 5=缩小 6=旋转 7=六排全暗。
def opOf (code : Nat) : Option Nat :=
  if code == 33 then some 0 else if code == 2 then some 1 else if code == 3 then some 2
  else if code == 34 || code == 1 then some 3 else if code == 8 then some 4 else if code == 9 then some 5
  else if code == 5 then some 6 else if code == 63 then some 7 else none
def slotCode (bits k : Nat) : Nat := (bits >>> (6 * k)) &&& 63
def decodeOps (bits : Nat) : Nat → Nat → Option (List Nat)
  | 0, _ => some []
  | n + 1, k =>
    match opOf (slotCode bits k), decodeOps bits n (k + 1) with
    | some o, some rest => some (o :: rest)
    | _, _ => none
/-- _lib.panel_ops：从左到右逐槽解码；空槽或认不出的码整串读不出。 -/
def panelOps (s : State) : Option (List Nat) :=
  match s.rpanel_bits with
  | some b => decodeOps b.toNat (nSlots s) 0
  | none => none
def isMove (o : Nat) : Bool := o < 4

-- ---------- 件的形态与 run_program（不核墙）----------
structure Pose where
  i : Int
  j : Int
  sc : Nat
  nt : Nat

def poseEq (p q : Pose) : Bool := p.i == q.i && p.j == q.j && p.sc == q.sc && p.nt == q.nt
def dI (o : Nat) : Int := if o == 0 then -1 else if o == 2 then 1 else 0
def dJ (o : Nat) : Int := if o == 1 then 1 else if o == 3 then -1 else 0
/-- 一条指令：平移一格；放大 ×2、缩小 ÷2（左上角不动，奇数倍缩不了）；旋转 南↔东、北↔西（对合，码 3−n）；六排全暗什么都不做。 -/
def stepPose (p : Pose) (o : Nat) : Option Pose :=
  if o < 4 then some { p with i := p.i + dI o, j := p.j + dJ o }
  else if o == 4 then some { p with sc := 2 * p.sc }
  else if o == 5 then (if p.sc % 2 == 1 then none else some { p with sc := p.sc / 2 })
  else if o == 6 then some { p with nt := 3 - p.nt }
  else some p
def runFree : Pose → List Nat → Option Pose
  | p, [] => some p
  | p, o :: rest =>
    match stepPose p o with
    | some q => runFree q rest
    | none => none
def scaleRun : Nat → List Nat → Option Nat
  | sc, [] => some sc
  | sc, o :: rest =>
    if o == 4 then scaleRun (2 * sc) rest
    else if o == 5 then (if sc % 2 == 1 then none else scaleRun (sc / 2) rest)
    else scaleRun sc rest
/-- _lib.program_result 的格数：件 4s×4s 缺 2s×s，14·s² 格。 -/
def progResultN (s : State) : Option Int :=
  match panelOps s, s.piece_scale with
  | some os, some sc => (scaleRun sc.toNat os).map (fun z => 14 * (z : Int) * (z : Int))
  | _, _ => none

  | _, _, _, _ => none

-- ---------- 棋盘：墙、出界、跳过读法 ----------
/-- 左上角在格 (i,j)、倍数 sc 的件盖住的 sc×sc 格里有一格出界或是墙（_lib.piece_footprint + board_walls）。 -/
def blocked (s : State) (sc : Nat) (i j : Int) : Bool :=
  let w := kn s.board_walls_bits
  let nr := kv s.board_geom_2
  let nc := kv s.board_geom_3
  (List.range sc).any (fun di => (List.range sc).any (fun dj =>
    let a := i + (di : Int)
    let b := j + (dj : Int)
    !(0 ≤ a && a < nr && 0 ≤ b && b < nc) || cellOn w a b))

/-- _lib.walk_path_skip：逐条平移，走不通的那一条跳过、后面照走；返回真走到的格（不含起点）。 -/
def walkSkip (s : State) (sc : Nat) : Int × Int → List Nat → List (Int × Int)
  | _, [] => []
  | (i, j), o :: rest =>
    let ni := i + dI o
    let nj := j + dJ o
    if blocked s sc ni nj then walkSkip s sc (i, j) rest else (ni, nj) :: walkSkip s sc (ni, nj) rest
def lastOr (p : Int × Int) : List (Int × Int) → Int × Int
  | [] => p
  | q :: rest => lastOr q rest

/-- _lib.program_trace + piece_footprint_blocks：不跳过、逐条走，有一步件盖住的格压墙或出界返回 some true；
    中途缩不了（奇数倍再缩）返回 none（整串做不完，那一条不管）。 -/
def traceBlocked (s : State) : Pose → List Nat → Option Bool
  | _, [] => some false
  | p, o :: rest =>
    match stepPose p o with
    | none => none
    | some q =>
      match traceBlocked s q rest with
      | none => none
      | some t => some (t || blocked s q.sc q.i q.j)

-- ---------- 记号盘（色 14）----------
def marked (s : State) : Bool := match s.marked_cells_n with | some n => decide (0 < n) | none => false
def markRow (s : State) (i : Int) : Bool := s.marked_cells_0_r == some i || s.marked_cells_1_r == some i
def maxRunAux : Option Nat → Nat → Nat → List Nat → Nat
  | _, _, best, [] => best
  | prev, cur, best, o :: rest =>
    let c := if prev == some o then cur + 1 else 1
    maxRunAux (some o) c (max best c) rest
def maxRun (os : List Nat) : Nat := maxRunAux none 0 0 os
def distinctN (os : List Nat) : Nat := ([0, 1, 2, 3].filter (fun d => os.contains d)).length
def dwell (s : State) (path : List (Int × Int)) : Nat := (path.filter (fun p => markRow s p.1)).length

-- ---------- 一道全平移程序交上去（disc_runs_program 一族）----------
def movesOf (s : State) : Option (List Nat × Pose) :=
  match panelOps s, piecePose s with
  | some os, some p => if 0 < os.length && os.all isMove then some (os, p) else none
  | _, _ => none
def pathOf (s : State) (os : List Nat) (p : Pose) : List (Int × Int) := walkSkip s p.sc (p.i, p.j) os
def endOf (s : State) (os : List Nat) (p : Pose) : Int × Int := lastOr (p.i, p.j) (pathOf s os p)
def isPad (s : State) (e : Int × Int) : Bool := cellOn (kn s.board_pads_bits) e.1 e.2
def endFits (s : State) (os : List Nat) (p : Pose) : Bool :=
  match cavPose s with
  | some cv => let e := endOf s os p; poseEq { p with i := e.1, j := e.2 } cv
  | none => false
/-- 记号盘上连四以上（four_in_a_row_voids）。 -/
def runsVoid (s : State) (os : List Nat) : Bool := marked s && decide (4 ≤ maxRun os)
/-- 记号盘上 marked_prog_class ≠ 'ok'：连四、码少于三种、或路径在记号行落两步以上。 -/
def markTrouble (s : State) (os : List Nat) (p : Pose) : Bool :=
  marked s && (decide (4 ≤ maxRun os) || decide (distinctN os < 3) || decide (1 < dwell s (pathOf s os p)))
/-- few_ops_and_dwell_voids：记号盘、1 倍件、不连四、码少于三种且记号行落两步以上。 -/
def fewDwell (s : State) (os : List Nat) (p : Pose) : Bool :=
  marked s && p.sc == 1 && !runsVoid s os && decide (distinctN os < 3) && decide (1 < dwell s (pathOf s os p))
/-- disc_runs_program 管的那一片：不在记号盘的麻烦区、算出来不入位、终点是花格。返回终点格。 -/
def moveTarget (s : State) : Option (Int × Int) :=
  match movesOf s with
  | some (os, p) =>
    if markTrouble s os p || endFits s os p || !isPad s (endOf s os p) then none else some (endOf s os p)
  | none => none
/-- offpad_end_voids 管的那一片：不连四（记号盘）、不入位、终点不是花格。 -/
def offPad (s : State) : Bool :=
  match movesOf s with
  | some (os, p) => !runsVoid s os && !endFits s os p && !isPad s (endOf s os p)
  | none => false

-- ---------- 带非平移码的程序 ----------
/-- allblank_voids_program：四方向框关卡、面板读得出、有一槽不是平移码（空槽、全暗、或别的码）。 -/
def allblankVoid (s : State) : Bool :=
  s.four_dir_boxes && s.rpanel_bits.isSome && 0 < nSlots s &&
    (match panelOps s with | some os => !os.all isMove | none => true)
def footprintVoid (s : State) : Bool :=
  match panelOps s, piecePose s with
  | some os, some p => 0 < os.length && !os.all isMove && traceBlocked s p os == some true
  | _, _ => false

/-- _lib.submit_fits：不核墙的 run_program 算出来是否恰好填满空腔；读不出件或空腔返回 none。 -/
def submitFits (s : State) : Option Bool :=
  match piecePose s, cavPose s with
  | some p, some cv =>
    some (match panelOps s with
      | some os => (match runFree p os with | some q => poseEq q cv | none => false)
      | none => false)
  | _, _ => none
/-- disc_fit_clears_level._fits：全平移按跳过读法，其余按 submit_fits。 -/
def discFits (s : State) : Option Bool :=
  match submitFits s with
  | none => none
  | some base =>
    match movesOf s with
    | some (os, p) => some (endFits s os p)
    | none => some base
/-- blank_slot_is_nop：五框关卡里把六排全暗的槽去掉、算出来恰好填满空腔。 -/
def nopFits (s : State) : Bool :=
  match panelOps s, piecePose s, cavPose s with
  | some os, some p, some cv =>
    !s.four_dir_boxes && os.contains 7 &&
      (match runFree p (os.filter (fun o => o != 7)) with | some q => poseEq q cv | none => false)
  | _, _, _ => false

-- ---------- 交卷：件动不动、挪到哪 ----------
/-- some none = 整帧不动；some (some e) = 件挪到 e；none = 规则说不出（入位过关查表，或册子闭嘴的那片）。 -/
def discOutcome (s : State) : Option (Option (Int × Int)) :=
  match movesOf s with
  | some (os, p) =>
    if runsVoid s os then some none                              -- four_in_a_row_voids
    else if fewDwell s os p then some none                       -- few_ops_and_dwell_voids
    else if endFits s os p then none                             -- 入位：过关整屏重画，查表
    else if !isPad s (endOf s os p) then some none               -- offpad_end_voids
    else if markTrouble s os p then none                         -- 'unsure' 那一片，册子闭嘴
    else some (some (endOf s os p))                              -- disc_runs_program
  | none =>
    if allblankVoid s then some none                             -- allblank_voids_program
    else if footprintVoid s then some none                       -- piece_footprint_blocks
    else none

/-- 件挪走后：终点那格花格被压住看不见，起点那格露出来——是不是花格看本关花格总数 k_pads 减去看得见的。 -/
def moved (s : State) (e : Int × Int) : Option State :=
  match s.piece_cell_r, s.piece_cell_c, s.k_pads with
  | some i, some j, some kp =>
    let vis := kn s.board_pads_bits
    let under : Int := kp - (popc 64 vis : Int)
    let eb := cellBitOf e.1 e.2
    let vis1 := if vis.testBit (8 * e.1 + e.2).toNat then vis ^^^ eb else vis
    if under == 0 || under == 1 then
      let vis2 := if under == 1 then vis1 ||| cellBitOf i j else vis1
      some { s with piece_cell_r := some e.1, piece_cell_c := some e.2,
                    board_pads_bits := some (vis2 : Int), board_pads_n := some (popc 64 vis2 : Int) }
    else none
  | _, _, _ => none

-- ---------- 顶栏与点灯 ----------
/-- 每个 ACTION6：本关周期 k_pay 为 1 时扣一格；为 2 时本关第偶数个动作（h_acts = 1）扣一格。h_acts 翻一次。 -/
def payOf (s : State) : Int := if s.k_pay == some 1 || s.h_acts == 1 then 1 else 0
def payCounter (s : State) : State :=
  { s with counter_left := s.counter_left - payOf s, counter_head := s.counter_head - payOf s,
           h_acts := (s.h_acts + 1) % 2 }
def withL0 (s : State) (b : Nat) : State :=
  let m := kn s.l0_panel_bits ^^^ bit b
  { s with l0_panel_bits := some (m : Int), panel_all_off := m == 1023 }
def withPanel (s : State) (b : Nat) : State :=
  let s1 := { s with rpanel_bits := some ((kn s.rpanel_bits ^^^ bit b : Nat) : Int) }
  { s1 with panel_ops_n := (panelOps s1).map (fun os => (os.length : Int)), program_result_n := progResultN s1 }

def stepRules (s : State) (a : Act) : Option State :=
  match clickRC a, s.k_pay with
  | some (r, c), some _ =>
    let s1 := payCounter s
    if s.level == 0 then
      match l0Hit r c with
      | some b => some (withL0 s1 b)
      | none => if onDisc s r c && s.panel_all_off then none else some s1
    else
      match rHit s r c with
      | some b => some (withPanel s1 b)
      | none =>
        if onDisc s r c then
          match discOutcome s with
          | some none => some s1
          | some (some e) => moved s1 e
          | none => none
        else none
  | _, _ => none

-- ---------- 规则：一条概念一条 Rule ----------
/-- 除顶栏外整帧不动：关卡、件、花格、面板灯都不变。 -/
def boardSame (s s' : State) : Bool :=
  s'.level == s.level && s'.piece_cell_r == s.piece_cell_r && s'.piece_cell_c == s.piece_cell_c
    && s'.piece_scale == s.piece_scale && s'.piece_notch == s.piece_notch
    && s'.board_pads_bits == s.board_pads_bits && s'.rpanel_bits == s.rpanel_bits

def l0HitA (a : Act) : Option Nat := match clickRC a with | some (r, c) => l0Hit r c | none => none
def rHitA (s : State) (a : Act) : Option Nat := match clickRC a with | some (r, c) => rHit s r c | none => none
def isA6 (a : Act) : Bool := (clickRC a).isSome

def gStem (s : State) (a : Act) : Bool := s.level == 0 && (match l0HitA a with | some b => decide (5 ≤ b) | none => false)
/-- 竖杆那一位翻转，其余 9 盏不动。 -/
def eStem (s : State) (a : Act) (s' : State) : Bool :=
  match l0HitA a with
  | some b => s'.level == 0 && s'.l0_panel_bits == some ((kn s.l0_panel_bits ^^^ bit b : Nat) : Int)
  | none => false

def gTop (s : State) (a : Act) : Bool := s.level == 0 && (match l0HitA a with | some b => decide (b < 5) | none => false)
/-- 横条那一位翻转，其余 9 盏不动。 -/
def eTop (s : State) (a : Act) (s' : State) : Bool :=
  match l0HitA a with
  | some b => s'.level == 0 && kn s'.l0_panel_bits ^^^ kn s.l0_panel_bits == bit b && s'.l0_panel_bits.isSome
  | none => false

def gLevel0Win (s : State) (a : Act) : Bool := s.level == 0 && onDiscA s a && decide (1 ≤ s.counter_head)
/-- 面板全暗：过关、顶栏满 61；否则只掉一格。 -/
def eLevel0Win (s : State) (_a : Act) (s' : State) : Bool :=
  if s.panel_all_off then s'.level == 1 && s'.counter_left == FULL
  else s'.level == 0 && s'.counter_left == s.counter_left - 1

def gPanelToggle (s : State) (a : Act) : Bool := s.level != 0 && s.rpanel_bits.isSome && (rHitA s a).isSome
/-- 命中的那盏灯翻转、别的灯与棋盘不动。 -/
def ePanelToggle (s : State) (a : Act) (s' : State) : Bool :=
  match rHitA s a with
  | some b => s'.level == s.level && s'.rpanel_bits == some ((kn s.rpanel_bits ^^^ bit b : Nat) : Int)
      && s'.piece_cell_r == s.piece_cell_r && s'.piece_cell_c == s.piece_cell_c && s'.board_pads_bits == s.board_pads_bits
  | none => false

def gCounterSpends (s : State) (a : Act) : Bool := isA6 a && !onDiscA s a && decide (1 ≤ s.counter_head)
/-- 同一关之内顶栏最多掉最右端那一格。 -/
def eCounterSpends (s : State) (_a : Act) (s' : State) : Bool :=
  if s'.level != s.level then true
  else (s.counter_left - s'.counter_left == 0 || s.counter_left - s'.counter_left == 1)
    && s.counter_head - s'.counter_head == s.counter_left - s'.counter_left

def gPaysEveryOther (s : State) (a : Act) : Bool := isA6 a && s.k_pay.isSome && decide (1 ≤ s.counter_head)
/-- 本关周期 k_pay：为 2 时本关第 1、3、5… 个动作不扣、第 2、4、6… 个扣一格；为 1 时每个都扣。换关那一步不管。 -/
def ePaysEveryOther (s : State) (_a : Act) (s' : State) : Bool :=
  if s'.level != s.level then true
  else s'.counter_left == s.counter_left - payOf s && s'.h_acts == (s.h_acts + 1) % 2

def gClearRefills (s : State) (a : Act) : Bool := onDiscA s a
/-- 真换关（不是通关）那一步，顶栏刷回满 61 格。 -/
def eClearRefills (s : State) (_a : Act) (s' : State) : Bool :=
  if s'.level != s.level && s'.level < 7 then s'.counter_left == FULL && s'.counter_head == FULL else true

def gDiscFit (s : State) (a : Act) : Bool := onDiscA s a && discFits s == some false
/-- 算出来填不满空腔就不过关。 -/
def eDiscFit (s : State) (_a : Act) (s' : State) : Bool := s'.level == s.level

def gRunsProgram (s : State) (a : Act) : Bool := onDiscA s a && (moveTarget s).isSome
/-- 件挪到跳过读法算出的终点格；终点那格花格被压住；起点那格露出来的是底色或一格花格，别的花格不变。 -/
def eRunsProgram (s : State) (_a : Act) (s' : State) : Bool :=
  match moveTarget s, s.piece_cell_r, s.piece_cell_c with
  | some e, some i, some j =>
    let vis := kn s.board_pads_bits
    let vis1 := if vis.testBit (8 * e.1 + e.2).toNat then vis ^^^ cellBitOf e.1 e.2 else vis
    s'.level == s.level && s'.piece_cell_r == some e.1 && s'.piece_cell_c == some e.2
      && (kn s'.board_pads_bits ||| cellBitOf i j) == (vis1 ||| cellBitOf i j)
  | _, _, _ => false

def gOffpad (s : State) (a : Act) : Bool := onDiscA s a && offPad s
def eOffpad (s : State) (_a : Act) (s' : State) : Bool := boardSame s s'

def gFourInRow (s : State) (a : Act) : Bool :=
  onDiscA s a && (match movesOf s with | some (os, _) => runsVoid s os | none => false)
def eFourInRow (s : State) (_a : Act) (s' : State) : Bool := boardSame s s'

def gFewDwell (s : State) (a : Act) : Bool :=
  onDiscA s a && (match movesOf s with | some (os, p) => fewDwell s os p | none => false)
def eFewDwell (s : State) (_a : Act) (s' : State) : Bool := boardSame s s'

def gAllblank (s : State) (a : Act) : Bool := onDiscA s a && allblankVoid s
def eAllblank (s : State) (_a : Act) (s' : State) : Bool := boardSame s s'

def gFootprint (s : State) (a : Act) : Bool := onDiscA s a && footprintVoid s
def eFootprint (s : State) (_a : Act) (s' : State) : Bool := boardSame s s'

def gBlankNop (s : State) (a : Act) : Bool := onDiscA s a && nopFits s
/-- 去掉全暗槽后正好填满，这一下就过关。 -/
def eBlankNop (s : State) (_a : Act) (s' : State) : Bool := s'.level == s.level + 1

def gEvenSlots (s : State) (a : Act) : Bool :=
  isA6 a && s.four_dir_boxes && s.rpanel_bits.isSome && nSlots s % 2 == 0 && (piecePose s).isSome
/-- 同一块盘上（关卡没换），件所在格「行+列」的奇偶不变。 -/
def eEvenSlots (s : State) (_a : Act) (s' : State) : Bool :=
  if s'.level != s.level then true
  else
    match s.piece_cell_r, s.piece_cell_c, s'.piece_cell_r, s'.piece_cell_c with
    | some a, some b, some c, some d => (a + b) % 2 == (c + d) % 2
    | _, _, _, _ => false

def gBoardFrozen (s : State) (a : Act) : Bool := s.level == 0 && !(onDiscA s a && s.panel_all_off)
/-- 第 0 关之内（「全暗时点大圆」那一下除外）棋盘区一格不变：比的是那块像素的粗指纹 l0_board_sum。 -/
def eBoardFrozen (s : State) (_a : Act) (s' : State) : Bool :=
  s'.level == 0 && s'.l0_board_sum.isSome && s'.l0_board_sum == s.l0_board_sum

def gResetRefills (s : State) (a : Act) : Bool := a == Act.reset && (s.level == 0 || s.level == 7)
/-- RESET（第 0 关之内，或通关之后）：回到第 0 关，顶栏 61 格刷满，面板复原成开局态 暗亮暗亮亮
    （横条与竖杆同色：第 1、3 槽暗 -> 位 0、2、5、7 = 165），不是全暗。 -/
def eResetRefills (_s : State) (_a : Act) (s' : State) : Bool :=
  s'.level == 0 && s'.counter_left == FULL && s'.counter_head == FULL
    && s'.l0_panel_bits == some 165 && !s'.panel_all_off

def rules : List Rule :=
  [ { name := "stem_click_toggles",       guard := gStem,           effect := eStem }
  , { name := "top_click_toggles",        guard := gTop,            effect := eTop }
  , { name := "level0_win",               guard := gLevel0Win,      effect := eLevel0Win }
  , { name := "panel_click_toggles",      guard := gPanelToggle,    effect := ePanelToggle }
  , { name := "counter_spends",           guard := gCounterSpends,  effect := eCounterSpends }
  , { name := "counter_pays_every_other", guard := gPaysEveryOther, effect := ePaysEveryOther }
  , { name := "clear_refills_counter",    guard := gClearRefills,   effect := eClearRefills }
  , { name := "disc_fit_clears_level",    guard := gDiscFit,        effect := eDiscFit }
  , { name := "disc_runs_program",        guard := gRunsProgram,    effect := eRunsProgram }
  , { name := "offpad_end_voids",         guard := gOffpad,         effect := eOffpad }
  , { name := "four_in_a_row_voids",      guard := gFourInRow,      effect := eFourInRow }
  , { name := "few_ops_and_dwell_voids",  guard := gFewDwell,       effect := eFewDwell }
  , { name := "allblank_voids_program",   guard := gAllblank,       effect := eAllblank }
  , { name := "piece_footprint_blocks",   guard := gFootprint,      effect := eFootprint }
  , { name := "blank_slot_is_nop",        guard := gBlankNop,       effect := eBlankNop }
  , { name := "even_slots_keep_parity",   guard := gEvenSlots,      effect := eEvenSlots }
  , { name := "board_frozen",             guard := gBoardFrozen,    effect := eBoardFrozen }
  , { name := "reset_refills",            guard := gResetRefills,   effect := eResetRefills } ]

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

/-- 第 6 关（第七块盘）过关那一下关卡号到 7、状态 WIN；此后只有 RESET。 -/
def isGoal (s : State) : Bool := s.level == 7
