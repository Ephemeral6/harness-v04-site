structure Lay where
  sol : Nat
  plg : Nat
  head : Nat
def Lay.all (l : Lay) : Nat := l.sol ||| l.plg ||| l.head
def Lay.shift (l : Lay) (d : Int) : Lay := ⟨shiftI l.sol d, shiftI l.plg d, shiftI l.head d⟩

/-- 画法（dock_weld / overlap_draws_over / swap_selection / swap_sleep 的共同画法）：
    静置层：选中色画法下各件显真色、插头显 8、两个插头重合显 3；沉睡画法下静置件连插头通体显 4。
    上层件盖在最上面：实心显它的画色（选中色画法 0，沉睡画法真色），笔头显 0，插头落在静置插头上显 3、否则显 8。 -/
def render (sl : Bool) (st : List Pc) (act : Option (Lay × Nat)) : Frame :=
  let oc := plugCount (st.map (·.plg))
  let base : Frame :=
    if sl then Frame.zero.orC 4 (st.foldl (fun m p => m ||| p.sol ||| p.plg) 0)
    else
      let f := st.foldl (fun f p => f.orC p.col (andNot p.sol oc.1)) Frame.zero
      (f.orC 8 (andNot oc.1 oc.2)).orC 3 oc.2
  match act with
  | none => base
  | some (l, col) =>
    let f := base.clear l.all
    let f := f.orC (if sl then col else 0) l.sol
    let f := f.orC 0 l.head
    let f := f.orC 3 (l.plg &&& oc.1)
    f.orC 8 (andNot l.plg oc.1)

/-- 上层件：帧上与静置层画法不一致的格。选中色画法下真色记在 h_sel（-1 = 开关时那个，见 k_sel）。 -/
def activeOf (s : State) : Option (Lay × Nat) :=
  let sl := sleepy s
  let v := visFrame s
  let d := Frame.xor v (render sl (statics s) none) &&& nat s.cells_m
  if d == 0 then none else
  let plg := d &&& (v.c8 ||| v.c3)
  let head := if sl then d &&& v.c0 else 0
  let sol := andNot d (plg ||| head)
  let col := if sl then (PC.find? (fun c => (v.get c &&& sol) != 0)).getD 0
             else if 0 ≤ s.h_sel then nat s.h_sel else optN s.k_sel
  some (⟨sol, plg, head⟩, col)

-- ---- 几何 ----
def colProj : Nat → Nat → Nat
  | 0, _ => 0
  | f+1, m => if m == 0 then 0 else (m &&& 1048575) ||| colProj f (m >>> 32)
/-- 外框：(最上行, 最左列, 最下行) -/
def bbox (m : Nat) : Nat × Nat × Nat :=
  (Nat.log2 (lowBit m) / 32, Nat.log2 (lowBit (colProj 20 m)), Nat.log2 m / 32)

def rotBits (r0 c0 h : Nat) : Nat → Nat → Nat
  | 0, _ => 0
  | f+1, m =>
    if m == 0 then 0 else
    let i := Nat.log2 (lowBit m)
    let r := i / 32
    let c := i % 32
    bitAt ((r0 + (c - c0)) * 32 + (c0 + (h - 1) - (r - r0))) ||| rotBits r0 c0 h f (m &&& (m - 1))
/-- rot_cw：绕整件（实心 + 插头）外框左上角顺时针转 90°，左上角不动。 -/
def Lay.rot (l : Lay) : Lay :=
  let b := bbox l.all
  let h := b.2.2 + 1 - b.1
  ⟨rotBits b.1 b.2.1 h 400 l.sol, rotBits b.1 b.2.1 h 400 l.plg, rotBits b.1 b.2.1 h 400 l.head⟩

-- ---- 笔 / 伸缩件的节（节表是 k_a<色>，见 _lib.LEVEL_CONST 的编码说明）----
def addCode (d : Nat) (l : Lay) : Lay :=
  let k := d / 1024
  let b := bitAt (d % 1024)
  if k == 1 then { l with sol := l.sol ||| b } else if k == 2 then { l with plg := l.plg ||| b } else { l with head := l.head ||| b }
def stageGo : Nat → Nat → Nat → Lay → Lay
  | 0, _, _, acc => acc
  | f+1, tab, i, acc =>
    if tab == 0 then acc else
    let d := tab % 4096
    let rest := tab / 4096
    if d == 0 then (if i == 0 then acc else stageGo f rest (i - 1) acc)
    else if i == 0 then stageGo f rest 0 (addCode d acc) else stageGo f rest i acc
def stageLay (tab i : Nat) : Lay := stageGo 400 tab i ⟨0, 0, 0⟩
/-- 按了 n 次之后在第几节：走到最后一节就折返（三角波）。 -/
def tri (n L : Nat) : Nat := if L == 0 then 0 else if n % (2 * L) ≤ L then n % (2 * L) else 2 * L - n % (2 * L)
/-- 把格集 src 平移到外框左上角与 dst 对齐所需的位移（位下标差）。 -/
def offTo (src dst : Nat) : Int :=
  let a := bbox dst
  let b := bbox src
  ((a.1 : Int) - b.1) * 32 + ((a.2.1 : Int) - b.2.1)
def isAnim (s : State) (c : Nat) : Bool := sleepy s && kL s c > 0
/-- stepRules 用的笔 / 伸缩件前进：第 n 次按 ACTION5 之后该件的格局 = 节表 k_a 的第 tri(n+1) 节，放在它当前所在的位置。
    **这是查表**（节表逐字是动作后果，取自帧），只给 faithful 重放用；pen_step 规则不引用它。 -/
def penNext (s : State) (l : Lay) (col : Nat) : Lay :=
  let n := hN s col
  let L := kL s col
  let a0 := stageLay (kA s col) (tri n L)
  (stageLay (kA s col) (tri (n + 1) L)).shift (offTo a0.all l.all)

-- ---- 点选 ----
def clickBit (x y : Nat) : Nat :=
  if x < 2 || y < 2 then 0 else
  if (y - 2) / 3 < 20 && (x - 2) / 3 < 20 then bitAt ((y - 2) / 3 * 32 + (x - 2) / 3) else 0
/-- 点中的静置件：先看实心，再看插头；几件都中取色号最小的（语料只有 seq 425 一例，那一格是两件插头重合）。 -/
def hitCol (st : List Pc) (b : Nat) : Option Pc :=
  match st.filter (fun p => (p.sol &&& b) != 0) with
  | p :: _ => some p
  | [] => (st.filter (fun p => (p.plg &&& b) != 0)).head?
/-- 被点醒（或被选中）的件回到上层时的格局：普通件就是它静置时的格局；笔 / 伸缩件按它的节数画。 -/
def wakeLay (s : State) (q : Pc) : Lay :=
  if isAnim s q.col then
    let a0 := stageLay (kA s q.col) (tri (hN s q.col) (kL s q.col))
    a0.shift (offTo a0.all (q.sol ||| q.plg))
  else ⟨q.sol, q.plg, 0⟩

def moveD : Act → Option Int

/-- 换人：原上层件回静置层（记下它此刻的格局），被点中的件 q 上到最上层。 -/
def switchTo (s0 : State) (st : List Pc) (ac : Option (Lay × Nat)) (q : Pc) : Out :=
  let p1 : List Pc × State := match ac with
    | some (l, col) => (upd st col (l.sol ||| l.head) l.plg, recordH s0 col (l.sol ||| l.head) l.plg)
    | none => (st, s0)
  let s2 := recordH p1.2 q.col 0 0
  let s3 := if sleepy s0 then s2 else { s2 with h_sel := (q.col : Int) }
  ⟨upd p1.1 q.col 0 0, some (wakeLay s0 q, q.col), s3⟩

def predict (s : State) (a : Act) : Option Out :=
  let st := statics s
  let ac := activeOf s
  let s0 := { s with h_steps := s.h_steps + 1 }
  match a with
  | .a6 x y =>
    let b := clickBit x y
    let self : Bool := match ac with | some (l, _) => b != 0 && (l.all &&& b) != 0 | none => false
    if self then
      match ac with
      | some (l, col) =>
        if sleepy s then some ⟨upd st col (l.sol ||| l.head) l.plg, none, recordH s0 col (l.sol ||| l.head) l.plg⟩
        else none
      | none => none
    else
      match hitCol st b with
      | some q => some (switchTo s0 st ac q)
      | none => some ⟨st, ac, s0⟩
  | .a5 =>
    match ac with
    | none => some ⟨st, none, s0⟩
    | some (l, col) =>
      let s1 := if sleepy s then incN s0 col else s0
      if isAnim s col then some ⟨st, some (penNext s l col, col), s1⟩
      else some ⟨st, some (l.rot, col), s1⟩
  | a' =>
    match moveD a' with
    | some d => some ⟨st, ac.map (fun p => (p.1.shift d, p.2)), s0⟩
    | none => none

  let oc := plugCount ((nb m).map (· &&& m))
  (m &&& andNot oc.1 oc.2, m &&& oc.2)

/-- move_*：可动件整体平移一格，别的件不动，按画法重画。 -/
def moveClaim (d : Int) (s s' : State) : Bool :=
  match activeOf s with
  | some (l, col) => s'.level == s.level && frameIs (render (sleepy s) (statics s) (some (l.shift d, col))) s'
  | none => false
def moveGuard (a0 : Act) (s : State) (a : Act) : Bool := a == a0 && hasAct s && inLevel s a

def ruleMoveUp : Rule := { name := "move_up", guard := moveGuard .a1, effect := fun s _ s' => moveClaim (-32) s s' }
def ruleMoveDown : Rule := { name := "move_down", guard := moveGuard .a2, effect := fun s _ s' => moveClaim 32 s s' }
def ruleMoveLeft : Rule := { name := "move_left", guard := moveGuard .a3, effect := fun s _ s' => moveClaim (-1) s s' }
def ruleMoveRight : Rule := { name := "move_right", guard := moveGuard .a4, effect := fun s _ s' => moveClaim 1 s s' }

/-- no_block：ACTION1-4 在有可动件时一定改变帧。 -/
def ruleNoBlock : Rule :=
  { name := "no_block", guard := fun s a => isMove a && hasAct s && inLevel s a,
    effect := fun s _ s' => Frame.xor (visFrame s) (visFrame s') != 0 }

/-- overlap_draws_over：平移后可动件每个实心块的落点都显它的画色，不管底下原来是什么。 -/
def ruleOverlap : Rule :=
  { name := "overlap_draws_over", guard := fun s a => isMove a && hasAct s && inLevel s a,
    effect := fun s a s' =>
      match activeOf s, moveD a with
      | some (l, col), some d => sub (shiftI l.sol d) (vis s' (drawCol s col))
      | _, _ => false }

/-- rot_cw：普通件按 ACTION5 绕外框左上角顺时针转 90°，外框左上角不变，别的件不动。 -/
def ruleRot : Rule :=
  { name := "rot_cw",
    guard := fun s a => a == .a5 && inLevel s a && (match activeOf s with | some (_, c) => !isAnim s c | none => false),
    effect := fun s _ s' =>
      match activeOf s with
      | some (l, col) =>
        let r := l.rot
        (bbox r.all).1 == (bbox l.all).1 && (bbox r.all).2.1 == (bbox l.all).2.1 &&
        frameIs (render (sleepy s) (statics s) (some (r, col))) s'
      | none => false }

/-- 从种子 seed 出发、在 region 里 4-连通地长满（种子本身总算在内）。 -/
def grow (region : Nat) : Nat → Nat → Nat
  | 0, m => m
  | f+1, m =>
    let m' := (m ||| ((nb m).foldl (· ||| ·) 0 &&& region))
    if m' == m then m else grow region f m'
/-- 帧上「非底色、非 4」的格：笔所在的那一团只在这里面长。 -/
def awakeRegion (t : State) : Nat := andNot (nat t.cells_m) (vis t 4)
/-- 笔头那一团：从笔头出发在 awakeRegion 里长满。 -/
def penComp (t : State) (h : Nat) : Nat := grow (awakeRegion t) 40 h
/-- 那一团里（除笔头、插头外）唯一的实心色。 -/
def penCol (s : State) (comp h : Nat) : Option Nat :=
  let body := andNot comp (h ||| vis s 8 ||| vis s 3)
  match PC.filter (fun c => (vis s c &&& body) != 0) with
  | [c] => some c
  | _ => none

/-- pen_step（概念原文的结构断言，不用节表）：prev 帧上恰有一个笔头（色 0 的格）时按 ACTION5——
    ① 原笔头那格下一帧显该件实心色或底色；② 全帧至多一个笔头；
    ③ 新笔头若在、原笔头那格又没变底色，两者在下一帧属于同一团；④ 该件的格数一定改变；
    ⑤ 两团之外的格一格不变；另有记账：该件按键计数加一。**新笔头落在哪、画出什么，本条不说。** -/
def rulePen : Rule :=
  { name := "pen_step",
    guard := fun s a => a == .a5 && popc 400 (vis s 0) == 1 &&
      (penCol s (penComp s (vis s 0)) (vis s 0)).isSome,
    effect := fun s _ s' =>
      let h := vis s 0
      let comp := penComp s h
      match penCol s comp h with
      | some col =>
        let nh := vis s' 0
        let hBg := (nat s'.cells_m &&& h) == 0
        let ncomp := if popc 400 nh == 1 then penComp s' nh else penComp s' h
        s'.level == s.level &&
        (hBg || (vis s' col &&& h) != 0) &&
        popc 400 nh ≤ 1 &&
        (popc 400 nh != 1 || hBg || (sub h ncomp && sub nh ncomp)) &&
        popc 400 ncomp != popc 400 comp &&
        sub (Frame.xor (visFrame s) (visFrame s')) (comp ||| ncomp) &&
        hN s' col == hN s col + 1
      | none => false }

def clickHit (s : State) (a : Act) : Option Pc :=
  match a with
  | .a6 x y =>
    let b := clickBit x y
    let self := match activeOf s with | some (l, _) => (l.all &&& b) != 0 | none => false
    if b == 0 || self then none else hitCol (statics s) b
  | _ => none

/-- select_click：选中色画法下点中一个件的实心，它整件变成色 0（插头照旧），原来的色 0 件露出真色，谁都不动。 -/
def ruleSelect : Rule :=
  { name := "select_click",
    guard := fun s a => !sleepy s && hasAct s && (clickHit s a).isSome,
    effect := fun s a s' =>
      match clickHit s a, activeOf s with
      | some q, some (l, col) =>
        let st := upd (upd (statics s) col (l.sol ||| l.head) l.plg) q.col 0 0
        vis s' 0 == q.sol &&
        frameIs (render false st (some (⟨q.sol, q.plg, 0⟩, q.col))) s'
      | _, _ => false }

/-- swap_selection：点选是身份交换——非底色格集一格不变，变了颜色的格全在这两件上。 -/
def ruleSwapSel : Rule :=
  { name := "swap_selection",
    guard := fun s a => !sleepy s && hasAct s && (clickHit s a).isSome,
    effect := fun s a s' =>
      match clickHit s a, activeOf s with
      | some q, some (l, _) =>
        s'.cells_m == s.cells_m && sub (Frame.xor (visFrame s) (visFrame s')) (l.all ||| q.sol ||| q.plg)
      | _, _ => false }

/-- sleep_wake：沉睡画法下点中沉睡件：非底色格集一格不变，被点的那格不再是 4。 -/
def ruleSleepWake : Rule :=
  { name := "sleep_wake",
    guard := fun s a => sleepy s && (clickHit s a).isSome,
    effect := fun s a s' =>
      match a with
      | .a6 x y => s'.cells_m == s.cells_m && (vis s' 4 &&& clickBit x y) == 0
      | _ => false }

/-- 沉睡画法下点在可动件自己身上。 -/
def clickSelf (s : State) (a : Act) : Bool :=
  match a, activeOf s with
  | .a6 x y, some (l, _) => clickBit x y != 0 && (l.all &&& clickBit x y) != 0
  | _, _ => false

/-- swap_sleep：被点中的件整件醒来（真色 + 插头），原来醒着的件整件沉成 4，两件一格没挪，别的格子不变。
    **扩展分支**（原概念对此返回 NA，因为点完没有醒着的件）：点在醒着的件自己身上时，只剩后半句——
    它整件沉成 4、一格不挪，别的格子不变，场上不再有醒着的件（seq 101、107、470 等）。 -/
def ruleSwapSleep : Rule :=
  { name := "swap_sleep",
    guard := fun s a => sleepy s && ((clickHit s a).isSome || clickSelf s a),
    effect := fun s a s' =>
      let st1 := match activeOf s with
        | some (l, col) => upd (statics s) col (l.sol ||| l.head) l.plg
        | none => statics s
      match clickHit s a with
      | some q => frameIs (render true (upd st1 q.col 0 0) (some (wakeLay s q, q.col))) s'
      | none => frameIs (render true st1 none) s' && (activeOf s').isNone }

/-- wake_plugs：点醒的件醒来后显 8 的自由插头，全都落在它自己格集里 4-邻接度为 1 的格上。 -/
def ruleWakePlugs : Rule :=
  { name := "wake_plugs",
    guard := fun s a => sleepy s && (clickHit s a).isSome,
    effect := fun s a s' =>
      match clickHit s a with
      | some q => let w := (wakeLay s q).all; sub (vis s' 8 &&& w) (degs w).1
      | none => false }

/-- plug_at_deg1：醒着的件恰有 2 个度 1 格、也恰有 2 个可见插头时，插头正好在那两格上。 -/
def rulePlugDeg1 : Rule :=
  { name := "plug_at_deg1",
    guard := fun s _ =>
      (match activeOf s with
       | some (l, _) => popc 400 (degs l.all).1 == 2 && popc 400 l.plg == 2
       | none => false),
    effect := fun s _ _ =>
      match activeOf s with
      | some (l, _) => l.plg == (degs l.all).1
      | none => false }

/-- plug_near_sleeper：平移后插头落在贴着沉睡件（没压上）、又不是自己上一帧占着的格上，那一格显 8。 -/
def nearSleeper (s : State) (a : Act) : Nat :=
  match activeOf s, moveD a with
  | some (l, _), some d =>
    let sc := sleeperCells s
    let adj := (nb sc).foldl (· ||| ·) 0
    andNot (andNot (shiftI l.plg d) sc) l.all &&& adj
  | _, _ => 0
def rulePlugNear : Rule :=
  { name := "plug_near_sleeper",
    guard := fun s a => sleepy s && isMove a && inLevel s a && nearSleeper s a != 0,
    effect := fun s a s' => sub (nearSleeper s a) (vis s' 8) }

/-- sleep_overlay：平移后落在沉睡件格上的实心显可动件真色；落在沉睡件度 ≥ 2 的格上的插头显 8。 -/
def ruleSleepOverlay : Rule :=
  { name := "sleep_overlay",
    guard := fun s a => sleepy s && isMove a && inLevel s a &&
      (match activeOf s, moveD a with
       | some (l, _), some d => (shiftI l.all d &&& sleeperCells s) != 0
       | _, _ => false),
    effect := fun s a s' =>
      match activeOf s, moveD a with
      | some (l, col), some d =>
        let sc := sleeperCells s
        let n := l.shift d
        sub (n.sol &&& sc) (vis s' col) && sub (n.plg &&& (degs sc).2) (vis s' 8)
      | _, _ => false }

/-- uncover：沉睡画法下可动件平移让出的格，下一帧只显底色或 4。 -/
def ruleUncover : Rule :=
  { name := "uncover",
    guard := fun s a => sleepy s && isMove a && hasAct s && inLevel s a,
    effect := fun s a s' =>
      match activeOf s, moveD a with
      | some (l, _), some d => sub (andNot l.all (shiftI l.all d) &&& nat s'.cells_m) (vis s' 4)
      | _, _ => false }

/-- dock_weld：插头只重合不消耗——ACTION1–5 前后，帧上 8 的个数加 3 的个数的两倍相等。
    按原概念的 NA 条件闭嘴：换关；不是 ACTION1–5；没有可动件；笔 / 伸缩件的 ACTION5（它造节撤节）；
    可动件前后任一帧有插头被它的实心盖住（看得见的插头少了，世界里的没少）；
    沉睡画法下可动件前后任一帧碰到沉睡件的格（sleeper_touch：隐藏插头会显形成 3）；
    prev 帧上有贴着沉睡件的焊格（_weld_with_hidden：撕开后隐藏插头回到不可见）。
    （主会话按复审补注）与原概念并不完全相同：① 「前后任一帧」查 touchesSleeper / hiddenPlug 比原文（只查落位 new）更严，
    使原概念判 TRUE 的 seq 581、582、640 三行漏判（3/354）；② 原概念另两个 NA——插头归属含糊（mover_plug_ambiguous）与让出格贴沉睡件——
    Lean 没写、照判，多判 60 行（21 + 39），语料上 0 失败，是多说不是少说。 -/
def plugTotal (t : State) : Nat := popc 400 (vis t 8) + 2 * popc 400 (vis t 3)
def hiddenPlug (s : State) : Bool :=
  match activeOf s with
  | some (l, _) => ((plugCount ((statics s).map (·.plg))).1 &&& (l.sol ||| l.head)) != 0
  | none => false
def touchesSleeper (s : State) : Bool :=
  sleepy s && (match activeOf s with | some (l, _) => (l.all &&& sleeperCells s) != 0 | none => false)
def weldNearSleeper (s : State) : Bool :=
  sleepy s && (vis s 3 &&& (nb (vis s 4)).foldl (· ||| ·) 0) != 0
def isKey (a : Act) : Bool := isMove a || a == .a5
def ruleDockWeld : Rule :=
  { name := "dock_weld",
    guard := fun s a => isKey a && hasAct s && inLevel s a &&
      !(a == .a5 && (match activeOf s with | some (_, c) => isAnim s c | none => false)) &&
      !hiddenPlug s && !touchesSleeper s && !weldNearSleeper s &&
      (match stepRules s a with | some t => !hiddenPlug t && !touchesSleeper t | none => false),
    effect := fun s _ s' => plugTotal s' == plugTotal s }

/-- goal_dock_wins：这一步之后全场没有落单插头，就换关。 -/
def ruleGoalWins : Rule :=
  { name := "goal_dock_wins",
    guard := fun s a => match predict s a with | some o => paired o | none => false,
    effect := fun s _ s' => s'.level == s.level + 1 }

/-- goal_needs_dock：这一步之后还有落单插头，就绝不换关。 -/
def ruleGoalNeeds : Rule :=
  { name := "goal_needs_dock",
    guard := fun s a => inLevel s a,
    effect := fun s _ s' => s'.level == s.level }

/-- hud_rate：关内每步顶栏花格 = floor(步数·32/budget + 1/2)（所以每步至多多花一格）；
    换关那步顶栏续满（花格 0），但通关整局那一步照普通步走。
    **只验了花格计数**：读数 hud_spent 只存色 0 的个数，「0 连续在左、4 在右」这个排布结构本条不验；
    花满 32 格（溢出）的分支语料里没走到过。 -/
def ruleHud : Rule :=
  { name := "hud_rate",
    guard := fun s a => (predict s a).isSome,
    effect := fun s _ s' =>
      let n := spent (nat s.h_steps + 1) (optN s.k_budget)
      if s'.level == s.level then s'.hud_spent == (n : Int) && s'.h_steps == s.h_steps + 1
      else if s'.level < 6 then s'.hud_spent == 0
      else s'.hud_spent == (n : Int) }

def rules : List Rule :=

   ruleSleepOverlay, ruleUncover, ruleDockWeld, ruleGoalWins, ruleGoalNeeds, ruleHud]
