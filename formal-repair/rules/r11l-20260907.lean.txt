def digs : Nat → Int → List Int
  | 0, _ => []
  | n + 1, x => x % 16 :: digs n (x / 16)

  | [] => 0
  | d :: ds => d + 16 * undigs ds

def selOf (s : State) : Int × Int := (s.selected_r, s.selected_c)
def hits (gs : List Grp) (p : Int × Int) : List (Int × Int) :=

def centroid (cs : List (Int × Int)) : Int × Int :=
  let n : Int := cs.length
  (cs.foldl (fun a q => a + q.1) 0 / n, cs.foldl (fun a q => a + q.2) 0 / n)

-- 选中那颗挪到 p 之后它那一组：锚点换掉，罐子中心 = 新锚点的下取整质心（m_is_centroid）
def movedGroup (g : Grp) (sel p : Int × Int) : Grp :=
  let an := g.an.map (fun x => if tcell x == sel then (p.1, p.2, x.2.2) else x)
  let m := centroid (an.map tcell)
  { mr := m.1, mc := m.2, ink := g.ink, an := an }

-- 点在空处的一击：(第几组, 挪之后的这一组)。点在菱形上、或选中那颗不在任何组里，给 none。
def movePlan (s : State) (p : Int × Int) : Option (Nat × Grp) :=
  let gs := groupsOf s
  if !(hits gs p).isEmpty then none
  else match gs.findIdx? (fun g => g.cells.contains (selOf s)) with
    | none => none
    | some i => some (i, movedGroup (gs.getD i ⟨0, 0, 0, []⟩) (selOf s) p)

-- 菱形 13 格踩硬墙（色2）、或罐子 21 格踩硬墙或软地（色10）⇒ 驳回（move_rejected_on_blocked_terrain）
-- 硬墙读自这一帧（walls_m），软地是这一关的常数（k_soft，见 _lib 的 LEVEL_CONST）
def wallBlocked (s : State) (p : Int × Int) (g : Grp) : Bool :=
  (shift p diamondOffs).any (wallAt s.walls_m) ||
    (shift g.mid discOffs).any (fun q => wallAt s.walls_m q || wallAt s.k_soft q)

def blobAt (s : State) (c : Int × Int) : Option (Int × Int × Int) := (blobsOf s).find? (fun b => tcell b == c)

def onLocked (rs : List (Int × Int × Int)) (g : Grp) : Bool := rs.any (fun t => tcell t == g.mid && t.2.2 == 0)
def allLocked (gs : List Grp) (rs : List (Int × Int × Int)) : Bool := !gs.isEmpty && gs.all (onLocked rs)

def moveStep (s : State) (p : Int × Int) : Option State :=
  match movePlan s p with
  | none => none
  | some (i, g) =>
    if wallBlocked s p g then some s
    else
      let g2 : Grp := match blobAt s g.mid with
        | some b => { g with ink := overwrite g.ink b.2.2 }
        | none => g
      let bs := (blobsOf s).filter (fun b => tcell b != g.mid)
      let rs := (retsOf s).map (fun t =>
        if tcell t == g2.mid && t.2.2 == colorSet g2.ink then (t.1, t.2.1, 0) else t)
      let gs := (groupsOf s).set i g2
      let s2 := withGroups (withBlobs (withRets s (encTriples rs 8)) (encTriples bs 9)) (encGroups gs)
      some { s2 with selected_r := p.1, selected_c := p.2,
                     level := if allLocked gs rs then s.level + 1 else s.level }

def stepRules (s : State) (a : Act) : Option State :=
  match clickRC a, barNext s.bar_rows_n with
  | some p, some b =>
    let s1 := { s with bar_rows_n := b }
    match hits (groupsOf s) p with
    | [] => moveStep s1 p
    | [q] => if q == selOf s then none else some { s1 with selected_r := q.1, selected_c := q.2 }
    | _ => none
  | _, _ => none

  | none => false

      (((groupsOf s).flatMap Grp.cells).filter (fun q => q != selOf s)).all
        (fun q => ((groupsOf t).flatMap Grp.cells).contains q) },
  -- 每个罐子的中心 = 本组菱形中心的下取整质心（罐子中心是从像素读的，组是虚线认的）
  { name := "m_is_centroid"
    guard := fun _ a => isA6 a
    effect := fun _ _ t => (groupsOf t).all (fun g => g.mid == centroid g.cells) },
  -- 没取色的一击：罐子环上的墨跟着平移、不变形（各组的墨原样）
  { name := "circle_ring_translates"
    guard := fun s a => moveOpen s a && (match planMid s a with | some m => (blobAt s m).isNone | none => false)
    effect := fun s _ t => t.level != s.level || inkList t == inkList s },
  -- 罐子中心落到漆块中心上：漆块的墨盖到罐子上，漆块消失
  { name := "circle_absorbs_blob_ink"
    guard := fun s a => moveOpen s a && (match planMid s a with | some m => (blobAt s m).isSome | none => false)
    effect := fun s a t => match clickRC a with
      | none => false
      | some p => match movePlan s p with
        | none => false
        | some (_, g) => match blobAt s g.mid with
          | none => false
          | some b => t.level != s.level ||
              ((groupsOf t).any (fun h => h.mid == g.mid && h.ink == overwrite g.ink b.2.2)
                && (blobAt t g.mid).isNone) },
  -- 没被取走的漆块一格不变（墨里色1 已按结构记回色10，见 _lib）；一步最多少一块
  { name := "decoy_silhouettes_are_fixed"
    guard := fun _ a => isA6 a
    effect := fun s _ t => t.level != s.level ||
      ((blobsOf t).all (fun b => (blobsOf s).contains b) && (blobsOf s).length ≤ (blobsOf t).length + 1) },
  -- 锁上的靶环一直锁着；而且任何一帧里都没有罐子压在一个没锁、颜色集合相同的靶心上
  { name := "lock_turns_reticle_zero"
    guard := fun _ a => isA6 a
    effect := fun s _ t => t.level != s.level ||
      ((lockedCells s).all (fun c => (lockedCells t).contains c) &&
       (groupsOf t).all (fun g => !(retsOf t).any (fun r => tcell r == g.mid && r.2.2 != 0 && r.2.2 == colorSet g.ink))) },
  -- 换关 ⟺ 这一击之后每一组都压在锁上的靶心上
  { name := "levelup_iff_mids_on_targets"
    guard := fun _ a => isA6 a
    effect := fun s a t =>
      if t.level == s.level then !allLocked (groupsOf t) (retsOf t)
      else match stepRules s a with
        | some u => allLocked (groupsOf u) (retsOf u)
        | none => false },
  -- 没锁、环上没有空格（色0）的罐子，恰好配得上一个靶环：臂色集合 = 环上颜色集合
  { name := "target_pairs_by_sector_colors"
    guard := fun _ a => isA6 a
    effect := fun _ _ t => (groupsOf t).all (fun g =>
      onLocked (retsOf t) g || (inkDigits g.ink).contains 0 ||
        ((retsOf t).filter (fun r => r.2.2 != 0 && r.2.2 == colorSet g.ink)).length == 1) },
  -- 一组里读得到的核色都相同
  { name := "cluster_is_within_core_color"
    guard := fun _ a => isA6 a
    effect := fun _ _ t => (groupsOf t).all (fun g =>
      let ks := (g.an.map (fun x => x.2.2)).filter (fun k => k ≥ 0)
      ks.all (fun k => k == ks.headD k)) },
  -- 点在空处的一击：踩禁地（菱形踩色2 / 罐子踩色2 或色10）⟺ 驳回（除竖条外一格不改）；没踩就一定被收下
  { name := "move_rejected_on_blocked_terrain"
    guard := fun s a => match clickRC a with
      | some p => (movePlan s p).isSome
      | none => false
    effect := fun s a t => match clickRC a with
      | some p =>
        if moveWall s a then sameExceptBar s t && barNext s.bar_rows_n == some t.bar_rows_n
        else t.level != s.level || selOf t == p
      | none => false },
  -- 新落点跟别的动件几何重叠、又没踩禁地：照样被收下
  { name := "object_overlap_does_not_reject_move"
    guard := fun s a => moveOpen s a && overlapsOthers s a
    effect := fun s a t => match clickRC a with
      | some p => t.level != s.level || selOf t == p
      | none => false },
  -- 一关之内硬墙不变（软地是每关常数、空地不在状态里，只比得出硬墙这一层）
  { name := "terrain_is_static"
    guard := fun _ a => isA6 a
    effect := fun s _ t => t.level != s.level || t.walls_m == s.walls_m },
  -- 扇区色 ≥ 2 的靶环，若每种扇区色恰好有一块漆带它，这几块漆的墨色格并起来铺满环上 20 格
  { name := "reticle_colors_have_tiling_inks"
    guard := fun _ a => isA6 a
    effect := fun _ _ t => (retsOf t).all (fun r =>
      let cols := ((List.range 16).map Int.ofNat).filter (fun c => bitAt r.2.2 c)
      let who := cols.map (fun c => (c, (blobsOf t).filter (fun b => c != 0 && (inkDigits b.2.2).contains c)))
      r.2.2 == 0 || cols.length < 2 || !(who.all (fun x => x.2.length == 1)) ||
        who.foldl (fun m x => bor m (posMask ((x.2.headD (0, 0, 0)).2.2) x.1)) 0 == fullRing) },
  -- 两墨盘（有漆块、靶环 ≤ 2 色、罐子 ≤ 2 种非空色）：环上的色块形状只有互补的两半（各 10 格）或整圈
  { name := "blob_inks_are_complementary_halves"
    guard := fun _ a => isA6 a
    effect := fun _ _ t =>
      let inks := (blobsOf t).map (fun b => b.2.2) ++ (groupsOf t).map (·.ink)
      let colorsOf := fun (ink : Int) => ((List.range 16).map Int.ofNat).filter (fun c => (inkDigits ink).contains c)
      let shapes := inks.flatMap (fun ink => (colorsOf ink).map (posMask ink))
      let halves := (shapes.filter (· != fullRing)).eraseDups
      (blobsOf t).isEmpty || (retsOf t).any (fun r => popcount r.2.2 > 2) ||
        (groupsOf t).any (fun g => ((colorsOf g.ink).filter (· != 0)).length > 2) ||
        (match halves with
         | [] => true
         | [h] => popcount h == 10
         | [h1, h2] => band h1 h2 == 0 && bor h1 h2 == fullRing
         | _ => false) }
]
