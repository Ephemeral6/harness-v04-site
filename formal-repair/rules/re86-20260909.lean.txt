-- ---------------------------------------------------------------- 一件东西
structure Pc where
  kind : Nat      -- 0 十字（横杠 + 竖杠）/ 1 空心方框 / 2 斜叉 / 3 菱形
  col : Nat
  hol : Bool      -- hub 那一格不画本色（十字恒 false）
  t : Nat
  l : Nat
  b : Nat
  r : Nat
  hr : Nat        -- hub 行（十字 = 横杠所在行）
  hc : Nat        -- hub 列（十字 = 竖杠所在列）
deriving DecidableEq, Repr

def dflt : Pc := ⟨0, 0, false, 0, 0, 0, 0, 0, 0⟩

def fld (n i : Nat) : Nat := (n >>> (7 + 8 * i)) &&& 255

def decN (n : Nat) : Pc :=
  ⟨n &&& 3, (n >>> 2) &&& 15, (n &&& 3) != 0 && ((n >>> 6) &&& 1) == 1,
   fld n 0, fld n 1, fld n 2, fld n 3, fld n 4, fld n 5⟩

def pOf : Int → Option Pc
  | .ofNat n => some (decN n)
  | .negSucc _ => none

def kOf : Option Int → Option Pc
  | some (.ofNat n) => some (decN n)
  | _ => none

def encP (p : Pc) : Int :=
  Int.ofNat (p.kind ||| (p.col <<< 2) ||| ((if p.kind != 0 && p.hol then 1 else 0) <<< 6)
    ||| (p.t <<< 7) ||| (p.l <<< 15) ||| (p.b <<< 23) ||| (p.r <<< 31) ||| (p.hr <<< 39) ||| (p.hc <<< 47))

def ctrR (p : Pc) : Nat := (p.t + p.b + 1) / 2
def ctrC (p : Pc) : Nat := (p.l + p.r + 1) / 2

-- ---------------------------------------------------------------- 格集合 = 位掩码（盘格 (R,C) 占第 (R-64)*64+(C-64) 位；第 63 行 HUD 另算）
-- 热路径上的比较一律直接用 Nat.ble / Nat.beq + cond（不经 Decidable，内核里省得多）
def leN (a b : Nat) : Bool := Nat.ble a b
def eqN (a b : Nat) : Bool := Nat.beq a b
def mn (a b : Nat) : Nat := cond (Nat.ble a b) a b
def mx (a b : Nat) : Nat := cond (Nat.ble a b) b a

def onB (r c : Nat) : Bool := leN 64 r && leN r 126 && leN 64 c && leN c 127
def bitAt (r c : Nat) : Nat := cond (onB r c) (1 <<< ((r - 64) * 64 + (c - 64))) 0
def rep (st k : Nat) : Nat := ((1 <<< (st * k)) - 1) / ((1 <<< st) - 1)
def colM : Nat := rep 64 63          -- 一整列（63 格，每隔 64 位一格）
def dg65 : Nat := rep 65 64          -- 右下斜线
def dg63 : Nat := rep 63 64          -- 左下斜线
def hits (m n : Nat) : Bool := !Nat.beq (m &&& n) 0
def minus (m n : Nat) : Nat := m ^^^ (m &&& n)

-- 横段 (r, c0..c1)、竖段 (r0..r1, c)，裁到盘内
def hsegK (r lo hi : Nat) : Nat :=
  cond (leN 64 r && leN r 126 && leN lo hi) (((1 <<< (hi - lo + 1)) - 1) <<< ((r - 64) * 64 + (lo - 64))) 0
def hseg (r c0 c1 : Nat) : Nat := hsegK r (mx c0 64) (mn c1 127)

def vsegK (c lo hi : Nat) : Nat :=
  cond (leN 64 c && leN c 127 && leN lo hi) ((colM >>> ((63 - (hi - lo + 1)) * 64)) <<< ((lo - 64) * 64 + (c - 64))) 0
def vseg (c r0 r1 : Nat) : Nat := vsegK c (mx r0 64) (mn r1 126)

-- 右下斜段 (r0+k, c0+k)、左下斜段 (r0+k, c0-k)，k = 0..len-1，裁到盘内
def dRK (r0 c0 len kmin : Nat) : Nat :=
  cond (Nat.blt kmin len && leN (r0 + kmin) 126 && leN (c0 + kmin) 127)
    ((dg65 >>> (65 * (64 - (mn (len - 1) (mn (126 - r0) (127 - c0)) - kmin + 1))))
      <<< ((r0 + kmin - 64) * 64 + (c0 + kmin - 64))) 0
def dR (r0 c0 len : Nat) : Nat := dRK r0 c0 len (mx (64 - r0) (64 - c0))

def dLK (r0 c0 len kmin : Nat) : Nat :=
  cond (Nat.blt kmin len && leN (r0 + kmin) 126 && leN (64 + kmin) c0)
    ((dg63 >>> (63 * (64 - (mn (len - 1) (mn (126 - r0) (c0 - 64)) - kmin + 1))))
      <<< ((r0 + kmin - 64) * 64 + (c0 - kmin - 64))) 0
def dL (r0 c0 len : Nat) : Nat := dLK r0 c0 len (mx (64 - r0) (c0 - 127))

-- 理想形状（伸出盘外的部分不画 —— edge_clips_arm）
def exBody (hr hc L : Nat) : Nat :=
  dR (hr - L) (hc - L) L ||| dR (hr + 1) (hc + 1) L ||| dL (hr - L) (hc + L) L ||| dL (hr + 1) (hc - 1) L
def diBody (hr hc L : Nat) : Nat :=
  dL (hr - L) hc (L + 1) ||| dR (hr - L) hc (L + 1) ||| dR hr (hc - L) (L + 1) ||| dL hr (hc + L) (L + 1)

def body (p : Pc) : Nat :=
  cond (eqN p.kind 0) (hseg p.hr p.l p.r ||| vseg p.hc p.t p.b)
    (cond (eqN p.kind 1) (hseg p.t p.l p.r ||| hseg p.b p.l p.r ||| vseg p.l p.t p.b ||| vseg p.r p.t p.b)
      (cond (eqN p.kind 2) (exBody p.hr p.hc (p.hr - p.t)) (diBody p.hr p.hc (p.hr - p.t))))

def cells (p : Pc) : Nat := body p ||| cond (!eqN p.kind 0 && !p.hol) (bitAt p.hr p.hc) 0

def sameShape (p q : Pc) : Bool :=
  p.kind == q.kind && p.t == q.t && p.l == q.l && p.b == q.b && p.r == q.r && p.hr == q.hr && p.hc == q.hc

-- ---------------------------------------------------------------- 每关常数
def ringsOf (v : Nat) : List (Nat × Nat × Nat) :=      -- 环心 (行, 列, 归属色)
  (List.range (v &&& 15)).map fun j =>
    let x := v >>> (4 + 16 * j)
    ((x &&& 63) + 64, ((x >>> 6) &&& 63) + 64, (x >>> 12) &&& 15)

def rowsMask (t k c0 c1 : Nat) : Nat := (List.range k).foldl (fun m i => m ||| hseg (t + i) c0 c1) 0

def swOne (x : Nat) : Nat × Nat × Nat × Nat :=         -- (整片区域, 框边, 内部, 内部颜色)
  let t := (x &&& 127) + 56
  let l := ((x >>> 7) &&& 127) + 56
  let n := (x >>> 14) &&& 15
  let area := rowsMask t n l (l + n - 1)
  let inner := rowsMask (t + 1) (n - 2) (l + 1) (l + n - 2)
  (area, minus area inner, inner, (x >>> 18) &&& 15)

def swsOf (v : Nat) : List (Nat × Nat × Nat × Nat) :=
  (List.range (v &&& 15)).map fun j => swOne (v >>> (4 + 24 * j))

def blkOf (v : Nat) : Nat :=
  (List.range (v &&& 15)).foldl (fun m j =>
    let x := v >>> (4 + 80 * j)
    m ||| (List.range 8).foldl (fun m2 rr =>
      m2 ||| (((x >>> (8 * rr)) &&& 255) <<< ((((x >>> 64) &&& 63) + rr) * 64 + ((x >>> 70) &&& 63)))) 0) 0

def ringEdge (x : Nat × Nat × Nat) : Nat :=
  hseg (x.1 - 1) (x.2.1 - 1) (x.2.1 + 1) ||| bitAt x.1 (x.2.1 - 1) ||| bitAt x.1 (x.2.1 + 1)
    ||| hseg (x.1 + 1) (x.2.1 - 1) (x.2.1 + 1)

-- 不动的那一层（环 / 色块 / 挡块），按颜色 0..15 各一个掩码
def stillOf (ring sw blk : Nat) : List Nat :=
  (List.range 16).map fun c =>
    (ringsOf ring).foldl (fun m x => m ||| (if c == 4 then ringEdge x else 0) ||| (if x.2.2 == c then bitAt x.1 x.2.1 else 0)) 0
    ||| (swsOf sw).foldl (fun m x => m ||| (if c == 2 then x.2.1 else 0) ||| (if x.2.2.2 == c then x.2.2.1 else 0)) 0
    ||| (if c == 1 then blkOf blk else 0)

structure Lay where

  fill : Nat                                  -- 第 i 位：第 i 件（斜叉）换下之后 hub 画本色
  bud : Nat                                   -- 预算 B
  rings : List (Nat × Nat × Nat)
  sws : List (Nat × Nat × Nat × Nat)
  blk : Nat
  still : List Nat

def mkLay (lay fill bud ring sw blk np : Nat) : Lay :=
  { ord := (List.range np).map fun k => (lay >>> (2 * k)) &&& 3
    fill := fill, bud := bud, rings := ringsOf ring, sws := swsOf sw, blk := blkOf blk,
    still := stillOf ring sw blk }

def noLay : Lay := ⟨[], 0, 1, [], [], 0, []⟩

def layOf (s : State) : Lay :=
  match s.k_lay, s.k_fill, s.k_B, s.k_ring, s.k_sw, s.k_blk, s.k_p2 with

      mkLay a b c d e f 3
  | some (.ofNat a), some (.ofNat b), some (.ofNat c), some (.ofNat d), some (.ofNat e), some (.ofNat f), _ =>
      mkLay a b c d e f 2
  | _, _, _, _, _, _, _ => noLay

def isMove : Act → Bool

  | _ => false

def shiftP (p : Pc) : Act → Pc
  | Act.a1 => { p with t := p.t - 3, b := p.b - 3, hr := p.hr - 3 }
  | Act.a2 => { p with t := p.t + 3, b := p.b + 3, hr := p.hr + 3 }
  | Act.a3 => { p with l := p.l - 3, r := p.r - 3, hc := p.hc - 3 }
  | Act.a4 => { p with l := p.l + 3, r := p.r + 3, hc := p.hc + 3 }
  | _ => p

-- 整件平移之后有没有一格落在挡块上（will_block）
def blockedP (p : Pc) (a : Act) (blk : Nat) : Bool := hits (cells (shiftP p a)) blk

-- 十字撞挡块：两条杠各自判，会压到挡块的那条不动，另一条平移；交点 = 横杠的行 × 竖杠的列。
-- 只有一条杠时（横线 / 竖线）交点跟着那条杠走。
def crossPush (p : Pc) (a : Act) (blk : Nat) : Pc :=
  let q := shiftP p a
  let hasH := decide (p.l < p.r)
  let hasV := decide (p.t < p.b)
  let hb := hasH && hits (hseg q.hr q.l q.r) blk
  let vb := hasV && hits (vseg q.hc q.t q.b) blk
  let rh := if hb || (!hasH && vb) then p.hr else q.hr
  let cv := if vb || (!hasV && hb) then p.hc else q.hc
  let l := if hasH then (if hb then p.l else q.l) else cv
  let r := if hasH then (if hb then p.r else q.r) else cv
  let t := if hasV then (if vb then p.t else q.t) else rh
  let b := if hasV then (if vb then p.b else q.b) else rh
  { p with t := min t rh, l := min l cv, b := max b rh, r := max r cv, hr := rh, hc := cv }

-- 方框撞挡块：迎面边不动、背面边推进 3、垂直方向长 3（加在新中心离旧中心近的那头，一样近加在下 / 右）；
-- 换完短边 < 4 就整件不动（rect_trade_needs_room）
def roomP (p : Pc) : Act → Bool
  | Act.a3 | Act.a4 => decide (6 ≤ p.r - p.l)
  | _ => decide (6 ≤ p.b - p.t)

def distN (x y : Nat) : Nat := if y ≤ x then x - y else y - x

def mkBox (p : Pc) (t l b r : Nat) : Pc :=
  { p with t := t, l := l, b := b, r := r, hr := (t + b + 1) / 2, hc := (l + r + 1) / 2 }

def boxPush (p : Pc) (a : Act) : Pc :=
  if !roomP p a then p
  else if a == Act.a3 || a == Act.a4 then
    let far := distN ((p.t + p.b + 4) / 2) (ctrR p)
    let near := distN ((p.t + p.b - 2) / 2) (ctrR p)
    let l := if a == Act.a4 then p.l + 3 else p.l
    let r := if a == Act.a4 then p.r else p.r - 3
    if near < far then mkBox p (p.t - 3) l p.b r else mkBox p p.t l (p.b + 3) r
  else
    let far := distN ((p.l + p.r + 4) / 2) (ctrC p)
    let near := distN ((p.l + p.r - 2) / 2) (ctrC p)
    let t := if a == Act.a2 then p.t + 3 else p.t
    let b := if a == Act.a2 then p.b else p.b - 3
    if near < far then mkBox p t (p.l - 3) b p.r else mkBox p t p.l b (p.r + 3)

def pushP (p : Pc) (a : Act) (blk : Nat) : Pc :=
  if !blockedP p a blk then shiftP p a
  else if p.kind == 0 then crossPush p a blk
  else if p.kind == 1 then boxPush p a
  else p

-- 碰色块（整片框区域，含框边）就整件染成那个色块的颜色；碰到几个取左上角最靠前的
def touchCol (L : Lay) (m : Nat) : Option Nat := (L.sws.find? fun x => hits x.1 m).map (·.2.2.2)

def recolor (L : Lay) (p : Pc) : Pc :=
  match touchCol L (cells p) with
  | some c => { p with col := c }
  | none => p

def posIn : List Nat → Nat → Nat
  | [], _ => 0
  | x :: xs, y => cond (Nat.beq x y) 0 (posIn xs y + 1)

-- ---------------------------------------------------------------- ACTION5：按形状次序表轮到下一件，包围盒中心在盘外的跳过
-- 形状名从几何算（a5_cycle_order 的七个名字）：十字族 竖杠长 0 ⇒ hline、四臂等长 ⇒ plus、否则 cross；
-- 方框族 高 = 宽 ⇒ square、否则 rect；ex / diamond 就是自己。
-- 次序表：plus → hline → ex → diamond → square → rect → cross，同形状按颜色升序，到末尾绕回。不用图层常数 k_lay。
-- 与概念原文的差别（照实写）：原文 PRIORITY 是 square → rect → plus → hline → ex → diamond → cross。这里只调了概念自己
-- 列为「先后没被观测确定过」（UNORDERED）的对——把 plus、hline 挪到 square、rect 前面；概念列为已确定的
-- plus<ex<diamond、hline<ex<diamond、square<cross、rect<cross 全部保留。原因：第 6 关三件（square 色 12 /
-- cross 色 7 或 8 / plus 色 10）实测轮法是 square → cross → plus（语料第 1115、1116、1155、1482、1483、1522、1864、1865、
-- 1904 行共 9 次，后 3 次在路线上），原文次序在这 9 次上全判错；调整后全语料 93 次 ACTION5 全对
-- （_scratch/re86-20260909/p_shapekey.py）。所以「plus 在 square 之前」是第 6 关数据逼出来的，不是原文的主张。
def shapeRank (p : Pc) : Nat :=
  cond (eqN p.kind 0)
    (cond (eqN p.t p.b) 1
      (cond (eqN (p.hr - p.t) (p.b - p.hr) && eqN (p.hr - p.t) (p.hc - p.l) && eqN (p.hr - p.t) (p.r - p.hc)) 0 6))
    (cond (eqN p.kind 1) (cond (eqN (p.b - p.t) (p.r - p.l)) 4 5) (cond (eqN p.kind 2) 2 3))

def a5key (w : W) (j : Nat) : Nat := (shapeRank (nthP w.ps j) * 16 + (nthP w.ps j).col) * 4 + j

def minKey (w : W) : List Nat → Option Nat
  | [] => none
  | j :: js => match minKey w js with
    | none => some j
    | some i => cond (Nat.blt (a5key w j) (a5key w i)) (some j) (some i)

def succ (w : W) (_L : Lay) : Nat :=
  let ring := (List.range w.ps.length).filter fun j => j == w.act || onB (ctrR (nthP w.ps j)) (ctrC (nthP w.ps j))
  match minKey w (ring.filter fun j => Nat.blt (a5key w w.act) (a5key w j)) with
  | some j => j
  | none => (minKey w ring).getD w.act

-- 被换下的那件斜叉：hub 画不画改成这一关的常数（换下之前它一直压在光标底下，看不见）
def release (L : Lay) (i : Nat) (p : Pc) : Pc :=
  if p.kind == 2 then { p with hol := !(L.fill.testBit i) } else p

-- ---------------------------------------------------------------- 一步之后的局面
def advance (s : State) (a : Act) : Option (W × Lay) :=
  match world s with
  | none => none
  | some w =>
    match a with
    | Act.a5 => some (⟨w.ps.set w.act (release (layOf s) w.act (cur w)), succ w (layOf s), w.n + 1⟩, layOf s)
    | Act.a1 | Act.a2 | Act.a3 | Act.a4 =>
        some (⟨w.ps.set w.act (recolor (layOf s) (pushP (cur w) a (layOf s).blk)), w.act, w.n + 1⟩, layOf s)
    | _ => none

-- 过关：每个环心都落在一件与归属色同色的东西的格子上（空着的 hub 不算）
def covered (w : W) (r c : Nat) (o : Option Nat) : Bool :=
  w.ps.any fun p => hits (cells p) (bitAt r c) && (o.all (· == p.col))

def wins (w : W) (L : Lay) : Bool :=
  !L.rings.isEmpty && L.rings.all fun x => covered w x.1 x.2.1 (some x.2.2)

def inLv (s : State) (a : Act) : Bool :=
  match advance s a with
  | some (w, L) => !wins w L
  | none => false

-- ---------------------------------------------------------------- 画整帧：不动层打底，东西按图层从下往上盖，光标贴在选中那件那一层
-- 逐色写：第 c 色 = 不动层的 c 色减去被东西 / 光标盖住的格，并上每件 c 色东西露出来的格
-- （它的格减去压在它上面的：图层更高的东西；选中那件及其以下的还要减去光标那一格），c = 0 时再并上露出来的光标。
-- （先前按「从下往上逐层涂 16 色」写过一版，逐帧结果相同，但内核里贵得多，已换掉。）
def cursorBit (w : W) : Nat := bitAt (ctrR (cur w)) (ctrC (cur w))

def coverAll (w : W) : Nat := w.ps.foldl (fun m p => m ||| cells p) (cursorBit w)

-- 图层高于 k 的那些件的格（k = 图层里的位置）
def higher (L : Lay) (k : Nat) : List Pc → Nat → Nat
  | [], _ => 0
  | p :: ps, j => cond (Nat.blt k (posIn L.ord j)) (cells p) 0 ||| higher L k ps (j + 1)

def aboveOf (w : W) (L : Lay) (i : Nat) : Nat :=
  higher L (posIn L.ord i) w.ps 0 ||| cond (leN (posIn L.ord i) (posIn L.ord w.act)) (cursorBit w) 0

def chanGo (w : W) (L : Lay) (c : Nat) : List Pc → Nat → Nat
  | [], _ => 0
  | p :: ps, i => cond (eqN p.col c) (minus (cells p) (aboveOf w L i)) 0 ||| chanGo w L c ps (i + 1)

def nthN : List Nat → Nat → Nat
  | [], _ => 0
  | x :: _, 0 => x
  | _ :: xs, n + 1 => nthN xs n

def chan (w : W) (L : Lay) (c : Nat) : Nat :=
  chanGo w L c w.ps 0 ||| minus (nthN L.still c) (coverAll w)
    ||| cond (eqN c 0) (minus (cursorBit w) (higher L (posIn L.ord w.act) w.ps 0)) 0

-- HUD（第 63 行）：从右往左 spent 格色 1，其余色 15
def hud (sp c : Nat) : Nat :=
  cond (eqN c 1) (((1 <<< sp) - 1) <<< (4096 - sp))
    (cond (eqN c 15) (((1 <<< (64 - sp)) - 1) <<< 4032) 0)

def prime : Nat := 2 ^ 62 - 57

def spentOf (L : Lay) (n : Nat) : Nat := (128 * n + L.bud) / (2 * L.bud)

def inkC (w : W) (L : Lay) (c : Nat) : Int := Int.ofNat ((chan w L c ||| hud (spentOf L w.n) c) % prime)
def encAt : List Pc → Nat → Int
  | [], _ => -1
  | p :: _, 0 => encP p
  | _ :: ps, n + 1 => encAt ps n
def seenAt (w : W) (L : Lay) (x : Nat) : Option Int := cond (hits (chan w L 0) (cursorBit w)) (some (Int.ofNat x - 64)) none

def emit (s : State) (w : W) (L : Lay) : State :=
  { s with h_p0 := encAt w.ps 0, h_p1 := encAt w.ps 1, h_p2 := encAt w.ps 2, h_act := Int.ofNat w.act, h_n := Int.ofNat w.n,
           ink_0 := inkC w L 0, ink_1 := inkC w L 1, ink_2 := inkC w L 2, ink_3 := inkC w L 3, ink_4 := inkC w L 4,
           ink_5 := inkC w L 6, ink_6 := inkC w L 7, ink_7 := inkC w L 8, ink_8 := inkC w L 9, ink_9 := inkC w L 10,
           ink_10 := inkC w L 11, ink_11 := inkC w L 12, ink_12 := inkC w L 13, ink_13 := inkC w L 14,
           ink_14 := inkC w L 15,
           cursor_r := seenAt w L (ctrR (cur w)), cursor_c := seenAt w L (ctrC (cur w)),
           spent := Int.ofNat (spentOf L w.n) }

def stepRules (s : State) (a : Act) : Option State :=
  (advance s a).map fun x => emit s x.1 x.2

-- ---------------------------------------------------------------- 规则（每条一个概念）
-- 下一帧的局面（数据里 hidden_step 读出来的）
def w' (s' : State) : W := (world s').getD ⟨[], 0, 0⟩
def piece' (s' : State) (i : Nat) : Pc := (w' s').ps.getD i dflt
-- 第 c 色的指纹字段（ink_i 按 0,1,2,3,4,6,…,15 排，没有底色 5）
def inkField (s' : State) (c : Nat) : Int :=
  match c with
  | 0 => s'.ink_0 | 1 => s'.ink_1 | 2 => s'.ink_2 | 3 => s'.ink_3 | 4 => s'.ink_4 | 6 => s'.ink_5
  | 7 => s'.ink_6 | 8 => s'.ink_7 | 9 => s'.ink_8 | 10 => s'.ink_9 | 11 => s'.ink_10 | 12 => s'.ink_11
  | 13 => s'.ink_12 | 14 => s'.ink_13 | _ => s'.ink_14
-- 按下一帧的局面（几何 + 选中 + 步数）重画一遍整帧。所有「画得对不对」的规则都比这同一个 redraw s'，各看各的那一色。
def redraw (s' : State) : State :=
  match world s' with
  | some w => emit s' w (layOf s')
  | none => s'
def drawn (s' : State) (c : Nat) : Bool := inkField s' c == inkField (redraw s') c

def moveOK (s : State) (a : Act) (d : Act) : Bool := a == d && inLv s a &&
  (match world s with | some w => !blockedP (cur w) a (layOf s).blk | none => false)
def translated (s : State) (a : Act) (s' : State) : Bool :=
  match world s with
  | some w => (w' s').act == w.act && sameShape (piece' s' w.act) (shiftP (cur w) a)
  | none => false

def blockedKind (s : State) (a : Act) (k : Nat) (room : Option Bool) : Bool :=
  isMove a && inLv s a && (match world s with
    | some w => blockedP (cur w) a (layOf s).blk && (cur w).kind == k && room.all (· == roomP (cur w) a)
    | none => false)

def arms (p : Pc) : List Nat := [p.hr - p.t, p.b - p.hr, p.hc - p.l, p.r - p.hc]
