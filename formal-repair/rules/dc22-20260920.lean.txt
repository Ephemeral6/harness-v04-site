-- ==== 位平面工具（整帧 = 4 个 4096 位 Nat；像素 (r,c) 占第 r*64+c 位） ====
def FULL : Nat := 2 ^ 4096 - 1
def rrep (h : Nat) : Nat := (2 ^ (64 * h) - 1) / (2 ^ 64 - 1)
def C0 : Nat := rrep 64
def C63 : Nat := C0 <<< 63
def mn (a b : Nat) : Nat := a ^^^ (a &&& b)
def rect (r0 r1 c0 c1 : Int) : Nat :=
  if r0 < 0 || c0 < 0 || r1 > 63 || c1 > 63 || r1 < r0 || c1 < c0 then 0
  else ((2 ^ (c1 - c0 + 1).toNat - 1) * rrep (r1 - r0 + 1).toNat) <<< (r0 * 64 + c0).toNat
def bt (r c : Int) : Nat := if r < 0 || c < 0 || r > 63 || c > 63 then 0 else 2 ^ (r * 64 + c).toNat
def cel (r c : Int) : Nat := rect r (r + 1) c (c + 1)
def east (m : Nat) : Nat := mn m C63 <<< 1
def west (m : Nat) : Nat := mn m C0 >>> 1
def south (m : Nat) : Nat := (m <<< 64) &&& FULL
def north (m : Nat) : Nat := m >>> 64
def dil (m : Nat) : Nat := m ||| east m ||| west m ||| south m ||| north m
def flood : Nat → Nat → Nat → Nat
  | 0, m, _ => m
  | k + 1, m, a => let m2 := dil m &&& a; if m2 == m then m else flood k m2 a
def lb (m : Nat) : Nat := m &&& (m ^^^ (m - 1))
def cnt : Nat → Nat → Nat
  | 0, _ => 0
  | k + 1, m => if m == 0 then 0 else cnt k (m ^^^ lb m) + 1
def cfold (m : Nat) : Nat :=
  let x := m ||| m >>> 2048
  let x := x ||| x >>> 1024
  let x := x ||| x >>> 512
  let x := x ||| x >>> 256
  let x := x ||| x >>> 128
  (x ||| x >>> 64) % 2 ^ 64
def shiftM (m : Nat) (dr dc : Int) : Nat :=
  let m := if dc > 0 then (List.range dc.toNat).foldl (fun x _ => east x) m else (List.range (-dc).toNat).foldl (fun x _ => west x) m
  if dr ≥ 0 then (m <<< (64 * dr.toNat)) &&& FULL else m >>> (64 * (-dr).toNat)
def spread (a : Nat) : Nat := a ||| east a ||| south a ||| south (east a)
-- EVEN：行、列都是偶数的像素（2×2 格的左上角）；ODDP：r+c 为奇数的像素
def A64 : Nat := (2 ^ 64 - 1) / 3
def EVR : Nat := (2 ^ 4096 - 1) / (2 ^ 128 - 1)
def EVEN : Nat := A64 * EVR
def ODDP : Nat := (A64 * 2) * EVR ||| (A64 * EVR) <<< 64

def q4 (p i : Nat) : Nat := let x := p >>> i % 2 ^ 66; x % 2 * 4096 + x / 2 % 2 * 256 + x / 2 ^ 64 % 2 * 16 + x / 2 ^ 65 % 2

def nib (v i : Nat) : Nat := v >>> (4 * (3 - i)) % 16
def pk4 (v k : Nat) : Nat := nib v 0 >>> k % 2 + 2 * (nib v 1 >>> k % 2) + 2 ^ 64 * (nib v 2 >>> k % 2) + 2 ^ 65 * (nib v 3 >>> k % 2)

-- ==== 概念 cell_enterable / 垫子 ====
def PAIRS : List (Nat × Nat) := [(6, 7), (13, 8), (12, 11), (9, 10)]
-- icon_recolors_pad 的花色循环 (6,7)→(13,8)→(12,11)→(9,10)→(6,7)，主色（前一个）位置仍放主色
def nextPair (p : Nat × Nat) : Nat × Nat :=
  if p == (6, 7) then (13, 8) else if p == (13, 8) then (12, 11) else if p == (12, 11) then (9, 10) else (6, 7)
def isPad (v : Nat) : Bool :=
  nib v 0 == nib v 3 && nib v 1 == nib v 2 && nib v 0 != nib v 1 &&
    PAIRS.any (fun p => (nib v 0 == p.1 && nib v 1 == p.2) || (nib v 0 == p.2 && nib v 1 == p.1))
def isMarker (v : Nat) : Bool := cnt4 v 0 == 2 && cnt4 v 1 == 1 && all4 v (fun x => x != 3 && x != 4 && x != 5)
def enterable (v : Nat) : Bool :=
  if all4 v (fun x => x != 3 && x != 4 && x != 5) then
    if has4 v 0 then isMarker v
    else all4 v (fun x => x == nib v 0) || has4 v 2 || isPad v || cnt4 v 8 ≥ 2
  else false

-- ==== card8_transposes_arms / card6_transposes_color7_arms：枢纽 4×4 + 相连的臂，过枢纽主对角线转置 ====
def transp : Nat → Nat → Int → Int → Nat → Nat
  | 0, _, _, _, acc => acc
  | k + 1, m, r0, c0, acc =>
    if m == 0 then acc else
      let i := Nat.log2 (lb m)
      let r : Int := i / 64
      let c : Int := i % 64
      transp k (m ^^^ lb m) r0 c0 (acc ||| bt (r0 + c - c0) (c0 + r - r0))
-- 结果码：0 没有这套装置，1 做了，2 被拒（玩家踩在臂上）
def arms (e : Env) (B : Fr) (ka kh : Nat) : Fr × Nat :=
  let hub := colm B kh &&& e.mapm
  if hub == 0 || cnt 17 hub != 16 then (B, 0) else
  let r0 : Int := Nat.log2 (lb hub) / 64
  let c0 : Int := Nat.log2 (lb (cfold hub))
  if hub != rect r0 (r0 + 3) c0 (c0 + 3) then (B, 0) else
  let ac := colm B ka &&& e.mapm
  let arm := flood 40 (dil hub &&& ac) ac
  if arm == 0 then (B, 0) else if arm &&& e.pm != 0 then (B, 2) else
  (pnt (pnt B arm 4) (transp 200 arm r0 c0 0) ka, 1)

-- ==== card_phase_swap_by_color：实心 k 格 ↔ 4/k 棋盘格（k 在 r+c 奇的位置） ====
def phase (e : Env) (B : Fr) (k : Nat) : Fr × Nat :=
  let m := colm B k &&& e.mapm
  let w := colm B 4 &&& e.mapm
  let sp := spread (m &&& west m &&& north m &&& north (west m) &&& EVEN)
  let cp := spread (west m &&& north m &&& w &&& north (west w) &&& EVEN)
  if sp ||| cp == 0 then (B, 0) else if sp &&& e.pm != 0 then (B, 2) else
  (pnt (pnt B (mn sp ODDP) 4) cp k, 1)

  let ma := colm B a &&& e.mapm
  let mb := colm B b &&& e.mapm
  (ma &&& west mb &&& north mb &&& north (west ma) ||| mb &&& west ma &&& north ma &&& north (west mb)) &&& EVEN
def flipPads (e : Env) (B : Fr) : Fr :=
  PAIRS.foldl (fun F p =>
    let anc := padAnc e B p.1 p.2
    let anc := if p == (6, 7) then anc else mn anc e.hub
    let sp := spread anc
    pnt (pnt F (colm B p.1 &&& sp) p.2) (colm B p.2 &&& sp) p.1) B
def teleDest (e : Env) (B : Fr) (r c : Int) (u : Nat) : Option (Int × Int) :=
  if !isPad u then none else
  match PAIRS.find? (fun p => (nib u 0 == p.1 && nib u 1 == p.2) || (nib u 0 == p.2 && nib u 1 == p.1)) with
  | none => none
  | some p =>
    let anc := mn (padAnc e B p.1 p.2) (bt r c)
    if anc != 0 && anc == lb anc then some ((Nat.log2 anc / 64 : Nat), (Nat.log2 anc % 64 : Nat)) else none

-- ==== card11_*：横条 (行, 轨道起列, 开局档)；第 n 次生效后档位 (开局档+n) mod 6，最右档色 12，色 1 压在色 12 上 ====
def bar3 (x : Nat) : Int × Int × Nat := ((fld x 0 6 : Nat), (fld x 6 6 : Nat), fld x 12 3)
def barsPaint (B : Fr) (bs : List Nat) (n : Nat) : Fr :=
  let B := bs.foldl (fun F x => let b := bar3 x; pnt F (rect b.1 (b.1 + 1) b.2.1 (b.2.1 + 15)) 4) B
  let draw := fun (want : Nat) (F : Fr) => bs.foldl (fun F x =>
    let b := bar3 x
    let sl := (b.2.2 + n) % 6
    let c := if sl == 5 then 12 else 1
    if c == want then pnt F (rect b.1 (b.1 + 1) (b.2.1 + 2 * sl) (b.2.1 + 2 * sl + 5)) c else F) F
  draw 1 (draw 12 B)
-- ==== card2 / 横带 (行, 起列, 开局端)：第 n 次后填充 f = 4·tri(n mod 8)（开局全闭的取 16-f），两端各 f/2，满 16 整条色 12 ====
def bandsPaint (B : Fr) (bs : List Nat) (n : Nat) : Fr :=
  let k := n % 8
  let f0 := 4 * (if k ≤ 4 then k else 8 - k)
  bs.foldl (fun F x =>
    let r : Int := fld x 0 6
    let c0 : Int := fld x 6 6
    let f := if fld x 12 1 == 1 then 16 - f0 else f0
    let F := pnt F (rect r (r + 1) c0 (c0 + 15)) 5
    if f == 16 then pnt F (rect r (r + 1) c0 (c0 + 15)) 12
    else if f == 0 then F
    else pnt F (rect r (r + 1) c0 (c0 + (f / 2 : Nat) - 1) ||| rect r (r + 1) (c0 + 16 - (f / 2 : Nat)) (c0 + 15)) 1) B

-- ==== card_pickup_clears_icon：走进压着 .XX./XXXX 图标的格，图标涂成地板 2，面板出现对应的卡 ====
def pickCard (F : Fr) (v : Nat) : Fr :=
  let r0 : Int := fld v 4 6
  let c0 : Int := fld v 10 6
  let w := fld v 16 4
  let h := fld v 20 3
  (List.range (w * h)).foldl (fun G k =>
    let x := fld v (24 + 4 * k) 4
    if x == 0 then G else pnt G (bt (r0 + (k / w : Nat)) (c0 + (k % w : Nat))) x) F
def iconAt (wr wc : Int) : Nat := rect wr wr (wc + 1) (wc + 2) ||| rect (wr + 1) (wr + 1) wc (wc + 3)
def pickup (e : Env) (s : State) (B : Fr) (tr tc : Int) : Fr :=
  let v := cv B tr tc
  if !has4 v 2 || all4 v (fun x => x == 0 || x == 2 || x == 3 || x == 4 || x == 5 || x == 14) then B else
  let good := FULL ^^^ [0, 2, 3, 4, 5, 14].foldl (fun m k => m ||| colm B k) 0
  let two := colm B 2
  let anc := two &&& west (west (west two)) &&& west good &&& west (west good) &&&
    north (good &&& west good &&& west (west good) &&& west (west (west good))) &&& e.mapm
  let hit := cel tr tc &&& good
  let cand := (List.range 24).foldl (fun acc k =>
    let wr := tr - 1 + (k / 8 : Nat)
    let wc := tc - 3 + (k % 8 : Nat)
    if anc &&& bt wr wc != 0 && iconAt wr wc &&& hit != 0 then acc ||| bt wr wc else acc) 0
  if cand == 0 || cand != lb cand then B else
  let wr : Int := Nat.log2 cand / 64
  let wc : Int := Nat.log2 cand % 64
  let key := pxc B wr (wc + 1)
  let B := pnt B (iconAt wr wc) 2
  let v1 := kd s.k_pk1
  let v2 := kd s.k_pk2
  if s.k_pk1.isSome && v1 % 16 == key then pickCard B v1
  else if s.k_pk2.isSome && v2 % 16 == key then pickCard B v2 else B

-- ==== 第 4 关爪子与杆：card_l4_claw_slides / card_l4_claw_waist_peek / l4_card8_closes_clamp ====
def dv (d : Nat) : Int × Int := if d == 1 then (-4, 0) else if d == 2 then (4, 0) else if d == 3 then (0, -4) else (0, 4)
def clawOf (e : Env) (B : Fr) : Option (Int × Int × Bool × Nat) :=
  let C := colm B 8 &&& e.mapm
  if C == 0 then none else
  let r0 : Int := Nat.log2 (lb C) / 64
  let r1 : Int := Nat.log2 C / 64
  let sh := r1 - r0 == 5
  some ((if sh then r0 - 1 else r0), (Nat.log2 (lb (cfold C)) : Nat), sh, C)
def rodOf (e : Env) (s : State) (B : Fr) : Nat :=
  let m := colm B 12 &&& e.mapm
  let m := (klist (kd s.k_bars) 15).foldl (fun x b => mn x (rect (fld b 0 6) (fld b 0 6 + 1) (fld b 6 6) (fld b 6 6 + 15))) m
  (klist (kd s.k_bands) 13).foldl (fun x b => mn x (rect (fld b 0 6) (fld b 0 6 + 1) (fld b 6 6) (fld b 6 6 + 15))) m
-- 结果码：0 没有爪子，1 动了，2 被拒（玩家踩在要动的东西上），3 推不动（腰格落点不全是底图的轨道色 0）
def clawMove (e : Env) (s : State) (B : Fr) (d : Nat) : Fr × Nat :=
  match clawOf e B with
  | none => (B, 0)
  | some (R, C0, sh, cm) =>
    let wr := R + 2 + (dv d).1
    let wc := C0 + 1 + (dv d).2
    let w := rect wr (wr + 3) wc (wc + 3)
    if wr < 0 || wc < 0 || wr + 3 > 61 || w == 0 || e.z &&& w != w then (B, 3) else
    let rod := if sh then rodOf e s B else 0
    if (cm ||| rod) &&& e.pm != 0 then (B, 2) else
    (pnt (pnt (restore e B (cm ||| rod)) (shiftM rod (dv d).1 (dv d).2) 12) (shiftM cm (dv d).1 (dv d).2) 8, 1)
def clawShut (e : Env) (B : Fr) : Fr × Nat :=
  match clawOf e B with
  | none => (B, 0)
  | some (R, C0, sh, cm) =>
    if sh || colm B 12 &&& rect (R + 2) (R + 5) (C0 + 7) (C0 + 7) == 0 then (B, 3) else
    let nw := (cm &&& rect (R + 2) (R + 5) 0 63) ||| south (cm &&& rect R (R + 1) 0 63) ||| north (cm &&& rect (R + 6) (R + 7) 0 63)
    (pnt (restore e B cm) nw 8, 1)

-- ==== 第 5 关十字件与方框：card6_rotates_piece / lit_diamond_moves_box / card8_clamps_box_on_piece / clamped_piece_slides ====
-- 件：环左上角 (rr,rc)，o = 0 横放；臂 = 环的 4 行 × 列 rc-8..rc+11，环上下各两行凸起；竖放是关于环的转置
def pR (rr rc : Int) (o : Nat) (a b c d : Int) : Nat := if o == 0 then rect (rr + a) (rr + b) (rc + c) (rc + d) else rect (rr + c) (rr + d) (rc + a) (rc + b)
def ringM (rr rc : Int) : Nat := mn (rect rr (rr + 3) rc (rc + 3)) (rect (rr + 1) (rr + 2) (rc + 1) (rc + 2))
def pieceM (p : Int × Int × Nat) : Nat :=
  let (rr, rc, o) := p
  mn (pR rr rc o 0 3 (-8) 11 ||| pR rr rc o (-2) (-2) (-1) 4 ||| pR rr rc o (-1) (-1) 0 3 ||| pR rr rc o 4 4 0 3 ||| pR rr rc o 5 5 (-1) 4) (ringM rr rc)
def pieceOk (p : Int × Int × Nat) : Bool :=
  let (rr, rc, o) := p
  if o == 0 then rr ≥ 2 && rr + 5 ≤ 53 && rc ≥ 8 && rc + 11 ≤ 39 else rc ≥ 2 && rc + 5 ≤ 39 && rr ≥ 8 && rr + 11 ≤ 53
def boxM (b : Int × Int × Nat) : Nat :=
  let (br, bc, sh) := b
  if sh == 0 then mn (rect br (br + 7) bc (bc + 7)) (rect (br + 1) (br + 6) (bc + 1) (bc + 6))
  else mn (rect (br + 1) (br + 6) (bc + 1) (bc + 6)) (rect (br + 3) (br + 4) (bc + 3) (bc + 4))
def boxIn (b : Int × Int × Nat) : Nat := let (br, bc, sh) := b; if sh == 0 then rect br (br + 7) bc (bc + 7) else rect (br + 1) (br + 6) (bc + 1) (bc + 6)
def pAll (p : Option (Int × Int × Nat)) : Nat := match p with | some q => pieceM q ||| ringM q.1 q.2.1 | none => 0
def render6 (e : Env) (B : Fr) (op nq : Option (Int × Int × Nat)) (ob nb : Option (Int × Int × Nat)) : Fr :=
  let B := restore e B (pAll op ||| (match ob with | some b => boxM b | none => 0))
  let B := match nq with | some q => pnt (pnt B (pieceM q) 12) (ringM q.1 q.2.1) 8 | none => B
  match nb with | some b => pnt B (boxM b) 8 | none => B
def mvP (p : Int × Int × Nat) (d : Int × Int) : Int × Int × Nat := (p.1 + d.1, p.2.1 + d.2, p.2.2)
-- 亮菱形推方框：返回 (底图, 件, 框, 结果码 1 动了 / 3 推不动)
def boxStep (e : Env) (B : Fr) (pc : Option (Int × Int × Nat)) (bx : Int × Int × Nat) (d : Nat) :
    Fr × Option (Int × Int × Nat) × (Int × Int × Nat) × Nat :=
  let nb := mvP bx (dv d)
  let sh := bx.2.2 == 1
  if nb.1 < 0 || nb.2.1 < 0 || nb.1 + 7 > 53 || nb.2.1 + 7 > 39 then (B, pc, bx, 3) else
  let under := render6 e B pc (if sh then none else pc) (some bx) none
  if colm under 4 &&& rect (nb.1 + 3) (nb.1 + 4) (nb.2.1 + 3) (nb.2.1 + 4) != 0 then (B, pc, bx, 3) else
  let npc := if sh then pc.map (fun p => mvP p (dv d)) else pc
  let blocked := match sh, pc, npc with
    | true, some p, some q => !pieceOk q || e.f &&& mn (pAll (some q)) (pAll (some p)) != 0
    | _, _, _ => false
  if blocked then (B, pc, bx, 3) else (render6 e B pc npc (some bx) (some nb), npc, nb, 1)
def clamp6 (e : Env) (B : Fr) (pc : Option (Int × Int × Nat)) (bx : Int × Int × Nat) : Fr × (Int × Int × Nat) × Nat :=
  let (br, bc, sh) := bx
  match pc with
  | none => (B, bx, 3)
  | some p =>
    let ctr := rect (br + 3) (br + 4) (bc + 3) (bc + 4)
    if sh == 1 || pieceM p &&& ctr != ctr || mn (rect (br + 2) (br + 5) (bc + 2) (bc + 5)) ctr != ringM p.1 p.2.1 then (B, bx, 3)
    else (render6 e B pc pc (some bx) (some (br, bc, 1)), (br, bc, 1), 1)
def dias (s : State) : List (Int × Int × Nat × Nat) :=
  if s.k_dia.isNone then [] else (List.range 4).map (fun i => let x := fld (kd s.k_dia) (19 * i) 19; ((fld x 0 6 : Nat), (fld x 6 6 : Nat), fld x 12 4, fld x 16 3))
-- legend_marker_lights_diamond：脚下是标记格 [1,0/0,X] 时 X 那颗整块亮，其余灭（15 个色 4 + 右下角 X）
def diamonds (s : State) (F : Fr) (u : Nat) : Fr :=
  let lit := if nib u 0 == 1 && nib u 1 == 0 && nib u 2 == 0 && nib u 3 > 1 then nib u 3 else 99
  (dias s).foldl (fun G d =>
    let b4 := rect d.1 (d.1 + 3) d.2.1 (d.2.1 + 3)
    if d.2.2.1 == lit then pnt G b4 lit else pnt (pnt G b4 4) (bt (d.1 + 3) (d.2.1 + 3)) d.2.2.1) F

-- ==== 卡：同色四连通块 ≥ 20 像素、外圈只有色 0 / 色 5（面板底与卡框） ====
def MAPROWS : Nat := rect 0 61 0 63
def cardOf (F : Fr) (r c : Int) : Option Nat :=
  if r < 0 || r > 61 || c < 0 || c > 63 then none else
  let v := pxc F r c
  if v == 0 || v == 4 || v == 5 then none else
  let g := flood 40 (bt r c) (colm F v &&& MAPROWS)
  if cnt 20 g < 20 then none else
  if mn (dil g &&& MAPROWS) g &&& (FULL ^^^ (colm F 0 ||| colm F 5)) != 0 then none else some v

-- ==== HUD：本关已走 t 拍，色 3 像素 = min(64, ceil(64t/tb))；拍数（拟合）见头部说明 ====
def hudRow (t tb : Nat) : Nat := (2 ^ min 64 ((64 * t + tb - 1) / tb) - 1) <<< 4032
def HUDM : Nat := (2 ^ 64 - 1) <<< 4032
def setHud (F : Fr) (h : Nat) : Fr := ⟨mn F.p0 HUDM ||| h, mn F.p1 HUDM ||| h, mn F.p2 HUDM, mn F.p3 HUDM⟩
def hudFull (F : Fr) : Bool := colm F 3 &&& rect 63 63 0 63 == rect 63 63 0 63

def oP (a b c : Option Int) : Option (Int × Int × Nat) := match a, b, c with | some x, some y, some z => some (x, y, z.toNat) | _, _, _ => none
def mkEnv (s : State) : Env :=
  ⟨rect 0 61 0 (kd s.k_pc - 1 : Nat), rect 0 61 (kd s.k_pc : Nat) 63, bgFix s (kd s.k_bgz), bgFix s (kd s.k_bgf), bgFix s (kd s.k_bgi),
    (if s.k_hub.isSome then bt (kd s.k_hub / 64 : Nat) (kd s.k_hub % 64 : Nat) else 0), cel s.me_r s.me_c⟩

-- ==== stepRules：一步。玩家先擦掉（脚下 u 补回），在底图上做装置的事，再把玩家画回去、重记 u ====
-- 拆成许多小函数：内核每步只展开走到的那一支（整段大 match 每步复制会把内存吃光）。
-- cls：0 移动 / 空点，1 卡生效，2 被拒，3 卡空点，4 爪子合口，5 按钮动 / 推不动，6 亮菱形推动，7 四色图标
def wt (cls : Nat) : Nat := if cls == 1 || cls == 6 || cls == 7 then 2 else if cls == 2 then 20 else if cls == 3 then 0 else 1
structure Rs where
  B : Fr
  r : Int
  c : Int
  pc : Option (Int × Int × Nat)
  bx : Option (Int × Int × Nat)
  cls : Nat
  n2 : Int
  n11 : Int
def frS (s : State) : Fr := ⟨s.fb0.toNat, s.fb1.toNat, s.fb2.toNat, s.fb3.toNat⟩
def bg0 (s : State) : Fr := put4 (frS s) s.me_r s.me_c s.h_u.toNat
def pcS (s : State) : Option (Int × Int × Nat) := oP s.pc6_0 s.pc6_1 s.pc6_2
def bxS (s : State) : Option (Int × Int × Nat) := oP s.bx6_0 s.bx6_1 s.bx6_2
def rs0 (s : State) (B : Fr) (cls : Nat) : Rs := ⟨B, s.me_r, s.me_c, pcS s, bxS s, cls, s.h_n2, s.h_n11⟩
def dOf (a : Act) : Int × Int := match a with | Act.a1 => (-2, 0) | Act.a2 => (2, 0) | Act.a3 => (0, -2) | Act.a4 => (0, 2) | _ => (0, 0)

def card8 (s : State) : Rs :=
  let a := arms (mkEnv s) (bg0 s) 8 13
  if a.2 != 0 then rs0 s a.1 a.2 else
  if s.k_btn.isSome then (let c := clawShut (mkEnv s) (bg0 s); rs0 s c.1 (if c.2 == 1 then 4 else 3)) else
  match bxS s with
  | some b => let r := clamp6 (mkEnv s) (bg0 s) (pcS s) b; { rs0 s r.1 r.2.2 with bx := some r.2.1 }
  | none => rs0 s (bg0 s) 3
def rot6 (s : State) (B : Fr) : Fr × Option (Int × Int × Nat) × Nat :=
  match pcS s with
  | some p =>
    let q := (p.1, p.2.1, 1 - p.2.2)
    let m := cel s.me_r s.me_c
    if !pieceOk q then (B, pcS s, 0)
    else if pAll (pcS s) &&& m != 0 && pAll (some q) &&& m != m then (B, pcS s, 2)
    else (render6 (mkEnv s) B (pcS s) (some q) (bxS s) (bxS s), some q, 1)
  | none => (B, none, 0)
def card6b (s : State) (B : Fr) : Rs :=
  let rt := rot6 s B
  if rt.2.2 == 2 then rs0 s (bg0 s) 2 else
  let d := (teleDest (mkEnv s) rt.1 s.me_r s.me_c s.h_u.toNat).getD (s.me_r, s.me_c)
  { rs0 s (flipPads (mkEnv s) rt.1) 1 with r := d.1, c := d.2, pc := rt.2.1 }
def card6 (s : State) : Rs :=
  let a := arms (mkEnv s) (bg0 s) 7 6
  if a.2 == 2 then rs0 s (bg0 s) 2 else card6b s a.1
def card11 (s : State) (B3 : Fr) : Rs :=
  if all4 s.h_u.toNat (fun z => z == 1 || z == 12) && !all4 (cv B3 s.me_r s.me_c) (fun z => z == 1 || z == 12) then rs0 s (bg0 s) 2
  else { rs0 s B3 1 with n11 := s.h_n11 + 1 }
def cardDev (s : State) (kc : Nat) : Rs :=
  if kc == 8 then card8 s else if kc == 6 then card6 s
  else if kc == 9 || kc == 15 then (let p := phase (mkEnv s) (bg0 s) kc; rs0 s p.1 (if p.2 == 0 then 3 else p.2))
  else if kc == 11 && s.k_bars.isSome then card11 s (barsPaint (bg0 s) (klist (kd s.k_bars) 15) (s.h_n11 + 1).toNat)
  else if kc == 2 && s.k_bands.isSome then { rs0 s (bandsPaint (bg0 s) (klist (kd s.k_bands) 13) (s.h_n2 + 1).toNat) 1 with n2 := s.h_n2 + 1 }
  else rs0 s (bg0 s) 3
-- panel_ring_fades_on_click：卡生效时面板上所有色 0 卡框褪成面板底色 5
def cardStep (s : State) (kc : Nat) : Rs :=
  let x := cardDev s kc
  if x.cls == 1 || x.cls == 4 then { x with B := pnt x.B (colm x.B 0 &&& (mkEnv s).panm) 5 } else x
def iconRe (s : State) (B : Fr) : Fr :=
  let hr : Int := kd s.k_hub / 64
  let hc : Int := kd s.k_hub % 64
  let v := cv B hr hc
  match PAIRS.find? (fun p => (nib v 0 == p.1 && nib v 1 == p.2) || (nib v 0 == p.2 && nib v 1 == p.1)) with
  | some p => let q := nextPair p; put4 B hr hc ([0, 1, 2, 3].foldl (fun acc k => acc * 16 + (if nib v k == p.1 then q.1 else q.2)) 0)
  | none => B
def diaStep (s : State) (d : Int × Int × Nat × Nat) (b : Int × Int × Nat) : Rs :=
  let b4 := rect d.1 (d.1 + 3) d.2.1 (d.2.1 + 3)
  if colm (frS s) d.2.2.1 &&& b4 != b4 then rs0 s (bg0 s) 0 else
  let r := boxStep (mkEnv s) (bg0 s) (pcS s) b d.2.2.2
  if r.2.2.2 == 1 then { rs0 s r.1 6 with pc := r.2.1, bx := some r.2.2.1 } else rs0 s (bg0 s) 5
def nonCard (s : State) (y x : Int) : Rs :=
  let br : Int := kd s.k_btn / 64
  let bc : Int := kd s.k_btn % 64
  match (List.range 4).find? (fun (i : Nat) => s.k_btn.isSome && rect br (br + 3) (bc + 5 * (i : Int)) (bc + 5 * (i : Int) + 3) &&& bt y x != 0) with
  | some i => let m := clawMove (mkEnv s) s (bg0 s) ([3, 1, 4, 2].getD i 0); rs0 s (if m.2 == 1 then m.1 else bg0 s) (if m.2 == 2 then 2 else 5)
  | none =>
    match (dias s).find? (fun d => rect d.1 (d.1 + 3) d.2.1 (d.2.1 + 3) &&& bt y x != 0), bxS s with
    | some d, some b => diaStep s d b
    | _, _ =>
      let ic := kd s.k_icon
      if s.k_icon.isSome && rect (ic / 64 : Nat) (ic / 64 + 3 : Nat) (ic % 64 : Nat) (ic % 64 + 3 : Nat) &&& bt y x != 0 && pxc (frS s) y x != 5
      then rs0 s (iconRe s (bg0 s)) 7 else rs0 s (bg0 s) 0
def moveStep (s : State) (d : Int × Int) : Rs :=
  let tr := s.me_r + d.1
  let tc := s.me_c + d.2
  if (d.1 != 0 || d.2 != 0) && tr ≥ 0 && tr ≤ 62 && tc ≥ 0 && tc ≤ 62 && enterable (cv (bg0 s) tr tc) then
    { rs0 s (pickup (mkEnv s) s (bg0 s) tr tc) 0 with r := tr, c := tc }
  else rs0 s (bg0 s) 0
def act (s : State) (a : Act) : Rs :=
  match a with
  | Act.a6 x y => match cardOf (frS s) y x with | some kc => cardStep s kc | none => nonCard s y x
  | _ => moveStep s (dOf a)
-- 画回玩家：件的环 / 方框的色 8 在玩家半步踏上它的外沿时画在玩家上面；图例菱形按脚下标记格亮灭；HUD
def finQ (s : State) (x : Rs) (t : Nat) : Fr :=
  let nu := cv x.B x.r x.c
  let cm := cel x.r x.c
  let tops : List (Nat × Nat) := (match x.pc with | some p => [(ringM p.1 p.2.1, rect p.1 (p.1 + 3) p.2.1 (p.2.1 + 3))] | none => []) ++
    (match x.bx with | some b => [(boxM b, boxIn b)] | none => [])
  let Q := tops.foldl (fun G y => if cm &&& y.1 != 0 && cm &&& y.2 != cm then pnt G (cm &&& y.1) 8 else G) (pnt x.B cm 14)
  let Q := if s.k_dia.isSome && (isMarker s.h_u.toNat || isMarker nu) then diamonds s Q nu else Q
  setHud Q (hudRow t (kd s.k_tb))
def stepRules (s : State) (a : Act) : Option State :=
  if hudFull (frS s) then some s else
  let x := act s a
  let t := s.h_t.toNat + wt x.cls
  let Q := finQ s x t

def live (s : State) : Bool := !hudFull (frameOf s)
def mapM (s : State) : Nat := rect 0 61 0 (kd s.k_pc - 1 : Nat)
def agree (G : Fr) (s' : State) (m : Nat) : Bool :=
  (G.p0 ^^^ s'.fb0.toNat) &&& m == 0 && (G.p1 ^^^ s'.fb1.toNat) &&& m == 0 &&
  (G.p2 ^^^ s'.fb2.toNat) &&& m == 0 && (G.p3 ^^^ s'.fb3.toNat) &&& m == 0

-- 某区域里除去玩家挪动前后那两格
def outP (s s' : State) (m : Nat) : Nat := mn (mn m (cel s.me_r s.me_c)) (cel s'.me_r s'.me_c)
def clk (s : State) (a : Act) : Option (Int × Int × Nat) := match a with | Act.a6 x y => some (y, x, pxc (frameOf s) y x) | _ => none

-- 点在面板上色 k 的卡上（不是菱形、不是四色图标——第 5 关它们也有 8/11/15 色的像素）
def onPan (s : State) (a : Act) (k : Nat) : Bool :=

  | some (y, x, v) => x ≥ kd s.k_pc && v == k && live s && !inBox4 s a s.k_icon &&

  | none => false
def meSame (s s' : State) : Bool := s'.me_r == s.me_r && s'.me_c == s.me_c

def bxOf (s : State) : Option (Int × Int × Nat) := bxS s
def diaHit (s : State) (a : Act) : Option ((Int × Int × Nat × Nat) × Bool) :=

  | some (y, x, _) => (dias s).find? (fun d => y ≥ d.1 && y ≤ d.1 + 3 && x ≥ d.2.1 && x ≤ d.2.1 + 3) |>.map (fun d =>

  | none => none
def btnHit (s : State) (a : Act) : Option Nat :=
