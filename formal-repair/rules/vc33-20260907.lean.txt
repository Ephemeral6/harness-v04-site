def stepRules (s : State) (a : Act) : Option State :=
  match a with
  | Act.a6 x y =>
    match tick s with
    | none => none
    | some t =>
      match btnAt s x y with
      | some b => if pullB s b then some (settle t (pulled s b) (czL s)) else some (settle t (frontsL s) (czL s))
      | none =>
        match gaugeAt s x y with
        | some g => if gaugeOn s g then some (settle t (frontsL s) (swapped s g)) else some (settle t (frontsL s) (czL s))
        | none => some (settle t (frontsL s) (czL s))
  | _ => none
