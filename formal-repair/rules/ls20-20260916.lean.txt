-- ---------- 基本量 ----------
def FULL : Int := 42
def kv (o : Option Int) : Int := o.getD (-1)
def kn (o : Option Int) : Nat := (o.getD 0).toNat
def bitOf (k : Nat) : Nat := 1 <<< k
def dropBit (m k : Nat) : Nat := if m.testBit k then m ^^^ (1 <<< k) else m
def minus (m n : Nat) : Nat := m - (m &&& n)          -- 集合差 m \ n

-- 格位：格号 k 的左上角像素 (5·(k/12), 4 + 5·(k%12))
def cellT (k : Nat) : Int := Int.ofNat (5 * (k / 12))
def cellL (k : Nat) : Int := Int.ofNat (4 + 5 * (k % 12))
def kOf (r c : Int) : Nat := (r.toNat / 5) * 12 + (c.toNat - 4) / 5
def kOfN (r c : Nat) : Nat := (r / 5) * 12 + (c - 4) / 5
def meK (s : State) : Nat := kOf s.me_r s.me_c

-- 按位过滤：保留 m 里满足 p 的位（逐个取最高位，次数 = 位数）
def filterBits (p : Nat → Bool) : Nat → Nat → Nat → Nat
  | 0, _, acc => acc
  | fuel + 1, m, acc =>
    if m == 0 then acc
    else
      let k := Nat.log2 m
      filterBits p fuel (m ^^^ (1 <<< k)) (if p k then acc ||| (1 <<< k) else acc)

-- ---------- 迷雾圆（fog_disc）----------
-- 像素 (r,c) 照得到 ⇔ 到 (me_r + 1.5, me_c + 1.5) 的距离 < 20：(2r − 2me_r − 3)² + (2c − 2me_c − 3)² < 1600。
-- 读数判据选成「物件里不含色 5 的像素全是期望值」，圆是凸的，所以一个物件读得出 ⇔ 它的几个角点都照得到。
def sq (x : Int) : Int := x * x
def lit (mr mc r c : Int) : Bool := decide (sq (2 * r - 2 * mr - 3) + sq (2 * c - 2 * mc - 3) < 1600)
def litAll (mr mc : Int) : List (Int × Int) → Bool
  | [] => true
  | p :: ps => lit mr mc p.1 p.2 && litAll mr mc ps
-- 开关 / 染色 / 补给：内 3×3 的四角
def innerLit (mr mc : Int) (k : Nat) : Bool :=
  litAll mr mc [(cellT k + 1, cellL k + 1), (cellT k + 1, cellL k + 3), (cellT k + 3, cellL k + 1), (cellT k + 3, cellL k + 3)]
-- 墙：格中心像素
def centerLit (mr mc : Int) (k : Nat) : Bool := lit mr mc (cellT k + 2) (cellL k + 2)
-- 框：以内 5×5 为中心的 9×9 的四角
def frameLit (mr mc : Int) (k : Nat) : Bool :=
  litAll mr mc [(cellT k - 2, cellL k - 2), (cellT k - 2, cellL k + 6), (cellT k + 6, cellL k - 2), (cellT k + 6, cellL k + 6)]
-- 直线（q = 144·方向 + 格号）：线的两个端点像素
def lineLit (mr mc : Int) (q : Nat) : Bool :=
  let k := q % 144
  let t := cellT k
  let l := cellL k
  match q / 144 with
  | 0 => litAll mr mc [(t, l - 1), (t + 4, l - 1)]
  | 1 => litAll mr mc [(t, l + 5), (t + 4, l + 5)]
  | 2 => litAll mr mc [(t - 1, l), (t - 1, l + 4)]
  | _ => litAll mr mc [(t + 5, l), (t + 5, l + 4)]

-- 这一帧看得见的格：玩家块盖住的那格读不出；迷雾关里还要照得到
def seen (fog : Bool) (km : Nat) (p : Nat → Bool) (m : Nat) : Nat :=
  if fog then filterBits (fun k => k != km && p k) 145 m 0 else dropBit m km

-- ---------- 会动的物件（mover_shifts / paint_orbits）----------
-- 开关编码：种类 + 2×(轴 + 2×(固定坐标 + 64×(lo + 64×(hi + 64×相位))))；轴 0 沿行走、1 沿列走。
-- 在 lo..hi 之间每步挪 5 像素、到端掉头：u = (t + 相位) mod 2n，坐标 = lo + 5·(u ≤ n ? u : 2n − u)。返回 (种类, 格号)。
def moverCell (code t : Nat) : Nat × Nat :=
  let v := code / 2
  let fixed := v / 2 % 64
  let lo := v / 128 % 64
  let hi := v / 8192 % 64
  let n := (hi - lo) / 5
  let u := (t + v / 524288) % (2 * n)
  let x := lo + 5 * (if u ≤ n then u else 2 * n - u)
  (code % 2, if v % 2 == 0 then kOfN fixed x else kOfN x fixed)

-- 染色物件轨道：绕中心格的 8 个邻格，次序 下、左下、左、左上、上、右上、右、右下，每步一格
def orbitOff : List Int := [12, 11, -1, -13, -12, -11, 1, 13]
def orbitCell (code t : Nat) : Nat := (Int.ofNat (code % 144) + orbitOff.getD ((code / 144 + t) % 8) 0).toNat

structure Objs where
  tog : Nat
  nc : Nat
  pnt : Nat

def addMover (o : Objs) (code : Int) (t : Nat) : Objs :=
  if code < 0 then o
  else
    let kc := moverCell code.toNat t
    if kc.1 == 0 then { o with tog := o.tog ||| bitOf kc.2 } else { o with nc := o.nc ||| bitOf kc.2 }

-- 第 t 步时的十字开关 / 非十字开关 / 染色物件（不动的 + 会动的）
def objsAt (s : State) (t : Nat) : Objs :=
  let o := addMover (addMover { tog := kn s.k_tog, nc := kn s.k_nc, pnt := kn s.k_pnt } (kv s.k_mv0) t) (kv s.k_mv1) t
  if kv s.k_orb < 0 then o else { o with pnt := o.pnt ||| bitOf (orbitCell (kv s.k_orb).toNat t) }

-- 还在的补给、还在的框
def trueRef (s : State) : Nat := minus (kn s.k_ref) s.h_rg.toNat
def framesLeft (s : State) (fg : Nat) : List Nat :=
  [kv s.k_frm0, kv s.k_frm1].filterMap (fun c => if c < 0 then none else if fg.testBit (c.toNat % 144) then none else some c.toNat)
def frameAt (frs : List Nat) (b : Nat) : Option Nat := frs.find? (fun c => c % 144 == b)

-- ---------- 读数怎么由真实局面决定（迷雾圆 + 玩家块遮挡）----------
def seenFrames (fog : Bool) (mr mc : Int) (frs : List Nat) : List Int :=
  (frs.filter (fun c => !fog || frameLit mr mc (c % 144))).map Int.ofNat

def render (s : State) : State :=
  let mr := s.me_r
  let mc := s.me_c
  let fog := kv s.k_fog == 1
  let km := kOf mr mc
  let o := objsAt s s.h_t.toNat
  let fr := seenFrames fog mr mc (framesLeft s s.h_fg.toNat)
  { s with
    wall_mask := Int.ofNat (seen fog km (centerLit mr mc) (kn s.k_wall)),
    line_mask := Int.ofNat (if fog then filterBits (lineLit mr mc) 600 (kn s.k_lines) 0 else kn s.k_lines),
    tog_mask := Int.ofNat (seen fog km (innerLit mr mc) o.tog),
    nc_mask := Int.ofNat (seen fog km (innerLit mr mc) o.nc),
    paint_mask := Int.ofNat (seen fog km (innerLit mr mc) o.pnt),
    refill_mask := Int.ofNat (seen fog km (innerLit mr mc) (trueRef s)),
    frame0 := fr.getD 0 (-1),
    frame1 := fr.getD 1 (-1) }

-- ---------- HUD ----------
def gb (p k : Nat) : Nat := if p.testBit k then 1 else 0
-- 顺时针转 90°：新 (i,j) = 旧 (2−j, i)，图样位 3i+j
def rotCW (p : Nat) : Nat :=
  gb p 6 + 2 * gb p 3 + 4 * gb p 0 + 8 * gb p 7 + 16 * gb p 4 + 32 * gb p 1 + 64 * gb p 8 + 128 * gb p 5 + 256 * gb p 2
def rotN : Nat → Nat → Nat
  | 0, p => p
  | n + 1, p => rotN n (rotCW p)
-- 图样册（非十字开关）：6 个基本形，按下非十字开关 = 同一朝向换成册里的下一形。
-- .#./.#./###、#.#/#.#/###、.##/#.#/.#.、.#./##./.##、###/..#/#.#、##./.##/#.#（6×4 个朝向两两不同，cat_check.py）
def CATALOG : List Nat := [466, 493, 174, 410, 359, 371]
def catAt (i : Nat) : Nat := CATALOG.getD (i % 6) 0
def catPairs : List (Nat × Nat) := (List.range 6).flatMap (fun i => (List.range 4).map (fun r => (i, r)))
def catNext (p : Nat) : Option Nat :=
  (catPairs.find? (fun ir => rotN ir.2 (catAt ir.1) == p)).map (fun ir => rotN ir.2 (catAt (ir.1 + 1)))
-- 染色：墨色 12 → 9 → 14 → 8 → 12
def inkNext (c : Nat) : Option Nat :=
  if c == 12 then some 9 else if c == 9 then some 14 else if c == 14 then some 8 else if c == 8 then some 12 else none
def hudPat (s : State) : Nat := s.hud.toNat % 512
def hudInk (s : State) : Nat := s.hud.toNat / 512

-- ---------- 落点（passable / move_*）----------
def delta : Act → Option (Int × Int)
  | .a1 => some (-1, 0)
  | .a2 => some (1, 0)
  | .a3 => some (0, -1)
  | .a4 => some (0, 1)
  | _ => none

def nbr (k : Nat) (d : Int × Int) : Option Nat :=
  let i := Int.ofNat (k / 12) + d.1
  let j := Int.ofNat (k % 12) + d.2
  if 0 ≤ i ∧ i < 12 ∧ 0 ≤ j ∧ j < 12 then some (i.toNat * 12 + j.toNat) else none

-- 落点贴着直线：左侧线推向右、右侧推向左、上侧推向下、下侧推向上
def pushOf (lines k : Nat) : Option (Int × Int) :=
  if lines.testBit k then some (0, 1)
  else if lines.testBit (144 + k) then some (0, -1)
  else if lines.testBit (288 + k) then some (1, 0)
  else if lines.testBit (432 + k) then some (-1, 0)
  else none

-- 沿推的方向一格格滑，下一格是墙、框或出界就停（进框只能直接一步走进去）
def slide (wall : Nat) (frs : List Nat) (d : Int × Int) : Nat → Nat → Nat
  | 0, x => x
  | fuel + 1, x =>
    match nbr x d with
    | none => x
    | some y => if wall.testBit y || (frameAt frs y).isSome then x else slide wall frs d fuel y

structure Plan where
  kind : Nat     -- 0 撞墙不动；1 撞框不动（HUD 与框不匹配）；2 挪动；3 掉命重开；4 走进最后一个框（换关）
  x : Nat        -- 落点格号
  ent : Bool     -- 这一步走进了一个匹配的框

def plan (s : State) (a : Act) : Option Plan :=
  match delta a with
  | none => none
  | some d =>
    let km := meK s
    let wall := kn s.k_wall
    let frs := framesLeft s s.h_fg.toNat
    match nbr km d with
    | none => some (if s.energy == 0 then { kind := 3, x := km, ent := false } else { kind := 0, x := km, ent := false })
    | some b =>
      if s.energy == 0 && !((trueRef s).testBit b && !wall.testBit b) then some { kind := 3, x := km, ent := false }
      else if wall.testBit b then some { kind := 0, x := km, ent := false }
      else
        match frameAt frs b with
        | some c =>
          if Int.ofNat (c / 144) == s.hud then some { kind := (if frs.length == 1 then 4 else 2), x := b, ent := true }
          else some { kind := 1, x := km, ent := false }
        | none =>
          match pushOf (kn s.k_lines) b with
          | none => some { kind := 2, x := b, ent := false }
          | some e => some { kind := 2, x := slide wall frs e 16 b, ent := false }

def planKind (s : State) (a : Act) : Nat := match plan s a with | some p => p.kind | none => 9
def planX (s : State) (a : Act) : Nat := match plan s a with | some p => p.x | none => meK s
def planEnt (s : State) (a : Act) : Bool := match plan s a with | some p => p.ent | none => false
def inLevel (s : State) (a : Act) : Bool := decide (planKind s a ≤ 2)
def moved (s : State) (a : Act) : Bool := planKind s a == 2
def tNext (s : State) (a : Act) : Nat := s.h_t.toNat + (if moved s a then 1 else 0)
def objsNext (s : State) (a : Act) : Objs := objsAt s (tNext s a)
def onCross (s : State) (a : Act) : Bool := moved s a && (objsNext s a).tog.testBit (planX s a)
def onNC (s : State) (a : Act) : Bool := moved s a && (objsNext s a).nc.testBit (planX s a)
def onPaint (s : State) (a : Act) : Bool := moved s a && (objsNext s a).pnt.testBit (planX s a)
def onRefill (s : State) (a : Act) : Bool := moved s a && (trueRef s).testBit (planX s a)

-- ---------- 各量的下一值 ----------
def energyNext (s : State) (a : Act) : Int :=
  if planKind s a == 1 then s.energy
  else if onRefill s a then FULL
  else if (onCross s a || onNC s a) && kv s.k_tfree == 1 then s.energy
  else max 0 (s.energy - kv s.k_cost)

def hudNext (s : State) (a : Act) : Option Int :=
  let pat : Option Nat := if onCross s a then some (rotCW (hudPat s)) else if onNC s a then catNext (hudPat s) else some (hudPat s)
  let ink : Option Nat := if onPaint s a then inkNext (hudInk s) else some (hudInk s)
  match pat, ink with
  | some p, some i => some (Int.ofNat (p + 512 * i))
  | _, _ => none

def rgNext (s : State) (a : Act) : Int := if onRefill s a then Int.ofNat (s.h_rg.toNat ||| bitOf (planX s a)) else s.h_rg
def fgNext (s : State) (a : Act) : Int := if planEnt s a then Int.ofNat (s.h_fg.toNat ||| bitOf (planX s a)) else s.h_fg

def afterDeath (s : State) : State :=
  let k := kn s.k_start
  render { s with lives := s.lives - 1, energy := FULL, hud := kv s.k_hud0, me_r := cellT k, me_c := cellL k,
                  h_t := 0, h_rg := 0, h_fg := 0 }

def afterMove (s : State) (a : Act) (hd : Int) : State :=
  let x := planX s a
  render { s with
    me_r := cellT x
    me_c := cellL x
    h_t := Int.ofNat (tNext s a)
    hud := hd
    energy := energyNext s a
    h_rg := rgNext s a
    h_fg := fgNext s a }

def stepRules (s : State) (a : Act) : Option State :=
  let k := planKind s a
  if k == 3 then some (afterDeath s)
  else if k ≤ 2 then (hudNext s a).map (afterMove s a)
  else none      -- 4：换关（新关布局是游戏内容，查 worldRows）；9：不是移动动作

-- ---------- 规则（一条概念一条）----------
-- 读数在 s' 里该是什么：按 s' 的玩家位置与隐藏量、本关布局重算（看得见的那部分）
def seenAfter (s' : State) (m : Nat) (p : Int → Int → Nat → Bool) : Nat :=
  seen (kv s'.k_fog == 1) (meK s') (p s'.me_r s'.me_c) m
def sameMain (s s' : State) : Bool :=
  s'.wall_mask == s.wall_mask && s'.line_mask == s.line_mask && s'.tog_mask == s.tog_mask && s'.nc_mask == s.nc_mask
    && s'.paint_mask == s.paint_mask && s'.refill_mask == s.refill_mask && s'.frame0 == s.frame0 && s'.frame1 == s.frame1
    && s'.hud == s.hud && s'.h_t == s.h_t

def meAt (s' : State) (x : Nat) : Bool := s'.me_r == cellT x && s'.me_c == cellL x

-- move_*：落点 = plan 算的格（撞墙 / 撞框原地；贴线就滑到底）；离开的那格里的持久物件（开关 / 染色）重新露出来
def leftRestored (s : State) (a : Act) (s' : State) : Bool :=
  let km := meK s
  let o := objsNext s a
  let vis := kv s'.k_fog != 1 || innerLit s'.me_r s'.me_c km
  !moved s a || (s'.tog_mask.toNat.testBit km == (vis && o.tog.testBit km)
                 && s'.nc_mask.toNat.testBit km == (vis && o.nc.testBit km)
                 && s'.paint_mask.toNat.testBit km == (vis && o.pnt.testBit km))
def moveEffect (s : State) (a : Act) (s' : State) : Bool := meAt s' (planX s a) && leftRestored s a s'

def rules : List Rule :=
  [ { name := "move_up", guard := fun s a => a == Act.a1 && inLevel s a, effect := moveEffect }
  , { name := "move_down", guard := fun s a => a == Act.a2 && inLevel s a, effect := moveEffect }
  , { name := "move_left", guard := fun s a => a == Act.a3 && inLevel s a, effect := moveEffect }
  , { name := "move_right", guard := fun s a => a == Act.a4 && inLevel s a, effect := moveEffect }
    -- 动 ⇔ 目标格可进（非墙、非不匹配的框）；不动时主区读数全不变（能量另管）
  , { name := "passable", guard := inLevel,
      effect := fun s a s' => if moved s a then !(s'.me_r == s.me_r && s'.me_c == s.me_c)
                               else s'.me_r == s.me_r && s'.me_c == s.me_c && sameMain s s' }
    -- 每步扣本关的 k_cost：撞墙也扣，撞框不扣，第 0 关踩开关不扣；到 0 为止
  , { name := "energy_cost", guard := fun s a => inLevel s a && !onRefill s a,
      effect := fun s a s' => s'.energy == energyNext s a }
    -- 挪进不是开关、不是补给的格：能量严格减少；撞框不动：能量不变
  , { name := "energy_tick", guard := fun s a => (moved s a || planKind s a == 1) && !onRefill s a && !onCross s a && !onNC s a,
      effect := fun s a s' => if moved s a then decide (s'.energy < s.energy) else s'.energy == s.energy }
    -- 能量 0 再动（目标不是补给）：命数减 1、能量回满、回到本关开局格、HUD 回开局、会动物件的钟归零，
    -- 整个局面（补给、框、开关位置、迷雾里的读数）回到本关开局——最后一项用 keyEq 比全部变化字段
  , { name := "death_reset", guard := fun s a => planKind s a == 3,
      effect := fun s _ s' => s'.lives == s.lives - 1 && s'.energy == FULL && meAt s' (kn s.k_start) && s'.level == s.level
                               && s'.hud == kv s.k_hud0 && s'.h_t == 0 && keyEq (afterDeath s) s' }
    -- 走进补给：能量回满（能量 0 时也一样，不掉命）
  , { name := "refill", guard := onRefill, effect := fun _ _ s' => s'.energy == FULL }
    -- 吃掉的补给记进 h_rg、从读数里消失；没吃的步补给不变
  , { name := "refill_vanishes", guard := inLevel,
      effect := fun s a s' => s'.h_rg == rgNext s a
                               && s'.refill_mask.toNat == seenAfter s' (minus (kn s.k_ref) (rgNext s a).toNat) innerLit }
    -- 踩十字开关：HUD 图样顺时针转 90°，墨色不变
  , { name := "toggle_rotates_hud", guard := onCross,
      effect := fun s _ s' => s'.hud == Int.ofNat (rotCW (hudPat s) + 512 * hudInk s) }
    -- 踩非十字开关：HUD 换成图样册里同朝向的下一形，墨色不变
  , { name := "toggle_maps_hud", guard := onNC,
      effect := fun s _ s' => (catNext (hudPat s)).map (fun p => Int.ofNat (p + 512 * hudInk s)) == some s'.hud }
    -- HUD 图样变 ⇔ 踩到开关（踩染色那步不在内）
  , { name := "toggle_changes_hud", guard := fun s a => inLevel s a && !onPaint s a,
      effect := fun s a s' => (s'.hud.toNat % 512 != hudPat s) == (onCross s a || onNC s a) }
    -- 踩染色：墨色换一档、图样不变；不踩：墨色不变
  , { name := "paint_recolors_hud", guard := inLevel,
      effect := fun s a s' => if onPaint s a then inkNext (hudInk s) == some (s'.hud.toNat / 512)
                                                && s'.hud.toNat % 512 == hudPat s
                               else s'.hud.toNat / 512 == hudInk s }
    -- 会动的开关：钟 h_t 只在挪动的步加一；读数里的开关在轨道算出的位置（被玩家盖住、被迷雾遮住的除外）
  , { name := "mover_shifts", guard := fun s a => inLevel s a && (decide (kv s.k_mv0 ≥ 0) || decide (kv s.k_mv1 ≥ 0)),
      effect := fun s a s' => s'.h_t == Int.ofNat (tNext s a)
                               && s'.tog_mask.toNat == seenAfter s' (objsNext s a).tog innerLit
                               && s'.nc_mask.toNat == seenAfter s' (objsNext s a).nc innerLit }
    -- 染色物件：不动的关原地；第 5 关沿轨道每步一格
  , { name := "paint_orbits", guard := fun s a => inLevel s a && (decide (kv s.k_orb ≥ 0) || decide (kv s.k_pnt > 0)),
      effect := fun s a s' => s'.paint_mask.toNat == seenAfter s' (objsNext s a).pnt innerLit }
    -- 走进匹配的框（不是最后一个）：那个框消失、HUD 不变；其余步框原样
  , { name := "frame_consumed", guard := fun s a => inLevel s a && decide (kv s.k_frm0 ≥ 0),
      effect := fun s a s' => s'.h_fg == fgNext s a
                               && [s'.frame0, s'.frame1] ==
                                  (seenFrames (kv s'.k_fog == 1) s'.me_r s'.me_c (framesLeft s (fgNext s a).toNat) ++ [-1, -1]).take 2
                               && (!planEnt s a || s'.hud == s.hud) }
    -- 换关 ⇔ 走进最后一个框且 HUD（图样 + 墨色）与它一致；换关时关卡号加一
  , { name := "levelup_iff_match", guard := fun s a => (delta a).isSome && decide (s.energy > 0),
      effect := fun s a s' => (s'.level != s.level) == (planKind s a == 4) && (planKind s a != 4 || s'.level == s.level + 1) }
    -- 迷雾关：读得出的墙、直线只有照得到的那些
  , { name := "fog_disc", guard := fun s a => kv s.k_fog == 1 && decide (planKind s a ≤ 3),
      effect := fun _ _ s' => s'.wall_mask.toNat == seenAfter s' (kn s'.k_wall) centerLit
                               && s'.line_mask.toNat == filterBits (lineLit s'.me_r s'.me_c) 600 (kn s'.k_lines) 0 }
  ]

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

-- 目标：第 6 关（最后一关）过完，关卡号到 7（WIN）
def isGoal (s : State) : Bool := s.level == 7
-- ==== 规则段 END ====

-- ==== 定理段（脚本生成）====
-- 规则与全部观测一致（内核判定，公理只剩 propext 一类）。
theorem consistent : checkRules rules corpus = true := by decide +kernel

-- 逐步重算整条路线：从第 0 关开局起，每一步 step 的结果在 keyEq 的全部变化字段上等于观测，最后一步到 WIN。
theorem faithful : (faithfulRun step start routeObs).map isGoal = some true := by decide +kernel
