def bitOf (r c : Int) : Nat := (r * 64 + c).toNat
-- 计数与质心用 SWAR：4096 位按行切成 64 个 64 位「格道」，一次大数运算同时处理 64 行。
-- 逐位循环（带累加器）在内核里会把未求值的累加式越堆越深，实测 consistent 吃到 10 GB 以上被杀。
def W64 : Nat := (1 <<< 64) - 1
def rep64 (n : Nat) : Nat := ((1 <<< (64 * n)) - 1) / W64          -- 每 64 位放一个 1、共 n 个
def lanes (w : Nat) : Nat := rep64 64 * w                            -- 64 位字 w 复制到 64 个格道
-- 每个格道（= 每一行）里 1 的个数，落在该格道的低字节
def rowCnt (x : Nat) : Nat :=
  let x1 := x - ((x >>> 1) &&& lanes (W64 / 3))
  let x2 := (x1 &&& lanes (W64 / 5)) + ((x1 >>> 2) &&& lanes (W64 / 5))
  let x3 := (x2 + (x2 >>> 4)) &&& lanes (W64 / 17)
  let x4 := x3 + (x3 >>> 8)
  let x5 := x4 + (x4 >>> 16)
  let x6 := x5 + (x5 >>> 32)
  x6 &&& lanes 255
-- 64 个格道的值加总（乘以全 1 格道模板后取第 63 格道）
def laneSum (y : Nat) : Nat := ((y * rep64 64) >>> (64 * 63)) &&& W64

def minus (a b : Nat) : Nat := a ^^^ (a &&& b)

-- 双倍相对坐标下的 90 度转：逆时针 (dr,dc)->(-dc,dr)（开口下标 +2），顺时针 (dr,dc)->(dc,-dr)
def ccwRel : Nat → Int × Int → Int × Int
  | 0, p => p
  | n + 1, p => ccwRel n (-p.2, p.1)
def cwRel : Nat → Int × Int → Int × Int
  | 0, p => p

  let x := ccwRel q (2 * r - hr2 s, 2 * c - hc2 s)
  ((hr2 s + x.1) / 2, (hc2 s + x.2) / 2)

-- 行段表整体绕容器中心转 q 个直角：每段两个端点转过去，仍是一段（横段转成竖段），一段一次运算

    let a := rotPt s q t.1 t.2.1

    after_pour_m_c9 := if pc == 9 then Int.ofNat hp else 0,
    after_pour_m_c11 := if pc == 11 then Int.ofNat hp else 0,
    after_pour_m_c12 := if pc == 12 then Int.ofNat hp else 0,
    after_pour_m_c14 := if pc == 14 then Int.ofNat hp else 0,
    after_pour_m_c15 := if pc == 15 then Int.ofNat hp else 0 }

-- 关卡内一步：
--   A1..A4  orbit_turn_rule 给新开口（死键不转），rot8_apply 给新姿势，小杯随工位出没；
--   A5      a5_paints_cup_side 的半平面涂进容器；
--   A6      点环 -> a6_swatch_sets_paint 换颜料 + underline_relocates_on_recolor 挪下划线；
--           点小杯描边 -> a6_spout_rim_stamps 盖章；其余 -> a6_dead_outside_swatches 死键；
--   每一步 bar_rule 的钟 h_k 加一。
--   levelup_needs_target_pic 判定这一步会过关（涂完的容器 == 面板目标图）时返回 none：
--   过关后新关的面板、色板是游戏内容，只能查 worldRows；所以路线上每个关卡内步都由 Lean 判定「不过关」。
--   RESET 不在这里算（路线上没有 RESET）。
def stepRules (s : State) (a : Act) : Option State :=
  let k := s.h_k + 1
  let d := s.open_dir
  let pc := s.paint_color
  let pm := paintRegion s a
  if hitsPic s pm pc then none else
  match a with
  | Act.a5 => some (mkNext s d pc k pm s.underline_1 s.underline_2)
  | Act.a6 x y =>
    match ringHit s (Int.ofNat y) (Int.ofNat x) with
    | some rc => some (mkNext s d rc.2 k 0 rc.1 (rc.1 + 4))
    | none => some (mkNext s d pc k pm s.underline_1 s.underline_2)
  | Act.reset => none
  | Act.other => none
  | _ =>
    match turnOf s a with
    | some nd => some (mkNext s nd pc k 0 s.underline_1 s.underline_2)
    | none => some (mkNext s d pc k 0 s.underline_1 s.underline_2)

-- ---------------- 规则用的判据小件 ----------------
def sameLv (s t : State) : Bool := s.level == t.level

-- 两张掩码只在容器矩形里不同
def sameOutside (s : State) (a b : Int) : Bool := within (M a ^^^ M b) (vrect s)

def panelEq (s t : State) : Bool :=
  s.panel_cells_m == t.panel_cells_m && s.panel_cells_m_c0 == t.panel_cells_m_c0
  && s.panel_cells_m_c4 == t.panel_cells_m_c4 && s.panel_cells_m_c5 == t.panel_cells_m_c5
  && s.panel_cells_m_c8 == t.panel_cells_m_c8 && s.panel_cells_m_c9 == t.panel_cells_m_c9
  && s.panel_cells_m_c11 == t.panel_cells_m_c11 && s.panel_cells_m_c12 == t.panel_cells_m_c12
  && s.panel_cells_m_c14 == t.panel_cells_m_c14 && s.panel_cells_m_c15 == t.panel_cells_m_c15

def ringEq (s t : State) : Bool :=
  rings s == rings t && s.swatches_n == t.swatches_n && s.paints_n == t.paints_n

-- 舞台上「所有杯」占的格：描边（含小杯）+ 大杯 + 小杯 + 各杯里的颜料
def cupsOf (s : State) : Nat := M s.ink_m ||| M s.obj_m ||| M s.spout_m ||| M s.cup_liquid_m

-- 除预算条（bar 与它的钟）之外，全部可比字段不变
def stillExceptBar (s t : State) : Bool := keyEq { t with bar := s.bar, h_k := s.h_k } s

-- 下划线恰好在「芯色 == 颜料色」的唯一那个环下
def underlineOk (g : State) : Bool :=
  match ringLeftsOf g g.paint_color with
  | [l] => g.underline_1 == l && g.underline_2 == l + 4
  | _ => false

-- 杯质心要的 (行和, 列和, 格数)：
--   行和 = Σ 行号 x 该行格数：各行格数（rowCnt）乘以「第 k 格道放 k」的模板，第 63 格道得 Σ (63 - 行号) x 格数；
--   列和 = Σ_j 2^j x #{格 : 列号第 j 位为 1}（按列号二进制位拆开数）。
def rampLanes : Nat := (List.range 64).foldl (fun a k => a ||| (k <<< (64 * k))) 0
def colBit (j : Nat) : Nat :=      -- 64 位字里「下标第 j 位为 1」的那些位
  if j == 0 then 2 * (W64 / 3) else if j == 1 then 4 * (W64 / 5) else if j == 2 then 16 * (W64 / 17)
  else if j == 3 then (W64 / 257) <<< 8 else if j == 4 then (W64 / 65537) <<< 16 else (W64 / ((1 <<< 32) + 1)) <<< 32
def cen (m : Int) : Int × Int × Int :=
  let x := M m
  let y := rowCnt x
  let n := laneSum y
  let rs := 63 * n - (((y * rampLanes) >>> (64 * 63)) &&& W64)
  let cs := (List.range 6).foldl (fun a j => a + (1 <<< j) * popN (x &&& lanes (colBit j))) 0
  (Int.ofNat rs, Int.ofNat cs, Int.ofNat n)

-- 与向量夹角最小的八方向（_lib.dir8）：比较带符号的 (dot/|u|)^2，两边乘 2 全是整数
def sgnSq (x : Int) : Int := if x < 0 then -(x * x) else x * x
def dirScore (v : Int × Int) (d : Int) : Int :=
  let u := dirVec d
  let dot := v.1 * u.1 + v.2 * u.2
  if u.1 != 0 && u.2 != 0 then sgnSq dot else 2 * sgnSq dot
def best8 (v : Int × Int) : Int :=
  [1, 2, 3, 4, 5, 6, 7].foldl (fun b i => if dirScore v b < dirScore v i then i else b) 0

-- open_faces_vessel：从杯质心指向容器中心的八方向 == 开口方向（乘 2n 去分母）
def facesHub (g : State) : Bool :=
  let q := cen g.obj_m
  q.2.2 == 0 || best8 (q.2.2 * hr2 g - 2 * q.1, q.2.2 * hc2 g - 2 * q.2.1) == g.open_dir

-- keys_are_screen_dirs：质心位移（乘 n1*n2 去分母）的主分量方向；两分量等长返回 (0,0)
def mainDir (s t : State) : Int × Int :=
  let p := cen s.obj_m
  let q := cen t.obj_m
  let dr := q.1 * p.2.2 - p.1 * q.2.2
  let dc := q.2.1 * p.2.2 - p.2.1 * q.2.2
  let adr := if dr < 0 then -dr else dr
  let adc := if dc < 0 then -dc else dc
  if adr < adc then (0, if 0 < dc then 1 else -1)
  else if adc < adr then (if 0 < dr then 1 else -1, 0)
  else (0, 0)

-- 先查换关表（worldRows 只有换关那几步），查不到交给 stepRules
def step (s : State) (a : Act) : Option State :=
  match worldRows.find? (fun w => keyEq w.s s && w.a == a) with
  | some w => some w.s'
  | none => stepRules s a

theorem consistent : checkRules rules corpus = true := by decide +kernel

-- 从第 0 关开局逐步重算整条路线，每一步都在 keyEq 的全部变化字段上等于观测，末步到达 WIN。
theorem faithful : (faithfulRun step start routeObs).map isGoal = some true := by decide +kernel
