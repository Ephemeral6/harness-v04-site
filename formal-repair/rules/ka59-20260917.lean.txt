-- ---------- 解码 ----------
def decObj (v : Int) : Pc :=
  let n := v.toNat
  let pos := n / 64
  ⟨Int.ofNat (pos / 64), Int.ofNat (pos % 64), 3 * ((n / 16) % 4), 3 * ((n / 4) % 4), 0, n % 4, 0, 0, 0, false⟩

def decBare (v : Int) : Pc :=
  let n := v.toNat
  let rest := n >>> 81
  let pos := rest / 256
  ⟨Int.ofNat (pos / 64), Int.ofNat (pos % 64), (rest / 16) % 16, rest % 16, n &&& ((1 <<< 81) - 1), 3, 0, 0, 0, false⟩

def decGate (v : Int) : Pc :=
  let n := v.toNat
  let pos := n >>> 21
  ⟨Int.ofNat (pos / 64), Int.ofNat (pos % 64), (n / 32768) % 64, (n / 512) % 64, 0, 4, (n / 64) % 8, 0, n % 64, false⟩

-- ---------- 格子掩码（第 r*64+c 位；棋盘 63 行 64 列） ----------
def ones (k : Nat) : Nat := (1 <<< k) - 1
/-- h 行、每行最低位一个 1。 -/
def rep (h : Nat) : Nat := ((1 <<< (64 * h)) - 1) / ((1 <<< 64) - 1)

def patRows (pat w : Nat) : Nat → Nat
  | 0 => 0
  | i + 1 => patRows pat w i ||| (((pat >>> (i * w)) &&& ones w) <<< (64 * i))

def maskP (p : Pc) : Nat :=
  let body := if p.pat == 0 then ones p.w * rep p.h else patRows p.pat p.w p.h
  body <<< (p.r * 64 + p.c).toNat

def inb (p : Pc) : Bool := 0 ≤ p.r && 0 ≤ p.c && p.r + p.h ≤ 63 && p.c + p.w ≤ 64

/-- 朝 d 挪 k 格（1上 2下 3左 4右）。 -/
def mvk (k : Int) (p : Pc) (d : Nat) : Pc :=
  if d == 1 then { p with r := p.r - k } else if d == 2 then { p with r := p.r + k }
  else if d == 3 then { p with c := p.c - k } else if d == 4 then { p with c := p.c + k } else p

/-- 一个格位 = 3 格。 -/
def mv (p : Pc) (d : Nat) : Pc := mvk 3 p d

/-- 包围盒朝 d 外侧紧贴的一排（出画面的不要；第 63 行是预算条，不算棋盘）。 -/
def stripM (p : Pc) (d : Nat) : Nat :=
  if d == 1 then (if p.r ≥ 1 then ones p.w <<< ((p.r - 1) * 64 + p.c).toNat else 0)
  else if d == 2 then (if p.r + p.h ≤ 62 then ones p.w <<< ((p.r + p.h) * 64 + p.c).toNat else 0)
  else if d == 3 then (if p.c ≥ 1 then rep p.h <<< (p.r * 64 + p.c - 1).toNat else 0)
  else (if p.c + p.w ≤ 63 then rep p.h <<< (p.r * 64 + p.c + p.w).toNat else 0)

def minus (a b : Nat) : Nat := a ^^^ (a &&& b)

/-- 朝 d 迈一个格位：出画面，或新占的格子里有 stop 色。 -/
def aheadBlocked (p : Pc) (d stop : Nat) : Bool :=
  let q := mv p d
  !inb q || minus (maskP q) (maskP p) &&& stop != 0

-- ---------- 推法（push_kick / push_shove / bare_pushed / gate_pushed / gate_strike 共用） ----------
def KICK : Nat := 5
def STRIKE : Nat := 3
def REACH : Int := 9      -- 扫击够得着的空隙：3 个格位以内

def kickN (d stop : Nat) : Nat → Pc → Pc
  | 0, p => p
  | k + 1, p => if aheadBlocked p d stop then p else kickN d stop k (mv p d)

/-- 落点压着色15 就接着同向滑，直到完全离开带子（途中撞色2 / 画面边就停）。 -/
def slide (d stop band : Nat) : Nat → Pc → Pc
  | 0, p => p
  | k + 1, p => if maskP p &&& band == 0 then p else if aheadBlocked p d stop then p else slide d stop band k (mv p d)

/-- 被推（或被扫）的东西：同向走至多 k 个格位，撞色2 / 画面边即停；落点压带就滑出带子。 -/
def kick (d stop band k : Nat) (p : Pc) : Pc := slide d stop band 25 (kickN d stop k p)

/-- 落点压到的东西朝同向让开：让到不重叠、自己也不压带为止（撞墙就停）。 -/
def makeWay (d stop band land : Nat) : Nat → Pc → Pc
  | 0, p => p
  | k + 1, p =>
    let m := maskP p
    if m &&& land == 0 && m &&& band == 0 then p
    else if aheadBlocked p d stop then p else makeWay d stop band land k (mv p d)

/-- 连锁让位：队列里每个刚落地的东西，压到谁谁就让开，让开的再进队。 -/
def chain (d stop band : Nat) : Nat → List Nat → List Pc → List Pc
  | 0, _, ps => ps
  | _ + 1, [], ps => ps
  | k + 1, src :: q, ps =>
    let land := maskP (ps.getD src pc0)
    let st := (List.range ps.length).foldl (fun (acc : List Pc × List Nat) j =>
      if j == src then acc else
      let pj := acc.1.getD j pc0
      if maskP pj &&& land != 0 then (acc.1.set j (makeWay d stop band land 25 pj), acc.2 ++ [j]) else acc) (ps, q)
    chain d stop band k st.2 st.1

/-- 贴着推手那一排的东西按 kick 走，再连锁让位。others 是被操纵物体之外的带壳物体与裸物体。 -/
def pushPieces (d stop band out : Nat) (others : List Pc) : List Pc :=
  let idx := List.range others.length
  let hit := idx.filter (fun j => maskP (others.getD j pc0) &&& out != 0)
  let kicked := hit.foldl (fun (ps : List Pc) j => ps.set j (kick d stop band KICK (ps.getD j pc0))) others
  chain d stop band 40 hit kicked

-- ---------- 门带：数拍子、扫击 ----------
def slices (g : Pc) : Nat := if g.d == 1 || g.d == 2 then g.h else g.w

/-- 每个方向动作长一层；差一层就长满时归零。 -/
def tick (g : Pc) : Pc :=
  if g.f + 1 ≥ slices g then { g with f := 0, wrap := true } else { g with f := g.f + 1, wrap := false }

def lowRow (m : Nat) : Nat := Nat.log2 (m ^^^ (m &&& (m - 1))) / 64
def highRow (m : Nat) : Nat := Nat.log2 m / 64

/-- 一片格子在第 r0.. 行的 h 行里，按列压成一行。 -/
def colsOf (m : Nat) (r0 : Nat) : Nat → Nat
  | 0 => 0
  | i + 1 => colsOf m r0 i ||| ((m >>> (64 * (r0 + i))) &&& ones 64)

def rectM (r0 r1 c0 c1 : Int) : Nat :=
  if r1 < r0 || c1 < c0 then 0 else (ones (c1 - c0 + 1).toNat * rep (r1 - r0 + 1).toNat) <<< (r0 * 64 + c0).toNat

/-- 这东西在归零带子长过去那一头的前方、空隙不到 3 个格位、中间没有色2/色15 隔着。
    loose：只看和带子重叠的那几列（横带则那几行）上的格子（gate_idle_strike 的读法；对长方形两种读法一样）。 -/
def frontClear (g p : Pc) (walls : Nat) (loose : Bool) : Bool :=
  let gr1 := g.r + g.h - 1
  let gc1 := g.c + g.w - 1
  let pr1 := p.r + p.h - 1
  let pc1 := p.c + p.w - 1
  if g.d == 1 || g.d == 2 then
    if pc1 < g.c || p.c > gc1 then false else
    let a := max g.c p.c
    let b := min gc1 pc1
    let m := if loose then maskP p &&& ((ones (b - a + 1).toNat <<< a.toNat) * rep 63) else 0
    if loose && m == 0 then false else
    let lo : Int := if loose then Int.ofNat (lowRow m) else p.r
    let hi : Int := if loose then Int.ofNat (highRow m) else pr1
    let front := if g.d == 2 then lo > gr1 else hi < g.r
    let gap := if g.d == 2 then p.r - gr1 - 1 else g.r - pr1 - 1
    let btw := if g.d == 2 then rectM (gr1 + 1) (p.r - 1) a b else rectM (pr1 + 1) (g.r - 1) a b
    front && gap < REACH && btw &&& walls == 0
  else
    if pr1 < g.r || p.r > gr1 then false else
    let a := max g.r p.r
    let b := min gr1 pr1
    let m := if loose then colsOf (maskP p &&& (ones (64 * (b - a + 1).toNat) <<< (64 * a.toNat))) p.r.toNat p.h else 0
    if loose && m == 0 then false else
    let lo : Int := if loose then Int.ofNat (Nat.log2 (m ^^^ (m &&& (m - 1)))) else p.c
    let hi : Int := if loose then Int.ofNat (Nat.log2 m) else pc1
    let front := if g.d == 4 then lo > gc1 else hi < g.c
    let gap := if g.d == 4 then p.c - gc1 - 1 else g.c - pc1 - 1
    let btw := if g.d == 4 then rectM a b (gc1 + 1) (p.c - 1) else rectM a b (pc1 + 1) (g.c - 1)
    front && gap < REACH && btw &&& walls == 0

/-- 归零的带子把前方够得着的东西朝带子长的方向扫 3 个格位（同一条推法）。先被操纵物体，再别的带壳物体，再裸物体。 -/
def strikeAll (stop band walls : Nat) (gs : List Pc) (sel : Pc) (os bs : List Pc) : Pc × List Pc × List Pc :=
  gs.foldl (fun (acc : Pc × List Pc × List Pc) g =>
    if !g.wrap then acc else
    let hitP := fun (p : Pc) (loose : Bool) => if frontClear g p walls loose then kick g.d stop band STRIKE p else p
    (hitP acc.1 false, acc.2.1.map (fun p => hitP p true), acc.2.2.map (fun p => hitP p true))) (sel, os, bs)

  | none => sc
  | some q => ⟨{ q with code := 0 }, sc.os.map (fun p => if p == q then { sc.sel with code := 2 } else p), sc.bs, sc.gs⟩

/-- 方向动作的前半：贴着东西就推（自己不动），否则挪一个格位（扫到色2/色15 或出画面就不动）。 -/
def movePhase (sc : Scene) (d walls band : Nat) : Scene :=
  let stop := minus walls band
  let out := stripM sc.sel d
  let occ := occOf (sc.os ++ sc.bs ++ sc.gs)
  if out &&& occ != 0 then
    let ps := pushPieces d stop band out (sc.os ++ sc.bs)
    ⟨sc.sel, ps.take sc.os.length, ps.drop sc.os.length,
     sc.gs.map (fun g => if maskP g &&& out != 0 then kick d stop band KICK g else g)⟩
  else
    let q := mv sc.sel d
    ⟨if inb q && minus (maskP q) (maskP sc.sel) &&& walls == 0 then q else sc.sel, sc.os, sc.bs, sc.gs⟩

/-- 方向动作：前半（推或挪）之后，门带数拍子，再归零扫击。 -/
def doDir (sc : Scene) (d walls band : Nat) : Scene :=
  let stop := minus walls band
  let moved := movePhase sc d walls band
  let gs := moved.gs.map tick
  let st := strikeAll stop band walls gs moved.sel moved.os moved.bs
  ⟨st.1, st.2.1, st.2.2, gs⟩

def gdkOf (s : State) : Nat := (s.k_gd.getD 0).toNat

/-- 动作之后的全部东西（点击或方向动作；别的动作 none）。 -/
def nextScene (s : State) (a : Act) : Option Scene :=
  match sceneOf s (gdkOf s) with
  | none => none
  | some sc =>
    match clickOf a with
    | some yx => some (doClick sc yx.1 yx.2)
    | none => if dirOf a == 0 then none else some (doDir sc (dirOf a) s.walls_m.toNat s.band_m.toNat)

/-- 被带壳物体 / 裸物体 / 门带盖着的格。 -/
def coverOf (sc : Scene) : Nat := occOf (sc.sel :: sc.os ++ sc.bs ++ sc.gs)

def stepRules (s : State) (a : Act) : Option State :=
  match s.k_T, s.k_wall, nextScene s a with
  | some T, some wall, some sc' =>
      let isClick := (clickOf a).isSome
      let ov := sortI ((sc'.sel :: sc'.os).map encObj)
      let bv := sortI (sc'.bs.map encBare)
      let gv := sortG (sc'.gs.map (fun g => (encGate g, if isClick then g.kn else g.d)))
      let gd := (List.range gv.length).foldl (fun acc i => acc ||| (((gv.getD i (0, 0)).2) <<< (3 * i))) 0
      let n := s.h_n.toNat + 1
      some { s with
        objs_0 := padI ov 0, objs_1 := padI ov 1, objs_2 := padI ov 2, objs_3 := padI ov 3,
        bare_0 := padI bv 0, bare_1 := padI bv 1, bare_2 := padI bv 2,
        gates_0 := padI (gv.map Prod.fst) 0, gates_1 := padI (gv.map Prod.fst) 1, gates_2 := padI (gv.map Prod.fst) 2,
        h_gd := Int.ofNat gd, h_n := Int.ofNat n,
        lit := Int.ofNat (litOf sc'.sel (occOf (sc'.os ++ sc'.bs ++ sc'.gs))),
        budget := Int.ofNat (bar n T.toNat),
        frame4 := Int.ofNat (minus wall.toNat (coverOf sc')) }
  | _, _, _ => none

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

-- ---------- 过关（level_solved） ----------
/-- 一片格子紧贴外侧的一圈（4-邻域）都在棋盘内、且都在 ok 里。 -/
def seatedP (p : Pc) (ok : Nat) : Bool :=
  let m := maskP p
  1 ≤ p.r && 1 ≤ p.c && p.r + p.h ≤ 62 && p.c + p.w ≤ 63 &&
    minus (minus ((m <<< 1) ||| (m >>> 1) ||| (m <<< 64) ||| (m >>> 64)) m) ok == 0

/-- 这一步之后每个物体（带壳 + 裸）都嵌在框里：紧贴的一圈全是框壁（这一帧露着的色4）、门格，或这一步前被物体盖着的格。 -/
def solves (s : State) (a : Act) : Bool :=
  match sceneOf s (gdkOf s), nextScene s a with
  | some sc, some sc' =>
    let ok := s.frame4.toNat ||| occOf sc.gs ||| occOf (sc.sel :: sc.os ++ sc.bs)
    (sc'.sel :: sc'.os ++ sc'.bs).all (fun p => seatedP p ok)
  | _, _ => false

-- ---------- 概念规则（guard = 概念 SCOPE 管这一步；effect = 概念的断言，由上面同一套原语算） ----------

def selEnc (s : State) : List Int := (present [s.objs_0, s.objs_1, s.objs_2, s.objs_3]).filter (fun v => v % 4 == 0)
def objsV (s : State) : List Int := present [s.objs_0, s.objs_1, s.objs_2, s.objs_3]
def bareV (s : State) : List Int := present [s.bare_0, s.bare_1, s.bare_2]
def gatesV (s : State) : List Int := present [s.gates_0, s.gates_1, s.gates_2]
def sameLevel (s t : State) : Bool := s.level == t.level

/-- 这一步被操纵物体朝 d 那一排贴着别的东西（带壳物体 / 裸物体 / 门带）。 -/
def touches (s : State) (d : Nat) (withGates : Bool) : Bool :=
  match sceneOf s (gdkOf s) with
  | none => false
  | some sc => stripM sc.sel d &&& occOf (sc.os ++ sc.bs ++ (if withGates then sc.gs else [])) != 0

def touchesGate (s : State) (d : Nat) : Bool :=
  match sceneOf s (gdkOf s) with
  | none => false
  | some sc => stripM sc.sel d &&& occOf sc.gs != 0

/-- 这一步有门带归零、而且（动作生效之后的）被操纵物体在它（推完之后的位置）扫得到的地方。 -/
def selStruck (s : State) (d : Nat) : Bool :=
  match sceneOf s (gdkOf s) with
  | none => false
  | some sc =>
    let m := movePhase sc d s.walls_m.toNat s.band_m.toNat
    d != 0 && (m.gs.map tick).any (fun g => g.wrap && frontClear g m.sel s.walls_m.toNat false)

/-- 这一步有门带归零、（推完之后）正前方有别的（没被操纵的）东西够得着。 -/
def idleStruck (s : State) (d : Nat) : Bool :=
  match sceneOf s (gdkOf s) with
  | none => false
  | some sc =>
    let m := movePhase sc d s.walls_m.toNat s.band_m.toNat
    d != 0 && (m.gs.map tick).any (fun g => g.wrap && (m.os ++ m.bs).any (fun p => frontClear g p s.walls_m.toNat true))

def anyWrap (s : State) : Bool := (gatesOf s (gdkOf s)).any (fun g => (tick g).wrap)

def rules : List Rule := [
  -- core_step：方向动作、没贴着东西、没被归零的带子扫到 → 被操纵物体挪一个格位或不动，落点由 stepRules 定死
  { name := "core_step",
    guard := fun s a => dirOf a != 0 && !touches s (dirOf a) true && !selStruck s (dirOf a),
    effect := fun s a t => !sameLevel s t || match stepRules s a with
      | some p => selEnc p == selEnc t
      | none => false },
  -- push_kick：贴着带壳物体 / 裸物体推 → 紧贴推手的那些按「走 5 格位、撞色2 停、压带滑出」落地，且都在下一帧的表里
  { name := "push_kick",
    guard := fun s a => dirOf a != 0 && touches s (dirOf a) false && !anyWrap s,
    effect := fun s a t => !sameLevel s t ||
      match sceneOf s (gdkOf s) with
      | none => false
      | some sc =>
        let d := dirOf a
        let out := stripM sc.sel d
        let stop := minus s.walls_m.toNat s.band_m.toNat
        let vs := (sc.os ++ sc.bs).filter (fun p => maskP p &&& out != 0)
        vs.all (fun p => let q := kick d stop s.band_m.toNat KICK p
          (if p.code == 3 then bareV t else objsV t).contains (if p.code == 3 then encBare q else encObj q)) },
  -- push_shove：同上，整张表（被压到的挨个让开）= 下一帧的带壳物体表与裸物体表；推手自己不动
  { name := "push_shove",
    guard := fun s a => dirOf a != 0 && touches s (dirOf a) false && !anyWrap s,
    effect := fun s a t => !sameLevel s t || match stepRules s a with
      | some p => objsV p == objsV t && bareV p == bareV t && selEnc s == selEnc t
      | none => false },
  -- bare_pushed：有裸物体、方向动作、这一步没有带子归零 → 裸物体只在被推时按推法走，否则不动
  { name := "bare_pushed",
    guard := fun s a => dirOf a != 0 && !(bareV s).isEmpty && !anyWrap s,
    effect := fun s a t => !sameLevel s t || match stepRules s a with
      | some p => bareV p == bareV t
      | none => false },
  -- gate_pushed：被操纵物体贴着门带发方向动作 → 自己不动，门带按同一条推法走
  { name := "gate_pushed",
    guard := fun s a => dirOf a != 0 && touchesGate s (dirOf a),
    effect := fun s a t => !sameLevel s t || match stepRules s a with
      | some p => gatesV p == gatesV t && selEnc s == selEnc t
      | none => false },
  -- gate_tick：每个方向动作每条门带长一层、差一层就长满时归零；点击时一格不变（位置由推法定）
  { name := "gate_tick",
    guard := fun s a => (dirOf a != 0 || (clickOf a).isSome) && !(gatesV s).isEmpty,
    effect := fun s a t => !sameLevel s t || match stepRules s a with
      | some p => gatesV p == gatesV t && p.h_gd == t.h_gd
      | none => false },
  -- gate_end：一条带子没挪、前后都看得出朝向时，朝向不变
  { name := "gate_end",
    guard := fun s a => a != Act.reset && !(gatesV s).isEmpty,
    effect := fun s _ t => !sameLevel s t ||
      ((gatesV s).map decGate).all (fun g => ((gatesV t).map decGate).all (fun g' =>
        !(g.r == g'.r && g.c == g'.c && g.h == g'.h && g.w == g'.w && g.d != 0 && g'.d != 0) || g.d == g'.d)) },
  -- gate_strike：归零那一步、动作生效后正对着带子的被操纵物体，再被朝带子长的方向扫 3 个格位
  { name := "gate_strike",
    guard := fun s a => dirOf a != 0 && selStruck s (dirOf a),
    effect := fun s a t => !sameLevel s t || match stepRules s a with
      | some p => selEnc p == selEnc t
      | none => false },
  -- gate_idle_strike：归零那一步、正前方没被操纵的东西也被扫 3 个格位（扫击不挑对象，只挑位置）
  { name := "gate_idle_strike",
    guard := fun s a => dirOf a != 0 && idleStruck s (dirOf a),
    effect := fun s a t => !sameLevel s t || match stepRules s a with
      | some p => objsV p == objsV t && bareV p == bareV t
      | none => false },
  -- contact_face：被操纵物体的一条边亮着，当且仅当那一排外侧紧贴着带壳物体 / 裸物体 / 门带（前后两帧都成立）
  { name := "contact_face",
    guard := fun _ _ => true,
    effect := fun s _ t =>
      (match sceneOf s (gdkOf s) with
        | some sc => s.lit == Int.ofNat (litOf sc.sel (occOf (sc.os ++ sc.bs ++ sc.gs)))
        | none => false) &&
      (match sceneOf t (gdkOf t) with
        | some sc => t.lit == Int.ofNat (litOf sc.sel (occOf (sc.os ++ sc.bs ++ sc.gs)))
        | none => false) },
  -- click_selects：点中别的带壳物体就换它当被操纵的（原来的核心变色4），点在别处不换；物体都不挪
  { name := "click_selects",
    guard := fun _ a => (clickOf a).isSome,
    effect := fun s a t => !sameLevel s t || match stepRules s a with
      | some p => objsV p == objsV t && p.lit == t.lit
      | none => false },
  -- level_solved：方向动作或点击这一步，关卡号 +1 ⟺ 动作之后（stepRules 同一套走法算的落点）每个物体都嵌在框里。
  --   同一条规则顺带断言框壁布局按关不变：关内每一帧露着的框壁 = 本关框壁布局 k_wall − 被物体 / 门带盖着的格。
  { name := "level_solved",
    guard := fun _ a => dirOf a != 0 || (clickOf a).isSome,
    effect := fun s a t => (t.level == s.level + 1) == solves s a &&
      (!sameLevel s t || match s.k_wall, sceneOf t (gdkOf t) with
        | some wall, some sc => t.frame4 == Int.ofNat (minus wall.toNat (coverOf sc))
        | _, _ => false) },
  -- budget_tick：本关第 n 个非 RESET 动作之后预算条 = 64 - round_half_up(64n/T)
  --   guard 另要求这一步之前条与计数对得上（状态自洽型 guard）。语料里唯一的失同步：文件第 1522 行那条 seq 1520 RESET
  --   属于另一个 guid（c4bd10d3，meta 带 recard），被插进 e9890b40 这局的流里，这局其实从未重置；生成器照样在那里把
  --   h_n 归零，于是 seq 1522–1562 共 41 步计数与条失同步，被这条 guard 挡掉。这条 guard 以后会挡住任何 h_n 失同步，
  --   本语料只有这一处（独立审计复核：把那条 RESET 当空操作时，3958 条非 RESET 行的预算条残差为 0）。
  { name := "budget_tick",
    guard := fun s a => a != Act.reset && (match s.k_T with
      | some T => s.budget == Int.ofNat (bar s.h_n.toNat T.toNat)
      | none => false),
    effect := fun s _ t => !sameLevel s t || match s.k_T with
      | some T => t.budget == Int.ofNat (bar (s.h_n.toNat + 1) T.toNat) && t.h_n == s.h_n + 1
      | none => false }
]

def isGoal (s : State) : Bool := s.level ≥ 7
