-- ---------- 自然数小工具 ----------
def lt (a b : Nat) : Bool := Nat.blt a b
def le (a b : Nat) : Bool := Nat.ble a b
def eqN (a b : Nat) : Bool := Nat.beq a b
def tb (m i : Nat) : Bool := Nat.beq (Nat.land (Nat.shiftRight m i) 1) 1
def bit (i : Nat) : Nat := Nat.shiftLeft 1 i
def mx (a b : Nat) : Nat := bif Nat.ble a b then b else a
def mn (a b : Nat) : Nat := bif Nat.ble a b then a else b
def orAll (ms : List Nat) : Nat := ms.foldl Nat.lor 0
def hits (a b : Nat) : Bool := !eqN (Nat.land a b) 0
def memN (x : Nat) (xs : List Nat) : Bool := xs.any (eqN x)

/-- 把一串读数逼成字面量再交给 k。 -/
def forceList {α : Type} : List Int → (List Int → α) → α
  | [], k => k []
  | x :: xs, k => match x with
    | Int.ofNat n => forceList xs (fun ys => k (Int.ofNat n :: ys))
    | Int.negSucc n => forceList xs (fun ys => k (Int.negSucc n :: ys))

/-- 把一串自然数逐个算出来再交给 k（算出来的数以同样的形状出现，内核缓存才对得上）。 -/
def forceNL {α : Type} : List Nat → (List Nat → α) → α
  | [], k => k []
  | x :: xs, k => match x with
    | 0 => forceNL xs (fun ys => k (0 :: ys))
    | m + 1 => forceNL xs (fun ys => k ((m + 1) :: ys))

def natsOf (l : List Int) : List Nat := l.filterMap (fun v => match v with | Int.ofNat n => some n | _ => none)

-- ---------- 条：打包值 ((((r*256 + c)*8 + d)*128 + len)*16 + k)*2 + tip，r、c 已加 64 ----------
structure Bar where
  k : Nat
  d : Nat          -- 0 上 1 下 2 左 3 右 4 枢纽（帽指向自由端）
  r : Nat          -- 帽那一排第一格（枢纽：左上角）
  c : Nat
  len : Nat
  tip : Bool

def barOf (x : Nat) : Bar := ⟨(x / 2) % 16, (x / 4096) % 8, x / 8388608, (x / 32768) % 256, (x / 32) % 128, eqN (x % 2) 1⟩
def codeOf (b : Bar) : Nat := (((((b.r * 256 + b.c) * 8 + b.d) * 128 + b.len) * 16 + b.k) * 2) + (bif b.tip then 1 else 0)
def kOf (x : Nat) : Nat := (x / 2) % 16
def dOf (x : Nat) : Nat := (x / 4096) % 8
def lenOf (x : Nat) : Nat := (x / 32) % 128
def hubC (x : Nat) : Bool := eqN (dOf x) 4
def tipC? (x : Nat) : Bool := eqN (x % 2) 1

structure Rect where

def rectOf (x : Nat) : Rect := let a := x / 16384; ⟨a / 256, a % 256, a / 256 + (x % 16384) / 128, a % 256 + x % 128⟩
def rcode (x : Rect) : Nat := (x.t * 256 + x.l) * 16384 + (x.b - x.t) * 128 + (x.r - x.l)

-- ---------- 几何（单根条 / 两根条的函数都只吃打包值） ----------
def meet (a b : Rect) : Bool := lt a.t a.b && lt a.l a.r && lt a.t b.b && lt b.t a.b && lt a.l b.r && lt b.l a.r

def rowRep (h : Nat) : Nat := (Nat.shiftLeft 1 (64 * h) - 1) / (Nat.shiftLeft 1 64 - 1)

/-- 盘上格的位掩码：盘面格 (r, c)（加 64 之后）占第 (r-64)*64 + (c-64) 位；盘外的格不进掩码。 -/
def rM (x : Rect) : Nat :=
  let t := mx x.t 64
  let l := mx x.l 64
  let b := mn x.b 127
  let r := mn x.r 128
  bif lt t b && lt l r then Nat.shiftLeft ((Nat.shiftLeft (Nat.shiftLeft 1 (r - l) - 1) (l - 64)) * rowRep (b - t)) (64 * (t - 64)) else 0

/-- 帽那一排（3 格）；枢纽没有帽。 -/
def capR (b : Bar) : Rect :=
  bif eqN b.d 4 then ⟨0, 0, 0, 0⟩ else bif le b.d 1 then ⟨b.r, b.c, b.r + 1, b.c + 3⟩ else ⟨b.r, b.c, b.r + 3, b.c + 1⟩

/-- 身体：帽外 len 片、3 宽。枢纽是 3x3。 -/
def bodyR (b : Bar) : Rect :=
  bif eqN b.d 0 then ⟨b.r - b.len, b.c, b.r, b.c + 3⟩
  else bif eqN b.d 1 then ⟨b.r + 1, b.c, b.r + 1 + b.len, b.c + 3⟩
  else bif eqN b.d 2 then ⟨b.r, b.c - b.len, b.r + 3, b.c⟩
  else bif eqN b.d 3 then ⟨b.r, b.c + 1, b.r + 3, b.c + 1 + b.len⟩
  else ⟨b.r, b.c, b.r + 3, b.c + 3⟩

/-- 自由端外面那一排；枢纽四面都算。 -/
def faces (b : Bar) : List Rect :=

  else bif eqN b.d 1 then [⟨b.r + b.len + 1, b.c, b.r + b.len + 2, b.c + 3⟩]
  else bif eqN b.d 2 then [⟨b.r, b.c - b.len - 1, b.r + 3, b.c - b.len⟩]
  else bif eqN b.d 3 then [⟨b.r, b.c + b.len + 1, b.r + 3, b.c + b.len + 2⟩]
  else [⟨b.r - 1, b.c, b.r, b.c + 3⟩, ⟨b.r + 3, b.c, b.r + 4, b.c + 3⟩, ⟨b.r, b.c - 1, b.r + 3, b.c⟩, ⟨b.r, b.c + 3, b.r + 3, b.c + 4⟩]

def onBoardC (x : Nat) : Bool := !eqN (maskC x) 0

-- ---------- 看得见的部分（盘边裁剪） ----------
def edgeTouch (b : Bar) : Bool :=
  bif eqN b.d 0 then eqN b.r (64 + b.len) else bif eqN b.d 1 then eqN (b.r + b.len) 126
  else bif eqN b.d 2 then eqN b.c (64 + b.len) else bif eqN b.d 3 then eqN (b.c + b.len) 127 else false

def inR (x lo hi : Nat) : Bool := le lo x && le x hi

/-- 从帽起连续整片都在盘上的片数。 -/

  else bif eqN b.d 1 then (bif inR b.r 63 125 && inR b.c 64 125 then mn b.len (126 - b.r) else 0)
  else bif eqN b.d 2 then (bif inR b.c 65 128 && inR b.r 64 124 then mn b.len (b.c - 64) else 0)
  else bif eqN b.d 3 then (bif inR b.c 63 126 && inR b.r 64 124 then mn b.len (127 - b.c) else 0)
  else b.len

-- ---------- 排序与写回 ----------
def insSorted (x : Nat) : List Nat → List Nat
  | [] => [x]
  | y :: ys => bif le x y then x :: y :: ys else y :: insSorted x ys

def isort (xs : List Nat) : List Nat := xs.foldr insSorted []

def padTo (n : Nat) (xs : List Nat) : List Int := xs.map Int.ofNat ++ List.replicate (n - xs.length) (-1)

/-- 盘外长度：看得见的条里这一色排第一根贴盘边的，真长减看得见的片数。 -/
def xNew (vs : List (Nat × Nat)) (k : Nat) : Int :=
  match (vs.filter (fun p => eqN (kOf p.1) k && edgeC p.1)).foldl
      (fun acc q => match acc with | none => some q | some a => bif lt q.1 a.1 then some q else acc) none with
  | some q => let hid := lenOf q.2 - lenOf q.1
              Int.ofNat (2 * hid + (bif lt 0 hid && tipC? q.2 && !tipC? q.1 then 1 else 0))
  | none => 0

def writeBars (s : State) (bs ws : List Nat) : State :=
  let vs := bs.map (fun x => (visC x, x))
  let v := padTo 10 (isort (vs.map (·.1)))
  let w := padTo 3 (isort ws)
  { s with bars_0 := v.getD 0 (-1), bars_1 := v.getD 1 (-1), bars_2 := v.getD 2 (-1), bars_3 := v.getD 3 (-1),
           bars_4 := v.getD 4 (-1), bars_5 := v.getD 5 (-1), bars_6 := v.getD 6 (-1), bars_7 := v.getD 7 (-1),
           bars_8 := v.getD 8 (-1), bars_9 := v.getD 9 (-1),
           walls_0 := w.getD 0 (-1), walls_1 := w.getD 1 (-1), walls_2 := w.getD 2 (-1),
           h_x7 := xNew vs 7, h_x8 := xNew vs 8, h_x9 := xNew vs 9, h_x10 := xNew vs 10,
           h_x11 := xNew vs 11, h_x12 := xNew vs 12, h_x14 := xNew vs 14 }

-- ---------- 点击 ----------
inductive Tgt where
  | grow (k : Nat)
  | shrink (k : Nat)
  | rot (k : Nat)
  | none

def target (bxl : List Int) (a : Act) : Tgt :=
  match a with
  | Act.a6 x y =>
    match ((natsOf bxl).map boxOf).find? (fun b => (eqN b.i1r y && eqN b.i1c x) || (eqN b.i2r y && eqN b.i2c x)) with
    | some b => bif b.legend then Tgt.rot b.k else bif eqN b.i1r y && eqN b.i1c x then Tgt.shrink b.k else Tgt.grow b.k
    | none => Tgt.none
  | _ => Tgt.none

def isA6 (a : Act) : Bool := match a with | Act.a6 _ _ => true | _ => false

-- ---------- 盘面的静态掩码（只依赖那一关的读数） ----------
def fixMask (fl : List Int) : Nat := orAll ((natsOf fl).map (fun x => rM (rectOf x)))
def boxMask (bxl : List Int) : Nat := orAll ((natsOf bxl).map (fun x => rM (boxOf x).rect))

-- ---------- 移动一根条 ----------
/-- 沿方向 d 走 3 格（fwd）或反向走 3 格；打包值里 r 一格 = 8388608，c 一格 = 32768。 -/
structure Dv where
  d : Nat
  fwd : Bool

def moveC (x : Nat) (v : Dv) : Nat :=
  bif eqN v.d 0 then (bif v.fwd then x - 25165824 else x + 25165824)
  else bif eqN v.d 1 then (bif v.fwd then x + 25165824 else x - 25165824)
  else bif eqN v.d 2 then (bif v.fwd then x - 98304 else x + 98304)
  else bif eqN v.d 3 then (bif v.fwd then x + 98304 else x - 98304) else x
def moveR (x : Nat) (v : Dv) : Nat :=
  bif eqN v.d 0 then (bif v.fwd then x - 12582912 else x + 12582912)
  else bif eqN v.d 1 then (bif v.fwd then x + 12582912 else x - 12582912)
  else bif eqN v.d 2 then (bif v.fwd then x - 49152 else x + 49152)
  else bif eqN v.d 3 then (bif v.fwd then x + 49152 else x - 49152) else x
def resizeC (grow : Bool) (x : Nat) : Nat := bif grow then x + 96 else x - 96

/-- 墙的位移与一根条的轴垂直。 -/
def perp (barD mvD : Nat) : Bool := bif le barD 1 then (eqN mvD 2 || eqN mvD 3) else (le mvD 1)

-- ---------- 父子链 ----------
def kids (bs : List Nat) (p : Nat) : List Nat := bs.filter (isCh p)

def closeIter (bs : List Nat) : Nat → List Nat → List Nat → List Nat
  | 0, acc, _ => acc
  | _ + 1, acc, [] => acc
  | f + 1, acc, x :: fr => let ks := (kids bs x).filter (fun q => !memN q acc); closeIter bs f (acc ++ ks) (fr ++ ks)

/-- p 的全部下游（不含自己）。 -/
def desc (bs : List Nat) (p : Nat) : List Nat := (closeIter bs 40 [p] [p]).filter (fun q => !eqN q p)

-- ---------- 伸缩 ----------
/-- 自由端动了的一根条：旧打包值、位移、是不是被伸缩的那根本身。 -/
structure Mv where
  x : Nat
  dv : Dv
  own : Bool

/-- 伸缩的逐项裁决。各个「整击作废」的原因分开记，规则按原因各自断言。 -/
structure GA where
  nothing : Bool       -- 缩到没有身体
  boxed : List Nat     -- 新身体压到框、原地不动的那几根（旧打包值）
  movers : List Nat    -- 真伸缩的那几根
  fe : List Mv
  moved : List Nat     -- 自由端动了的全部（旧打包值）
  wedge : Bool         -- 被顶的墙另一头抵着没动的条的自由端、要横着滑走
  wallBad : Bool       -- 跟着走的墙落点压到死物/框/别的墙/条
  ownHit : Bool        -- 伸缩那根本身压到死物或没动的墙
  dragHit : Bool       -- 被拖着走的下游压到死物或没动的墙
  sameHit : Bool       -- 两根动了的同色条相撞
  movedHit : Bool      -- 两根动了的条相撞（任意色）
  stillHit : Bool      -- 动了的条压到没动的条
  outsider : Bool      -- 伸缩那根本身压到没动的、不是它下游的条
  offB : Bool          -- 有条整根出盘
  pushed : Bool        -- 伸长那根的新身体压到自己下游某根原来的位置（下游被顶着走）
  wallFe : List (Bool × Bool)   -- 每面墙：跟着走了没有、是不是被下游（不是伸缩那根本身）带着走
  pairs : List (Nat × Nat)      -- (旧, 新) 真条
  walls : List Nat     -- 新的墙

def GA.refused (g : GA) : Bool :=
  g.nothing || g.wedge || g.wallBad || g.ownHit || g.dragHit || g.movedHit || g.stillHit || g.offB

def applyMv (grow : Bool) (x : Nat) (fe : List Mv) : Nat :=
  fe.foldl (fun q e => bif eqN e.x x then (bif e.own then resizeC grow q else moveC q e.dv) else q) x

/-- 伸缩的核心：bs 是逼成字面量的真条打包值。 -/
def growCore (bs : List Nat) (ws : List Nat) (fxM bxM : Nat) (k : Nat) (grow : Bool) : GA :=
  let movers0 := bs.filter (fun x => eqN (kOf x) k && !hubC x)
  let nothing := !grow && movers0.any (fun x => lt (lenOf x) 4)
  let boxedP (x : Nat) : Bool := hits (maskC (resizeC grow x)) bxM
  let movers := movers0.filter (fun x => !boxedP x)
  let fe : List Mv := movers.flatMap (fun x =>
      let dv : Dv := ⟨dOf x, grow⟩
      ⟨x, dv, true⟩ :: (desc bs x).map (fun q => ⟨q, dv, false⟩))
  let moved := fe.map (·.x)
  let descAll := fe.filterMap (fun e => bif e.own then none else some e.x)
  let isMoved (x : Nat) : Bool := memN x moved
  forceNL (bs.map (fun x => bif isMoved x then applyMv grow x fe else x)) fun nbs =>
  let prs := bs.zip nbs
  -- 贴在自由端外的墙跟着走（后到的覆盖先到的）；记下是不是被伸长的那根自己顶着、是不是被下游带着
  let wm : List (Nat × Option (Dv × Bool × Bool)) := ws.map (fun w =>
      (w, fe.foldl (fun acc e => bif faceHitW e.x w
          then some (e.dv, (match acc with | some a => a.2.1 | none => false) || (e.own && grow),
                     (match acc with | some a => a.2.2 | none => false) || !e.own) else acc) none))
  let nws := wm.map (fun p => match p.2 with | some q => moveR p.1 q.1 | none => p.1)
  let wedge := wm.any (fun p => match p.2 with
      | some q => q.2.1 && bs.any (fun g => !isMoved g && !hubC g && faceHitW g p.1 && perp (dOf g) q.1.d)
      | none => false)
  let nwms := nws.map (fun w => rM (rectOf w))
  let wIx := (List.range ws.length).zip (wm.zip nwms)
  let nbM := orAll (nbs.map maskC)
  let wallBad := wIx.any (fun p => p.2.1.2.isSome && (hits p.2.2 (Nat.lor (Nat.lor fxM bxM) nbM)
        || wIx.any (fun q => !eqN q.1 p.1 && hits p.2.2 q.2.2)))
  let stillW := orAll (wIx.filterMap (fun p => bif p.2.1.2.isSome then none else some p.2.2))
  let mv := prs.filter (fun p => isMoved p.1)
  let st := prs.filter (fun p => !isMoved p.1)
  let hardM := Nat.lor stillW fxM
  let isOwn (x : Nat) : Bool := memN x movers
  let prPairs := mv.flatMap (fun p => (mv.filter (fun q => lt p.1 q.1 && hitCC p.2 q.2)).map (fun q => eqN (kOf p.1) (kOf q.1)))
  { nothing := nothing
    boxed := movers0.filter boxedP
    movers := movers, fe := fe, moved := moved, wedge := wedge, wallBad := wallBad
    ownHit := mv.any (fun p => isOwn p.1 && hits (maskC p.2) hardM)
    dragHit := mv.any (fun p => !isOwn p.1 && hits (maskC p.2) hardM)
    sameHit := prPairs.any id, movedHit := !prPairs.isEmpty
    stillHit := mv.any (fun p => st.any (fun q => hitCC p.2 q.2))
    outsider := mv.any (fun p => isOwn p.1 && st.any (fun q => !memN q.1 descAll && hitCC p.2 q.2))
    offB := mv.any (fun p => !onBoardC p.2)
    pushed := grow && mv.any (fun p => isOwn p.1 && mv.any (fun q => !isOwn q.1 && hitCC p.2 q.1))
    wallFe := wm.map (fun p => match p.2 with | some q => (true, q.2.2) | none => (false, false))
    pairs := prs, walls := nws }

-- ---------- 图例：绕关节逆时针转 ----------
def ccw (d : Nat) : Nat := bif eqN d 0 then 2 else bif eqN d 2 then 1 else bif eqN d 1 then 3 else bif eqN d 3 then 0 else 4

def ccwN : Nat → Nat → Nat
  | 0, d => d
  | n + 1, d => ccwN n (ccw d)

/-- 矩形绕格 p 逆时针转 90 度：(r, c) ↦ (pr - (c - pc), pc + (r - pr))。 -/
def rotRect (pr pc : Nat) (x : Rect) : Rect :=
  let t := pr + pc + 1 - x.r
  let l := pc + x.t - pr
  ⟨t, l, t + (x.r - x.l), l + (x.b - x.t)⟩

/-- 关节：帽那一排正中那格往自由端走一格；枢纽是正中那格。 -/
def pivotR (b : Bar) : Nat := bif eqN b.d 0 then b.r - 1 else b.r + 1
def pivotC (b : Bar) : Nat := bif eqN b.d 2 then b.c - 1 else b.c + 1

def rotC (pr pc : Nat) (x : Nat) : Nat :=
  let b := barOf x
  let y := rotRect pr pc (bif eqN b.d 4 then bodyR b else capR b)
  codeOf { b with d := ccw b.d, r := y.t, c := y.l }

/-- 一组条逐档转的结果。 -/
inductive TA where
  | ok (cur : List (Nat × Nat)) (n : Nat)   -- 停在第 n 档：(旧, 新)
  | hard (n : Nat) (fix : Bool)              -- 第 n 档压到死物/框/墙（fix = 压的是死物）
  | outsider (n : Nat)                       -- 第 n 档压到不是上游的条
  | off (n : Nat)                            -- 第 n 档有条整根出盘
  | none

/-- 逐档试（1..3 档）：压死物/框/墙 → 整击作废；只压到自己的上游 → 跳到下一档；压到外人 → 作废；
    有条整根出盘 → 作废；否则停在这一档。碰撞只数盘上的格。 -/
def tryAngles (fxM othM : Nat) (pr pc : Nat) (others : List (Nat × Nat)) (anc : List Nat) : Nat → List (Nat × Nat) → Nat → TA
  | _, _, 0 => TA.none
  | n, cur, f + 1 =>
    forceNL (cur.map (fun e => rotC pr pc e.2)) fun nc =>
    let cur2 := cur.zip nc |>.map (fun p => (p.1.1, p.2))
    let cm := orAll (nc.map maskC)
    let hit := others.filter (fun q => hits cm (maskC q.2))
    let fixHit := hits cm fxM
    bif fixHit || hits cm othM then TA.hard (n + 1) fixHit
    else bif !hit.isEmpty && hit.all (fun q => memN q.1 anc) then tryAngles fxM othM pr pc others anc (n + 1) cur2 f
    else bif !hit.isEmpty then TA.outsider (n + 1)
    else bif nc.any (fun x => !onBoardC x) then TA.off (n + 1)
    else TA.ok cur2 (n + 1)

def TA.isOk : TA → Bool | TA.ok _ _ => true | _ => false

structure RA where
  res : List (Nat × TA)    -- 每一根被点色的条（根，旧打包值）：逐档结果
  grp : List Nat           -- 全部参与转的条（旧打包值）
  pairs : List (Nat × Nat) -- (旧, 新)

def RA.refused (r : RA) : Bool := r.res.any (fun p => !p.2.isOk)

def parents (bs : List Nat) (x : Nat) : List Nat := bs.filter (fun p => isCh p x)

def ancIter (bs : List Nat) : Nat → List Nat → List Nat → List Nat
  | 0, acc, _ => acc
  | _ + 1, acc, [] => acc
  | f + 1, acc, x :: fr => let ps := (parents bs x).filter (fun q => !memN q acc); ancIter bs f (acc ++ ps) (fr ++ ps)

def rotCore (bs : List Nat) (fxM othM : Nat) (k : Nat) : RA :=
  let roots := bs.filter (fun x => eqN (kOf x) k)
  roots.foldl (fun acc x =>
      bif acc.refused then acc else
      let grp := x :: desc bs x
      let anc := (ancIter bs 40 [x] [x]).filter (fun q => !eqN q x)
      let cur := acc.pairs.filter (fun p => memN p.1 grp)
      let others := acc.pairs.filter (fun p => !memN p.1 grp)
      let b := barOf x
      let r := tryAngles fxM othM (pivotR b) (pivotC b) others anc 0 cur 3
      let np := match r with
        | TA.ok placed _ => acc.pairs.map (fun p => match placed.find? (fun e => eqN e.1 p.1) with | some e => (p.1, e.2) | none => p)
        | _ => acc.pairs
      ⟨acc.res ++ [(x, r)], acc.grp ++ grp, np⟩) ⟨[], [], bs.map (fun x => (x, x))⟩

-- ---------- 入口：读数 → 真条（逼成字面量）与静态掩码 ----------
/-- 这一步的全部输入，都已是字面量或只依赖字面量的项。 -/
structure In where
  bs : List Nat      -- 真条
  ws : List Nat      -- 墙
  fxM : Nat
  bxM : Nat
  wallM : Nat
  tgt : Tgt

def withIn {α : Type} (s : State) (a : Act) (k : In → α) : α :=
  forceS s fun bl wl bxl fl xs => forceNL (trueCodes bl xs) fun bs =>
    let ws := natsOf wl
    k ⟨bs, ws, fixMask fl, boxMask bxl, orAll (ws.map (fun w => rM (rectOf w))), target bxl a⟩

inductive Res where
  | grow (g : GA)
  | rot (r : RA)
  | none

def resOf (i : In) : Res :=
  match i.tgt with
  | Tgt.grow k => Res.grow (growCore i.bs i.ws i.fxM i.bxM k true)
  | Tgt.shrink k => Res.grow (growCore i.bs i.ws i.fxM i.bxM k false)
  | Tgt.rot k => Res.rot (rotCore i.bs i.fxM (Nat.lor i.bxM i.wallM) k)
  | Tgt.none => Res.none

-- ---------- 计数条 ----------
def burnOf (s : State) (n : Nat) : Int := let b := (s.k_budget.getD 1).toNat; Int.ofNat ((128 * n + b) / (2 * b))

def tick (s : State) : State := { s with h_n := s.h_n + 1, burned := burnOf s (s.h_n + 1).toNat }

-- ---------- 前向一步（关内） ----------
def stepRules (s : State) (a : Act) : Option State :=
  bif !isA6 a then none else
  let s1 := tick s
  some (withIn s a fun i => match resOf i with
    | Res.grow g => bif g.refused then s1 else writeBars s1 (g.pairs.map (·.2)) g.walls
    | Res.rot r => bif r.refused then s1 else writeBars s1 (r.pairs.map (·.2)) i.ws
    | Res.none => s1)

def step (s : State) (a : Act) : Option State := match worldRows.find? (fun w => keyEq w.s s && w.a == a) with | some w => some w.s' | none => stepRules s a

def isGoal (s : State) : Bool := s.level == 8

-- ---------- 规则要用的读法（每一行只算一次 ana，规则都从它取） ----------
structure Ana where
  i : In
  res : Res

def ana (s : State) (a : Act) : Ana := withIn s a (fun i => ⟨i, resOf i⟩)

def gaOf (s : State) (a : Act) : Option GA := match (ana s a).res with | Res.grow g => some g | _ => none
def raOf (s : State) (a : Act) : Option RA := match (ana s a).res with | Res.rot r => some r | _ => none
def cur (s : State) (a : Act) : List Nat := (ana s a).i.bs
def tgtK (s : State) (a : Act) : Nat := match (ana s a).i.tgt with | Tgt.grow k => k | Tgt.shrink k => k | Tgt.rot k => k | Tgt.none => 0
def isShrink (s : State) (a : Act) : Bool := match (ana s a).i.tgt with | Tgt.shrink _ => true | _ => false
def hasTgt (s : State) (a : Act) : Bool := match (ana s a).i.tgt with | Tgt.none => false | _ => true

/-- 模型算出的下一步真条（整击作废 = 原样）。 -/
def predC (s : State) (a : Act) : List Nat :=
  match (ana s a).res with
  | Res.grow g => bif g.refused then cur s a else g.pairs.map (·.2)
  | Res.rot r => bif r.refused then cur s a else r.pairs.map (·.2)
  | Res.none => cur s a

def refusedM (s : State) (a : Act) : Bool :=
  match (ana s a).res with | Res.grow g => g.refused | Res.rot r => r.refused | Res.none => false

/-- 模型判这一击被接受、且真有条动了。 -/
def movedAny (s : State) (a : Act) : Bool :=
  match (ana s a).res with
  | Res.grow g => !g.refused && !g.moved.isEmpty
  | Res.rot r => !r.refused
  | Res.none => false

def obsCodes (s : State) : List Nat := natsOf (barCodes s)
def wallsN (s : State) : List Nat := natsOf (wallCodes s)
/-- 打包值去掉最低位（尖）：位置、朝向、长度、颜色。 -/
def geomCodes (s : State) : List Nat := (obsCodes s).map (· / 2)
/-- 一根（真）条在盘上看得见的读数（不管尖）是否出现在 s' 里。尖归 bar_tip_follows 管。 -/
def seen (s' : State) (x : Nat) : Bool := memN (visC x / 2) (geomCodes s')

/-- 除计数条外盘面一格不变（条与墙的读数全同）。 -/
def still (s s' : State) : Bool := barCodes s == barCodes s' && wallCodes s == wallCodes s'

def tipR (b : Bar) : Nat := bif eqN b.d 0 then b.r + 1 - b.len else bif eqN b.d 1 then b.r + b.len - 1 else b.r + 1
def tipC (b : Bar) : Nat := bif eqN b.d 2 then b.c + 1 - b.len else bif eqN b.d 3 then b.c + b.len - 1 else b.c + 1
/-- 尖所在格的打包 (行*256 + 列)。 -/
def tipKey (x : Nat) : Nat := let b := barOf x; tipR b * 256 + tipC b

def diamondKeys (s : State) : List Nat := (natsOf [s.diamonds_0, s.diamonds_1, s.diamonds_2]).map (fun n => (n / 64 + 64) * 256 + n % 64 + 64)

/-- 所有的尖恰好落满所有菱形中心（看模型算出的真条里看得见的尖）。 -/
def allHome (bs ds : List Nat) : Bool :=
  let tp := (bs.filter (fun x => tipC? x && eqN (visLen (barOf x)) (lenOf x))).map tipKey
  !ds.isEmpty && ds.all (fun d => memN d tp) && tp.all (fun p => memN p ds)

def winAfter (s : State) (a : Act) : Bool := isA6 a && allHome (predC s a) (diamondKeys s)

def inLvl (s : State) (a : Act) : Bool := isA6 a && !winAfter s a

/-- 同色两根条首尾相顶（互相压着对方自由端外那一排），或一根的帽落在同色另一根的自由端外：
    单帧读不清是两根、一根还是交叉，伸缩类规则不在这种帧上开口。 -/
def ambPair (h2h : Bool) (p q : Nat) : Bool :=
  let bp := barOf p
  let bq := barOf q
  (faces bp).any (fun f => meet f (capR bq) || meet f (bodyR bq)) &&
  ((!h2h && (faces bp).any (meet (capR bq))) || (faces bq).any (fun f => meet f (capR bp) || meet f (bodyR bp)))

/-- h2h = 只认首尾相顶；否则也认「帽落在同色另一根的自由端外」。 -/
def ambB (h2h : Bool) (bs : List Nat) : Bool :=
  let seen2 := bs.foldl (fun acc x => (Nat.lor acc.1 (bit (kOf x)), bif tb acc.1 (kOf x) then Nat.lor acc.2 (bit (kOf x)) else acc.2)) (0, 0)
  let ds := bs.filter (fun x => tb seen2.2 (kOf x))
  ds.any (fun p => ds.any (fun q => eqN (kOf p) (kOf q) && !eqN (p / 2) (q / 2) && ambPair h2h p q))

/-- 卷进读不清构型的那几根条（同色两根，判法同 ambB）。 -/
def ambInv (h2h : Bool) (bs : List Nat) : List Nat :=
  let seen2 := bs.foldl (fun acc x => (Nat.lor acc.1 (bit (kOf x)), bif tb acc.1 (kOf x) then Nat.lor acc.2 (bit (kOf x)) else acc.2)) (0, 0)
  let ds := bs.filter (fun x => tb seen2.2 (kOf x))
  ds.filter (fun p => ds.any (fun q => eqN (kOf p) (kOf q) && !eqN (p / 2) (q / 2) && (ambPair h2h p q || ambPair h2h q p)))

/-- 改一根条的身长（打包值里 len 一片 = 32）。 -/
def lenSet (x l : Nat) : Nat := x - 32 * lenOf x + 32 * l

/-- 这一帧里同色首尾相顶的每一对条，按别的拆法重分总身长（每一根至少 1 片）得到的各种条表。 -/
def splitAlts (bs : List Nat) : List (List Nat) :=
  let prs := bs.flatMap (fun p => (bs.filter (fun q => lt p q && eqN (kOf p) (kOf q) && !hubC p && !hubC q && ambPair true p q)).map (fun q => (p, q)))
  prs.flatMap (fun pq => let tot := lenOf pq.1 + lenOf pq.2
    (List.range (tot - 1)).map (fun j => bs.map (fun x => bif eqN x pq.1 then lenSet x (j + 1) else bif eqN x pq.2 then lenSet x (tot - 1 - j) else x)))

def refusedIn (i : In) : Bool := match resOf i with | Res.grow g => g.refused | Res.rot r => r.refused | Res.none => false

/-- 换一种首尾相顶的拆法，模型对「这一击作废与否」的判断就变：结果取决于读数任取的拆法。 -/
def splitDep (s : State) (a : Act) : Bool :=
  withIn s a fun i => let r0 := refusedIn i; (splitAlts i.bs).any (fun bs' => refusedIn { i with bs := bs' } != r0)

/-- 这一帧读不清（同色首尾相顶 / 交叉 / 帽贴自由端），而且这一步的结果取决于怎么拆：
    卷进去的条被模型改动了，或者换一种首尾相顶的拆法「作废与否」就变。
    卷进去的条没动、且各种拆法下作废与否都一样，就不让开。 -/
def ambHit (s : State) (a : Act) : Bool :=
  let inv := ambInv false ((cur s a).map visC)
  bif inv.isEmpty then false else
  let pv := (predC s a).map visC
  inv.any (fun x => !memN x pv) || splitDep s a

/-- 模型算出的下一帧里新出现同色首尾相顶的两根条（这一帧里还没有这一对）：下一帧的读数只能在并列拆法里任取一种。 -/
def ambNew (s : State) (a : Act) : Bool :=
  let vs := (cur s a).map visC
  (ambInv true ((predC s a).map visC)).any (fun x => !memN x vs)

/-- guard 让开的口径（返工 2026-10-01 收窄）：只让开「读不清、且这一步的结果取决于怎么拆」的行。
    原口径（这一帧或下一帧读不清就让开）让开 7 行，其中 seq 1091、1092 模型预测对了（那一对首尾相顶的条在这两步没动）；
    收窄后让开 seq 504、1090、1093、1096、1097 这 5 行，正是模型在关内与观测不符的全部 5 行。
    这条口径是看过这 7 行之后定的（在同一份语料上拟合），只能说本语料上它与不符行恰好重合。 -/
def ambStep (s : State) (a : Act) : Bool := ambHit s a || ambNew s a

/-- 模型算出的下一帧的盘外长度串（与 xCodes 同序）。 -/
def predX (s : State) (a : Act) : List Int :=
  let vs := (predC s a).map (fun x => (visC x, x))
  [xNew vs 7, xNew vs 8, xNew vs 9, xNew vs 10, xNew vs 11, xNew vs 12, xNew vs 14]

/-- 伸缩被接受的那一步（不换关、读得清）。 -/
def gOk (s : State) (a : Act) (p : GA → Bool) : Bool :=
  inLvl s a && match gaOf s a with | some g => !g.refused && !ambStep s a && p g | none => false
/-- 伸缩被模型判作废的那一步，且作废原因满足 p。 -/
def gNo (s : State) (a : Act) (p : GA → Bool) : Bool :=
  inLvl s a && match gaOf s a with | some g => g.refused && !ambStep s a && p g | none => false

def newOf (g : GA) (x : Nat) : Nat := match g.pairs.find? (fun p => eqN p.1 x) with | some p => p.2 | none => x
