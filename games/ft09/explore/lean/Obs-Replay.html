def emptyBoard : Board := ⟨[], [], [], 0⟩
def boardAt (i : Nat) : Board := boards.getD i emptyBoard

def Obs.before (o : Obs) : World := ⟨boardAt o.bPrev, o.cPrev⟩
def Obs.after  (o : Obs) : World := ⟨boardAt o.bNext, o.cNext⟩

/-- 越界给一个空步。越界本身是 bug；空步在任何判据下都会立刻露馅（它的 `act` 是 `none`）。 -/
def dummy : Obs := ⟨0, 0, 0, .reset, 0, [], none, 0, []⟩

/-- 按 **seq** 取。不按下标取：下标会随筛掉的条数整体错位，取到另一条真转移而不报错。 -/
def bySeq (n : Nat) : Obs := (steps.find? (fun o => o.seq == n)).getD dummy

/-- 从 seq `want` 那一条起连着取 `len` 条，做成 (动作, 看到的下一帧)。
    取不到就是空表——**空表会让 `traceOk` 平凡为真**，所以每一段另配一条 `_len` 定理。 -/
def playFrom (want len : Nat) : List (Act × World) :=
  ((steps.dropWhile (fun o => o.seq != want)).take len).map (fun o => (o.act.getD .void, o.after))
