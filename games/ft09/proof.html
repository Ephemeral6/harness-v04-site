证明

核心规则被写成 Lean 4 里的定义。对每一关，Lean 从认证那一遍的真实开局出发，照规则推演每一次点击，逐帧和录像比对，直到过关那一步；整个过程由 Lean 的内核检查。

验证

  • 这一局的通关由 ARC Prize 官方认证：6/6 关，100 分（官方记录 ↗）。证明用的就是这一遍的操作。
  • 规则在录像全部 168 步上逐步检验，0 次说错；每一步都至少有一条规则表态。
  • 同一批 167 步，Lean 版规则与 Python 版规则逐步对账，结论完全一致。

def _uniform(g, r, c, n):
    if r < 0 or c < 0 or r + n > len(g) or c + n > len(g[0]):
        return None
    v = g[r][c]
    for dr in range(n):
        row = g[r + dr]
        for dc in range(n):
            if row[c + dc] != v:
                return None
    return v


def _subgrid(g, t, l):
    """(t,l) 起的 6x6 读成 3x3 个 2x2 小格；有小格不单色（或越界）返回 None。"""
    if t < 0 or l < 0 or t + TILE > len(g) or l + TILE > len(g[0]):
        return None
    out = []
    for a in range(3):
        row = []
        for b in range(3):
            v = _uniform(g, t + a * SUB, l + b * SUB, SUB)
            if v is None:
                return None
            row.append(v)
        out.append(row)
    return out


def _is_center(g, t, l):
    p = _subgrid(g, t, l)
    if p is None or p[1][1] in PAT:
        return False
    return all(p[i][j] in PAT for i in range(3) for j in range(3) if (i, j) != (1, 1))


def _read_slot(g, t, l, bg):
    """一个槽位读成 ('tile', base, marks) / ('empty',) / ('junk',)。中心另外认。"""
    v = _uniform(g, t, l, TILE)
    if v is not None:
        return ("empty",) if v == bg else ("tile", v, ())
    p = _subgrid(g, t, l)
    if p is None:
        return ("junk",)
    base = p[0][0]
    if any(p[a][b] != base for a, b in BASE_SUB) or base == bg:
        return ("junk",)
    marks = []
    for (a, b), d in EDGE.items():
        if p[a][b] == base:
            continue
        if p[a][b] != MARK:
            return ("junk",)
        marks.append(d)
    return ("tile", base, tuple(d for d in DIRS if d in marks))

def _layout(g):
    """返回 (背景色, 中心列表[(t,l)], 相位 → 格阵序号, 槽位表 {(lat,r,c): 读法}, 可用行数)。"""
    H, W = len(g), len(g[0])
    rows = H - 1                                        # 最后一行是计数条，不属于版面
    bg = Counter(v for row in g[:rows] for v in row).most_common(1)[0][0]
    area = g[:rows]
    centers = [(t, l) for t in range(rows - TILE + 1) for l in range(W - TILE + 1)
               if _is_center(area, t, l)]
    phases = sorted({(t % PITCH, l % PITCH) for t, l in centers})
    lat_of = {ph: i for i, ph in enumerate(phases)}
    slots = {}
    cset = set(centers)
    for (pr, pc), k in lat_of.items():
        for t in range(pr, rows - TILE + 1, PITCH):
            for l in range(pc, W - TILE + 1, PITCH):
                key = (k, t // PITCH, l // PITCH)
                slots[key] = ("center",) if (t, l) in cset else _read_slot(area, t, l, bg)
    return bg, centers, lat_of, slots, rows


def _bar(g):
    row = g[-1]
    if any(v not in (BAR_FULL, BAR_USED) for v in row):
        raise ParseError("G1", "计数条里出现了两色之外的颜色")
    n = sum(1 for v in row if v == BAR_USED)
    if any(v != BAR_USED for v in row[len(row) - n:]):
        raise ParseError("G1", "计数条已用的那一段没有贴着最右端")
    return n


def _palette(g, bg, rows, colors):
    """右上角图例从上往下；没有图例退回清点（那些关只有两色，顺序无所谓）。"""
    top = g[0]
    runs, i = [], 0
    while i < len(top):
        if top[i] == bg:
            i += 1
            continue
        j = i
        while j < len(top) and top[j] == top[i]:
            j += 1
        runs.append((i, j - i))
        i = j
    seq = []
    if len(runs) == 1 and runs[0][1] == LEGEND_W:
        c0, r = runs[0][0], 0
        while r + LEGEND_W <= rows:
            v = _uniform(g, r, c0, LEGEND_W)
            if v is None or v == bg:
                break
            seq.append(v)
            r += LEGEND_W
    if not seq:
        seq = sorted(colors)
    return tuple(seq)


def parse(frame) -> World:
    """一帧像素 → 一个 World。读不出来抛 `ParseError`。"""
    g = frame
    if not g or not g[0]:
        raise ParseError("G0", "空帧")
    bar = _bar(g)
    bg, centers, lat_of, slots, rows = _layout(g)
    if not centers:
        raise ParseError("G0", "一个网格中心都没找到")

    tiles, bases = [], []
    for (k, r, c), s in sorted(slots.items()):
        if s[0] == "tile":
            tiles.append(Tile(Cell(k, r, c), s[2]))
            bases.append(s[1])

    grids = []
    for t, l in centers:
        p = _subgrid(g, t, l)
        k = lat_of[(t % PITCH, l % PITCH)]
        cen = Cell(k, t // PITCH, l // PITCH)
        pattern = tuple(PAT[p[1 + dr][1 + dc]] for dr, dc in RING)
        grid = Grid(cen, p[1][1], pattern)
        for cell, want in grid.ring():
            s = slots.get((cell.lat, cell.row, cell.col), ("off",))
            if s[0] == "junk":
                raise ParseError("G5", f"网格 {cen} 的外环位置 {cell} 读不出来")
            has_tile = s[0] == "tile"
            if (want == NONE) == has_tile:
                raise ParseError("G2", f"网格 {cen} 图样在 {cell} 写 {want}，版面上"
                                 + ("有" if has_tile else "没有") + "瓦片")
        grids.append(grid)

    for tl in tiles:
        for dr, dc in tl.marks:
            n = tl.cell.shift(dr, dc)
            s = slots.get((n.lat, n.row, n.col), ("off",))
            if s[0] == "junk":
                raise ParseError("G4", f"瓦片 {tl.cell} 的记号指向读不出来的槽位 {n}")

    colors = set(bases) | {gr.lit for gr in grids}
    palette = _palette(g, bg, rows, colors)
    if len(palette) < 2:
        raise ParseError("G3", "调色板少于两色")
    bad = sorted(c for c in colors if c not in palette)
    if bad:
        raise ParseError("G3", f"底色 / 点亮色 {bad} 不在调色板 {list(palette)} 里")

    return World(palette=palette, tiles=tuple(tiles), grids=tuple(sorted(grids)),
                 colors=tuple(bases), bar_used=bar,
                 backdrop=_backdrop(g, rows, [px for (k, r, c), s in slots.items()
                                              if s[0] in ("tile", "center")
                                              for px in [_slot_px(lat_of, k, r, c)]]))


def _slot_px(lat_of, k, r, c):
    pr, pc = next(ph for ph, kk in lat_of.items() if kk == k)
    return r * PITCH + pr, c * PITCH + pc


def _backdrop(g, rows, covered_tl):
    """画面上**不归任何瓦片、中心、计数条**的那些像素，压成一个不透明的指纹。

    它是「底板」：背景、空槽、图例、方框线……模型不逐格描述它们，
    但「走一步它们一格不动」是一句能判错的话（卡 05 对账时新册拒绝集比旧册小了 9 个变异，
    全是改在这些像素上的——旧册按像素判，新册的世界里原来没有它们）。
    **只许判相等**：不许比大小、不许拆开读、不许跨关比。
    """
    cov = set()
    for t, l in covered_tl:
        for dy in range(TILE):
            for dx in range(TILE):
                cov.add((t + dy, l + dx))
    h = hashlib.sha1()
    for r in range(rows):
        h.update(bytes((255 if (r, c) in cov else v) for c, v in enumerate(g[r])))
    return h.hexdigest()[:16]


def parse_action(frame, action: str) -> Act:
    """原始动作 → `Act`。点击坐标（x=列, y=行）落在哪个槽位要靠这一帧的格阵几何。

    落在某个槽位的 6x6 之内（瓦片或中心）→ CLICK；落在所有这类槽位之外 → VOID。
    这一帧读不成世界时照样抛 `ParseError`。
    """
    name, _, coords = str(action).partition(":")
    name = name.strip().upper()
    if name == "RESET":
        return Act(RESET)
    if name != "ACTION6":
        raise ParseError("A0", f"没见过的动作 {action!r}")
    x, y = (int(v) for v in coords.split(","))
    g = frame
    _, centers, lat_of, slots, _ = _layout(g)
    hits = []
    for (pr, pc), k in lat_of.items():
        if y < pr or x < pc:
            continue
        r, c = (y - pr) // PITCH, (x - pc) // PITCH
        if (y - pr) % PITCH >= TILE or (x - pc) % PITCH >= TILE:
            continue
        s = slots.get((k, (pr + r * PITCH) // PITCH, (pc + c * PITCH) // PITCH))
        if s and s[0] in ("tile", "center"):
            hits.append(Cell(k, (pr + r * PITCH) // PITCH, (pc + c * PITCH) // PITCH))
    if len(hits) > 1:
        raise ParseError("A1", f"一下点击落在两张格阵的槽位上：{hits}")
    return Act(CLICK, hits[0]) if hits else Act(VOID)

一个网格的外环到齐了没有：grid_done(w, colors, g)；全部到齐：all_done(w, colors)。
"""一个网格的外环**到齐了**没有：`grid_done(w, colors, g)`；全部到齐：`all_done(w, colors)`。

由 `World.grids`（题面）+ `World.tiles`（外环上是哪几块）+ 一张底色表组合而来：

    图样写 ON 的位置    那块瓦片的底色 = 这个网格的点亮色
    图样写 OFF 的位置   那几块瓦片的底色**彼此相同**，而且**不是**点亮色
    图样写 NONE 的位置  那里没有瓦片（解析的 G2 保证），不管

「OFF 那几块要彼此相同」是一句**可判错的话**，不是随手的写法：
更弱的说法是「每块各自不是点亮色就行」。两种说法在两色的关上没有差别，
只在三色的关上才分得开——那一关语料里能不能分开，读数写在 `analysis/goal.txt`。

底色表单独传进来而不是直接读 `w.colors`：换关那一步要问的是**点完那一刻**
到齐了没有，那张表由 `effect.colors_after` 算出来。
"""
from __future__ import annotations


def grid_done(w, colors, g) -> bool:
    off = set()
    for cell, want in g.ring():
        if want == "none":
            continue
        i = w.index_of(cell)
        if i is None:
            return False
        c = colors[i]
        if want == "on":
            if c != g.lit:
                return False
        else:
            if c == g.lit:
                return False
            off.add(c)
    return len(off) <= 1


def all_done(w, colors) -> bool:
    return all(grid_done(w, colors, g) for g in w.grids)

    i = w.index_of(cell)

    for dr, dc in w.tiles[i].marks:
        j = w.index_of(cell.shift(dr, dc))
        if j is not None:
            out.append(j)
    return out

  match k with
  | .same     => w.step a == o && !(w.step a).isGoal
  | .win      => w.step a == o && (w.step a).isGoal
  | .newBoard => (w.step a).isGoal

# Keys —— 「不可做」的钥匙，每把配三条防线

这一局语料里六关都走通了，**没有任何一关被宣告不可做**。这一份不是在给某一关下结论，
是在证明这套模型**说得出**「做不了」——而且说的时候有防线，不是对谁都说不可做。

## 钥匙一：调色板造不出点亮色

某个网格的点亮色**不在**本关调色板里，而它外环上有一块 ON 位置的瓦片此刻不是点亮色
⇒ 那块瓦片永远变不成点亮色（点一下只会沿调色板前进，造不出调色板外的色）⇒ 永远到不了。

它不是「挡住了」（这一局没有任何阻挡）：它是一个**守恒的性质**——「这块不是点亮色」每走一步都还成立。
证明对动作串做归纳（骨架 `unsolvable_of_invariant`），**不枚举任何状态**，与盘面大小无关。

                           钥匙套不上去是因为那张盘不是那种盘，不是因为我没找到证明

## 边界写在明处

`step` 没有重开那一支（重开不进模型），所以 `scene0_unsolvable` 的准确意思是

/-- 一步之后第 i 块的底色：在「被动」名单里就前进一位，否则原样。 -/
theorem step_color (w : World) (a : Act) (i : Nat) :
    (w.step a).colors[i]? =
      w.colors[i]?.map (fun c => if (w.board.moved a).contains i then advance w.board.palette c else c) := by
  simp only [World.step, step, List.getElem?_mapIdx]

/-- 调色板外的色，在这个世界的调色板上前进一位也到不了它。 -/
theorem advance_ne (w : World) (c lit : Nat) (hlit : lit ∉ w.board.palette) (hc : c ≠ lit) :
    advance w.board.palette c ≠ lit := by
  unfold advance
  split
  · rw [List.getD_eq_getElem?_getD]
    cases h : w.board.palette[(_ + 1) % w.board.palette.length]? with
    | none => simpa using hc
    | some x =>
      simp only [Option.getD_some]
      intro hx
      exact hlit (hx ▸ List.mem_of_getElem? h)
  · exact hc

/-- 钥匙一的不变量：网格 `g` 在盘上、点亮色不在调色板里、外环有一块 ON 位置的瓦片此刻不是点亮色。 -/
def Key (g : Grid) (w : World) : Prop :=
  g ∈ w.board.grids ∧ g.lit ∉ w.board.palette ∧
    ∃ p ∈ g.ring, p.2 = .on ∧ ∃ c, p.1.bind w.colorAt = some c ∧ c ≠ g.lit

theorem key_keep (g : Grid) : ∀ w a, Act.legal w a = true → Key g w → Key g (w.step a) := by
  intro w a _ ⟨hg, hlit, p, hp, hon, c, hc, hne⟩
  refine ⟨hg, hlit, p, hp, hon, ?_⟩
  cases hp1 : p.1 with
  | none => simp [hp1] at hc
  | some cell =>
    simp only [hp1, Option.bind_some, World.colorAt] at hc ⊢
    cases hi : w.board.indexOf cell with
    | none => simp [hi] at hc
    | some i =>
      simp only [hi, Option.bind_some] at hc
      have hb : (w.step a).board = w.board := rfl
      simp only [hb, hi, Option.bind_some, step_color, hc, Option.map_some]
      split
      · exact ⟨_, rfl, advance_ne w _ _ hlit hne⟩
      · exact ⟨_, rfl, hne⟩

theorem key_no_goal (g : Grid) (w : World) (h : Key g w) : w.isGoal = false := by
  obtain ⟨hg, _, p, hp, hon, c, hc, hne⟩ := h
  have hdone : w.gridDone g = false := by
    simp only [World.gridDone, Bool.and_eq_false_iff]
    left; left; left
    rw [List.all_eq_false]
    refine ⟨(p.1.bind w.colorAt, p.2), List.mem_map.mpr ⟨p, hp, rfl⟩, ?_⟩
    simp [hon, hc, hne]
  simp only [World.isGoal, List.all_eq_false]
  exact ⟨g, hg, by simp [hdone]⟩

theorem key_no_win (g : Grid) : ∀ w a, Act.legal w a = true → Key g w → (w.step a).isGoal = false :=
  fun w a hl h => key_no_goal g (w.step a) (key_keep g w a hl h)

/-- **钥匙一挂上骨架**：满足它的局面，只靠点击怎么走都到不了。 -/
theorem unsolvable_by_key (g : Grid) (w : World) (h : Key g w) : Unsolvable w :=
  unsolvable_of_invariant (Key g) (key_keep g) (key_no_win g) w h

  ∀ as : List Act, as.all (Act.legal w) = true → everGoal w as = false

/-- 一步之内版面不变，所以「合法」这件事走一步之后照样成立（它只看版面）。 -/
theorem legal_step (w : World) (a b : Act) : Act.legal (w.step a) b = Act.legal w b := by
  cases b <;> rfl

/-- **骨架**：Φ 每走一步都还成立（`hkeep`），满足 Φ 的局面这一步不可能到齐（`hwin`）
    ⇒ 从满足 Φ 的局面出发，**任何**合法动作串都到不了。 -/
theorem unsolvable_of_invariant (P : World → Prop)
    (hkeep : ∀ w a, Act.legal w a = true → P w → P (w.step a))
    (hwin : ∀ w a, Act.legal w a = true → P w → (w.step a).isGoal = false) :
    ∀ w, P w → Unsolvable w := by
  intro w hp as
  induction as generalizing w with
  | nil => intro _; rfl
  | cons a t ih =>
    intro hall
    simp only [List.all_cons, Bool.and_eq_true] at hall
    obtain ⟨ha, hrest⟩ := hall
    simp only [everGoal, hwin w a ha hp, Bool.false_or]
    apply ih (w.step a) (hkeep w a ha hp)
    rw [List.all_eq_true] at hrest ⊢
    intro b hb
    rw [legal_step]
    exact hrest b hb

theorem not_both (w : World) (hw : Winnable w) (hu : Unsolvable w) : False := by
  obtain ⟨as, hl, h⟩ := hw
  rw [hu as hl] at h
  exact Bool.false_ne_true h

def PAT_MID : Nat := 0     -- 该位置应涂本网格点亮色
def PAT_OTHER : Nat := 2   -- 该位置应涂调色板里别的某一色（同一网格统一一个）
def PAT_NONE : Nat := 3    -- 该位置没有可涂瓦片，没有目标

-- 外环第 k 个位置（跳过中心）在网格里的 (行, 列)
def slotIJ (k : Nat) : Nat × Nat :=
  if k < 3 then (0, k) else if k == 3 then (1, 0) else if k == 4 then (1, 2) else (2, k - 5)

-- 一条要求：物理位置 (r, c) 上的瓦片，按第 gi 个网格的图样该涂点亮色（on）还是「另一色」
structure Want where
  r : Int
  c : Int
  gi : Nat
  on : Bool

def wantsOf (gi : Nat) (g : Grid) : List Want :=
  ((List.range 8).zip g.codes).filterMap (fun p =>
    if p.2 == PAT_NONE then none
    else some { r := g.r + 8 * Int.ofNat (slotIJ p.1).1, c := g.c + 8 * Int.ofNat (slotIJ p.1).2,
                gi := gi, on := p.2 == PAT_MID })

def product : List (List Nat) → List (List Nat)
  | [] => [[]]
  | xs :: rest => xs.flatMap (fun v => (product rest).map (v :: ·))

-- 给每个网格选定「另一色」x 之后，一条要求的颜色
def wantColor (mids x : List Nat) (w : Want) : Nat := if w.on then mids.getD w.gi 0 else x.getD w.gi 0

-- 共用瓦片两边一致：同一物理位置上的所有要求颜色相同
def sharedAgree (ws : List Want) (mids x : List Nat) : Bool :=
  ws.all (fun w => match ws.find? (fun v => v.r == w.r && v.c == w.c) with
    | some v => wantColor mids x v == wantColor mids x w
    | none => true)

-- 目标色表 [(行, 列, 色)]：图样取值只许 0/2/3、点亮色要在调色板里；
-- 每个网格的「另一色」在调色板除点亮色之外的颜色里挑，组合里恰好一种满足共用瓦片一致才算定死，否则说不出话（none）
def ringTargets (gs : List Grid) (pal : List Nat) : Option (List (Int × Int × Nat)) :=
  if gs.isEmpty || !gs.all (fun g => pal.contains g.mid && g.codes.all (fun v => v == PAT_MID || v == PAT_OTHER || v == PAT_NONE))
  then none
  else
    let mids := gs.map (·.mid)
    let ws := ((List.range gs.length).zip gs).flatMap (fun p => wantsOf p.1 p.2)
    let cands := gs.map (fun g => pal.filter (fun v => v != g.mid))
    match ws, (product cands).filter (sharedAgree ws mids) with
    | [], _ => none
    | _, [x] => some (ws.map (fun w => (w.r, w.c, wantColor mids x w)))
    | _, _ => none

def BAR_W : Nat := 64          -- 计数条 = 最后一行，全宽
def BAR_FULL : Int := 12
def BAR_USED : Int := 11

-- Python round() 的半数取偶：num/den 恰为 .5 时取偶数那一边
def roundHalfEven (num den : Nat) : Nat :=
  let q := num / den
  let r := num % den
  if 2 * r > den then q + 1 else if 2 * r == den then q + q % 2 else q

-- 已用 u 格时最后一行的游程清点（_lib.bar_runs，生成器按 (色, 长度) 排序后拆成两槽）
def withBar (s : State) (u : Nat) : State :=
  if u == 0 then
    { s with bar_used := 0, bar_runs_n := 1, bar_runs_0_r := BAR_FULL, bar_runs_0_c := Int.ofNat BAR_W,
             bar_runs_1_r := none, bar_runs_1_c := none }
  else if u ≥ BAR_W then
    { s with bar_used := Int.ofNat BAR_W, bar_runs_n := 1, bar_runs_0_r := BAR_USED, bar_runs_0_c := Int.ofNat BAR_W,
             bar_runs_1_r := none, bar_runs_1_c := none }
  else
    { s with bar_used := Int.ofNat u, bar_runs_n := 2, bar_runs_0_r := BAR_USED, bar_runs_0_c := Int.ofNat u,
             bar_runs_1_r := some BAR_FULL, bar_runs_1_c := some (Int.ofNat (BAR_W - u)) }

-- bar_is_full_then_used 能在这份状态上核的部分：只有「未用」「已用」两色、至多两段，已用格数与游程一致。
-- （左右次序在生成器按色号排序拆槽时已经丢了，这半句问不出来。）
def barShapeOk (s : State) : Bool :=
  match s.bar_runs_n, s.bar_runs_1_r, s.bar_runs_1_c with
  | 1, none, none => (s.bar_runs_0_r == BAR_FULL && s.bar_used == 0)
                      || (s.bar_runs_0_r == BAR_USED && s.bar_used == s.bar_runs_0_c)
  | 2, some r1, some c1 => s.bar_runs_0_r == BAR_USED && r1 == BAR_FULL && s.bar_used == s.bar_runs_0_c
                            && s.bar_runs_0_c + c1 == Int.ofNat BAR_W
  | _, _, _ => false

-- ---------- 一步 ----------
def clickStep (s : State) (x y : Nat) : Option State :=
  let tl := tileList s
  match hitTile tl x y with
  | none => none
  | some h =>
    if !inActive s x y then some s
    else match allSome (tl.map (advance (palette s tl) h)), s.k_bar_num, s.k_bar_den with
      | some tl', some num, some den =>
        -- 凑齐目标的那一下会换关、整屏重画：新布局规则说不出，交给查表
        if matchedWith s tl' != some false then none
        else
          let c := s.h_clicks + 1
          let s1 := withBar { setTiles s (tl'.map encTile) with h_clicks := c }
                      (roundHalfEven (c.toNat * num.toNat) den.toNat)
          let g := pairOf (gridSet s1 tl')
          let p := pairOf (palette s1 tl')
          some { s1 with grid_palette_colors_r := g.1, grid_palette_colors_c := g.2,
                         palette_seq_r := p.1, palette_seq_c := p.2 }
      | _, _, _ => none

-- h 的各个记号指向的物理位置（上、下、左、右相距 8 像素；可以出界）
def markTargets (h : Tile) : List (Int × Int) :=
  let t : Int := h.t
  let l : Int := h.l
  (if hasBit h.marks 0 then [(t - 8, l)] else []) ++ (if hasBit h.marks 1 then [(t + 8, l)] else [])
  ++ (if hasBit h.marks 2 then [(t, l - 8)] else []) ++ (if hasBit h.marks 3 then [(t, l + 8)] else [])

def tileHere (tl : List Tile) (p : Int × Int) : Bool := tl.any (fun d => (d.t : Int) == p.1 && (d.l : Int) == p.2)

-- mark_directs_linkage 的另一半：记号指向的位置上点击前没有可涂瓦片（出界 / 空槽 / 别的网格中心）时，
-- 那个方向无效——点击后那里也没长出瓦片，且若那里是某个网格的中心，该网格（位置、点亮色、图样）原样不变。
def markFrozenOk (s s' : State) (h : Tile) : Bool :=
  (markTargets h).all (fun p =>
    tileHere (tileList s) p
    || (!tileHere (tileList s') p
        && (gridList s).all (fun g => !(g.r + 8 == p.1 && g.c + 8 == p.2) || (gridList s').contains g)))

-- nothing_else_changes：除底色外瓦片表逐块相同（位置、归属、记号都不动），且这一步不推的瓦片底色不变。
-- 只在读数层面核（瓦片表、网格槽位、图例、方框等读数不变），不逐像素核「别处一格不动」。
def othersKept (s s' : State) (h : Option Tile) : Bool :=
  let tl := tileList s
  let tl' := tileList s'
  tl.length == tl'.length
  && (tl.zip tl').all (fun p => skeleton p.1 == skeleton p.2
        && (match h with
            | some hh => affects hh p.1 || p.1.base == p.2.base
            | none => p.1.base == p.2.base))
