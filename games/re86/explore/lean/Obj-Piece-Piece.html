# Obj/Piece/Piece —— 会动的东西（对应 `objects/piece/piece.py`）

一件东西 = 几何族 + 颜色 + 包围盒 + hub + hub 画不画本色。
**全是定义，一条定理都没有。**

和 Python 那一份的差别只有一处，而且是有意的：这里没有 `known`。
`known` 是**观测的残缺**（某一边读不到底），不是世界的属性——它住在 `Obs/Check.lean` 的 `SeenPiece` 里。
模型里的东西每一边都是确定的。

格子归属用**算术**判（`mem`），不列格子：Python 那一份用格子集合，这里各自独立写，
两份的判决由逐帧核对（`Obs/Check.traceOk`）对账——**不许由一份导出另一份**。
-/

namespace Game

inductive Kind where
  | cross | box | ex | diamond
  deriving DecidableEq, Repr

structure Piece where
  kind   : Kind
  color  : Nat
  top    : Int
  left   : Int
  bottom : Int
  right  : Int
  hr     : Int      -- hub 行：cross 是两条杠的交点，ex / diamond 是对称中心
  hc     : Int      -- hub 列
  hollow : Bool     -- hub 那一格不画本色（ex / diamond / box）
  deriving DecidableEq, Repr

def iabs (x : Int) : Int := if x < 0 then -x else x

/-- ex / diamond 的半径 L（包围盒是正方形，边长 2L+1）。 -/
def Piece.radius (p : Piece) : Int := p.hr - p.top

/-- 包围盒中心，偶数边长取靠下靠右那一格。光标画在这一格。 -/
def Piece.cr (p : Piece) : Int := (p.top + p.bottom + 1) / 2
def Piece.cc (p : Piece) : Int := (p.left + p.right + 1) / 2

/-- (r, c) 这一格是不是它的（理想形状，不裁盘）。 -/
def Piece.mem (p : Piece) (r c : Int) : Bool :=
  match p.kind with
  | .cross =>
      (r == p.hr && decide (p.left ≤ c) && decide (c ≤ p.right)) ||
      (c == p.hc && decide (p.top ≤ r) && decide (r ≤ p.bottom))
  | .box =>
      (decide (p.top ≤ r) && decide (r ≤ p.bottom) && decide (p.left ≤ c) && decide (c ≤ p.right) &&
       (r == p.top || r == p.bottom || c == p.left || c == p.right)) ||
      (r == p.hr && c == p.hc && !p.hollow)
  | .ex =>
      let dr := iabs (r - p.hr)
      let dc := iabs (c - p.hc)
      (dr == dc && decide (1 ≤ dr) && decide (dr ≤ p.radius)) ||
      (dr == 0 && dc == 0 && !p.hollow)
  | .diamond =>
      let dr := iabs (r - p.hr)
      let dc := iabs (c - p.hc)
      (dr + dc == p.radius) || (dr == 0 && dc == 0 && !p.hollow)

/-- 整件平移。 -/
def Piece.shift (p : Piece) (dr dc : Int) : Piece :=
  { p with top := p.top + dr, left := p.left + dc, bottom := p.bottom + dr,
           right := p.right + dc, hr := p.hr + dr, hc := p.hc + dc }

end Game
