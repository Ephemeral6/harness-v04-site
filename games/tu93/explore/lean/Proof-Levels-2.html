/-! ## 第 2 关：认证段 seq 408–426，19 步（成绩单 level_actions = 19） -/

/-- 第 2 关的真实开局：seq 408 那一行动作之前的那一帧，按整关版面读出来。 -/
def L2_start : St := startAt 2 408

/-- 开局到过关前一步为止，18 步（动作, 下一帧）。过关那一脚不在这里：它的下一帧是第 3 关的盘。 -/
def L2_play : List (Dir × Obs) := playFrom 408 18

/-- 全部 19 个动作，最后一个是过关那一脚。 -/
def L2_acts : List Dir := actsFrom 408 19

theorem L2_play_len : L2_play.length = 18 := by decide +kernel
theorem L2_acts_len : L2_acts.length = 19 := by decide +kernel

/-- 开局读出来的是「一步都没走」（预算带满格）——明写的那条假设在这一关的落点。
    （写成 `St.used L_start` 而不写 `L_start.used`：装置的「长在世界上」按名字认，
    `L_start.used` 这个带点的名字它拆不出 `L_start`，会误判成「跟这一局无关」。） -/
theorem L2_start_fresh : St.used L2_start = 0 := by decide +kernel

#eval match firstSplit (levelBoard 2) boardOf L2_start L2_play 0 with
  | none   => "L2  一处都没分家（共 " ++ toString L2_play.length ++ " 步）"
  | some i => "L2  第 " ++ toString i ++ " 步分家"

/-- **模型沿着这条路一步不差。** 只有开局那一帧是观测给的，后面全是它自己算的。 -/
theorem L2_trace : traceOk (levelBoard 2) boardOf L2_start L2_play = true := by decide +kernel

/-- **这一关走得通。** 见证是认证段里真人走的那 19 步。 -/
theorem L2_winnable : Winnable (levelBoard 2) L2_start := ⟨L2_acts, by decide +kernel⟩
