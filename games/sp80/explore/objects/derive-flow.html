它是什么（derive/flow.py 的文件头）

`flow(w)` —— 一发水从发射口出发会怎么走：喂到哪些杯口、有没有水头落空。
由 发射口 / 件 / 杯口 / 画出来的颜色 组合而来。

## 水头的四种走法（起点册子最后一版的水流模型，那一版从两段流水动画逐帧读出）

「下」= 发射口的方向（发射器指向球）；「上」反之。一格一格走，一个水头一次走一格：

    落    下面一格空就往下落。下面是件：
            若左右恰有一侧是**同一件**（这一件的内角），就朝另一侧出一股射流；
            否则左右空的各出一股沿面流。
          下面是墙、杯：停（落空）。下面出了盘：流走。
    沿面  下面空就落；否则前面空就往前；前面出了盘：流走；否则停（落空）。——所以件两端会把水漏下去。
    射流  不看下面，前面空就一直走；前面出了盘：流走；撞墙/杯：停（落空）；撞件就在自己那一列上下各分一股「爬」。
    爬    侧前方一空就回到射流、朝原方向走；否则沿竖向继续；都走不了：停（落空）。

水头走进杯口，那一口就算喂到了，那个水头停下。

## 过关的判据（`fire_outcome`）

**每个杯口都被喂到，并且没有一个水头在盘里落空。** 流出盘外的不算落空。

「不许落空」这半句来自起点册子的 levelup_iff_branch_tree_feeds_every_mouth（v3）：
它在起点语料上三盘「每个杯口都接到了水、但另有一支落在杯口外的地板上」都没过关，
把这半句去掉就三条假阳。而起点册子最后一版的水流模型（只管第 5 关）没有这半句，
遇到水落在杯壁上干脆闭嘴。这里把两版合成一版：六关一条判据。
「流出盘外不算落空」同样来自那一版（「出界的那一支消失，不算失败」），
本批语料第 2 关三发过关每一发都有两股沿面流从盘的左右边缘流走，世界照样过关。

## 水从哪里出发

每颗球往「下」一格。**另外**：件里嵌着的记号格（色4）也是一个发射口——
记号往「下」第二格出水，前提是记号往「下」第一格是空的（那一格常驻一颗只在动画里看得见的球）。

## 说不出话的情形（`unsure` 非空）

**射流或爬**的水头走进已经湿了的格——一股射流横穿水柱时怎么走没有实测
（起点册子只有一条旁证：一发射流穿两道落柱、四口全中的摆法没过关）。
模型在这里不表态：`fire_outcome` 返回 None，规则判 NA，Lean 那边是「说不出话」那一支。

起点册子原来把**沿面流**走进湿格也算「交叉」而闭嘴。这批语料改了它：
第 2/3/4 关一共 9 发过关，每一发都有两股沿面流在同一件的面上汇合，世界 9 发全过——
沿面流汇合不是交叉。

## 「同一件」怎么认

**按画出来的连通块认**（件格的四连通块），不按件的身份认。
起点册子 overlap_reads_as_one_component：重叠能造出真的内拐角；
这批语料第 5 关的过关摆法正是两件叠在一起。

它的 Lean（Obj/Composite/Flow.lean，身份：定义）

/-
# Flow —— 一发水怎么走（对应 `derive/flow.py`，那一份的文件头是这一份的讲解）

「下」= 发射口的方向。水头四种：落 / 沿面 / 射流 / 爬，一轮每个水头走一格；
走进杯口就算喂到；走不动时要么是**流出盘外**（不算），要么是**在盘里落空**（记一笔）。
射流或爬走进已经湿了的格：**模型说不出话**（`unsure`）。

过关 ⟺ 每个杯口都被喂到 ∧ 没有一个水头在盘里落空 ∧ 模型没有说不出话。

这是一个**有燃料的模拟**：最多走 100 轮，燃料用完还有水头就算说不出话。
这一份没有定理；它是「一发水」的定义，被 `Rule/Win/FireWins.lean` 用。
-/
import Obj.Composite.Mouths

namespace Game

inductive HK where
  | fall | surf | jet | climb
deriving DecidableEq, Repr

/-- 一个水头：种类、所在格、横向朝右吗、（爬的时候）竖向朝行号大的一侧吗。 -/
structure Head where
  kind : HK
  pos : Pos
  right : Bool
  vdown : Bool
deriving DecidableEq, Repr

def HK.code : HK → Nat
  | .fall => 0 | .surf => 1 | .jet => 2 | .climb => 3

/-- 去重用的键：同一格、同一种、同一方向就是同一个水头。 -/
def Head.key (h : Head) : Nat :=
  h.pos * 16 + h.kind.code * 4 + (if h.right then 2 else 0) + (if h.vdown then 1 else 0)

def hstep (right : Bool) (p : Pos) : Pos := if right then p + 1 else p - 1
def vstep (down : Bool) (p : Pos) : Pos := if down then p + 64 else p - 64

def adj (p q : Pos) : Bool := q == p + 1 || p == q + 1 || q == p + 64 || p == q + 64

/-- 件格里含 `seed` 的那一块画出来的连通块（最多扩 `fuel` 轮）。 -/
def compOf (cells : List Pos) : Nat → List Pos → List Pos
  | 0, acc => acc
  | f + 1, acc =>
      let more := cells.filter (fun q => !acc.contains q && acc.any (fun s => adj s q))
      if more.isEmpty then acc else compOf cells f (acc ++ dedup more)

/-- 这一轮某个水头的结果：下一步的水头，以及它是不是在盘里落空了。 -/
structure Out where
  next : List Head
  stray : Bool

section
variable (b : Board) (w : World) (down : Bool) (cells : List Pos)

def free (p : Pos) : Bool := b.grid.inside p && colorAt b w p == 12

def stepHead (h : Head) : Out :=
  let p := h.pos
  match h.kind with
  | .fall =>
    let below := vstep down p
    if free b w below then ⟨[⟨.fall, below, false, false⟩], false⟩
    else if !b.grid.inside below then ⟨[], false⟩
    else if !cells.contains below then ⟨[], true⟩
    else
      let comp := compOf cells cells.length [below]
      let lf := free b w (p - 1)
      let rf := free b w (p + 1)
      if comp.contains (p - 1) && rf then ⟨[⟨.jet, p + 1, true, false⟩], false⟩
      else if comp.contains (p + 1) && lf then ⟨[⟨.jet, p - 1, false, false⟩], false⟩
      else
        let out := (if lf then [⟨.surf, p - 1, false, false⟩] else []) ++
                   (if rf then [⟨.surf, p + 1, true, false⟩] else [])
        ⟨out, out.isEmpty && b.grid.inside (p - 1) && b.grid.inside (p + 1)⟩
  | .surf =>
    let below := vstep down p
    let front := hstep h.right p
    if free b w below then ⟨[⟨.fall, below, false, false⟩], false⟩
    else if free b w front then ⟨[⟨.surf, front, h.right, false⟩], false⟩
    else if !b.grid.inside front then ⟨[], false⟩
    else ⟨[], true⟩
  | .jet =>
    let front := hstep h.right p
    if free b w front then ⟨[⟨.jet, front, h.right, false⟩], false⟩
    else if !b.grid.inside front then ⟨[], false⟩
    else if !cells.contains front then ⟨[], true⟩
    else
      let out := (if free b w (p - 64) then [⟨.climb, p - 64, h.right, false⟩] else []) ++
                 (if free b w (p + 64) then [⟨.climb, p + 64, h.right, true⟩] else [])
      ⟨out, out.isEmpty⟩
  | .climb =>
    let side := hstep h.right p
    let up := vstep h.vdown p
    if free b w side then ⟨[⟨.jet, side, h.right, false⟩], false⟩
    else if free b w up then ⟨[⟨.climb, up, h.right, h.vdown⟩], false⟩
    else if !b.grid.inside up || !b.grid.inside side then ⟨[], false⟩
    else ⟨[], true⟩

end

/-- 模拟的状态。 -/
structure Sim where
  heads : List Head
  visited : List Nat
  wet : List Pos
  fed : List Pos
  strays : Nat
  unsure : Bool

/-- 把这一轮新长出来的水头一个一个收进来（去重、喂杯口、查交叉）。 -/
def admit (mo : List Pos) (s : Sim) (h : Head) : Sim :=
  if s.visited.contains h.key then s
  else
    let s := { s with visited := h.key :: s.visited }
    if mo.contains h.pos then { s with fed := if s.fed.contains h.pos then s.fed else h.pos :: s.fed }
    else
      let cross := s.wet.contains h.pos && (h.kind == .jet || h.kind == .climb)
      { s with unsure := s.unsure || cross, wet := h.pos :: s.wet, heads := s.heads ++ [h] }

def round (b : Board) (w : World) (down : Bool) (cells mo : List Pos) (s : Sim) : Sim :=
  let outs := s.heads.map (stepHead b w down cells)
  let strays := (outs.filter (·.stray)).length
  let nxt := outs.flatMap (·.next)
  nxt.foldl (admit mo) { s with heads := [], strays := s.strays + strays }

def run (b : Board) (w : World) (down : Bool) (cells mo : List Pos) : Nat → Sim → Sim
  | 0, s => if s.heads.isEmpty then s else { s with unsure := true }
  | f + 1, s => if s.heads.isEmpty then s else run b w down cells mo f (round b w down cells mo s)

/-- 水从哪里出发：每颗球往「下」一格；件里的记号往「下」第二格（第一格得是空的）。 -/
def springs (b : Board) (w : World) (down : Bool) : List Pos :=
  let balls := b.sources.map (fun s => vstep down s.ball)
  (w.pieces.flatMap (·.marks)).foldl
    (fun acc m =>
      let m1 := vstep down m
      let m2 := vstep down m1
      if b.grid.inside m1 && colorAt b w m1 == 12 && !acc.contains m2 then acc ++ [m2] else acc)
    balls

def start (b : Board) (w : World) (down : Bool) (mo : List Pos) : Sim :=
  (springs b w down).foldl
    (fun s p =>
      if !(free b w p) then s
      else if mo.contains p then { s with fed := if s.fed.contains p then s.fed else p :: s.fed }
      else { s with wet := p :: s.wet, heads := s.heads ++ [⟨.fall, p, false, false⟩],
                    visited := (Head.key ⟨.fall, p, false, false⟩) :: s.visited })
    ⟨[], [], [], [], 0, false⟩

/-- 这一发会不会过关：some true / some false；说不出话给 none。 -/
def fireOutcome (b : Board) (w : World) : Option Bool :=
  match b.sources with
  | [] => none
  | s :: rest =>
    let down := s.fallsDown
    let vertical (t : Source) : Bool := t.ball == t.launcher + 64 || t.launcher == t.ball + 64
    if rest.any (fun t => t.fallsDown != down) || !(b.sources.all vertical) then none
    else
      let mo := mouths b w
      let cells := dedup (pieceCells w)
      let fin := run b w down cells mo 100 (start b w down mo)
      if fin.unsure then none
      else some (!mo.isEmpty && mo.all (fun m => fin.fed.contains m) && fin.strays == 0)

end Game
