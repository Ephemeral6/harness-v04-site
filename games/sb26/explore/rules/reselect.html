def scope (L : Layout) (w : World) (t : Cell) : Bool :=
  match w.sel with
  | some s => s != t && (w.pieceAt t).isSome && !L.onBoard s && !L.onBoard t

def scope_(prev, action, nxt, ctx=None):
    return _scope(click_case(try_world(prev), try_world(nxt), action, ctx, unreadable(prev) or unreadable(nxt)))


def _scope(r):
    why, case = r
    if why:
        return why
    w, _nw, sel, tgt = case
    if sel is None:
        return "手上没料 —— 归 pick / hand_empty / void_click"
    if tgt is None:
        return "点空处 —— 归 void_click"
    if tgt == sel:
        return "点中它自己 —— 归 unpick"
    if w.piece_at(tgt) is None:
        return "点中空位 —— 归 place"
    if w.on_board(sel) or w.on_board(tgt):
        return "两位不都在料架上 —— 板上那半归 swap；跨侧没有规则"
    return None


def assert_(prev, action, nxt, ctx=None):
    r = click_case(try_world(prev), try_world(nxt), action, ctx, unreadable(prev) or unreadable(nxt))
    if _scope(r):
        return "NA"
    w, nw, _sel, tgt = r[1]
    return "TRUE" if nw == replace(w, selected=tgt) else "FALSE"
