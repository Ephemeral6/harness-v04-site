/-! ## 第 4 关 —— 认证段 seq 355–371：走 16 步、第 17 步按提交过关 -/

def L4_layout : Layout := Replay.layoutAt 4

/-- 第 4 关的真实开局：**从语料里读出来的**（seq 355 的前一帧），不是手写的盘。 -/
def L4_start : World := Replay.worldAt (Replay.bySeq 355).prev

/-- 开局到过关前一步，16 步，每步配观测到的下一帧。 -/
def L4_play : List (Act × World) := Replay.playFrom 355 16

theorem L4_len : (L4_play.map Prod.snd : List World).length = 16 := by decide

#eval match firstSplit L4_layout L4_start L4_play 0 with
  | none   => "L4 一处都没分家（共 " ++ toString L4_play.length ++ " 步）"
  | some i => "L4 第 " ++ toString i ++ " 步分家"

/-- **模型沿着这条路一步不差**：只有开局那一帧是观测给的，后面 16 帧全是模型自己算的。 -/
theorem L4_trace : traceOk L4_layout (L4_start : World) L4_play = true := by decide

/-- 观测到的过关那一步（seq 371）：模型在那一帧上按同一个键，也说「过关」。 -/
theorem L4_observed_win : (Replay.bySeq 371).obs = Replay.ObsIx.levelUp ∧
    step L4_layout (Replay.worldAt (Replay.bySeq 371).prev : World) (Replay.bySeq 371).act = Res.won := by
  decide

/-- **第 4 关走得通。** 见证：上面那 16 步，再加过关那一脚（提交）。 -/
theorem L4_winnable : Winnable L4_layout (L4_start : World) :=
  ⟨L4_play.map Prod.fst ++ [Act.submit], by decide⟩
