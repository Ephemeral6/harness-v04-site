/-
# World —— 这一局的全部状态（对应 `objects/world.py`），**拆成两半**

    Layout   版面：一关之内不变（target · boards · rack）
    World    局面：一步就可能变（pieces · selected · used）

Python 那一侧 `World` 是一个 dataclass 装六个字段；这里把版面单独拿出来住进 `Layout`，
`step` 的类型是 `Layout → World → Act → Res` —— 它**读**版面，但返回值里**没有**版面。
于是「走一步版面变了」这句话**表示不出来**，不用证（规则 `invariant/layout_frozen` 的身份因此是「推论」）。

动作与结果也在这里定义（它们是 `step` 的签名的一部分，不是规则）。
-/

import Obj.Geometry.Cell
import Obj.Piece.Piece
import Obj.Layout.Board

namespace Game

/-- 版面。`rack` 是料架上的位，从左到右。 -/
structure Layout where
  target : List Nat
  boards : List Board
  rack   : List Cell
deriving DecidableEq, Repr, Inhabited

/-- 局面。`pieces`：哪一位上放着哪块料（空位不出现，顺序无关）。 -/
structure World where
  pieces : List (Cell × Piece)
  sel    : Option Cell
  used   : Nat
deriving DecidableEq, Repr, Inhabited

/-- 动作：点一格 / 提交（ACTION5）/ 撤销（ACTION7）/ 重开（RESET）。 -/
inductive Act where
  | click (c : Cell)
  | submit
  | undo
  | reset
deriving DecidableEq, Repr, Inhabited

/-- 一步的结果。**`unknown` = 模型在这一步说不出话**（撤销、重开要历史；跨侧点料语料 0 次）。
    它不是「什么都没发生」—— 那是 `ok w`。 -/
inductive Res where
  | ok (w : World)
  | won
  | unknown
deriving DecidableEq, Repr, Inhabited

/-- 全部的位：各板的位 + 料架的位。 -/
def Layout.locs (L : Layout) : List Cell := L.boards.flatMap (fun b => b.slots) ++ L.rack

/-- 这一位在不在某一块板上（不在就在料架上）。 -/
def Layout.onBoard (L : Layout) (c : Cell) : Bool := L.boards.any (fun b => b.slots.contains c)

/-- 这一位上放着什么。 -/
def World.pieceAt (w : World) (c : Cell) : Option Piece :=
  (w.pieces.find? (fun x => x.1 == c)).map Prod.snd

/-- 两个局面相等（料的表按集合比：帧上读不出「料的表」的顺序）。 -/
def World.same (a b : World) : Bool :=
  a.sel == b.sel && a.used == b.used && a.pieces.length == b.pieces.length &&
    a.pieces.all (fun x => b.pieces.contains x)

end Game

"""`World` —— 这一局的**全部状态**。字段就是对象。

每个字段先回答判据 2：

> **去掉它之后，找不找得出一对世界状态，原来能区分、现在无论如何都区分不了？**

见证写在 `analysis/<字段>.txt`，分离表在 `analysis/field_split.txt`。

## 两半：版面 / 局面

    版面（一关之内不变）   target · boards · rack
    局面（一步就可能变）   pieces · selected · used

形式化那一侧把版面住进一个 `step` 碰不到的结构（`Board`），局面才是状态。

## 没进来的东西，以及为什么

* 关卡号、`WIN` / `NOT_FINISHED`：上游 API 给的，世界里再存一份是第二份副本（戒律 3）。
* 计数条的总长：整批语料里恒为 64，**一位信息都不带**，卡 03 删掉（见 `analysis/record.txt` 第 1 轮）。
* 面板重复画了几排：卡 06 删掉（消融 0、没有一条动态规则读它；它的用处只在解析里——两排相同只取一排）。
* 环底下那根管子：卡 06 删掉（消融 0；那个环在语料里从没被动过，「动它会怎样」进欠账）。
* 空位标记：它和「这一位上没有料」是**同一个量的两种读法**，写成解析的良构检查，不进字段。
* 板框、角标、管子的**像素**：解析把它们认领掉（「每一格非底色都要有人认领」），但不进字段。
"""
from __future__ import annotations

from dataclasses import dataclass

from .geometry.cell import Cell
from .layout.board import Board
from .piece.piece import Piece


@dataclass(frozen=True)
class World:
    """一帧世界。"""

    # ---- 版面：一关之内不变
    target: tuple[int, ...]                 # 上排面板各框的边框色，从左到右 —— 要凑出的那一串
    boards: tuple[Board, ...]               # 计数条上方的每一块板（按框色排）
    rack: tuple[Cell, ...]                  # 计数条下方料架上的位，从左到右

    # ---- 局面：一步就可能变
    pieces: tuple[tuple[Cell, Piece], ...]  # 哪个位上放着哪块料（按位排；空位不出现）
    selected: Cell | None                   # 被选中框套住的那一位；没有则 None
    used: int                               # 计数条已经吃掉几格

    # ---------------------------------------------------------------- 只读的便利
    def piece_at(self, loc: Cell) -> Piece | None:
        for c, p in self.pieces:
            if c == loc:
                return p
        return None

    def locations(self) -> tuple[Cell, ...]:
        """全部的位：各板上的 + 料架上的。**版面的一部分**，一关之内不变。"""
        out = [s for b in self.boards for s in b.slots] + list(self.rack)
        return tuple(sorted(out))

    def on_board(self, loc: Cell) -> bool:
        return any(loc in b.slots for b in self.boards)

    def board_of(self, loc: Cell) -> Board | None:
        for b in self.boards:
            if loc in b.slots:
                return b
        return None
