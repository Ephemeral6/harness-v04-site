"""帧 → 世界。**全局唯一碰像素的地方。** 上面每一层只认 `World` 的字段。

## 五个界面色（跨关不变，整批语料逐帧核过）

    BG=4     底色
    MARK=2   空位标记（2×2）与计数条的未用格
    USED=3   计数条的已用格
    SEL=0    选中框（6×6 空心 20 格）与起读板的四角角标
    PANEL=5  上排面板的底

它们是**界面**的颜色，不是料的颜色：料色、板框色、面板目标色一律按结构认，
不按色号认（册子里吃过「颜色白名单」的亏：某一关一块砖的颜色恰好等于另一关的框色）。

## 良构检查：同一个量的两种读法必须对得上，对不上就抛

* 计数条：已用格必须是从最右端起连续的一段（「用了几格」读成数和读成形状要一致）；
* 每个位上**恰好**一样东西：要么一块料，要么一个空位标记（「这一位空不空」两种读法）；
* 选中框套住的必须是一块料（「选中了谁」两种读法：框的位置 / 框里那块料）；
* 计数条上方的每个位**恰好**贴着一块板的底色（「这一位属于哪块板」读不出唯一答案就不硬猜）。

读不出来一律抛 `ParseError`，**这里只管抛，不管归类**（见 `ParseError`）。
"""
from __future__ import annotations

import sys
from pathlib import Path

from .geometry.cell import Cell
from .layout.board import Board
from .piece.piece import Piece
from .world import World

try:                                        # 跨局几何（纯计算，没有世界断言）
    from geo import bbox, cells_of, components
except ImportError:                         # 不经 settle 直接跑时，自己把它找进来
    sys.path.insert(0, str(next(p for p in Path(__file__).resolve().parents
                                if (p / "ledger" / "common" / "geo.py").is_file()) / "ledger" / "common"))
    from geo import bbox, cells_of, components

BG, MARK, USED, SEL, PANEL = 4, 2, 3, 0, 5


class ParseError(Exception):
    """这一帧读不成一个世界。**这是一条可判错的话**，不是一次异常处理。

    读不成的帧要归成两类，由拿得到 meta 的那一层（`run_rules.py`）归：

        (a) 不是局面   终局画面 —— 它画的不是一张可玩的盘
        (b) 读不出来   是一张盘而我读不出来 —— **只有这一种是欠账**
    """


# ---------------------------------------------------------------- 动作

def parse_action(action: str):
    """动作串 → (种类, 点击格)。`ACTION6:x,y` 的 x 是列、y 是行。

    种类：'click' / 'submit'（ACTION5）/ 'undo'（ACTION7）/ 'reset'（RESET）/ 'other'。
    """
    s = str(action)
    if s.startswith("ACTION6"):
        if ":" not in s:
            return ("other", None)
        try:
            x, y = (int(v) for v in s.split(":", 1)[1].split(","))
        except ValueError:
            return ("other", None)
        return ("click", Cell(y, x))
    if s == "ACTION5":
        return ("submit", None)
    if s == "ACTION7":
        return ("undo", None)
    if s.upper().startswith("RESET"):
        return ("reset", None)
    return ("other", None)


# ---------------------------------------------------------------- 帧

def parse(frame) -> World:
    """一帧 → 一个 World。读不出来抛 `ParseError`。"""
    g = frame
    H = len(g)
    if H == 0:
        raise ParseError("空帧")
    W = len(g[0])

    # ---- 计数条：整行只有「未用 / 已用」两色的那一行
    bars = [r for r in range(H) if len(g[r]) == W and all(v in (MARK, USED) for v in g[r])]
    if len(bars) != 1:
        raise ParseError(f"计数条：找到 {len(bars)} 行（要恰好 1 行）")
    bar = bars[0]
    used = sum(1 for v in g[bar] if v == USED)
    if any(v != USED for v in g[bar][W - used:]) or any(v != MARK for v in g[bar][:W - used]):
        raise ParseError("计数条：已用格不是从最右端起连续的一段")

    # ---- 面板：PANEL 底色那块矩形里的 6×6 空心框
    pcells = cells_of(g, PANEL)
    if not pcells:
        raise ParseError("面板：没有")
    pt, pb, pl, pr = bbox(pcells)
    boxes = []
    for v in sorted({u for row in g[pt:pb + 1] for u in row[pl:pr + 1]} - {BG, PANEL}):
        inside = [(r, c) for (r, c) in cells_of(g, v) if pt <= r <= pb and pl <= c <= pr]
        for comp in components(inside):
            t, b, l, r = bbox(comp)
            if len(comp) == 20 and (b - t, r - l) == (5, 5):
                boxes.append((t, l, v))
    if not boxes:
        raise ParseError("面板：一个框都没有")
    rows: dict[int, list] = {}
    for (t, l, v) in sorted(boxes):
        rows.setdefault(t, []).append(v)
    seqs = [tuple(rows[t]) for t in sorted(rows)]
    # 面板画了几排不进世界（卡 06 删）：两排一模一样是显示方式，只取一排。
    # 两排不同的面板语料里 0 次；那时按行续上去是猜法，没被检验过。
    target = seqs[0] if all(s == seqs[0] for s in seqs) else tuple(x for s in seqs for x in s)

    def outside_panel(r: int) -> bool:
        return r > pb

    # ---- 料：4×4 实心（砖）/ 4×4 空心单色环（指路环）
    found: dict[Cell, Piece] = {}
    for v in sorted({u for row in g for u in row} - {BG, MARK, USED, SEL, PANEL}):
        for comp in components(cells_of(g, v)):
            t, b, l, r = bbox(comp)
            if len(comp) == 16 and (b - t, r - l) == (3, 3) and outside_panel(t):
                found[Cell(t + 1, l + 1)] = Piece("brick", v)
    for t in range(pb + 1, H - 3):
        for l in range(W - 3):
            v = g[t][l]
            if v in (BG, MARK, USED, SEL, PANEL):
                continue
            if any(g[t + dr][l + dc] != v for dr in range(4) for dc in range(4)
                   if dr in (0, 3) or dc in (0, 3)):
                continue
            if any(g[t + dr][l + dc] != BG for dr in (1, 2) for dc in (1, 2)):
                continue
            loc = Cell(t + 1, l + 1)
            if loc in found:
                raise ParseError(f"料：{loc} 同时读成砖和环")
            found[loc] = Piece("link", v)

    # ---- 空位标记：2×2 的 MARK（计数条那一行不算）
    holes = set()
    for comp in components([(r, c) for (r, c) in cells_of(g, MARK) if r != bar]):
        t, b, l, r = bbox(comp)
        if len(comp) == 4 and (b - t, r - l) == (1, 1):
            holes.add(Cell(t, l))
        else:
            raise ParseError(f"空位标记：一块 {len(comp)} 格的 MARK 不是 2×2")
    both = holes & set(found)
    if both:
        raise ParseError(f"位 {sorted(both)[0]} 上同时有料和空位标记")

    # ---- 选中框：SEL 色的 6×6 空心 20 格
    selected = None
    ring_cells: set = set()
    sel_comps = components(cells_of(g, SEL))
    for comp in sel_comps:
        t, b, l, r = bbox(comp)
        if len(comp) == 20 and (b - t, r - l) == (5, 5):
            if selected is not None:
                raise ParseError("选中框：不止一个")
            selected = Cell(t + 2, l + 2)
            ring_cells = set(comp)
    if selected is not None and selected not in found:
        raise ParseError(f"选中框套住的 {selected} 上没有料")

    # ---- 位 = 料的位 ∪ 空位标记的位；按计数条分上下两侧
    locs = sorted(set(found) | holes)
    above = [x for x in locs if x.row < bar]
    below = [x for x in locs if x.row > bar]
    rack = tuple(sorted(below, key=lambda x: (x.col, x.row)))

    # ---- 板：每个上侧的位贴着哪一块底色连通块
    def foot(x: Cell) -> set:
        if x in found:
            s = {(x.row - 1 + dr, x.col - 1 + dc) for dr in range(4) for dc in range(4)}
        else:
            s = {(x.row + dr, x.col + dc) for dr in range(2) for dc in range(2)}
        if x == selected:
            s |= ring_cells
        return s

    bg_comps = components(cells_of(g, BG))
    owner = {}
    for i, comp in enumerate(bg_comps):
        for cell in comp:
            owner[cell] = i
    occupied = set()
    for x in locs:
        occupied |= foot(x)
    groups: dict[int, list] = {}
    for x in above:
        f = foot(x)
        touching = set()
        for (r, c) in f:
            for (rr, cc) in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if (rr, cc) not in f and (rr, cc) in owner:
                    touching.add(owner[(rr, cc)])
        if len(touching) != 1:
            raise ParseError(f"位 {x} 贴着 {len(touching)} 块底色，认不出它属于哪块板")
        i = touching.pop()
        if i == 0:
            raise ParseError(f"位 {x} 贴着的是外面那一大块底色，不在任何板里")
        groups.setdefault(i, []).append(x)

    boards = []
    for i, slots in groups.items():
        comp = bg_comps[i]
        votes: dict[int, int] = {}
        for (r, c) in comp:
            for (rr, cc) in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if not (0 <= rr < H and 0 <= cc < W):
                    continue
                if (rr, cc) in owner or (rr, cc) in occupied or g[rr][cc] == SEL:
                    continue
                votes[g[rr][cc]] = votes.get(g[rr][cc], 0) + 1
        if not votes:
            raise ParseError(f"板：第 {i} 块底色四周一格框都没有")
        frame_color = max(votes.items(), key=lambda kv: (kv[1], -kv[0]))[0]
        t, _b, l, _r = bbox(comp)
        root = (0 <= t - 2 and 0 <= l - 2 and g[t - 2][l - 2] == SEL
                and g[t - 2][l - 1] == SEL and g[t - 1][l - 2] == SEL)
        boards.append(((t, l), Board(frame_color, root, tuple(sorted(slots)))))
    boards.sort()
    frames = [b.frame for _, b in boards]
    if len(set(frames)) != len(frames):
        raise ParseError("两块板同一个框色 —— 环指过来分不清是哪块")

    # ---- 环底下的管子：只认领像素，不进世界（卡 06 删；那个环从没被动过）
    pipe_cells: set = set()
    for x, p in found.items():
        if p.kind == "link" and x.row + 3 < H:
            if g[x.row + 3][x.col] == p.color and g[x.row + 3][x.col + 1] == p.color:
                r = x.row + 3
                while r < H and g[r][x.col] == p.color and g[r][x.col + 1] == p.color:
                    pipe_cells |= {(r, x.col), (r, x.col + 1)}
                    r += 1

    # ---- 良构：**每一格非底色都要有人认领**
    # 册子那一侧有「面板 8 行一格不动」这种按像素说的话；对象层接手之后，那句话的内容
    # 只剩下「面板那一串没变」，而面板里多出一格杂色、板框上缺一格，世界都读不出来 ——
    # 于是一批「改了几格」的变异被旧册子拒掉、被新规则放过（compare_book.py 实测 7 个）。
    # 所以：认得出的对象把自己占的格认领走，剩下还有非底色的格 → 这一帧读不成世界。
    claimed: set = {(bar, c) for c in range(W)}
    claimed |= {(r, c) for r in range(pt, pb + 1) for c in range(pl, pr + 1)
                if g[r][c] == PANEL}
    for (t, l, _v) in boxes:
        claimed |= {(t + dr, l + dc) for dr in range(6) for dc in range(6) if dr in (0, 5) or dc in (0, 5)}
    for x in locs:
        claimed |= foot(x)
    claimed |= ring_cells | pipe_cells
    for (t, l), b in boards:
        comp = bg_comps[next(i for i, s in groups.items() if s and s[0] in b.slots)]
        it, ib, il, ir = bbox(comp)
        for c in range(il - 1, ir + 2):
            for r in (it - 1, ib + 1):
                if g[r][c] == b.frame or (r, c) in pipe_cells:
                    claimed.add((r, c))
        for r in range(it, ib + 1):
            for c in (il - 1, ir + 1):
                if g[r][c] == b.frame:
                    claimed.add((r, c))
        if b.root:
            for (r0, c0, dr, dc) in ((it - 2, il - 2, 1, 1), (it - 2, ir + 2, 1, -1),
                                     (ib + 2, il - 2, -1, 1), (ib + 2, ir + 2, -1, -1)):
                for (r, c) in ((r0, c0), (r0, c0 + dc), (r0, c0 + 2 * dc), (r0 + dr, c0), (r0 + 2 * dr, c0)):
                    if 0 <= r < H and 0 <= c < W and g[r][c] == SEL:
                        claimed.add((r, c))
    stray = [(r, c) for r in range(H) for c in range(W) if g[r][c] != BG and (r, c) not in claimed]
    if stray:
        raise ParseError(f"有 {len(stray)} 格非底色没人认领（第一格 {stray[0]}，色 {g[stray[0][0]][stray[0][1]]}）")

    return World(
        target=target,
        boards=tuple(b for _, b in boards),
        rack=rack,
        pieces=tuple(sorted(found.items())),
        selected=selected,
        used=used,
    )
