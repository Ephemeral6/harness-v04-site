# Obj/Palette/Swatch —— 一个色环（对应 `objects/palette/swatch.py`）

5×5 的按钮，芯是一种颜料色。点中它（整块 5×5 都算）换颜料。
-/

import Obj.Geometry.Box

namespace Game

structure Swatch where
  box   : Box
  color : Nat
  deriving DecidableEq, Repr

/-- 点在 (r, c) 点中了哪个色环；没点中返回 `none`。按列从左到右找第一个。 -/
def swatchHit : List Swatch → Nat → Nat → Option Swatch
  | [],      _, _ => none
  | s :: ss, r, c => if s.box.contains r c then some s else swatchHit ss r c

end Game
