/-- 第 2 关的真实开局：认证段 seq 351 那一帧（**从语料的帧里读出来的**，不是手写的盘）。 -/
def L2_start : World := worldOf (rowAt 351)

/-- 开局到过关前一步，20 步（seq 351..370）。 -/
def L2_play : List (Act × St) := pathIn 351 370

/-- 过关那一脚（seq 371）。它的下一帧是下一关的盘，观测对不上，所以不在路里。 -/
def L2_win : Act := (rowAt 371).act

#eval match firstSplit step L2_start L2_play 0 with
  | none   => "L2  一处都没分家（共 " ++ toString L2_play.length ++ " 步）"
  | some i => "L2  第 " ++ toString i ++ " 步分家"

/-- **模型沿这条 20 步的路一步不差**：只有开局是观测给的，后面全是它自己算的。 -/
theorem L2_trace : L2_play.length = 20 ∧ traceWith step (L2_start : World) L2_play = true := by
  decide

/-- **第 2 关走得通。** 见证 = 那 20 步 + 过关那一脚。 -/
theorem L2_winnable : Winnable (L2_start : World) :=
  ⟨L2_play.map Prod.fst ++ [L2_win], by decide⟩

#print axioms L2_trace
#print axioms L2_winnable
