记号盘上多两条限制：同一条指令连着 4 次及以上作废；用到的不同指令少于 3 种作废。
（三种判决相同的读法里取的一种；分辨实验见 `analysis/gaps.txt` G9。）
-/
import Obj.Compose.Program
import Obj.Compose.Walk
import Obj.Frame.Layout
import Obj.Frame.World

namespace Game

inductive Outcome
  | win
  | move (f : Form)
  | void
deriving DecidableEq, Repr

/-- 最长连段：同一条指令连着出现最多几次。 -/
def maxRunAux : Option Op → Nat → Nat → List Op → Nat
  | _,    _,   best, []        => best
  | prev, cur, best, op :: ops =>
      let cur' := if prev = some op then cur + 1 else 1
      maxRunAux (some op) cur' (max best cur') ops

def maxRun (ops : List Op) : Nat := maxRunAux none 0 0 ops

def distinctCount (ops : List Op) : Nat := ops.eraseDups.length

def strictVoid (b : Board) (ops : List Op) : Bool :=
  !b.marks.isEmpty && (decide (4 ≤ maxRun ops) || decide (distinctCount ops < 3))

def outcome (L : Layout) (w : World) : Outcome :=
  match decode w.panel L.toolbox with
  | none     => .void
  | some ops =>
      if strictVoid L.board ops then .void
      else
        let e := walk L.board w.piece ops
        if e = L.board.socket then .win
        else if e.footprint.all (fun x => (visiblePads L.pads w.piece).contains x) then .move e
        else .void

end Game

"""交卷的结果。组合自：`panel` + `layout.toolbox`（读成程序）、`piece` + `layout.board`（走程序）、
`pads`（终点能不能停）、`layout.board.socket`（填没填满）、`layout.board.marks`（记号盘多两条限制）。

    ("win",  终点的形)    终点的形和插槽空腔一模一样
    ("move", 终点的形)    终点盖住的格全是看得见的花格
    ("void", 理由)        其余一切：件一格不挪

## 记号盘上多两条限制（整局只有一块带记号的盘）

    同一条指令连着出现 4 次及以上 → 整道作废
    整道程序用到的不同指令少于 3 种 → 整道作废

这是三种在本批语料上判决完全相同的读法里取的一种（`python alt_readings.py`，14 次记号盘交卷三种都 14/14）。
取它的理由写在明处：它条件最少；另一种（第一版用过的「记号挡路 + 光带那一行只许停一格」）
对「件在 (6,2) 交六个北」给出「过关」，而起点册子的改版史记着另一次跑里一道「连六」的程序在这块盘上作废——
那条记录不在本批语料里、核不了，只拿来在分不开的读法之间挑，不当证据用。分辨实验在 `analysis/gaps.txt` G9。
"""
from __future__ import annotations

from .program import decode
from .walk import walk

STRICT_RUN = 4                               # 记号盘：同一条指令连到这个数就作废
STRICT_KINDS = 3                             # 记号盘：用到的不同指令少于这个数就作废


def max_run(ops) -> int:
    best = cur = 0
    prev = None
    for op in ops:
        cur = cur + 1 if op == prev else 1
        prev = op
        best = max(best, cur)
    return best


def strict_void(board, ops):
    """记号盘上这道程序作不作废：作废返回理由，不作废（或不是记号盘）返回 None。"""
    if not board.marks:
        return None
    if max_run(ops) >= STRICT_RUN:
        return f"记号盘：同一条指令连了 {max_run(ops)} 次"
    if len(set(ops)) < STRICT_KINDS:
        return f"记号盘：只用到 {len(set(ops))} 种指令"
    return None


def outcome(w):
    board = w.layout.board
    ops, why = decode(w.panel, w.layout.toolbox)
    if ops is None:
        return ("void", why)
    why = strict_void(board, ops)
    if why is not None:
        return ("void", why)
    end, _trail = walk(board, w.piece, ops)
    if end == board.socket:
        return ("win", end)
    if end.footprint() <= w.pads.cells:
        return ("move", end)
    return ("void", "终点既不是花格也不是插槽空腔")
