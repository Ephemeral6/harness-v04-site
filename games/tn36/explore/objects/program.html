# Program —— 面板 → 一串指令。对应 `derive/program.py`（复合概念：面板 + 底栏）。

**这一份是照着世界独立写的，不是从 Python 导出来的**：两份密码本各写一遍，
由逐条那一层（`Obs/Corpus.lean`）与 `verify.py` 的逐条对账核它们判得一样。

读不成程序（有一槽没写码、码不在密码本里、指令不是底栏给了的）⇒ `none`，交卷整道作废。
-/
import Obj.Panel.Panel
import Obj.Panel.Toolbox

namespace Game

/-- 密码本：一个码是「这一竖列里哪几个 (横/竖, 组) 暗着」。西移有两个码。 -/
def codebook : List (List (Kind × Nat) × Op) :=
  [ ([(.bar, 0), (.stem, 0)], .S),
    ([(.bar, 0), (.stem, 2)], .N),
    ([(.stem, 0), (.stem, 2)], .W),
    ([(.bar, 0)], .W),
    ([(.stem, 0)], .E),
    ([(.stem, 1)], .GROW),
    ([(.bar, 0), (.stem, 1)], .SHRINK),
    ([(.bar, 0), (.bar, 1)], .ROT),
    ([(.bar, 0), (.bar, 1), (.bar, 2), (.stem, 0), (.stem, 1), (.stem, 2)], .NOP) ]

/-- 这一块面板上所有的 (横/竖, 组)。 -/
def Panel.pairs (p : Panel) : List (Kind × Nat) :=
  (List.range p.groups).flatMap fun g => [(.bar, g), (.stem, g)]

/-- 第 k 槽那一竖列恰好写着 code 这个码吗（集合相等：码里的都暗、码外的都亮、码不越出组数）。 -/
def Panel.columnIs (p : Panel) (k : Nat) (code : List (Kind × Nat)) : Bool :=
  code.all (fun q => decide (q.2 < p.groups)) &&
  p.pairs.all fun q => p.isDark q.1 q.2 k == code.contains q

def lookupOp (p : Panel) (k : Nat) : List (List (Kind × Nat) × Op) → Option Op
  | []             => none
  | (c, op) :: cs  => if p.columnIs k c then some op else lookupOp p k cs

/-- 读成程序。 -/
def decode (p : Panel) (tb : Toolbox) : Option (List Op) :=
  (List.range p.slots).foldr (fun k acc =>
    match acc, lookupOp p k codebook with
    | some ops, some op =>
        match tb with
        | none     => some (op :: ops)
        | some ops' => if ops'.contains op then some (op :: ops) else none
    | _, _ => none) (some [])

end Game

from __future__ import annotations

S, N, W, E = "S", "N", "W", "E"
TRANSLATIONS = {N: (-1, 0), S: (1, 0), W: (0, -1), E: (0, 1)}

CODEBOOK = {
    frozenset({("bar", 0), ("stem", 0)}): S,
    frozenset({("bar", 0), ("stem", 2)}): N,
    frozenset({("stem", 0), ("stem", 2)}): W,
    frozenset({("bar", 0)}): W,
    frozenset({("stem", 0)}): E,
    frozenset({("stem", 1)}): "GROW",
    frozenset({("bar", 0), ("stem", 1)}): "SHRINK",
    frozenset({("bar", 0), ("bar", 1)}): "ROT",
    frozenset({("bar", 0), ("bar", 1), ("bar", 2),
               ("stem", 0), ("stem", 1), ("stem", 2)}): "NOP",
}


def decode(panel, toolbox):
    """返回 (指令列表, None) 或 (None, 读不成的理由)。"""
    ops = []
    for k in range(panel.slots):
        code = panel.column(k)
        if not code:
            return None, f"第 {k} 槽一个码都没写"
        op = CODEBOOK.get(code)
        if op is None:
            return None, f"第 {k} 槽的码不在密码本里"
        if toolbox.ops is not None and op not in toolbox.ops:
            return None, f"第 {k} 槽的指令 {op} 不是底栏给了的"
        ops.append(op)
    return ops, None
