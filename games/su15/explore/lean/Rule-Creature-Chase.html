/-
# Chase —— 会动的东西盯点击之前最近的那一件（认身份），最多 4 小步，压上就停

对应 Python `rule/会动的东西/creature_chases_target.py`（算法在 `derive/creature/chase.py`）。**只有定义。**
坐标 ×104 算整数：件心偏 (边长−1)·52，会动的东西的重心行偏 `rowOff`、列偏 208；一小步里某一轴差够半格（52）才挪。
-/

import Obj.World

namespace Game

def cCentre (c : Creature) : Int × Int := (c.row * 104 + rowOff c.kind, c.col * 104 + 208)
def pCentre (p : Piece) : Int × Int := (p.row * 104 + (p.sideI - 1) * 52, p.col * 104 + (p.sideI - 1) * 52)

def dist2 (a b : Int × Int) : Int := (a.1 - b.1) * (a.1 - b.1) + (a.2 - b.2) * (a.2 - b.2)

/-- 半格以上朝它挪一格；**正好 +½ 挪、正好 −½ 不挪**（= floor(差 + ½)，语料量出来的）。 -/
def sgn (d : Int) : Int := if 52 ≤ d then 1 else if d < -52 then -1 else 0

/-- 这一只压没压在这一件上。 -/
def touches (c : Creature) (p : Piece) : Bool := c.cells.any (fun x => p.has x.1 x.2)

/-- 点击之前离它最近的那一件；并列或没有件 ⇒ `none`。 -/
def target (c : Creature) (ps : List Piece) : Option Piece :=
  match ps.map (fun p => dist2 (cCentre c) (pCentre p)) with
  | [] => none
  | d :: ds =>
      let best := ds.foldl min d
      match ps.filter (fun p => dist2 (cCentre c) (pCentre p) == best) with
      | [p] => some p
      | _ => none

def chaseN : Nat → Creature → Piece → Creature
  | 0, c, _ => c
  | n + 1, c, t =>
      if touches c t then c
      else if sgn ((pCentre t).1 - (cCentre c).1) = 0 ∧ sgn ((pCentre t).2 - (cCentre c).2) = 0 then c
      else chaseN n { c with row := c.row + sgn ((pCentre t).1 - (cCentre c).1),
                             col := c.col + sgn ((pCentre t).2 - (cCentre c).2) } t

def chase (c : Creature) (t : Piece) : Creature := chaseN 4 c t

end Game
