/-
# Reachable —— 这个局面走得通 / 走不通

    Winnable   w  :=  ∃ as, 每一下合法 ∧ 照着走中途赢过
    Unsolvable w  :=  ∀ as, 每一下合法 → 照着走从没赢过

**写在明处的削弱**：`everWon` 碰到模型说不出话的那一支（撤销、RESET、甩飞、并、目标并列……）返回 `false`。
于是 `Winnable` 的意思是「**不经过那几支**就走得通」——见证路线里一步 `na` 都没有，这一条是被内核算出来的；
而 `Unsolvable` 的意思会变成「**不经过那几支**就赢不了」。**这一局没有宣告任何一关走不通**
（全部 9 关都有人通过，成绩单 9/9），所以下面那条骨架只是骨架：它在这里一次都没被用来说「不可做」。
-/

import Step

namespace Game

/-- 合法的一下：点在 64×64 的帧里。谈可不可做时必须限死，否则「一下点到帧外某处」就成了一种走法。 -/
def Act.legal : Act → Bool
  | .click r c => decide (0 ≤ r) && decide (r < 64) && decide (0 ≤ c) && decide (c < 64)
  | .undo => true
  | .reset => true

def everWon (b : Board) (lim : Nat) : World → List Act → Bool
  | _, [] => false
  | w, a :: as =>
      match step b lim w a with
      | .won => true
      | .ok w' => everWon b lim w' as
      | _ => false

def Winnable (b : Board) (lim : Nat) (w : World) : Prop :=
  ∃ as : List Act, (as.all Act.legal = true) ∧ everWon b lim w as = true

def Unsolvable (b : Board) (lim : Nat) (w : World) : Prop :=
  ∀ as : List Act, as.all Act.legal = true → everWon b lim w as = false

/-- 两句话不能同时成立。 -/
theorem not_both (b : Board) (lim : Nat) (w : World) : Winnable b lim w → Unsolvable b lim w → False := by
  intro ⟨as, hl, hw⟩ hu
  have := hu as hl
  rw [this] at hw
  exact Bool.false_ne_true hw

/-- 骨架：不变量 ⇒ 不可做。**全文唯一做归纳的地方**，与盘面大小、动作串多长都无关，不枚举任何状态。 -/
theorem unsolvable_of_invariant (b : Board) (lim : Nat) (P : World → Prop)
    (hkeep : ∀ w w' a, Act.legal a = true → P w → step b lim w a = Res.ok w' → P w')
    (hwin : ∀ w a, P w → step b lim w a ≠ Res.won) :
    ∀ w, P w → Unsolvable b lim w := by
  intro w hp as
  induction as generalizing w with
  | nil => intro _; rfl
  | cons a t ih =>
    intro hall
    simp only [List.all_cons, Bool.and_eq_true] at hall
    simp only [everWon]
    cases hstep : step b lim w a with
    | won => exact absurd hstep (hwin w a hp)
    | over => rfl
    | na => rfl
    | ok w' => exact ih w' (hkeep w w' a hall.1 hp hstep) hall.2

end Game
