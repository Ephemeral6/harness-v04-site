# World —— 状态：件、会动的东西、预算条上剩几格

对应 Python `objects/world.py`，**少了 `board` 那一位**：版面搬进了 `Board`（`Obj/Board/Board.lean`），

import Obj.Piece.Piece
import Obj.Creature.Creature
import Obj.Board.Board

namespace Game

structure World where
  pieces    : List Piece
  creatures : List Creature
  budget    : Nat
deriving DecidableEq, Repr, Inhabited

/-- 插入排序（按给定的 `lt`）。 -/
def insertBy {α : Type} (lt : α → α → Bool) (x : α) : List α → List α
  | [] => [x]
  | y :: ys => if lt y x then y :: insertBy lt x ys else x :: y :: ys

def sortBy {α : Type} (lt : α → α → Bool) : List α → List α
  | [] => []
  | x :: xs => insertBy lt x (sortBy lt xs)

/-- 规范形：两个列表各自排好序。模型每算出一个世界都过一遍这里。 -/
def World.norm (w : World) : World :=
  { w with pieces := sortBy Piece.lt w.pieces, creatures := sortBy Creature.lt w.creatures }

end Game
