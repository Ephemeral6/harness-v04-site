def dummy : Step := { seq := 0, lvl := 0, board := 0, act := none, prev := [], next := none, cert := false }

/-- 按 **seq** 取。越界给空步——空步在任何判据下都会立刻露馅。 -/
def bySeq (n : Nat) : Step := (steps.find? (fun s => s.seq == n)).getD dummy

/-- 从 seq `want` 那一行起连着取 `len` 行，变成 (动作, 看到的下一帧颜色)。
    取不到就给空表——所以每一段都另配一条长度定理钉住它到底有几步。 -/
def playFrom (want len : Nat) : List (Act × Colors) :=
  ((steps.dropWhile (fun s => s.seq != want)).take len).filterMap
    (fun s => match s.act, s.next with
      | some a, some o => some (a, o)
      | _, _ => none)

/-- 从 seq `want` 起连着取 `len` 行的动作。 -/
def actsFrom (want len : Nat) : List Act :=
  ((steps.dropWhile (fun s => s.seq != want)).take len).filterMap (fun s => s.act)

/-- 这一行动作之前的世界：这一帧读到的盘 + 这一帧的颜色。 -/
def worldOf (s : Step) : World := { board := boardAt s.board, colors := s.prev }
