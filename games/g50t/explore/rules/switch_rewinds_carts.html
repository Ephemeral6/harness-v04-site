import Obj.Board.Rail

namespace G50t

def switchRewindsCarts (tr : Track) (occ0 : List Cell) (pos roundPos : Cell) : Cell :=
  match tr.kind with
  | .spring => tr.home
  | .latch  => if occ0.contains tr.button then pos else roundPos
  | .other  => pos

end G50t

def scope_(prev, action, nxt, ctx):
    s = common_skip(prev, nxt, ctx)
    if s:
        return s
    if action != "ACTION5":
        return "不是 ACTION5"
    if ctx.get("hist") is None:
        return "这一关的历史不全"
    ts = tracks(ctx["hist"].start(prev))
    if not ts:
        return "场上没有车"
    if match(ts, prev) is None or match(ts, nxt) is None:
        return "车对不上开局的轨道"
    return None


def assert_(prev, action, nxt, ctx):
    h = ctx["hist"]
    ts = tracks(h.start(prev))
    fs = h.frames(prev)
    r0 = match(ts, fs[h.round_start()])
    a, b = match(ts, prev), match(ts, nxt)
    occ = occupied(prev)
    for t in ts:
        k = KIND.get(t.color)
        if k == SPRING:
            want = t.home
        elif k == LATCH:
            want = a[t] if t.button in occ else r0[t]
        else:
            want = a[t]
        if b[t] != want:
            return False
    return True
