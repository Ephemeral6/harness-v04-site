import Obj.Board.Rail

namespace G50t

def springFollowsButton (tr : Track) (occ0 occ1 : List Cell) (pos : Cell) : Cell :=
  if occ1.contains tr.button && !(occ0.contains tr.button) then tr.fwd
  else if occ0.contains tr.button && !(occ1.contains tr.button) then tr.home
  else pos

end G50t

def _springs(ctx, prev):
    return [t for t in tracks(ctx["hist"].start(prev)) if KIND.get(t.color) == SPRING]


def scope_(prev, action, nxt, ctx):
    s = common_skip(prev, nxt, ctx)
    if s:
        return s
    if action not in MOVE_ACTIONS:
        return "不是方向键"
    if ctx.get("hist") is None:
        return "这一关的历史不全"
    ts = tracks(ctx["hist"].start(prev))
    if not any(KIND.get(t.color) == SPRING for t in ts):
        return "场上没有弹簧车"
    if match(ts, prev) is None or match(ts, nxt) is None:
        return "车对不上开局的轨道"
    return None


def assert_(prev, action, nxt, ctx):
    ts = tracks(ctx["hist"].start(prev))
    a, b = match(ts, prev), match(ts, nxt)
    o0, o1 = occupied(prev), occupied(nxt)
    for t in ts:
        if KIND.get(t.color) != SPRING:
            continue
        if t.button in o1 and t.button not in o0:
            want = t.fwd
        elif t.button in o0 and t.button not in o1:
            want = t.home
        else:
            want = a[t]
        if b[t] != want:
            return False
    return True
