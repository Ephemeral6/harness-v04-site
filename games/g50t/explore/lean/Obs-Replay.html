/-- 从 seq `a`（含）到 `b`（不含）那一段，**不含过关那一步**，每一步配上看到的下一帧。
    取不到的就是空表——所以每一段都另有一条 `_len` 定理钉住它有几步。 -/
def playRange (a b : Nat) : List (Act × Seen) :=
  (obsAll.filter (fun o => a ≤ o.seq && o.seq < b)).filterMap
    (fun o => match o.seen with
      | some s => if o.brk then none else some (o.act, s)
      | none => none)

/-- 同一段的动作，**连过关那一脚一起**——「走得通」的见证。 -/
def actsRange (a b : Nat) : List Act :=
  (obsAll.filter (fun o => a ≤ o.seq && o.seq < b)).map (·.act)
