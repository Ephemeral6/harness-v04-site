组：objects
这是个什么游戏回放对象规则形式化怎么给别人看

它是什么

帧 → 世界。全局唯一碰像素的地方。

上面每一层只认 World 的字段（格、块、框、门带、条），读不到帧。

## 读法（每一步读不出来都抛 ParseError，不返回半个世界）

    1. 预算条    第 63 行：左边一段色4、右边一段色0，色4 的格数就是 bar
    2. 块        带壳块从核心认：一块实心的 a×b（a,b ≤ 3）单色（5 / 4 / 0），
                 朝四周各扩 a 行 b 列是一整块外壳（色14）。核心色0 的那一块是被操纵的，
                 它的外壳上可能有整条边也被涂成色0（贴着东西的那一面）
                 裸十字：色11 的连通块
    3. 相位      所有块的左上角像素 mod 3 必须一致 —— 同一个量的两种读法，对不上就抛
    4. 门带      色12/13 的连通块：实心长方形，色12 是从一头起的整层
    5. 地形      每一格的 9 个像素（被块 / 门带盖住的不算）：有色2 或出了画面 → 墙；有色15 → 带
    6. 框        从框线（色4，不在块里的）推出框的内部（见 _slots）
    7. 良构      (a) 框线恰好是框内部的轮廓；(b) 被操纵块的外壳上亮起的边
                 恰好是「那一面紧贴着别的块或门带」的那几面 —— 都是同一件事的两种读法

## 读不成的帧归不归类，不在这里判

这里只管抛。「这一帧本来就不是局面（终局画面、过场）」还是「是局面而读不出来」，
由拿得到 meta 的那一层分（run_rules.py），它把两档分开报。

它的 Lean

不进 Lean：唯一碰像素的地方：Lean 那边从格起算，像素一个都不进模型

"""帧 → 世界。**全局唯一碰像素的地方。**

上面每一层只认 `World` 的字段（格、块、框、门带、条），读不到帧。

## 读法（每一步读不出来都抛 `ParseError`，不返回半个世界）

    1. 预算条    第 63 行：左边一段色4、右边一段色0，色4 的格数就是 `bar`
    2. 块        带壳块从核心认：一块实心的 a×b（a,b ≤ 3）单色（5 / 4 / 0），
                 朝四周各扩 a 行 b 列是一整块外壳（色14）。核心色0 的那一块是被操纵的，
                 它的外壳上可能有整条边也被涂成色0（贴着东西的那一面）
                 裸十字：色11 的连通块
    3. 相位      所有块的左上角像素 mod 3 必须一致 —— **同一个量的两种读法**，对不上就抛
    4. 门带      色12/13 的连通块：实心长方形，色12 是从一头起的整层
    5. 地形      每一格的 9 个像素（被块 / 门带盖住的不算）：有色2 或出了画面 → 墙；有色15 → 带
    6. 框        从框线（色4，不在块里的）推出框的内部（见 `_slots`）
    7. 良构      (a) 框线恰好是框内部的轮廓；(b) 被操纵块的外壳上亮起的边
                 恰好是「那一面紧贴着别的块或门带」的那几面 —— 都是同一件事的两种读法

## 读不成的帧归不归类，**不在这里判**

这里只管抛。「这一帧本来就不是局面（终局画面、过场）」还是「是局面而读不出来」，
由拿得到 meta 的那一层分（`run_rules.py`），它把两档分开报。
"""
from __future__ import annotations

from .board.slot import Slot
from .geometry.cell import Cell
from .movable.gate import Gate
from .movable.piece import Piece
from .world import World

ROWS = 63            # 第 0..62 行是版面
HUD = 63             # 第 63 行是预算条
COLS = 64
K = 3                # 一格 3 像素

BG, WALL, SEL, IDLE, RET, BARE, ON, OFF, SHELL, BAND = 1, 2, 0, 5, 4, 11, 12, 13, 14, 15

UP, DOWN, LEFT, RIGHT = (-1, 0), (1, 0), (0, -1), (0, 1)
DIRS4 = (UP, DOWN, LEFT, RIGHT)


class ParseError(Exception):
    """这一帧读不成一个世界。**这是一条可判错的话**，不是一次异常处理。

    读不成的帧要归成两类（归类在读数那一层做，不在这里）：
        不是局面   终局画面、过场、整屏重画 —— 有名字、有条数，不算欠账
        读不出来   是一张盘而我读不出来 —— **只有这一种是欠账，它要是 0 才算过**
    """


# ---------------------------------------------------------------- 小工具（只在这个文件里用）

def _components(cells):
    todo, out = set(cells), []
    while todo:
        seed = todo.pop()
        comp, stack = {seed}, [seed]
        while stack:
            r, c = stack.pop()
            for dr, dc in DIRS4:
                nb = (r + dr, c + dc)
                if nb in todo:
                    todo.discard(nb)
                    comp.add(nb)
                    stack.append(nb)
        out.append(comp)
    return out


def _bbox(cells):
    rs = [r for r, _ in cells]
    cs = [c for _, c in cells]
    return min(rs), max(rs), min(cs), max(cs)


def _rect(box):
    r0, r1, c0, c1 = box
    return {(r, c) for r in range(r0, r1 + 1) for c in range(c0, c1 + 1)}


def _in_board(r, c):
    return 0 <= r < ROWS and 0 <= c < COLS


# ---------------------------------------------------------------- 1. 预算条

def _bar(row) -> int:
    n = 0
    while n < len(row) and row[n] == RET:
        n += 1
    if any(v != SEL for v in row[n:]):
        raise ParseError(f"预算条不是「一段色4 + 一段色0」：{row}")
    return n


# ---------------------------------------------------------------- 2. 块

def _shell_box(g, core):
    """核心 → 外壳的包围盒；外壳要整块在版面里。"""
    r0, r1, c0, c1 = core
    a, b = r1 - r0 + 1, c1 - c0 + 1
    box = (r0 - a, r1 + a, c0 - b, c1 + b)
    if box[0] < 0 or box[2] < 0 or box[1] >= ROWS or box[3] >= COLS:
        return None
    return box


def _solid_cores(g, color):
    """某一种核心色的全部带壳块（未被操纵的那些：外壳整块色14）。
    认不成块的那一片同色格原样退回，由调用方决定它是什么。"""
    cells = [(r, c) for r in range(ROWS) for c in range(COLS) if g[r][c] == color]
    found, rest = [], []
    for comp in _components(cells):
        box = _bbox(comp)
        r0, r1, c0, c1 = box
        a, b = r1 - r0 + 1, c1 - c0 + 1
        sb = _shell_box(g, box) if len(comp) == a * b and a <= 3 and b <= 3 else None
        if sb is not None and all(g[r][c] == SHELL for (r, c) in _rect(sb) - comp):
            found.append((sb, box))
        else:
            rest.append(comp)
    return found, rest


def _lit_edges(g, box):
    """外壳四条边里整条是色0 的那几条（贴着东西的那几面）。"""
    r0, r1, c0, c1 = box
    sides = {UP: [(r0, c) for c in range(c0, c1 + 1)],
             DOWN: [(r1, c) for c in range(c0, c1 + 1)],
             LEFT: [(r, c0) for r in range(r0, r1 + 1)],
             RIGHT: [(r, c1) for r in range(r0, r1 + 1)]}
    return {d for d, s in sides.items() if all(g[r][c] == SEL for r, c in s)}, sides


def _selected(g, taken):
    """被操纵的那一块：(外壳盒, 核心盒, 亮着的边)。一个色0 都没有时 None。"""
    zeros = {(r, c) for r in range(ROWS) for c in range(COLS) if g[r][c] == SEL}
    if not zeros:
        return None
    best = []
    for (r0, c0) in zeros:
        for a in (1, 2, 3):
            for b in (1, 2, 3):
                core = (r0, r0 + a - 1, c0, c0 + b - 1)
                if not _rect(core) <= zeros:
                    continue
                box = _shell_box(g, core)
                if box is None:
                    continue
                R = _rect(box)
                if not zeros <= R or R & taken:
                    continue
                if any(g[r][c] not in (SEL, SHELL) for (r, c) in R):
                    continue
                lit, sides = _lit_edges(g, box)
                ring = set().union(*sides.values())
                litpx = set().union(*(set(sides[d]) for d in lit)) if lit else set()
                # 外壳上的色0 只许整条边地出现；核心以外、外壳之内的里层不许有色0
                if any((r, c) not in litpx for (r, c) in (zeros & ring)):
                    continue
                if (zeros - ring) != _rect(core):
                    continue
                best.append((a * b, box, core, frozenset(lit)))
    if not best:
        raise ParseError("有色0 的像素，但认不出被操纵的那一块")
    top = max(x[0] for x in best)
    tops = {(b, c, l) for (ar, b, c, l) in best if ar == top}
    if len(tops) != 1:
        raise ParseError(f"被操纵的那一块有 {len(tops)} 种同样好的读法")
    return next(iter(tops))


# ---------------------------------------------------------------- 4. 门带

def _gate(g, comp, pr, pc):
    box = _bbox(comp)
    if _rect(box) != set(comp):
        raise ParseError(f"门带不是实心长方形（被盖住了？）：{box}")
    r0, r1, c0, c1 = box
    if (r0 - pr) % K or (c0 - pc) % K or (r1 - r0 + 1) % K or (c1 - c0 + 1) % K:
        raise ParseError(f"门带没对齐格阵：{box}")
    cells = frozenset(Cell((r - pr) // K, (c - pc) // K)
                      for r in range(r0, r1 + 1, K) for c in range(c0, c1 + 1, K))
    hot = {(r, c) for (r, c) in comp if g[r][c] == ON}
    H, W = r1 - r0 + 1, c1 - c0 + 1
    if not hot:
        return Gate(cells, None, None, 0)
    rows = sorted({r for r, _ in hot})
    cols = sorted({c for _, c in hot})
    full_rows = len(hot) == len(rows) * W and rows == list(range(rows[0], rows[-1] + 1))
    full_cols = len(hot) == len(cols) * H and cols == list(range(cols[0], cols[-1] + 1))
    by_row = full_rows and len(rows) < H
    by_col = full_cols and len(cols) < W
    if by_row and by_col:
        raise ParseError(f"门带上的色12 横竖两种读法都成立：{box}")
    if by_row and rows[0] == r0:
        return Gate(cells, "row", +1, len(rows))
    if by_row and rows[-1] == r1:
        return Gate(cells, "row", -1, len(rows))
    if by_col and cols[0] == c0:
        return Gate(cells, "col", +1, len(cols))
    if by_col and cols[-1] == c1:
        return Gate(cells, "col", -1, len(cols))
    raise ParseError(f"门带上的色12 不是从一头起的整层：{box}")


# ---------------------------------------------------------------- 6. 框

def _slots(g, cand, hidden, ring, lo, hi, pr, pc):
    """从框线推出框的内部。

    框线画在**外侧那一格**紧贴内部的那一排像素上。所以对一格 u 和一个方向 d，
    看的是邻格 v = u+d 朝着 u 的那一排 3 个像素（`_line`）：

        R   看得见的全是框线（色4）         → u 在框里，v 不在同一个框里
        H   3 个全被块 / 门带盖住            → 这一面说不了话
        O   其余（看得见地面 / 墙 / 带）     → 若 u 在框里，v 也在同一个框里

    先从带 R 的格起种子，沿 O 扩；再把「四面都是 R / H / 朝框内的 O」的被盖住的格
    收进来（块整个或一半坐在框里时，框线在它身上看不见）；最后相邻两格之间没有 R 就并成一个框。
    """
    def line(u, d):
        (i, j), (di, dj) = u, d
        vi, vj = i + di, j + dj
        if di == 1:
            px = [(pr + K * vi, pc + K * vj + k) for k in range(K)]
        elif di == -1:
            px = [(pr + K * vi + K - 1, pc + K * vj + k) for k in range(K)]
        elif dj == 1:
            px = [(pr + K * vi + k, pc + K * vj) for k in range(K)]
        else:
            px = [(pr + K * vi + k, pc + K * vj + K - 1) for k in range(K)]
        if any(not _in_board(r, c) for r, c in px):
            return "X"
        vis = [p for p in px if p not in hidden]
        if not vis:
            return "H"
        if all(p in ring for p in vis):
            return "R"
        return "O"

    def corners(u):
        i, j = u
        top, left = pr + K * i, pc + K * j
        return [(top - 1, left - 1), (top - 1, left + K), (top + K, left - 1), (top + K, left + K)]

    L = {(u, d): line(u, d) for u in cand for d in DIRS4}
    inside = {u for u in cand if any(L[(u, d)] == "R" for d in DIRS4)}
    # 第二种种子：四面全被盖住（块压在框上），可是**对角那几个框线角**露在外面。
    # 十字块压住一个一格的框时就是这样：四条边都在十字的四条臂底下，只剩四个角看得见。
    # 看得见的角必须**全是**框线：块的一条臂伸进框里时，臂根那一格的角一半是框线一半是地面，它不在框里。
    def seen_corners(u):
        return [p for p in corners(u) if _in_board(*p) and p not in hidden]
    inside |= {u for u in cand if all(L[(u, d)] in ("R", "H") for d in DIRS4)
               and seen_corners(u) and all(p in ring for p in seen_corners(u))}
    stack = list(inside)
    while stack:
        u = stack.pop()
        for d in DIRS4:
            s = L[(u, d)]
            v = (u[0] + d[0], u[1] + d[1])
            if s == "X":
                raise ParseError(f"框内一格朝画面外：格 {u}")
            if s == "O":
                if v not in cand:
                    raise ParseError(f"框内一格 {u} 朝 {d} 是开的，而那边不是空地")
                if v not in inside:
                    inside.add(v)
                    stack.append(v)
    def corner_ok(u):
        """u 要算框内，它看得见的对角那几个像素要么是框线，要么落在已经算框内的格里。"""
        i, j = u
        out = []
        for (r, c), (di, dj) in zip(corners(u), ((-1, -1), (-1, 1), (1, -1), (1, 1))):
            if not _in_board(r, c) or (r, c) in hidden or (r, c) in ring:
                continue
            out.append((i + di, j + dj) in inside)
        return all(out)

    grew = True
    while grew:
        grew = False
        for u in cand - inside:
            nbs = [(d, (u[0] + d[0], u[1] + d[1])) for d in DIRS4]
            if not any(v in inside for _, v in nbs):
                continue
            if all(L[(u, d)] in ("R", "H") or (L[(u, d)] == "O" and v in inside) for d, v in nbs) \
                    and corner_ok(u):
                inside.add(u)
                grew = True
    # 相邻两格之间没有框线就是同一个框
    parent = {u: u for u in inside}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    for u in inside:
        for d in (DOWN, RIGHT):
            v = (u[0] + d[0], u[1] + d[1])
            if v in inside and L[(u, d)] != "R" and L[(v, (-d[0], -d[1]))] != "R":
                parent[find(u)] = find(v)
    groups: dict = {}
    for u in inside:
        groups.setdefault(find(u), set()).add(u)
    return [frozenset(s) for s in groups.values()]


def _outline8(pix):
    out = set()
    for (r, c) in pix:
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                q = (r + dr, c + dc)
                if q not in pix:
                    out.add(q)
    return out


# ---------------------------------------------------------------- 主函数

def parse(frame) -> World:
    g = frame
    if len(g) != 64 or any(len(row) != COLS for row in g):
        raise ParseError("帧不是 64×64")
    bar = _bar(g[HUD])

    idle, rest5 = _solid_cores(g, IDLE)
    if rest5:
        raise ParseError(f"有 {len(rest5)} 片色5 认不成带壳块的核心")
    retired, rest4 = _solid_cores(g, RET)
    taken = set()
    for sb, _ in idle + retired:
        taken |= _rect(sb)
    sel = _selected(g, taken)

    boxes = [(sb, "idle") for sb, _ in idle] + [(sb, "retired") for sb, _ in retired]
    if sel is not None:
        boxes.append((sel[0], "selected"))
    bare_px = [(r, c) for r in range(ROWS) for c in range(COLS) if g[r][c] == BARE]
    bares = _components(bare_px)
    if not boxes and not bares:
        raise ParseError("一块可推的东西都没有")

    # 3. 相位：所有块的左上角 mod 3 必须一致
    phases = {(b[0] % K, b[2] % K) for b, _ in boxes} | {(_bbox(p)[0] % K, _bbox(p)[2] % K) for p in bares}
    if len(phases) != 1:
        raise ParseError(f"块的相位对不上：{sorted(phases)}")
    pr, pc = next(iter(phases))

    def cells_of_px(pix):
        cs = {Cell((r - pr) // K, (c - pc) // K) for r, c in pix}
        full = {(pr + K * x.row + i, pc + K * x.col + j) for x in cs for i in range(K) for j in range(K)}
        if full != set(pix):
            raise ParseError("一块东西没铺满整格")
        return frozenset(cs)

    covered = set()
    pieces, selected, ret = [], None, set()
    for box, kind in boxes:
        R = _rect(box)
        if R & covered:
            raise ParseError("两块带壳块重叠")
        covered |= R
        p = Piece(cells_of_px(R), True)
        pieces.append(p)
        if kind == "selected":
            selected = p.anchor
        elif kind == "retired":
            ret.add(p.anchor)
    for comp in bares:
        if set(comp) & covered:
            raise ParseError("裸十字和带壳块重叠")
        covered |= set(comp)
        pieces.append(Piece(cells_of_px(comp), False))

    gate_px = [(r, c) for r in range(ROWS) for c in range(COLS) if g[r][c] in (ON, OFF)]
    gates = []
    for comp in _components(gate_px):
        gates.append(_gate(g, comp, pr, pc))
        covered |= set(comp)

    # 5. 地形：每一格 9 个像素里看得见的那些
    ring = set()
    for comp in rest4:
        ring |= set(comp)
    i_lo, i_hi = (0 - pr) // K, (ROWS - 1 - pr) // K
    j_lo, j_hi = (0 - pc) // K, (COLS - 1 - pc) // K
    walls, bands, cand = set(), set(), set()
    for i in range(i_lo, i_hi + 1):
        for j in range(j_lo, j_hi + 1):
            px = [(pr + K * i + a, pc + K * j + b) for a in range(K) for b in range(K)]
            if any(not _in_board(r, c) for r, c in px):
                walls.add(Cell(i, j))
                continue
            vis = [g[r][c] for r, c in px if (r, c) not in covered]
            bad = [v for v in vis if v not in (BG, WALL, BAND, RET)]
            if bad:
                raise ParseError(f"格 ({i},{j}) 里有认不得的颜色 {sorted(set(bad))}")
            if WALL in vis:
                walls.add(Cell(i, j))
            elif BAND in vis:
                bands.add(Cell(i, j))
            elif RET not in vis and not any((r, c) in covered and g[r][c] in (ON, OFF) for r, c in px):
                cand.add((i, j))

    # 画面之外一圈也记成墙：一步只挪一格、推 / 扫都是一格一格试，外面一圈就够挡住
    for i in range(i_lo - 1, i_hi + 2):
        walls.add(Cell(i, j_lo - 1))
        walls.add(Cell(i, j_hi + 1))
    for j in range(j_lo - 1, j_hi + 2):
        walls.add(Cell(i_lo - 1, j))
        walls.add(Cell(i_hi + 1, j))

    # 6. 框
    slot_sets = _slots(g, cand, covered, ring, i_lo, i_hi, pr, pc)
    slots = frozenset(Slot(frozenset(Cell(i, j) for i, j in s)) for s in slot_sets)

    # 7a. 良构：框线恰好是框内部的轮廓（被盖住的那几段不算）
    drawn = set()
    for s in slot_sets:
        pix = {(pr + K * i + a, pc + K * j + b) for i, j in s for a in range(K) for b in range(K)}
        drawn |= _outline8(pix)
    visible_outline = {p for p in drawn if _in_board(*p) and p not in covered}
    if visible_outline != ring:
        extra = len(ring - visible_outline)
        miss = len(visible_outline - ring)
        raise ParseError(f"框线和框内部的轮廓对不上：多出 {extra} 个框线像素、缺 {miss} 个")

    world = World(phase=(pr, pc), walls=frozenset(walls), bands=frozenset(bands), slots=slots,
                  pieces=tuple(sorted(pieces)), selected=selected, retired=frozenset(ret),
                  gates=tuple(sorted(gates)), bar=bar)

    # 7b. 良构：亮着的边 == 紧贴着别的块或门带的那几面
    if sel is not None:
        me = world.piece_at(selected)
        occ = set()
        for p in world.pieces:
            if p is not me:
                occ |= {(x.row, x.col) for x in p.cells}
        for gt in world.gates:
            occ |= {(x.row, x.col) for x in gt.cells}
        mine = {(x.row, x.col) for x in me.cells}
        want = set()
        for d in DIRS4:
            front = {(r + d[0], c + d[1]) for r, c in mine} - mine
            if front & occ:
                want.add(d)
        if want != set(sel[2]):
            raise ParseError(f"被操纵那一块亮着的边 {sorted(sel[2])} ≠ 紧贴着东西的那几面 {sorted(want)}")
    return world
