/-
# Levels —— 每一关一条定理（**自动生成，不要手改**；生成器 `export_levels.py`）

    L<n>_trace       这一段有几步（钉死，防空表）∧ 沿真人走过的那条路，模型每一帧都和观测对上
    L<n>_tail_trace  （中途掉过命的关）掉命之后那一段，同上，起点是「重来之后的开局」
    L<n>_winnable    这一关走得通 —— 见证就是真人那条路（或它掉命之后那一段）

**只有开局那一帧是观测给的，后面全是模型自己算的。** 观测一律从 `Obs/Replay.lean` 按 `seq` 切。

## 开局用的口径

    玩家、钥匙、能量、命      这一关第一帧（`Replay.f<k>`）
    版面、静止装置、补给、门   **整关取并集**（`Replay.board` / `Replay.statics`）
                             ——比玩家在第一帧看到的多知道几格（有雾的那一关尤其如此），这个代价写在明处
    会动的装置的方向          见各关的说明（端点上是定的；不在端点、或开局看不见 ⇒ 模型自己挑，候选数写在那一关）

## 一条明写的假设（不进模型）

**掉命之后，这一关从开局重来，命少一条**（`restart`，`rule/unmodeled/restart_to_start.py`）。
它不进 `step`（开局是历史）。有它，掉过命的那一关的尾段起点就是 `restart L<n>_start`；
尾段的逐帧核对**第一帧就会因为它错而红**——没红**不构成证明**，
但那是这条假设在这批语料上唯一一次有机会被驳倒。

## 一句削弱，写在明处

`everWon` 碰到掉命返回 `false`，所以 `_winnable` 的见证是一条**不经过掉命**的路；
这一局七关都有这样的路，没有一关需要靠掉命才过得去。
-/

import Obj.World
import Proof.Reachable
import Obs.Replay

set_option maxRecDepth 100000

-- 不开命名空间：装置的 `lean_check` 按短名 `#print axioms`。
open Game Game.Replay

/-- 开局：玩家、钥匙、能量、命取这一关第一帧；版面与静止的东西取整关并集；会动的装置另给。 -/
def mkStartAt (lv : Nat) (s : Seen) (pats : List PState) (orb : Nat) : World :=
  { me := s.me, key := s.key, energy := s.energy, lives := s.lives,
    goals := (statics lv).goals, rotators := (statics lv).rotators, shapers := (statics lv).shapers,
    painters := (statics lv).painters, refills := (statics lv).refills, pats := pats, orb := orb }

/-! ## 第 0 关

认证段 seq 1100–1112（成绩单记这一关 13 步）。这一关没有巡逻的装置。
-/

def L0_start : World := mkStartAt 0 f0 [] 0

/-- 开局到第一处断点为止，12 步（seq 1100–1111）。 -/
def L0_play : List (Act × Seen) := playFrom 1100 12

#eval match firstSplit (board 0) L0_start L0_play 0 with
  | none   => "LEVEL 0 play ok " ++ toString L0_play.length
  | some i => "LEVEL 0 play split at " ++ toString i

/-- **这一段有 12 步，而且模型沿着它一步不差。** 只有开局是观测给的。 -/
theorem L0_trace : L0_play.length = 12 ∧ traceOk (board 0) (L0_start : World) L0_play = true := by
  decide +kernel

#print axioms L0_trace

/-- 见证：整条真实走法，再加过关那一脚（seq 1112，它的下一帧是另一张盘，所以不在上面的路里）。 -/
def L0_acts : List Act := L0_play.map Prod.fst ++ [.a1]

/-- **第 0 关走得通**：从 `L0_start` 起照着走，中途赢了。 -/
theorem L0_winnable : Winnable (board 0) (L0_start : World) :=
  ⟨L0_acts, by decide +kernel⟩

#print axioms L0_winnable
