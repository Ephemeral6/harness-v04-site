/-
# Corpus —— 模型面对**每一条**真实转移（第三条腿的 ①）

每一步都从**观测到的那一帧**重新起算（按纤维：帧上读不出来的那几位全列出来），
算一步，和看到的下一帧比。RESET 不是用例（模型的动作里没有它），条数单列。

## 读数不是定理

下面那句 `#eval` 走的是**求值器**，不进可信基——它是读数。
所以另挑了几条（每类一条）用 `decide` 走内核真证出来；**两者的差别写在明处**：
全量的「反例 0」是读数，那几条是定理。
-/
import Obj.World
import Obs.Replay

set_option maxRecDepth 100000

-- 不开命名空间：装置的 `lean_check` 按短名 `#print axioms`。
open Game Game.Replay

def verdicts : List (Nat × Nat × Verdict) :=
  rows.filterMap (fun r => (caseOf r).map (fun c => (r.seq, r.kind, agrees (board c.lvl) c)))

def count (v : Verdict) : Nat := (verdicts.filter (fun x => x.2.2 == v)).length

-- 读数：一行，`verify.py` 按前缀 `CORPUS` 取。
#eval "CORPUS cases=" ++ toString verdicts.length ++ " yes=" ++ toString (count .yes)
  ++ " no=" ++ toString (count .no) ++ " na=" ++ toString (count .na)
  ++ " reset=" ++ toString (rows.length - verdicts.length)
  ++ " no_seqs=" ++ toString ((verdicts.filter (fun x => x.2.2 == .no)).map (·.1) |>.take 20)
  ++ " na_seqs=" ++ toString ((verdicts.filter (fun x => x.2.2 == .na)).map (·.1) |>.take 20)

/-- 一条转移上「有一个和这一帧相符的世界，`step` 从它算出来的正是下一帧」。 -/
def holdsAt (s : Nat) : Bool :=
  ((rowAt s).bind caseOf).any (fun c =>
    (fiber (board c.lvl) c.prev).any (fun w : World => judgeOne (board c.lvl) c w == .yes))

/-! ## 每类一条，走内核
