from __future__ import annotations

from derive.click import decode
from derive.step import predict_cached, touched, unreadable_input
from rule._common import find_bar, in_level, same_bar

CLAIM = "这一下既没伸缩、也没被拖、没被顶、没被转的条和墙，下一帧原封不动。"
SCOPE_NOTE = "只管模型预测真的动了点什么的步（点面板或点图例、不作废）。作废的步由 作废/ 各条判「整步不变」。"

def scope_(w, action, ctx=None):
    if not in_level(ctx):
        return "不是关内步"
    k, _ = decode(w.board, action)
    if k not in ("grow", "shrink", "rot"):
        return "不是点面板图标或图例"
    out = predict_cached(w, action)
    if out.kind not in ("moved", "rotated"):
        return "模型预测整步作废"
    why = unreadable_input(w, out)
    if why:
        return why
    return None


def assert_(w, action, nw, ctx=None) -> bool:
    out = predict_cached(w, action)
    moved = touched(out)
    for i, b in enumerate(w.bars):
        if i in moved:
            continue
        o = find_bar(nw, b)
        if o is None or not same_bar(b, o):
            return False
    mw = set(out.move.movedw) if out.move is not None else set()
    keep = {x for j, x in enumerate(w.walls) if j not in mw}
    return keep <= set(nw.walls)
