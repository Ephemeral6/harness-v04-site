import Obs.Check

namespace Game

theorem levelup_step_is_effective (w w' : World) (a : Act)
    (h : after w a = some w') (hg : isGoal w' = true) (h0 : isGoal w = false) :
    step w a ≠ .refused := by
  intro hr
  simp only [after, hr] at h
  cases h
  rw [h0] at hg
  exact Bool.noConfusion hg

end Game
