    /-
# Obj/Piece/Bar —— 条（对应 `objects/piece/bar.py`）

一排三格的帽 + 从帽往外的 `len` 条 3 格宽的身体；尖（可有可无）在离帽第 `len - 1` 条的正中。
**只记帽的位置、方向、长度、有没有尖**：「身体是实心的、从帽一路连到自由端」「尖永远在离自由端一格处」
在这个表示下表示不出反例。

`(r, c)` 是帽左上格（带偏移 `O`）。
-/

import Obj.Geometry.Grid

namespace Game

structure Bar where
  color : Nat
  r     : Nat
  c     : Nat
  dir   : Dir
  len   : Nat
  tip   : Bool
  deriving DecidableEq, Repr

/-- 离帽第 `a` 条到第 `b` 条（`a ≤ b`）合起来的长方形。 -/
def Bar.span (x : Bar) (a b : Nat) : Rect :=
  match x.dir with
  | .U => ⟨x.r - b, x.c, x.r - a, x.c + 2⟩
  | .D => ⟨x.r + a, x.c, x.r + b, x.c + 2⟩
  | .L => ⟨x.r, x.c - b, x.r + 2, x.c - a⟩
  | .R => ⟨x.r, x.c + a, x.r + 2, x.c + b⟩

def Bar.cap (x : Bar) : Rect := x.span 0 0
def Bar.body (x : Bar) : Rect := x.span 1 x.len
/-- 帽 + 身体：这根条占的全部格子（一块长方形）。 -/
def Bar.whole (x : Bar) : Rect := x.span 0 x.len
/-- 自由端外面紧贴的那一条。 -/
def Bar.face (x : Bar) : Rect := x.span (x.len + 1) (x.len + 1)

/-- 尖在哪一格（离帽第 `len - 1` 条的正中）。 -/
def Bar.tipCell (x : Bar) : Nat × Nat :=
  match x.dir with
  | .U => (x.r - (x.len - 1), x.c + 1)
  | .D => (x.r + (x.len - 1), x.c + 1)
  | .L => (x.r + 1, x.c - (x.len - 1))
  | .R => (x.r + 1, x.c + (x.len - 1))

/-- 整根平移：往 `d` 走 3 格。 -/
def Bar.shift (x : Bar) (d : Dir) : Bar :=
  match d with
  | .U => { x with r := x.r - 3 }
  | .D => { x with r := x.r + 3 }
  | .L => { x with c := x.c - 3 }
  | .R => { x with c := x.c + 3 }

end Game
