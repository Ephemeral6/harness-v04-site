"""格与方向。格就是 `(行, 列)` 两个整数，从 0 数起。

`strip(r, c, dir, k)`：一根条的帽的左上格在 `(r, c)`，身体朝 `dir` 长，
离帽第 `k` 条（`k = 0` 是帽自己）那一条横截面的三格，按「左上 → 右下」排。
**中间那一格（下标 1）**就是尖所在的那一列 / 那一行。

判据 4（对称）：四个方向不写四遍，统一成 `DIRS` 里一个单位位移；
逆时针转一档 = `CCW`，转 4 档回原处（`rot_pt` 是 4 阶循环，见 `lean2/Obj/Geometry`）。
"""
from __future__ import annotations

ROWS = 63          # 场地 0..62；第 63 行是计费行，不属于场地
COLS = 64

DIRS = {"U": (-1, 0), "D": (1, 0), "L": (0, -1), "R": (0, 1)}
# 屏幕上看的逆时针（行向下长）：朝下的转一档朝右。
CCW = {"U": "L", "L": "D", "D": "R", "R": "U"}


def inside(r: int, c: int) -> bool:
    return 0 <= r < ROWS and 0 <= c < COLS


def strip(r: int, c: int, d: str, k: int) -> list[tuple[int, int]]:
    """帽左上格 (r,c)、朝 d 长的条，离帽第 k 条的三格（可能落在盘外）。"""
    dr, dc = DIRS[d]
    if dc == 0:                       # 竖着长：帽是横的一排三格
        return [(r + dr * k, c + j) for j in range(3)]
    return [(r + j, c + dc * k) for j in range(3)]


def rot_pt(p: tuple[int, int], piv: tuple[int, int]) -> tuple[int, int]:
    """绕格 piv 逆时针转一档（屏幕坐标，行向下）。"""
    r, c = p
    pr, pc = piv
    return (pr - (c - pc), pc + (r - pr))


def rect_cells(r0: int, c0: int, r1: int, c1: int) -> set[tuple[int, int]]:
    return {(y, x) for y in range(r0, r1 + 1) for x in range(c0, c1 + 1)}
