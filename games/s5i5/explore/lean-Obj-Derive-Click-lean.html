# Obj/Derive/Click —— 这一下点中了什么（对应 `derive/click.py`）

点击坐标 `(x, y)` 是 API 给的（x 列、y 行，不带偏移）；这里加上偏移再和盘面比。
面板先按图标中心认（缩 / 伸），再按框认（框里别处），最后认图例框。
-/

import Obj.World

namespace Game

inductive Click where
  | shrink (K : Nat)
  | grow (K : Nat)
  | rot (K : Nat)
  | elsewhere
  deriving DecidableEq, Repr

def inBox (b : Rect) (r c : Nat) : Bool :=
  Nat.ble b.r0 r && Nat.ble r b.r1 && Nat.ble b.c0 c && Nat.ble c b.c1

def iconHit (r c : Nat) : List Panel → Option Click
  | [] => none
  | p :: ps =>
    if p.shrink == (r, c) then some (.shrink p.color)
    else if p.grow == (r, c) then some (.grow p.color)
    else iconHit r c ps

def boxHit (r c : Nat) : List Panel → Bool
  | [] => false
  | p :: ps => inBox p.box r c || boxHit r c ps

def legendHit (r c : Nat) : List Legend → Option Click
  | [] => none
  | l :: ls => if inBox l.box r c then some (.rot l.color) else legendHit r c ls

def decode (bd : Board) (x y : Nat) : Click :=
  match iconHit (y + O) (x + O) bd.panels with
  | some k => k
  | none =>
    if boxHit (y + O) (x + O) bd.panels then .elsewhere
    else (legendHit (y + O) (x + O) bd.legends).getD .elsewhere

end Game
