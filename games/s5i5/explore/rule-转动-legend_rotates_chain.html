它说什么：点 K 色图例：K 色那根条（或 K 色枢纽上挂着的几根）连同整条下游，绕关节逆时针转 90°。关节 = K 色条身体第 1 条的正中那一格（枢纽则是枢纽中心）。转到的那一档要是压着别的条、墙或框，就接着转 180°、270°，停在第一档压不着的。

什么时候不管：只管点在图例框里、模型预测真的转成了的步。转不成的归 作废/rotation_blocked_refuses；K 色条不止一根又没有枢纽的，模型不说话（语料 0 次）。

明确不主张：不主张顺时针；不主张转的时候扫过的格子要空（只看落点那一档）；不主张长度会变（那是 推论/rotation_keeps_length，由表示法推出）。

# Rule/Turn/LegendRotatesChain —— 定义（对应 `rule/转动/legend_rotates_chain.py`）

点 K 色图例：K 色那根条（或 K 色枢纽上挂着的几根）连同整条下游，绕关节逆时针转 90°；
关节 = K 色条身体第 1 条的正中那一格（枢纽则是枢纽中心）。转到的那一档压着别的东西就接着转，
逐档的结论交给 `Rule/Refuse/RotationBlocked` 的 `verdict`。

import Obj.Derive.Attach
import Rule.Refuse.RotationBlocked

namespace Game

/-- 一根条绕 `(pr, pc)` 逆时针转一档：帽那块长方形转过去，方向跟着转。 -/
def turnBar1 (pr pc : Nat) (b : Bar) : Bar :=
  let cap := b.cap.rot pr pc
  { b with r := cap.r0, c := cap.c0, dir := b.dir.ccw }

def turnBar (pr pc : Nat) : Nat → Bar → Bar
  | 0, b => b
  | k + 1, b => turnBar pr pc k (turnBar1 pr pc b)

/-- 在下游里的转 `k` 档，不在的原样。 -/
def turnAll (pr pc k : Nat) : List (Bar × Bool) → List Bar
  | [] => []
  | (b, m) :: rest => (if m then turnBar pr pc k b else b) :: turnAll pr pc k rest

/-- K 色恰好一根条时的关节：身体第 1 条的正中。 -/
def jointOf (b : Bar) : Nat × Nat :=
  match b.dir with
  | .U => (b.r - 1, b.c + 1)
  | .D => (b.r + 1, b.c + 1)
  | .L => (b.r + 1, b.c - 1)
  | .R => (b.r + 1, b.c + 1)

/-- 关节与根：有 K 色枢纽就绕枢纽中心、根是帽贴着枢纽的条；否则 K 色要恰好一根条。 -/
def pivotRoots (w : World) (K : Nat) : Option (Nat × Nat × List Bool) :=
  match w.board.hubs.find? (·.color == K) with
  | some h => some (h.r + 1, h.c + 1, w.bars.map (fun b => meetsAny b.cap h.ring))
  | none =>
    match w.bars.filter (·.color == K) with
    | [b] => some ((jointOf b).1, (jointOf b).2, w.bars.map (·.color == K))
    | _ => none

/-- 第 `k` 档的四个读数。 -/
def turnInfo (w : World) (pr pc k : Nat) (chain : List (Bar × Bool)) : Bool × Bool × Bool × Bool :=
  let moved := (chain.filter (·.2)).map (fun p => turnBar pr pc k p.1)
  let others := (chain.filter (fun p => !p.2)).map (fun p => p.1.whole)
              ++ w.board.hubs.map Hub.rect ++ w.walls.map (·.rect) ++ w.board.boxes
  (moved.any (fun b => meetsAny b.whole w.board.fixed),
   moved.any (fun b => meetsAny b.whole others),
   moved.any (fun b => !b.whole.onBoard),
   moved.any (fun b => !b.cap.onBoard))

/-- 逐档试，最多 `n` 档。 -/
def tryTurn (w : World) (pr pc : Nat) (chain : List (Bar × Bool)) : Nat → Nat → Res
  | _, 0 => .refused
  | k, n + 1 =>
    match turnInfo w pr pc k chain with
    | (fx, bl, off, co) =>
      match verdict k fx bl off co with
      | .refuse => .refused
      | .next => tryTurn w pr pc chain (k + 1) n
      | .accept => .ok { w with bars := turnAll pr pc k chain }

def rotate (w : World) (K : Nat) : Res :=
  match pivotRoots w K with
  | none => .silent
  | some (pr, pc, roots) => tryTurn w pr pc (w.bars.zip (downstream w.bars roots)) 1 3

end Game

def pivot_and_roots(w, K):
    hubs = [h for h in w.board.hubs if h.color == K]
    if hubs:
        h = hubs[0]
        return (h.r + 1, h.c + 1), on_hub(w.bars, h)
    roots = [i for i, b in enumerate(w.bars) if b.color == K]
    if len(roots) != 1:
        return None, roots
    b = w.bars[roots[0]]
    return strip(b.r, b.c, b.dir, 1)[1], roots


def turn(b, piv, k: int):
    """一根条绕 piv 逆时针转 k 档：帽的三格各自转，方向跟着转。"""
    cp = strip(b.r, b.c, b.dir, 0)
    d = b.dir
    for _ in range(k):
        cp = [rot_pt(p, piv) for p in cp]
        d = CCW[d]
    return replace(b, r=min(p[0] for p in cp), c=min(p[1] for p in cp), dir=d)


def rotate(w, K) -> Outcome:
    """effect：逐档试。每一档先问 作废/rotation_blocked_refuses 要不要整步作废。"""
    refuse = __import__("rule.作废.rotation_blocked_refuses", fromlist=["verdict"]).verdict
    piv, roots = pivot_and_roots(w, K)
    if piv is None:
        return Outcome("silent", None, by="同色条不止一根又没有枢纽")
    chain = downstream(w.bars, roots)
    others: set = set()
    for j, b in enumerate(w.bars):
        if j not in chain:
            others |= bar_cells(b)
    for h in w.board.hubs:
        others |= hub_cells(h)
    walls: set = set()
    for x in w.walls:
        walls |= wall_cells(x)
    boxes = box_cells(w.board)
    tried = []
    for k in (1, 2, 3):
        nb = list(w.bars)
        for j in chain:
            nb[j] = turn(w.bars[j], piv, k)
        occ: set = set()
        caps_off = False
        for j in chain:
            occ |= bar_cells(nb[j])
            caps_off |= any(not inside(*p) for p in strip(nb[j].r, nb[j].c, nb[j].dir, 0))
        info = {"k": k, "fixed": bool(occ & w.board.fixed),
                "blocked": bool(occ & others or occ & walls or occ & boxes),
                "off": any(not inside(*p) for p in occ), "cap_off": caps_off}
        tried.append(info)
        v = refuse(info)
        rot = {"chain": chain, "pivot": piv, "tried": tried}
        if v == "refuse":
            return Outcome("refused", w, by="作废.rotation_blocked_refuses", rot=rot)
        if v == "next":
            continue
        return Outcome("rotated", replace(w, bars=tuple(nb)), rot=rot)
    return Outcome("refused", w, by="作废.rotation_blocked_refuses",
                   rot={"chain": chain, "pivot": piv, "tried": tried})


def scope_(w, action, ctx=None):
    if not in_level(ctx):
        return "不是关内步"
    k, _ = decode(w.board, action)
    if k != "rot":
        return "不是点图例"
    out = predict_cached(w, action)
    if out.kind != "rotated":
        return "模型预测转不成" if out.kind == "refused" else "模型不说话"
    why = unreadable_input(w, out)
    if why:
        return why
    return None


def assert_(w, action, nw, ctx=None) -> bool:
    out = predict_cached(w, action)
    for j in out.rot["chain"]:
        o = find_bar(nw, out.world.bars[j])
        if o is None or not same_bar(out.world.bars[j], o):
            return False
    return True
