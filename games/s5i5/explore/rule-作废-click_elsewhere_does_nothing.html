"""点在图标中心和图例框以外的地方，什么都不发生。"""
from __future__ import annotations

from derive.click import decode
from rule._common import in_level, unchanged

CLAIM = "点在面板图标中心、图例框以外的任何地方（包括面板框里的别处），条和墙一格不动。"
SCOPE_NOTE = "只管这样的点击。"
NOT_CLAIMED = "不主张计费行（归 不进模型/counter_burns）。"
HISTORY = ["v1 新立。起点册子里没有这一条——这一类点击语料里只有 3 次，册子没给它起名。"]
EVIDENCE = [739]


def scope_(w, action, ctx=None):
    if not in_level(ctx):
        return "不是关内步"
    k, _ = decode(w.board, action)
    if k not in ("none", "panel"):
        return "点中了图标或图例" if k != "reset" else "RESET"
    return None


def assert_(w, action, nw, ctx=None) -> bool:
    return unchanged(w, nw)
