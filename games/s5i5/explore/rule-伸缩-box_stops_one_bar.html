def stopped(w, K: int, s: int) -> set:
    """effect：被框挡住的条的下标。"""
    if s <= 0:
        return set()
    bx = box_cells(w.board)
    return {i for i, b in enumerate(w.bars)
            if b.color == K and body_cells(replace(b, length=b.length + s)) & bx}


def scope_(w, action, ctx=None):
    if not in_level(ctx):
        return "不是关内步"
    k, _ = decode(w.board, action)
    if k != "grow":
        return "不是「伸」"
    out = predict_cached(w, action)
    if out.kind != "moved":
        return "模型预测整步作废"
    why = unreadable_input(w, out)
    if why:
        return why
    if not out.move.stopped:
        return "没有一根被框挡住"
    return None


def assert_(w, action, nw, ctx=None) -> bool:
    out = predict_cached(w, action)
    for i in out.move.stopped:
        o = find_bar(nw, w.bars[i])
        if o is None or not same_bar(w.bars[i], o):
            return False
    return True
