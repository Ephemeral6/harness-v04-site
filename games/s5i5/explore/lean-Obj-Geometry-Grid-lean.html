# Obj/Geometry/Grid —— 格、方向、长方形（对应 `objects/geometry/grid.py`）

**一格 = 一个像素。** 坐标一律加上偏移 `O = 100` 存成自然数：
盘面是 `100 ≤ 行 ≤ 162`、`100 ≤ 列 ≤ 163`；伸出盘外的条可以落到 `< 100` 或 `> 162`，
但不会落到 0 以下。这样全程只用 `Nat`，内核算 `decide` 时走的是带加速的自然数运算，
而且「减过头」这种事表示不出来（偏移留足了 100 格余量）。

**这一份里没有一条定理。** 全是定义。
-/

namespace Game

/-- 盘面坐标的偏移。 -/
def O : Nat := 100

/-- 身体朝哪边长。 -/
inductive Dir where
  | U | D | L | R
  deriving DecidableEq, Repr

/-- 屏幕上看的逆时针一档（行向下）：朝下的转一档朝右。 -/
def Dir.ccw : Dir → Dir
  | .U => .L | .L => .D | .D => .R | .R => .U

/-- 反方向。 -/
def Dir.opp : Dir → Dir
  | .U => .D | .D => .U | .L => .R | .R => .L

/-- 一个格往 `d` 走 `n` 格。 -/
def Dir.go (d : Dir) (n : Nat) (r c : Nat) : Nat × Nat :=
  match d with
  | .U => (r - n, c) | .D => (r + n, c) | .L => (r, c - n) | .R => (r, c + n)

/-- 闭区间长方形（含边）。 -/
structure Rect where
  r0 : Nat
  c0 : Nat
  r1 : Nat
  c1 : Nat
  deriving DecidableEq, Repr

/-- 两块长方形至少共一格。 -/
def Rect.meets (a b : Rect) : Bool :=
  Nat.ble a.r0 b.r1 && Nat.ble b.r0 a.r1 && Nat.ble a.c0 b.c1 && Nat.ble b.c0 a.c1

/-- 整块落在盘上。 -/
def Rect.onBoard (a : Rect) : Bool :=
  Nat.ble O a.r0 && Nat.ble a.r1 (O + 62) && Nat.ble O a.c0 && Nat.ble a.c1 (O + 63)

/-- 一格。 -/
def Rect.cell (r c : Nat) : Rect := ⟨r, c, r, c⟩

/-- 整块平移：往 `d` 走 3 格。 -/
def Rect.shift (a : Rect) (d : Dir) : Rect :=
  match d with
  | .U => ⟨a.r0 - 3, a.c0, a.r1 - 3, a.c1⟩
  | .D => ⟨a.r0 + 3, a.c0, a.r1 + 3, a.c1⟩
  | .L => ⟨a.r0, a.c0 - 3, a.r1, a.c1 - 3⟩
  | .R => ⟨a.r0, a.c0 + 3, a.r1, a.c1 + 3⟩

/-- 绕格 `(pr, pc)` 逆时针转一档（屏幕坐标）：`(r, c) ↦ (pr + pc - c, pc + r - pr)`。 -/
def rotPt (pr pc r c : Nat) : Nat × Nat := (pr + pc - c, pc + r - pr)

/-- 长方形绕一格逆时针转一档：两个对角转过去，再取行列的最小 / 最大。 -/
def Rect.rot (a : Rect) (pr pc : Nat) : Rect :=
  let p := rotPt pr pc a.r0 a.c0
  let q := rotPt pr pc a.r1 a.c1
  ⟨min p.1 q.1, min p.2 q.2, max p.1 q.1, max p.2 q.2⟩

/-- 一张长方形表里有没有一块和 `a` 相交。 -/
def meetsAny (a : Rect) : List Rect → Bool
  | [] => false
  | x :: xs => a.meets x || meetsAny a xs

end Game
