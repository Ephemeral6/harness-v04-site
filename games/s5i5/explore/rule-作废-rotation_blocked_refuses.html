import Obj.World

namespace Game

inductive Verdict where
  | refuse | next | accept
  deriving DecidableEq, Repr

/-- 一档的四个读数 → 结论。 -/
def verdict (k : Nat) (fixed blocked off capOff : Bool) : Verdict :=
  if fixed then .refuse
  else if blocked then .next
  else if capOff then .refuse
  else if off && Nat.blt 1 k then .refuse
  else .accept

end Game

def verdict(info: dict) -> str:
    """一档的结论：refuse / next / accept。"""
    if info["fixed"]:
        return "refuse"
    if info["blocked"]:
        return "next"
    if info["cap_off"]:
        return "refuse"
    if info["off"] and info["k"] > 1:
        return "refuse"
    return "accept"
