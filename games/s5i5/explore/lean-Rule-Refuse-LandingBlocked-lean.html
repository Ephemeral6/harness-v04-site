import Obj.World

namespace Game

/-- 一张长方形表里有没有两块相交。 -/
def anyPairMeets : List Rect → Bool
  | [] => false
  | x :: xs => meetsAny x xs || anyPairMeets xs

def landingBlocked (bd : Board) (bars : List Bar) (walls : List Wall) : Bool :=
  anyPairMeets (bars.map Bar.whole ++ walls.map (·.rect) ++ bd.hubs.map Hub.rect)

end Game
