theorem turnBar_keeps (pr pc : Nat) : ∀ (k : Nat) (b : Bar),
    (turnBar pr pc k b).len = b.len ∧ (turnBar pr pc k b).tip = b.tip ∧ (turnBar pr pc k b).color = b.color := by
  intro k
  induction k with
  | zero => intro b; simp [turnBar]
  | succ k ih =>
    intro b
    simp only [turnBar]
    have := ih (turnBar1 pr pc b)
    simp only [turnBar1] at this
    exact this

/-- 被转的每一根条（整串里的任何一根、转任何档）长度、尖、颜色不变。 -/
theorem rotation_keeps_length (pr pc k : Nat) (b : Bar) :
    (turnBar pr pc k b).len = b.len ∧ (turnBar pr pc k b).tip = b.tip ∧ (turnBar pr pc k b).color = b.color :=
  turnBar_keeps pr pc k b

from __future__ import annotations

from derive.click import decode
from derive.step import predict_cached, unreadable_input

def scope_(w, action, ctx=None):
    if not in_level(ctx):
        return "不是关内步"
    if decode(w.board, action)[0] != "rot":
        return "不是点图例"
    out = predict_cached(w, action)
    if out.kind != "rotated":
        return "模型预测没转成"
    return unreadable_input(w, out)


def assert_(w, action, nw, ctx=None) -> bool:
    out = predict_cached(w, action)
    for j in out.rot["chain"]:
        o = find_bar(nw, out.world.bars[j])
        if o is None:
            return False
        if o.clipped:
            if o.length > w.bars[j].length:
                return False
            continue
        if o.length != w.bars[j].length or o.tip != w.bars[j].tip:
            return False
    return True
