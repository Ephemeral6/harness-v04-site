from derive.click import decode
from derive.step import predict_cached, unreadable_input

def scope_(w, action, ctx=None):
    if not in_level(ctx):
        return "不是关内步"
    if decode(w.board, action)[0] not in ("grow", "shrink"):
        return "不是点面板图标"
    out = predict_cached(w, action)
    if out.kind != "moved":
        return "模型预测整步作废"
    if not any(w.bars[i].tip for i in out.move.grown):
        return "伸缩的条都不带尖"
    return unreadable_input(w, out)


def assert_(w, action, nw, ctx=None) -> bool:
    out = predict_cached(w, action)
    for i in out.move.grown:
        if not w.bars[i].tip:
            continue
        o = find_bar(nw, out.world.bars[i])
        if o is None or (not o.tip and not o.clipped):
            return False
    return True
