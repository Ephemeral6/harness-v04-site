公理表（0 条）
    /-
# Obj/Derive/Attach —— 谁挂在谁的自由端上（对应 `derive/attach.py`）

条 j 挂在条 i 的自由端上 ⇔ j 的帽和 i 的 `face` 相交，而且两根**不同色**。
下游 = 沿这个关系往下的全部条。这里用一张与条表等长的布尔表示「在不在下游里」。
-/

import Obj.World

namespace Game

def attached (i j : Bar) : Bool := j.color != i.color && j.cap.meets i.face

/-- 表里有没有一根被标记的条，让 `j` 挂在它的自由端上。 -/
def hangsOnMarked (j : Bar) : List (Bar × Bool) → Bool
  | [] => false
  | (i, m) :: rest => (m && attached i j) || hangsOnMarked j rest

/-- 往下扩一层。 -/
def growChain (all : List (Bar × Bool)) : List (Bar × Bool) :=
  all.map (fun (j, m) => (j, m || hangsOnMarked j all))

/-- 扩 `n` 层（`n` = 条数就够把整条下游扩全）。 -/
def growChainN : Nat → List (Bar × Bool) → List (Bar × Bool)
  | 0, xs => xs
  | n + 1, xs => growChainN n (growChain xs)

/-- 从一组根出发的整条下游（含根）。 -/
def downstream (bars : List Bar) (roots : List Bool) : List Bool :=
  (growChainN bars.length (bars.zip roots)).map (·.2)

end Game
