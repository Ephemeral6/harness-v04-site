def fixedBlocks (bd : Board) (bars : List Bar) (walls : List Wall) : Bool :=
  let bad := bd.fixed ++ bd.boxes
  bars.any (fun b => meetsAny b.whole bad) || walls.any (fun w => meetsAny w.rect bad)

def refuses(m) -> bool:
    bad = m.w.board.fixed | box_cells(m.w.board)
    for b in m.nb:
        if bar_cells(b) & bad:
            return True
    return any(wall_cells(x) & bad for x in m.nw)


def scope_(w, action, ctx=None):
    return refusal_scope(w, action, ME, ctx)


def assert_(w, action, nw, ctx=None) -> bool:
    return unchanged(w, nw)
